39void prune(const Target& target, Lam* fun) {
40 auto live = Liveness(target, fun);
41 // Rewriting a jump in place must not inline its target, e.g., once pruning made both targets of a branch equal.
42 for (const auto& block : live.blocks()) {
43 if (auto lam = block.lam; lam->filter() != fun->world().lit_ff()) {
44 auto body = lam->body();
45 lam->unset()->set(false, body);
46 }
47 }
48
49 auto count = DefMap<size_t>();
50 for_each_use(live, [&](const Def* u) { ++count[u]; });
51
52 auto unused = [&](const Def* d) {
53 auto n = fe::lookup(count, d);
54 return !n || *n == 0;
55 };
56 auto dead = DefSet();
57 for (bool todo = true; todo;) {
58 todo = false;
59 for (const auto& block : live.blocks()) {
60 auto ex = block.lam->body()->as<App>()->callee()->isa<Extract>();
61 auto cond = ex ? target.isa_sched_cc(ex->index()) : nullptr;
62 for (auto insn : block.insns | std::views::reverse) {
63 if (dead.contains(insn) || insn == cond || !target.is_removable(insn)) continue;
64 auto ds = live.defs(insn);
65 if (ds.empty() || !std::ranges::all_of(ds, unused)) continue;
66 dead.emplace(insn);
67 todo = true;
68 for (auto u : live.uses(insn))
69 --count[u];
70 }
71 }
72 }
73 if (dead.empty()) return;
74
75 auto p = Prune(target, fun->world(), dead);
76 rewrite_blocks(live, p);
77}

168void Colorer::nodes() {
169 auto used = DefSet();
170 for_each_use(live_, [&](const Def* u) { used.emplace(u); });
171
172 for (const auto& block : live_.blocks()) {
173 for (auto x : live_.params(block))
174 if (is_value(x) && used.contains(x)) add(x);
175 for (auto insn : block.insns) {
176 auto implied = DefSet();
177 for (auto [j, _] : target_.constraint(insn).results)
178 implied.emplace(Target::result(insn, j));
179 for (auto d : live_.defs(insn))
180 if (is_value(d) && (used.contains(d) || implied.contains(d))) add(d);
181 }

185void Colorer::constraints() {
186 for (const auto& block : live_.blocks()) {
187 for (auto insn : block.insns) {
188 auto c = target_.constraint(insn);
189 // Fixed operands, results, and clobbers of an instruction are general-purpose registers.
190 if (cls_ == Liveness::Gpr) {
191 for (auto [j, r] : c.fixed)
192 fix(c.ops[j], r);
193 for (auto [j, r] : c.results)
194 fix(Target::result(insn, j), r);
195 for (auto r : c.clobbers)
196 for (auto u : live_.uses(insn))
197 forbid(u, r);
198 }
199 if (c.tied && is_value(c.ops[*c.tied])) link(c.ops[*c.tied], Target::result(insn, 0), tie_);
200 }
201 }

237void Colorer::solve() {
238 for (size_t i = 0; i != values_.size(); ++i)
239 graph_.set_node_costs(i, costs_[i]);
240
241 auto priority = fe::Vector<size_t>(values_.size(), [](size_t i) { return i; });
242 auto sol = pbqp::solve(graph_, priority);
243 if (dot_) {
244 std::println(*dot_, "// fun {}", fun_);
245 for (size_t i = 0; i != values_.size(); ++i)
246 std::println(*dot_, "// n{} = {}", i, values_[i]);
247 pbqp::write_dot(*dot_, graph_, sol.choices);
248 }
249 if (!sol.feasible()) {
250 auto conflicts = DefVec();
251 for (size_t i = 0; i != values_.size(); ++i)
252 if (sol.conflicts[i]) conflicts.emplace_back(values_[i]);
253 fun_->blame("{} backend: cannot assign registers in `{}`; conflicting values: {}", target_.name(), fun_,
254 fe::Join(conflicts, ", "))
255 .bail();
256 }
257 for (size_t i = 0; i != values_.size(); ++i)
258 regs_[values_[i]] = sol.choices[i];

290void Checker::walk(const Def* def, const Def* at) {
291 if (def->isa_mut() || def->is_closed() || target_.is_mem(def)) return;
292 if (target_.is_in(def)) {
293 auto v = target_.strip(def);
294 if (target_.cls(v) != cls_) return;
295 auto r = Target::in_reg(def);
296 auto [i, ins] = regs_.emplace(v, r);
297 if (!live_.local(v) || !is_value(v)) fail(at, std::format("`{}` is no value", v));
298 if (target_.reserved(cls_, r)) fail(at, std::format("`{}` is in a reserved register", v));
299 if (!ins && i->second != r) fail(at, std::format("`{}` is in two registers", v));
300 return;
301 }
302 if (live_.local(def)) {
303 for (auto d : def->projs())
304 if (is_value(d)) fail(at, std::format("`{}` is used without a register", d));
305 return;
306 }
307 if (auto app = def->isa<App>(); app && target_.prev_mem(app)) fail(at, std::format("`{}` is off the chain", app));
308 for (auto op : def->ops())
309 walk(op, at);
310}

312void Checker::run() {
313 for (const auto& block : live_.blocks()) {
314 for (auto insn : block.insns) {
315 walk(insn->callee(), insn);
316 walk(insn->arg(), insn);
317 }
318 auto term = block.lam->body()->as<App>();
319 if (auto ex = term->callee()->isa<Extract>(); ex && target_.isa_sched_cc(ex->index())) {
320 walk(ex->tuple(), term);
321 walk(term->arg(), term);
322 } else {
323 walk(term, term);
324 }
325 }
326
327 for (const auto& block : live_.blocks()) {
328 for (auto insn : block.insns) {
329 auto c = target_.constraint(insn);
330 if (cls_ == Liveness::Gpr) {
331 for (auto [j, r] : c.results) {
332 auto d = Target::result(insn, j);
333 if (regs_.emplace(d, r).first->second != r)
334 fail(insn, std::format("`{}` is not in its fixed register", d));
335 }
336 for (auto [j, r] : c.fixed)
337 if (!target_.is_in(c.ops[j]) || Target::in_reg(c.ops[j]) != r)
338 fail(insn, std::format("operand `{}` is not in its fixed register", c.ops[j]));
339 for (auto r : c.clobbers)
340 for (auto u : live_.uses(insn))
341 if (reg(u) == r) fail(insn, std::format("operand `{}` is in a clobbered register", u));
342 }
343 if (c.tied && is_value(c.ops[*c.tied]) && reg(c.ops[*c.tied]) != reg(Target::result(insn, 0)))
344 fail(insn, std::format("`{}` is not in the register of operand `{}`", Target::result(insn, 0),
345 c.ops[*c.tied]));
346 }
347 }
348
349 for (const auto& block : live_.blocks()) {
350 interferences(
351 live_, block, [&](const Def* v) { return regs_.contains(v); },
352 [&](const Def* a, const Def* b) {
353 if (regs_[a] == regs_[b])
354 fail(block.lam, std::format("`{}` and `{}` interfere but share a register", a, b));
355 });
356 if (auto k = call_cont(live_, fun_, block))
357 for (auto v : k->live_in)
358 if (auto r = reg(v); r && target_.caller_saved(cls_, *r))
359 fail(block.lam, std::format("`{}` lives across a call in a caller-saved register", v));
360 }
361}
