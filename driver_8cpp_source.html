27fs::path adjust_libmim_path(const fs::path& libmim_path) {
28 if (has_plugin_dir(libmim_path)) return libmim_path;
29
30 auto dir = libmim_path.parent_path();
31 auto lib_name = libmim_path.filename();
32 while (!dir.empty()) {
33 if (dir == dir.root_path()) break;
34
35 std::error_code ignore;
36 auto candidate = dir / MIM_LIBDIR / "mim";
37 if (fs::is_directory(candidate, ignore) && !ignore) return candidate.parent_path() / lib_name;
38
39 dir = dir.parent_path();
40 }
41
42 return libmim_path;
43}
