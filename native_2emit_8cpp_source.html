43/// Sequentializes the parallel copies @p moves between registers and @p slot_moves between spill slots.
44void Emitter::parallel(Moves moves, fe::Vector<std::pair<size_t, size_t>> slot_moves) {
45 std::erase_if(slot_moves, [](auto m) { return m.first == m.second; });
46 for (auto [_, from] : slot_moves)
47 push_slot(from);
48 for (auto [to, _] : slot_moves | std::views::reverse)
49 pop_slot(to);
50
51 auto imms = Moves();
52 for (auto& m : moves)
53 if (m.second.imm) imms.emplace_back(m);
54 std::erase_if(moves, [](const auto& m) { return m.second.imm || m.first == m.second.reg; });
55
56 while (!moves.empty()) {
57 auto free = std::ranges::find_if(moves, [&](const auto& m) {
58 return std::ranges::none_of(moves, [&](const auto& n) { return n.second.reg == m.first; });
59 });
60 if (free != moves.end()) {
61 move(free->first, free->second.reg);
62 moves.erase(free);
63 continue;
64 }
65 // Only cycles are left.
66 auto [d, s] = moves.back();
67 moves.pop_back();
68 if (auto tmp = scratch(d.cls)) {
69 move(*tmp, d);
70 for (auto& [_, from] : moves)
71 if (from.reg == d) from.reg = *tmp;
72 move(d, s.reg);
73 } else {
74 swap(s.reg, d);
75 for (auto& [_, from] : moves)
76 if (from.reg == d)
77 from.reg = s.reg;
78 else if (from.reg == s.reg)
79 from.reg = d;
80 std::erase_if(moves, [](const auto& m) { return m.first == m.second.reg; });
81 }
82 }
83
84 for (auto [dst, s] : imms)
85 mov_imm(dst, s.imm);
86}

88void Emitter::copy(Lam* target, const Def* arg) {
89 auto moves = Moves();
90 auto slot_moves = fe::Vector<std::pair<size_t, size_t>>();
91 auto n = target->num_vars();
92 for (size_t i = 0; i != n; ++i) {
93 auto x = target->var(n, i);
94 auto a = arg->proj(n, i);
95 if (target_.is_mem(x)) continue;
96 if (auto cls = target_.cls(x); cls == Target::Stack)
97 slot_moves.emplace_back(spill_slot(x), spill_slot(a));
98 else if (auto r = fe::lookup(regs_, x))
99 moves.emplace_back(Reg{*r, cls}, source(a));
100 }
101 parallel(std::move(moves), std::move(slot_moves));

108void Emitter::call(Lam* callee, const Def* arg) {
109 auto n = arg->num_projs();
110 auto moves = Moves();
111 auto num = std::array<size_t, Target::Num_Classes>{};
112 for (size_t i = 0; i + 1 < n; ++i) {
113 if (auto x = arg->proj(n, i); !target_.is_mem(x)) {
114 auto cls = target_.cls(target_.strip(x));
115 auto& c = num[cls];
116 if (c == target_.num_arg_regs(cls))
117 arg->blame("{} backend: too many arguments are not supported yet", backend()).bail();
118 moves.emplace_back(Reg{target_.arg_reg(cls, c++), cls}, source(x));
119 }
120 }
121 parallel(std::move(moves), {});
122 call_insn(symbol(callee), num);
123
124 auto ret = arg->proj(n, n - 1);
125 if (ret == ret_) return epilogue();
126
127 auto k = Lam::isa_mut_basicblock(ret);
128 auto m = k->num_vars();
129 auto back = Moves();
130 for (size_t i = 0; i != m; ++i)
131 if (auto x = k->var(m, i); !target_.is_mem(x)) {
132 auto cls = target_.cls(x);
133 if (auto r = fe::lookup(regs_, x)) back.emplace_back(Reg{*r, cls}, Src{.reg = {target_.ret_reg(cls), cls}});
134 }
135 parallel(std::move(back), {});
136 jump(k);
137}

139void Emitter::insn(const App* app) {
140 const auto& ann = target_.annexes();
141 if (Target::is_axm(ann.sched, app)) {
142 auto f = Target::sched_insn(app);
143 auto res = app->proj(2, 1);
144 if (Target::is_axm(ann.spill, f, 1)) return spill(source(app->arg()), app->arg()->type(), spill_slot(res));
145 if (Target::is_axm(ann.reload, f, 1))
146 return reload(Reg{def_reg(res), target_.cls(res)}, res->type(), spill_slot(app->arg()));
147 } else if (Target::is_axm(ann.slot, app)) {
148 auto first = alloc_slots(nat(app->callee()->as<App>()->arg()));
149 return slot_addr(def_reg(app->proj(2, 1)), first, num_slots_ - first);
150 }
151 step(app);
152}

154void Emitter::block(const Liveness::Block& b) {
155 for (auto app : b.insns)
156 insn(app);
157
158 auto term = b.lam->body()->as<App>();
159 auto arg = term->arg();
160 auto callee = term->callee();
161 if (callee == ret_) {
162 auto moves = Moves();
163 for (auto a : term->args()) {
164 if (target_.is_mem(a)) continue;
165 auto cls = target_.cls(target_.strip(a));
166 moves.emplace_back(Reg{target_.ret_reg(cls), cls}, source(a));
167 }
168 if (moves.size() > 1) b.lam->blame("{} backend: `{}` returns more than one value", backend(), b.lam).bail();
169 parallel(std::move(moves), {});
170 epilogue();
171 } else if (auto f = Lam::isa_mut_returning(callee)) {
172 call(f, arg);
173 } else if (auto ex = callee->isa<Extract>()) {
174 auto targets = ex->tuple();
175 if (targets->num_projs() != 2)
176 b.lam->blame("{} backend: only two-way branches are supported yet", backend()).bail();
177 auto [e, t] = targets->projs<2>([](const Def* d) { return Lam::isa_mut_basicblock(d); });
178 if (!e || !t) b.lam->blame("{} backend: `{}` branches to something other than blocks", backend(), b.lam).bail();
179 auto c = ex->index();
180 auto trampoline = takes_values(target_, t);
181 auto t_lbl = trampoline ? label(b.lam) + ".t" : label(t);
182 if (auto app = target_.isa_sched_cc(c)) {
183 use_flags(app->arg());
184 branch_cc(std::get<0>(Axm::get(Target::sched_insn(app)))->sub(), t_lbl);
185 } else if (Lit::isa(c)) {
186 c->blame("{} backend: a branch on the literal `{}`", backend(), c).bail();
187 } else {
188 branch_if(use_reg(c), t_lbl);
189 }
190 copy(e, arg);
191 if (trampoline) {
192 jmp(label(e));
193 label_here(t_lbl);
194 copy(t, arg);
195 jump(t);
196 } else {
197 jump(e);
198 }
199 } else if (auto bb = Lam::isa_mut_basicblock(callee)) {
200 copy(bb, arg);
201 jump(bb);
202 } else {
203 b.lam->blame("{} backend: cannot emit the jump to `{}`", backend(), callee).bail();
204 }
