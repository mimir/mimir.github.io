24/// Pipeline phase for `%ll.emit`.
25/// Writes the LLVM IR of the fully lowered world to `<world>.ll` (or `a.ll` if the world is unnamed).
26/// The output path can be overridden on the command line via `-X ll:o=<file>` or `-X ll:output=<file>`.
27/// The runtime-wrapper linking mode is selected via `-X ll:rt=embed` (default) or `-X ll:rt=extern`.
28class Emit : public Phase {
29public:
30 Emit(World& world, flags_t annex)
31 : Phase(world, annex) {}
32
33 void start() override {
34 auto name = world().name() ? std::string(world().name().view()) : "a"s;
35 auto path = name + ".ll"s;
36 auto rt = Emitter::Rt::embed;
37 for (const auto& arg : args()) {
38 world().DLOG("ll backend arg: `{}`", arg);
39 if (arg.starts_with("o="))
40 path = arg.substr(2);
41 else if (arg.starts_with("output="))
42 path = arg.substr(7);
43 else if (arg == "rt=embed")
44 rt = Emitter::Rt::embed;
45 else if (arg == "rt=extern")
46 rt = Emitter::Rt::ext;
47 }
48 auto ofs = std::ofstream(path);
49 auto emitter = Emitter(world(), "llvm_emitter", ofs);
50 emitter.rt_mode(rt);
51 if (rt == Emitter::Rt::embed) emitter.load_rt_module("ll_rt.ll");
52 emitter.run();
53 }
54};

63std::string Emitter::convert_impl(const Def* type, bool simd) {
64 if (auto i = types_.find(type); i != types_.end()) return i->second;
65
66 if (Axm::isa<mem::M>(type)) fe::throwf("ll backend: cannot convert %mem.M type '{}'", type);
67 std::ostringstream s;
68 std::string name;
69
70 if (type->isa<Nat>()) {
71 return types_[type] = "i64";
72 } else if (Idx::isa(type)) {
73 return types_[type] = "i" + std::to_string(Idx::expect_bitwidth(type, "a statically-sized index type"));
74 } else if (auto w = math::isa_f(type)) {
75 switch (*w) {
76 case 16: return types_[type] = "half";
77 case 32: return types_[type] = "float";
78 case 64: return types_[type] = "double";
79 default: fe::throwf("ll backend: unsupported floating-point width {} in type '{}'", *w, type);
80 }
81 } else if (auto ptr = Axm::isa<mem::Ptr>(type)) {
82 auto [pointee, addr_space] = ptr->args<2>();
83 std::print(s, "{} addrspace({})*", convert(pointee, false), addr_space);
84 } else if (auto arr = type->isa<Arr>()) {
85 if (auto se = is_simd(arr); se && simd) {
86 auto [size, elem] = *se;
87 std::print(s, "<{} x {}>", size, convert(elem));
88 } else {
89 u64 size = 0;
90 if (auto arity = Lit::isa(arr->arity())) size = *arity;
91 std::print(s, "[{} x {}]", size, convert(arr->body(), false));
92 }
93 } else if (auto pi = type->isa<Pi>()) {
94 if (!Pi::isa_returning(pi)) fe::throwf("ll backend: cannot convert the type of a basic block: '{}'", pi);
95 std::print(s, "{} (", convert_ret_pi(pi->ret_pi()));
96
97 if (auto t = detail::isa_mem_sigma_2(pi->dom()))
98 s << convert(t);
99 else {
100 auto doms = pi->doms();
101 for (auto sep = ""; auto dom : doms.view().rsubspan(1)) {
102 if (Axm::isa<mem::M>(dom)) continue;
103 s << sep << convert(dom);
104 sep = ", ";
105 }
106 }
107 s << ")*";
108 } else if (auto t = detail::isa_mem_sigma_2(type)) {
109 return convert(t);
110 } else if (auto sigma = type->isa<Sigma>()) {
111 if (sigma->isa_mut()) {
112 name = id(sigma);
113 types_[sigma] = name;
114 std::print(s, "{} = type", name);
115 }
116
117 std::print(s, "{{");
118 for (auto sep = ""; auto t : sigma->ops()) {
119 if (Axm::isa<mem::M>(t)) continue;
120 s << sep << convert(t);
121 sep = ", ";
122 }
123 std::print(s, "}}");
124 } else {
125 fe::throwf("ll backend: cannot convert type '{}' to LLVM", type);
126 }
127
128 if (name.empty()) return types_[type] = s.str();
129
130 if (s.str().empty()) fe::throwf("ll backend: empty type declaration for '{}'", type);
131 type_decls_ << s.str() << '\n';
132 return types_[type] = name;
133}

137 for (const auto& [phi, args] : bb.phis) {

181void Emitter::emit_epilogue_impl(Lam* lam) {
182 auto app = lam->body()->expect<App>("an application in tail position");
183 auto& bb = lam2bb_[lam];
184 // A target-specific intrinsic in tail position (e.g. %gpu.launch) emits its own code and
185 // yields the continuation to branch to.
186 if (auto ret = isa_targetspecific_intrinsic(bb, app)) return bb.tail("br label {}", *ret);
187 if (app->callee() == root()->ret_var()) { // return
188 Vector<std::string> values;
189 DefVec types;
190 for (auto arg : app->args()) {
191 if (auto val = emit_unsafe(arg); !val.empty()) {
192 values.emplace_back(val);
193 types.emplace_back(arg->type());
194 }
195 }
196
197 switch (values.size()) {
198 case 0: return bb.tail("ret void");
199 case 1:
200 return Axm::isa<mem::M>(types[0]) ? bb.tail("ret void")
201 : bb.tail("ret {} {}", convert(types[0]), values[0]);
202 default: {
203 std::string type;
204 std::string prev;
205
206 if (auto se = is_simd_aggregate(types)) {
207 auto common_src = find_common_simd_src(app);
208 if (common_src) {
209 auto v_src = emit(common_src);
210 auto t = convert(common_src->type());
211 return bb.tail("ret {} {}", t, v_src);
212 }
213 auto [size, elem] = *se;
214 auto val_t = convert(elem);
215
216 type = std::format("<{} x {}>", size, val_t);
217 for (auto val : values) {
218 if (prev.empty())
219 prev = "<";
220 else
221 prev += ", ";
222 prev += std::format("{} {}", val_t, val);
223 }
224 prev += ">";
225 } else {
226 prev = "undef";
227 type = convert(world().sigma(types));
228 for (size_t i = 0, n = values.size(); i != n; ++i) {
229 if (auto mem = Axm::isa<mem::M>(types[i])) continue;
230 auto v_elem = values[i];
231 auto t_elem = convert(types[i]);
232 auto namei = "%ret_val." + std::to_string(i);
233 bb.tail("{} = insertvalue {} {}, {} {}, {}", namei, type, prev, t_elem, v_elem, i);
234 prev = namei;
235 }
236 }
237 bb.tail("ret {} {}", type, prev);
238 }
239 }
240
241 } else if (auto dispatch = Dispatch(app)) {
242 for (auto callee : dispatch.tuple()->projs([](const Def* def) { return def->isa_mut<Lam>(); }))
243 if (size_t n = callee->num_tvars(); n == 1 && is_simd(callee->var(0)->type()))
244 emit_phi(callee, callee->var(0), emit(app->arg(n, 0)), lam);
245 else
246 emit_phi_args(callee, app, lam);
247
248 auto v_index = emit(dispatch.index());
249 size_t n = dispatch.num_targets();
250 auto bbs = absl::FixedArray<std::string>(n);
251 for (size_t i = 0; i != n; ++i)
252 bbs[i] = emit(dispatch.target(i));
253
254 if (auto branch = Branch(app)) return bb.tail("br i1 {}, label {}, label {}", v_index, bbs[1], bbs[0]);
255
256 auto t_index = convert(dispatch.index()->type());
257 bb.tail("switch {} {}, label {} [ ", t_index, v_index, bbs[0]);
258 for (size_t i = 1; i != n; ++i)
259 std::print(bb.tail().back(), "{} {}, label {} ", t_index, std::to_string(i), bbs[i]);
260 std::print(bb.tail().back(), "]");
261 } else if (app->callee()->isa<Bot>()) {
262 return bb.tail("ret ; bottom: unreachable");
263 } else if (auto callee = Lam::isa_mut_basicblock(app->callee())) { // ordinary jump
264
265 if (auto common_src = find_common_simd_src(app)) {
266 auto v_src = emit(common_src);
267 auto callee_var = callee->var();
268 if (simd_phi_.find(callee) == simd_phi_.end()) simd_phi_[callee] = callee_var;
269 auto key = simd_phi_[callee];
270 emit_phi(callee, key, v_src, lam);
271 for (auto var : callee->vars())
272 locals_[var] = id(key);
273 locals_[callee_var] = id(key);
274 } else {
275 emit_phi_args(callee, app, lam);
276 }
277 return bb.tail("br label {}", id(callee));
278
279 } else if (auto longjmp = Axm::isa<clos::longjmp>(app)) {
280 declare("void @longjmp(i8*, i32) noreturn");
281
282 auto [mem, jbuf, tag] = app->args<3>();
283 emit_unsafe(mem);
284 auto v_jb = emit(jbuf);
285 auto v_tag = emit(tag);
286 bb.tail("call void @longjmp(i8* {}, i32 {})", v_jb, v_tag);
287 return bb.tail("unreachable");
288 } else if (auto mslot = Axm::isa<mem::mslot>(app)) {
289 // Continuation-based stack slot: allocate and jump to the passed continuation with the fresh pointer.
290 auto [Ta, rest] = mslot->uncurry_args<2>();
291 auto [pointee, addr_space] = Ta->projs<2>();
292 auto [msize, ret] = rest->projs<2>();
293 emit_unsafe(msize->proj(0)); // mem
294 // TODO array with size
295 auto ret_lam = ret->expect_mut<Lam>("a %mem.slot continuation");
296 auto ptr = ret_lam->var(2, 1);
297 auto v_ptr = emit_slot(bb, app, pointee, addr_space);
298 emit_phi(ret_lam, ptr, v_ptr, lam);
299 return bb.tail("br label {}", id(ret_lam));
300 } else if (Pi::isa_returning(app->callee_type())) { // function call
301 auto v_callee = emit(app->callee());
302
303 Vector<std::string> args;
304 auto app_args = app->args();
305 for (auto arg : app_args.view().rsubspan(1))
306 if (auto v_arg = emit_unsafe(arg); !v_arg.empty()) args.emplace_back(convert(arg->type()) + " " + v_arg);
307
308 if (app->args().back()->isa<Bot>()) {
309 // TODO: Perhaps it'd be better to simply η-wrap this prior to the BE...
310 if (convert_ret_pi(app->callee_type()->ret_pi()) != "void")
311 fe::throwf("ll backend: call with a ⊥ return continuation must return void, but '{}' does not", app);
312 bb.tail("call void {}({})", v_callee, fe::Join(args));
313 return bb.tail("unreachable");
314 }
315
316 auto ret_lam = app->args().back()->expect_mut<Lam>("a return continuation");
317 size_t n = 0;
318 for (auto var : ret_lam->vars())
319 if (!Axm::isa<mem::M>(var->type())) ++n;
320
321 if (n == 0) {
322 bb.tail("call void {}({})", v_callee, fe::Join(args));
323 } else {
324 auto name = "%" + app->unique_name() + "ret";
325 auto t_ret = convert_ret_pi(ret_lam->type());
326 bb.tail("{} = call {} {}({})", name, t_ret, v_callee, fe::Join(args));
327 emit_phi(ret_lam, ret_lam->var(), name, lam);
328 }
329
330 return bb.tail("br label {}", id(ret_lam));
331 }

334std::string Emitter::emit_bb_impl(BB& bb, const Def* def) {
335 if (auto lam = def->isa<Lam>()) return id(lam);
336
337 auto name = id(def);
338 std::string op;
339
340 auto emit_tuple = [&](const Def* tuple) {
341 if (detail::isa_mem_sigma_2(tuple->type())) {
342 emit_unsafe(tuple->proj(2, 0));
343 return emit(tuple->proj(2, 1));
344 }
345
346 if (tuple->is_closed()) {
347 bool is_array = tuple->type()->isa<Arr>();
348 auto simd_array = convert(tuple->type()).front() == '<'; // needed to respect pointer context
349 std::string s;
350 s += simd_array ? "<" : is_array ? "[" : "{";
351 auto sep = "";
352 for (size_t i = 0, n = tuple->num_projs(); i != n; ++i) {
353 auto e = tuple->proj(n, i);
354 if (auto v_elem = emit_unsafe(e); !v_elem.empty()) {
355 auto t_elem = convert(e->type());
356 s += sep + t_elem + " " + v_elem;
357 sep = ", ";
358 }
359 }
360
361 return s += simd_array ? ">" : is_array ? "]" : "}";
362 }
363
364 std::string prev = "undef";
365 auto t = convert(tuple->type());
366 for (size_t src = 0, dst = 0, n = tuple->num_projs(); src != n; ++src) {
367 auto e = tuple->proj(n, src);
368 if (auto elem = emit_unsafe(e); !elem.empty()) {
369 auto elem_t = convert(e->type());
370 // TODO: check dst vs src
371 auto namei = name + "." + std::to_string(dst);
372 if (t.front() == '<') // not using is_simd to respect the pointer context (Pointer Pointee case)
373 prev = bb.assign(namei, "insertelement {} {}, {} {}, {} {}", t, prev, elem_t, elem, elem_t, dst);
374 else
375 prev = bb.assign(namei, "insertvalue {} {}, {} {}, {}", t, prev, elem_t, elem, dst);
376 dst++;
377 }
378 }
379 return prev;
380 };
381
382 if (def->isa<Var>()) {
383 if (is_simd(def->type())) return id(def);
384 auto ts = def->type()->projs();
385 if (std::ranges::any_of(ts, [](auto t) { return Axm::isa<mem::M>(t); })) return {};
386 return emit_tuple(def);
387 }
388
389 auto emit_gep_index = [&](const Def* index) {
390 auto v_i = emit(index);
391 auto t_i = convert(index->type());
392
393 if (auto size = Idx::isa(index->type())) {
394 if (auto w = Idx::size2bitwidth(size); w && *w < 64) {
395 v_i = bb.assign(name + ".zext",
396 "zext {} {} to i{} ; add one more bit for gep index as it is treated as signed value",
397 t_i, v_i, *w + 1);
398 t_i = "i" + std::to_string(*w + 1);
399 }
400 }
401
402 return std::pair(v_i, t_i);
403 };
404
405 if (auto lit = def->isa<Lit>()) {
406 if (lit->type()->isa<Nat>() || Idx::isa(lit->type())) {
407 return std::to_string(lit->get());
408 } else if (auto w = math::isa_f(lit->type())) {
409 std::stringstream s;
410 u64 hex;
411
412 switch (*w) {
413 case 16:
414 s << "0xH" << std::setfill('0') << std::setw(4) << std::right << std::hex << lit->get<u16>();
415 return s.str();
416 case 32: {
417 hex = std::bit_cast<u64>(f64(lit->get<f32>()));
418 break;
419 }
420 case 64: hex = lit->get<u64>(); break;
421 default: fe::throwf("ll backend: unsupported floating-point width {} for literal '{}'", *w, def);
422 }
423
424 s << "0x" << std::setfill('0') << std::setw(16) << std::right << std::hex << hex;
425 return s.str();
426 }
427 fe::throwf("ll backend: cannot emit literal '{}' of type '{}'", def, def->type());
428 } else if (def->isa<Bot>()) {
429 return "undef";
430 } else if (auto top = def->isa<Top>()) {
431 if (Axm::isa<mem::M>(top->type())) return {};
432 // bail out to error below
433 } else if (auto tuple = def->isa<Tuple>()) {
434 return emit_tuple(tuple);
435 } else if (auto pack = def->isa<Pack>()) {

437 return emit_tuple(pack);
438 } else if (auto sel = Select(def)) {

441 auto cond_t = convert(sel.cond()->type());

444 } else if (auto extract = def->isa<Extract>()) {

451 // before emitting the index, as it might be a weird value for mem vars.

453 if (auto sigma = extract->type()->isa<Sigma>(); sigma && sigma->num_ops() == 0) return {};

482 auto v_tup = emit(insert->tuple());
483 auto v_val = emit(insert->value());
484 if (auto idx = Lit::isa(insert->index())) {
485 auto v_idx = emit(insert->index());
486 if (is_simd(insert->tuple()->type()))
487

489 else

492 } else {
493 if (is_simd(insert->tuple()->type())) {
494 auto v_i = emit(insert->index());
495 auto t_i = convert(insert->index()->type());
496 if (t_i != "i32") {
497 auto w_src = Idx::expect_bitwidth(insert->index()->type(), "an %insert index of known width");
498 v_i = bb.assign(name + ".idx", "{} {} {} to i32", w_src < 32 ? "zext" : "trunc", t_i, v_i);
499 }

501 }

503 auto [v_i, t_i] = emit_gep_index(insert->index());

505 "{}.alloca = alloca {} ; copy to alloca to emulate insert with store + gep + load", name, t_tup);
506 std::print(bb.body().emplace_back(), "store {} {}, {}* {}.alloca", t_tup, v_tup, t_tup, name);
507 std::print(bb.body().emplace_back(), "{}.gep = getelementptr inbounds {}, {}* {}.alloca, i64 0, {} {}",
508 name, t_tup, t_tup, name, t_i, v_i);
509 std::print(bb.body().emplace_back(), "store {} {}, {}* {}.gep", t_val, v_val, t_val, name);
510 return bb.assign(name, "load {}, {}* {}.alloca", t_tup, t_tup, name);
511 }
512 } else if (auto global = def->isa<Global>()) {
513 auto v_init = emit(global->init());
514 auto [pointee, addr_space] = Axm::expect<mem::Ptr>(global->type(), "a %mem.Ptr")->args<2>();
515 std::print(vars_decls_, "{} = global {} {}\n", name, convert(pointee), v_init);
516 return globals_[global] = name;
517 } else if (auto nat = Axm::isa<core::nat>(def)) {
518 auto [a, b] = nat->args<2>([this](auto def) { return emit(def); });
519
520 switch (nat.id()) {
521 case core::nat::add: return bb.assign(name, "add nsw nuw i64 {}, {}", a, b);
522 case core::nat::sub: {
523 // nat subtraction saturates at 0: cap result when b > a
524 auto ugt = bb.assign(name + ".ugt", "icmp ugt i64 {}, {}", b, a);
525 auto raw = bb.assign(name + ".raw", "sub i64 {}, {}", a, b);
526 return bb.assign(name, "select i1 {}, i64 0, i64 {}", ugt, raw);
527 }
528 case core::nat::mul: return bb.assign(name, "mul nsw nuw i64 {}, {}", a, b);
529 // %core.nat.div/mod define division/modulo by zero as `a / 0 = 0` and `a % 0 = a`.
530 // replace a zero divisor by 1 to keep udiv/urem well-defined, then select the
531 // defined result for the zero case (0 for div, a for mod).
532 case core::nat::div: {
533 auto bz = bb.assign(name + ".bz", "icmp eq i64 {}, 0", b);
