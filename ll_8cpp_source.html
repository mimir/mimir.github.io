24/// Pipeline phase for `%ll.emit`.
25/// Writes the LLVM IR of the fully lowered world to `<world>.ll` (or `a.ll` if the world is unnamed).
26/// The output path can be overridden on the command line via `-X ll:o=<file>` or `-X ll:output=<file>`.
27/// The runtime-wrapper linking mode is selected via `-X ll:rt=embed` (default) or `-X ll:rt=extern`.
28class Emit : public Phase {
29public:
30 Emit(World& world, flags_t annex)
31 : Phase(world, annex) {}
32
33 void start() override {
34 world().log().d("ll backend args: {}", fe::Join(args()));
35
36 auto name = world().name() ? world().name().str() : "a"s;
37 auto path = name + ".ll"s;
38 if (auto o = arg_value(args(), "o", "output")) path = *o;
39 auto rt = arg_value(args(), "rt") == "extern" ? Emitter::Rt::ext : Emitter::Rt::embed;
40
41 auto ofs = std::ofstream(path);
42 auto emitter = Emitter(world(), "llvm_emitter", ofs);
43 emitter.rt_mode(rt);
44 if (rt == Emitter::Rt::embed) emitter.load_rt_module("ll_rt.ll");
45 emitter.run();
46 }
47};

56std::string Emitter::convert_impl(const Def* type, bool simd) {
57 if (auto i = types_.find(type); i != types_.end()) return i->second;
58
59 if (Axm::isa<mem::M>(type)) fe::throwf(MIM_LL_BE "cannot convert `%mem.M` type `{}`", type);
60 std::ostringstream s;
61 std::string name;
62
63 if (type->isa<Nat>()) {
64 return types_[type] = "i64";
65 } else if (Idx::isa(type)) {
66 // `Idx 1` is zero bits wide information-theoretically, but LLVM has no i0: clamp to i1
67 // (its only value is 0, so any i1 zext/trunc of it stays correct).
68 auto w = Idx::expect_bitwidth(type, "a statically-sized index type");
69 return types_[type] = "i" + std::to_string(std::max<nat_t>(1, w));
70 } else if (auto w = math::isa_f(type)) {
71 switch (*w) {
72 case 16: return types_[type] = "half";
73 case 32: return types_[type] = "float";
74 case 64: return types_[type] = "double";
75 default: fe::throwf(MIM_LL_BE "unsupported floating-point width {} in type `{}`", *w, type);
76 }
77 } else if (auto ptr = Axm::isa<mem::Ptr>(type)) {
78 auto [pointee, addr_space] = ptr->args<2>();
79 std::print(s, "{} addrspace({})*", convert(pointee, false), addr_space);
80 } else if (auto arr = type->isa<Arr>()) {
81 if (auto se = is_simd(arr); se && simd) {
82 auto [size, elem] = *se;
83 std::print(s, "<{} x {}>", size, convert(elem));
84 } else {
85 u64 size = 0;
86 if (auto arity = Lit::isa(arr->arity())) size = *arity;
87 std::print(s, "[{} x {}]", size, convert(arr->body(), false));
88 }
89 } else if (auto pi = type->isa<Pi>()) {
90 if (!Pi::isa_returning(pi)) fe::throwf(MIM_LL_BE "cannot convert the type of a basic block: `{}`", pi);
91 std::print(s, "{} (", convert_ret_pi(pi->ret_pi()));
92
93 if (auto t = detail::isa_mem_sigma_2(pi->dom()))
94 s << convert(t);
95 else {
96 auto doms = pi->doms();
97 for (auto sep = ""; auto dom : doms.view().rsubspan(1)) {
98 if (Axm::isa<mem::M>(dom)) continue;
99 s << sep << convert(dom);
100 sep = ", ";
101 }
102 }
103 s << ")*";
104 } else if (auto t = detail::isa_mem_sigma_2(type)) {
105 return convert(t);
106 } else if (auto sigma = type->isa<Sigma>()) {
107 if (sigma->isa_mut()) {
108 name = id(sigma);
109 types_[sigma] = name;
110 std::print(s, "{} = type", name);
111 }
112
113 std::print(s, "{{");
114 for (auto sep = ""; auto t : sigma->ops()) {
115 if (Axm::isa<mem::M>(t)) continue;
116 s << sep << convert(t);
117 sep = ", ";
118 }
119 std::print(s, "}}");
120 } else {
121 fe::throwf(MIM_LL_BE "cannot convert type `{}` to LLVM", type);
122 }
123
124 if (name.empty()) return types_[type] = s.str();
125
126 if (s.str().empty()) fe::throwf(MIM_LL_BE "empty type declaration for `{}`", type);
127 type_decls_ << s.str() << '\n';
128 return types_[type] = name;
129}

133 for (const auto& [phi, args] : bb.phis) {

177void Emitter::emit_epilogue_impl(Lam* lam) {
178 auto app = lam->body()->expect<App>("an application in tail position");
179 auto& bb = lam2bb_[lam];
180 // A target-specific intrinsic in tail position (e.g. %gpu.launch) emits its own code and
181 // yields the continuation to branch to.
182 if (auto ret = isa_targetspecific_intrinsic(bb, app)) return bb.tail("br label {}", *ret);
183 if (app->callee() == root()->ret_var()) { // return
184 fe::Vector<std::string> values;
185 DefVec types;
186 for (auto arg : app->args()) {
187 if (auto val = emit_unsafe(arg); !val.empty()) {
188 values.emplace_back(val);
189 types.emplace_back(arg->type());
190 }
191 }
192
193 switch (values.size()) {
194 case 0: return bb.tail("ret void");
195 case 1:
196 return Axm::isa<mem::M>(types[0]) ? bb.tail("ret void")
197 : bb.tail("ret {} {}", convert(types[0]), values[0]);
198 default: {
199 std::string type;
200 std::string prev;
201
202 if (auto se = is_simd_aggregate(types)) {
203 auto common_src = find_common_simd_src(app);
204 if (common_src) {
205 auto v_src = emit(common_src);
206 auto t = convert(common_src->type());
207 return bb.tail("ret {} {}", t, v_src);
208 }
209 auto [size, elem] = *se;
210 auto val_t = convert(elem);
211
212 type = std::format("<{} x {}>", size, val_t);
213 for (auto val : values) {
214 if (prev.empty())
215 prev = "<";
216 else
217 prev += ", ";
218 prev += std::format("{} {}", val_t, val);
219 }
220 prev += ">";
221 } else {
222 prev = "undef";
223 type = convert(world().sigma(types));
224 for (size_t i = 0, n = values.size(); i != n; ++i) {
225 if (auto mem = Axm::isa<mem::M>(types[i])) continue;
226 auto v_elem = values[i];
227 auto t_elem = convert(types[i]);
228 auto namei = "%ret_val." + std::to_string(i);
229 bb.tail("{} = insertvalue {} {}, {} {}, {}", namei, type, prev, t_elem, v_elem, i);
230 prev = namei;
231 }
232 }
233 bb.tail("ret {} {}", type, prev);
234 }
235 }
236
237 } else if (auto dispatch = Dispatch(app)) {
238 for (auto callee : dispatch.tuple()->projs([](const Def* def) { return def->isa_mut<Lam>(); }))
239 if (size_t n = callee->num_tvars(); n == 1 && is_simd(callee->var(0)->type()))
240 emit_phi(callee, callee->var(0), emit(app->arg(n, 0)), lam);
241 else
242 emit_phi_args(callee, app, lam);
243
244 auto v_index = emit(dispatch.index());
245 size_t n = dispatch.num_targets();
246 auto bbs = absl::FixedArray<std::string>(n);
247 for (size_t i = 0; i != n; ++i)
248 bbs[i] = emit(dispatch.target(i));
249
250 if (auto branch = Branch(app)) return bb.tail("br i1 {}, label {}, label {}", v_index, bbs[1], bbs[0]);
251
252 auto t_index = convert(dispatch.index()->type());
253 bb.tail("switch {} {}, label {} [ ", t_index, v_index, bbs[0]);
254 for (size_t i = 1; i != n; ++i)
255 std::print(bb.tail().back(), "{} {}, label {} ", t_index, std::to_string(i), bbs[i]);
256 std::print(bb.tail().back(), "]");
257 } else if (app->callee()->isa<Bot>()) {
258 return bb.tail("ret ; bottom: unreachable");
259 } else if (auto callee = Lam::isa_mut_basicblock(app->callee())) { // ordinary jump
260
261 if (auto common_src = find_common_simd_src(app)) {
262 auto v_src = emit(common_src);
263 auto callee_var = callee->var();
264 if (simd_phi_.find(callee) == simd_phi_.end()) simd_phi_[callee] = callee_var;
265 auto key = simd_phi_[callee];
266 emit_phi(callee, key, v_src, lam);
267 for (auto var : callee->vars())
268 locals_[var] = id(key);
269 locals_[callee_var] = id(key);
270 } else {
271 emit_phi_args(callee, app, lam);
272 }
273 // A loop header whose exit condition is wrapped in `%ll.vec` (see affine's LowerFor)
274 // carries this loop's `!llvm.loop` vectorize hint on every branch into it; on the latch
275 // it lifts LLVM's tiny-trip-count bailout.
276 if (callee->is_set()) {
277 if (auto head = callee->body()->isa<App>()) {
278 if (auto dispatch = Dispatch(head); dispatch && Axm::isa<ll::vec>(dispatch.index())) {
279 auto [it, _] = loop_md_.emplace(callee, LoopMdBase + 1 + loop_md_.size());
280 return bb.tail("br label {}, !llvm.loop !{}", id(callee), it->second);
281 }
282 }
283 }
284 return bb.tail("br label {}", id(callee));
285
286 } else if (auto longjmp = Axm::isa<clos::longjmp>(app)) {
287 declare("void @longjmp(i8*, i32) noreturn");
288
289 auto [mem, jbuf, tag] = app->args<3>();
290 emit_unsafe(mem);
291 auto v_jb = emit(jbuf);
292 auto v_tag = emit(tag);
293 bb.tail("call void @longjmp(i8* {}, i32 {})", v_jb, v_tag);
294 return bb.tail("unreachable");
295 } else if (auto mslot = Axm::isa<mem::mslot>(app)) {
296 // Continuation-based stack slot: allocate and jump to the passed continuation with the fresh pointer.
297 auto [Ta, rest] = mslot->uncurry_args<2>();
298 auto [pointee, addr_space] = Ta->projs<2>();
299 auto [msize, ret] = rest->projs<2>();
300 emit_unsafe(msize->proj(0)); // mem
301 // TODO array with size
302 auto ret_lam = ret->expect_mut<Lam>("a %mem.slot continuation");
303 auto ptr = ret_lam->var(2, 1);
304 auto v_ptr = emit_slot(bb, app, pointee, addr_space);
305 emit_phi(ret_lam, ptr, v_ptr, lam);
306 return bb.tail("br label {}", id(ret_lam));
307 } else if (Pi::isa_returning(app->callee_type())) { // function call
308 auto v_callee = emit(app->callee());
309
310 fe::Vector<std::string> args;
311 auto app_args = app->args();
312 for (auto arg : app_args.view().rsubspan(1))
313 if (auto v_arg = emit_unsafe(arg); !v_arg.empty()) args.emplace_back(convert(arg->type()) + " " + v_arg);
314
315 if (app->args().back()->isa<Bot>()) {
316 // TODO: Perhaps it'd be better to simply η-wrap this prior to the BE...
317 if (convert_ret_pi(app->callee_type()->ret_pi()) != "void")
318 fe::throwf(MIM_LL_BE "call with a ⊥ return continuation must return void, but `{}` does not", app);
319 bb.tail("call void {}({})", v_callee, fe::Join(args));
320 return bb.tail("unreachable");
321 }
322
323 auto ret_lam = app->args().back()->expect_mut<Lam>("a return continuation");
324 size_t n = 0;
325 for (auto var : ret_lam->vars())
326 if (!Axm::isa<mem::M>(var->type())) ++n;
327
328 if (n == 0) {
329 bb.tail("call void {}({})", v_callee, fe::Join(args));
330 } else {
331 auto name = "%" + app->unique_name() + "ret";
332 auto t_ret = convert_ret_pi(ret_lam->type());
333 bb.tail("{} = call {} {}({})", name, t_ret, v_callee, fe::Join(args));
334 emit_phi(ret_lam, ret_lam->var(), name, lam);
335 }
336
337 return bb.tail("br label {}", id(ret_lam));
338 }
339}

341std::string Emitter::emit_tuple(BB& bb, const std::string& name, const Def* tuple) {
342 if (detail::isa_mem_sigma_2(tuple->type())) {
343 emit_unsafe(tuple->proj(2, 0));
344 return emit(tuple->proj(2, 1));
345 }
346
347 if (tuple->is_closed()) {
348 bool is_array = tuple->type()->isa<Arr>();
349 auto simd_array = convert(tuple->type()).front() == '<'; // needed to respect pointer context
350 std::string s;
351 s += simd_array ? "<" : is_array ? "[" : "{";
352 auto sep = "";
353 for (size_t i = 0, n = tuple->num_projs(); i != n; ++i) {
354 auto e = tuple->proj(n, i);
355 if (auto v_elem = emit_unsafe(e); !v_elem.empty()) {
356 auto t_elem = convert(e->type());
357 s += sep + t_elem + " " + v_elem;
358 sep = ", ";
359 }
360 }
361
362 return s += simd_array ? ">" : is_array ? "]" : "}";
363 }
364
365 std::string prev = "undef";
366 auto t = convert(tuple->type());
367 for (size_t src = 0, dst = 0, n = tuple->num_projs(); src != n; ++src) {
368 auto e = tuple->proj(n, src);
369 if (auto elem = emit_unsafe(e); !elem.empty()) {
370 auto elem_t = convert(e->type());
371 // TODO: check dst vs src
372 auto namei = name + "." + std::to_string(dst);
373 if (t.front() == '<') // not using is_simd to respect the pointer context (Pointer Pointee case)
374 prev = bb.assign(namei, "insertelement {} {}, {} {}, {} {}", t, prev, elem_t, elem, elem_t, dst);
375 else
376 prev = bb.assign(namei, "insertvalue {} {}, {} {}, {}", t, prev, elem_t, elem, dst);
377 dst++;
378 }
379 }
380 return prev;
381}

383std::pair<std::string, std::string> Emitter::emit_gep_index(BB& bb, const std::string& name, const Def* index) {
384 auto v_i = emit(index);
385 auto t_i = convert(index->type());
386
387 if (auto size = Idx::isa(index->type())) {
388 if (auto w = Idx::size2bitwidth(size); w && *w < 64) {
389 v_i = bb.assign(name + ".zext",
390 "zext {} {} to i{} ; add one more bit for gep index as it is treated as signed value", t_i,
391 v_i, *w + 1);
392 t_i = "i" + std::to_string(*w + 1);
393 }
394 }
395
396 return std::pair(v_i, t_i);

399std::string Emitter::emit_lit(const Def* def) {
400 if (auto lit = def->isa<Lit>()) {
401 if (lit->type()->isa<Nat>() || Idx::isa(lit->type())) {
402 return std::to_string(lit->get());
403 } else if (auto w = math::isa_f(lit->type())) {
404 std::stringstream s;
405 u64 hex;
406
407 switch (*w) {
408 case 16:
409 s << "0xH" << std::setfill('0') << std::setw(4) << std::right << std::hex << lit->get<u16>();
410 return s.str();
411 case 32: {
412 hex = std::bit_cast<u64>(f64(lit->get<f32>()));
413 break;
414 }
415 case 64: hex = lit->get<u64>(); break;
416 default: fe::throwf(MIM_LL_BE "unsupported floating-point width {} for literal `{}`", *w, def);
417 }
418
419 s << "0x" << std::setfill('0') << std::setw(16) << std::right << std::hex << hex;
420 return s.str();
421 }
422 fe::throwf(MIM_LL_BE "cannot emit literal `{}` of type `{}`", def, def->type());
423 }
424 fe::unreachable();
425}

427std::optional<std::string> Emitter::emit_builtin(BB& bb, const std::string& name, const Def* def) {
428 if (def->isa<Var>()) {
429 if (is_simd(def->type())) return id(def);
430 auto ts = def->type()->projs();
431 if (std::ranges::any_of(ts, [](auto t) { return Axm::isa<mem::M>(t); })) return std::string();
432 return emit_tuple(bb, name, def);
433 }
434
435 if (def->isa<Lit>()) {
436 return emit_lit(def);
437 } else if (def->isa<Bot>()) {
438 return "undef";
439 } else if (auto top = def->isa<Top>()) {
440 if (Axm::isa<mem::M>(top->type())) return std::string();
441 // bail out to the dispatcher's error
442 } else if (auto tuple = def->isa<Tuple>()) {
443 return emit_tuple(bb, name, tuple);
444 } else if (auto pack = def->isa<Pack>()) {
445 if (auto lit = Lit::isa(pack->body()); lit && *lit == 0) return "zeroinitializer";
446 return emit_tuple(bb, name, pack);
447 } else if (auto sel = Select(def)) {
448 auto t = convert(sel.extract()->type());
449 auto [elem_a, elem_b] = sel.pair()->projs<2>([&](auto e) { return emit_unsafe(e); });
450 auto cond_t = convert(sel.cond()->type());
451 auto cond = emit(sel.cond());
452 return bb.assign(name, "select {} {}, {} {}, {} {}", cond_t, cond, t, elem_b, t, elem_a);
453 } else if (auto extract = def->isa<Extract>()) {
454 auto tuple = extract->tuple();
455 auto index = extract->index();
456 auto v_tup = emit_unsafe(tuple);
457 if (is_simd(tuple->type()) && !Axm::isa<mem::M>(tuple->type())) return v_tup;
458
459 // this exact location is important: after emitting the tuple -> ordering of mem ops
460 // before emitting the index, as it might be a weird value for mem vars.
461 if (Axm::isa<mem::M>(extract->type())) return std::string();
462 if (auto sigma = extract->type()->isa<Sigma>(); sigma && sigma->num_ops() == 0) return std::string();
463
464 auto t_tup = convert(tuple->type());
465 if (auto li = Lit::isa(index)) {
466 if (detail::isa_mem_sigma_2(tuple->type())) return v_tup;
467 // Adjust index: convert() drops %mem.M elements from sigmas,
468 // so subtract the number of mem elements preceding the index.
469 auto v_i = *li;
470 if (auto sigma = tuple->type()->isa<Sigma>())
471 for (u64 i = 0; i < *li; ++i)
472 if (Axm::isa<mem::M>(sigma->op(i))) --v_i;
473
474 return bb.assign(name, "extractvalue {} {}, {}", t_tup, v_tup, v_i);
475 }
476
477 auto t_elem = convert(extract->type());
478 auto [v_i, t_i] = emit_gep_index(bb, name, index);
479
480 std::print(lam2bb_[root()].body().emplace_front(),
481 "{}.alloca = alloca {} ; copy to alloca to emulate extract with store + gep + load", name, t_tup);
482 std::print(bb.body().emplace_back(), "store {} {}, {}* {}.alloca", t_tup, v_tup, t_tup, name);
483 std::print(bb.body().emplace_back(), "{}.gep = getelementptr inbounds {}, {}* {}.alloca, i64 0, {} {}", name,
484 t_tup, t_tup, name, t_i, v_i);
485 return bb.assign(name, "load {}, {}* {}.gep", t_elem, t_elem, name);
486 } else if (auto insert = def->isa<Insert>()) {
487 if (Axm::isa<mem::M>(insert->tuple()->proj(0)->type()))
488 fe::throwf(MIM_LL_BE "cannot insert into a tuple with a `%mem.M` element: `{}`", insert);
489 auto t_tup = convert(insert->tuple()->type());
490 auto t_val = convert(insert->value()->type());
491 auto v_tup = emit(insert->tuple());
492 auto v_val = emit(insert->value());
493 if (auto idx = Lit::isa(insert->index())) {
494 auto v_idx = emit(insert->index());
495 if (is_simd(insert->tuple()->type()))
496
497 return bb.assign(name, "insertelement {} {}, {} {}, i32 {}", t_tup, v_tup, t_val, v_val, v_idx);
498 else
499
500 return bb.assign(name, " insertvalue {} {}, {} {}, {}", t_tup, v_tup, t_val, v_val, v_idx);
501 } else {
502 if (is_simd(insert->tuple()->type())) {
503 auto v_i = emit(insert->index());
504 auto t_i = convert(insert->index()->type());
505 if (t_i != "i32") {
506 auto w_src = Idx::expect_bitwidth(insert->index()->type(), "an `%insert` index of known width");
507 v_i = bb.assign(name + ".idx", "{} {} {} to i32", w_src < 32 ? "zext" : "trunc", t_i, v_i);
508 }
509 return bb.assign(name, "insertelement {} {}, {} {}, i32 {}", t_tup, v_tup, t_val, v_val, v_i);
510 }
511 auto t_elem = convert(insert->value()->type());
512 auto [v_i, t_i] = emit_gep_index(bb, name, insert->index());
513 std::print(lam2bb_[root()].body().emplace_front(),
514 "{}.alloca = alloca {} ; copy to alloca to emulate insert with store + gep + load", name, t_tup);
515 std::print(bb.body().emplace_back(), "store {} {}, {}* {}.alloca", t_tup, v_tup, t_tup, name);
516 std::print(bb.body().emplace_back(), "{}.gep = getelementptr inbounds {}, {}* {}.alloca, i64 0, {} {}",
517 name, t_tup, t_tup, name, t_i, v_i);
518 std::print(bb.body().emplace_back(), "store {} {}, {}* {}.gep", t_val, v_val, t_val, name);
519 return bb.assign(name, "load {}, {}* {}.alloca", t_tup, t_tup, name);
520 }
521 } else if (auto global = def->isa<Global>()) {
522 auto v_init = emit(global->init());
523 auto [pointee, addr_space] = Axm::expect<mem::Ptr>(global->type(), "a `%mem.Ptr`")->args<2>();
524 std::print(vars_decls_, "{} = global {} {}\n", name, convert(pointee), v_init);
525 return globals_[global] = name;
526 }
527 return std::nullopt;
528}

530std::optional<std::string> Emitter::emit_core(BB& bb, const std::string& name, const Def* def) {
531 std::string op;
532 if (auto nat = Axm::isa<core::nat>(def)) {
533 auto [a, b] = nat->args<2>([this](auto def) { return emit(def); });
534
535 switch (nat.id()) {
536 case core::nat::add: return bb.assign(name, "add nsw nuw i64 {}, {}", a, b);
537 case core::nat::sub: {
538 // nat subtraction saturates at 0: cap result when b > a
539 auto ugt = bb.assign(name + ".ugt", "icmp ugt i64 {}, {}", b, a);
540 auto raw = bb.assign(name + ".raw", "sub i64 {}, {}", a, b);
541 return bb.assign(name, "select i1 {}, i64 0, i64 {}", ugt, raw);
542 }
543 case core::nat::mul: return bb.assign(name, "mul nsw nuw i64 {}, {}", a, b);
544 // %core.nat.div/mod define division/modulo by zero as `a / 0 = 0` and `a % 0 = a`.
545 // replace a zero divisor by 1 to keep udiv/urem well-defined, then select the
546 // defined result for the zero case (0 for div, a for mod).
547 case core::nat::div: {
548 auto bz = bb.assign(name + ".bz", "icmp eq i64 {}, 0", b);
549 auto bsaf = bb.assign(name + ".bsafe", "select i1 {}, i64 1, i64 {}", bz, b);
550 auto q = bb.assign(name + ".q", "udiv i64 {}, {}", a, bsaf);
551 return bb.assign(name, "select i1 {}, i64 0, i64 {}", bz, q);
552 }
553 case core::nat::mod: {
554 auto bz = bb.assign(name + ".bz", "icmp eq i64 {}, 0", b);
555 auto bsaf = bb.assign(name + ".bsafe", "select i1 {}, i64 1, i64 {}", bz, b);
556 auto r = bb.assign(name + ".r", "urem i64 {}, {}", a, bsaf);
557 return bb.assign(name, "select i1 {}, i64 {}, i64 {}", bz, a, r);
558 }
559 }
560 } else if (auto ncmp = Axm::isa<core::ncmp>(def)) {
561 auto [a, b] = ncmp->args<2>([this](auto def) { return emit(def); });
562 op = "icmp ";
563
564 switch (ncmp.id()) {
565 // clang-format off
566 case core::ncmp::e: op += "eq" ; break;
567 case core::ncmp::ne: op += "ne" ; break;
568 case core::ncmp::g: op += "ugt"; break;
569 case core::ncmp::ge: op += "uge"; break;
570 case core::ncmp::l: op += "ult"; break;
571 case core::ncmp::le: op += "ule"; break;
572 // clang-format on
573 default: fe::throwf(MIM_LL_BE "unhandled `%core.ncmp` id in `{}`", def);
574 }
575
576 return bb.assign(name, "{} i64 {}, {}", op, a, b);
577 } else if (auto idx = Axm::isa<core::idx>(def)) {
578 auto x = emit(idx->arg());
579 auto s = Idx::expect_bitwidth(idx->type(), "a `%core.idx` result of known width");
580 auto t = convert(idx->type());
581 if (s < 64) return bb.assign(name, "trunc i64 {} to {}", x, t);
582 return x;
583 } else if (auto bit1 = Axm::isa<core::bit1>(def)) {
584 if (bit1.id() != core::bit1::neg) fe::throwf(MIM_LL_BE "unhandled `%core.bit1` id in `{}`", def);
585 auto x = emit(bit1->arg());
586 auto t = convert(bit1->type());
587 return bb.assign(name, "xor {} -1, {}", t, x);
588 } else if (auto bit2 = Axm::isa<core::bit2>(def)) {
589 auto [a, b] = bit2->args<2>([this](auto def) { return emit(def); });
590 auto t = convert(bit2->type());
591
592 auto neg = [&](std::string_view x) { return bb.assign(name + ".neg", "xor {} -1, {}", t, x); };
593
594 switch (bit2.id()) {
595 // clang-format off
596 case core::bit2::and_: return bb.assign(name, "and {} {}, {}", t, a, b);
597 case core::bit2:: or_: return bb.assign(name, "or {} {}, {}", t, a, b);
598 case core::bit2::xor_: return bb.assign(name, "xor {} {}, {}", t, a, b);
599 case core::bit2::nand: return neg(bb.assign(name, "and {} {}, {}", t, a, b));
600 case core::bit2:: nor: return neg(bb.assign(name, "or {} {}, {}", t, a, b));
601 case core::bit2::nxor: return neg(bb.assign(name, "xor {} {}, {}", t, a, b));
602 case core::bit2:: iff: return bb.assign(name, "and {} {}, {}", t, neg(a), b);
603 case core::bit2::niff: return bb.assign(name, "or {} {}, {}", t, neg(a), b);
604 // clang-format on
605 default: fe::throwf(MIM_LL_BE "unhandled `%core.bit2` id in `{}`", def);
606 }
607 } else if (auto shr = Axm::isa<core::shr>(def)) {
608 auto [a, b] = shr->args<2>([this](auto def) { return emit(def); });
609 auto t = convert(shr->type());
610
611 switch (shr.id()) {
612 case core::shr::a: op = "ashr"; break;
613 case core::shr::l: op = "lshr"; break;
614 }
615
616 return bb.assign(name, "{} {} {}, {}", op, t, a, b);
617 } else if (auto wrap = Axm::isa<core::wrap>(def)) {
618 auto [mode, ab] = wrap->uncurry_args<2>();
619 auto [a, b] = ab->projs<2>([this](auto def) { return emit(def); });
620 auto t = convert(wrap->type());
621 auto lmode = static_cast<core::Mode>(Lit::expect(mode, "a `%core.wrap` mode"));
622
623 switch (wrap.id()) {
624 case core::wrap::add: op = "add"; break;
625 case core::wrap::sub: op = "sub"; break;
626 case core::wrap::mul: op = "mul"; break;
627 case core::wrap::shl: op = "shl"; break;
628 }
629
630 if (fe::has_flag(lmode, core::Mode::nuw)) op += " nuw";
631 if (fe::has_flag(lmode, core::Mode::nsw)) op += " nsw";
632
633 return bb.assign(name, "{} {} {}, {}", op, t, a, b);
634 } else if (auto div = Axm::isa<core::div>(def)) {
635 auto [m, xy] = div->args<2>();
636 auto [x, y] = xy->projs<2>();
637 auto t = convert(x->type());
638 emit_unsafe(m);
639 auto a = emit(x);
640 auto b = emit(y);
641
642 switch (div.id()) {
643 case core::div::sdiv: op = "sdiv"; break;
644 case core::div::udiv: op = "udiv"; break;
645 case core::div::srem: op = "srem"; break;
646 case core::div::urem: op = "urem"; break;
647 }
648
649 return bb.assign(name, "{} {} {}, {}", op, t, a, b);
650 } else if (auto icmp = Axm::isa<core::icmp>(def)) {
651 auto [a, b] = icmp->args<2>([this](auto def) { return emit(def); });
652 auto t = convert(icmp->arg(0)->type());
653 op = "icmp ";
654
655 switch (icmp.id()) {
656 // clang-format off
657 case core::icmp::e: op += "eq" ; break;
658 case core::icmp::ne: op += "ne" ; break;
659 case core::icmp::sg: op += "sgt"; break;
660 case core::icmp::sge: op += "sge"; break;
661 case core::icmp::sl: op += "slt"; break;
662 case core::icmp::sle: op += "sle"; break;
663 case core::icmp::ug: op += "ugt"; break;
664 case core::icmp::uge: op += "uge"; break;
665 case core::icmp::ul: op += "ult"; break;
666 case core::icmp::ule: op += "ule"; break;
667 // clang-format on
668 default: fe::throwf(MIM_LL_BE "unhandled `%core.icmp` id in `{}`", def);
669 }
670
671 return bb.assign(name, "{} {} {}, {}", op, t, a, b);
672 } else if (auto extr = Axm::isa<core::extrema>(def)) {
673 auto [x, y] = extr->args<2>();
674 auto t = convert(x->type());
675 auto a = emit(x);
676 auto b = emit(y);
677 std::string f = "llvm.";
678 switch (extr.id()) {
679 case core::extrema::Sm: f += "smin."; break;
680 case core::extrema::SM: f += "smax."; break;
681 case core::extrema::sm: f += "umin."; break;
682 case core::extrema::sM: f += "umax."; break;
683 }
684 f += t;
685 declare("{} @{}({}, {})", t, f, t, t);
686 return bb.assign(name, "tail call {} @{}({} {}, {} {})", t, f, t, a, t, b);
687 } else if (auto abs = Axm::isa<core::abs>(def)) {
688 auto [m, x] = abs->args<2>();
689 auto t = convert(x->type());
690 auto a = emit(x);
691 std::string f = "llvm.abs." + t;
692 declare("{} @{}({}, {})", t, f, t, "i1");
693 return bb.assign(name, "tail call {} @{}({} {}, {} {})", t, f, t, a, "i1", "1");
694 } else if (auto conv = Axm::isa<core::conv>(def)) {
695 auto v_src = emit(conv->arg());
696 auto t_src = convert(conv->arg()->type());
697 auto t_dst = convert(conv->type());
698
699 nat_t w_src = Idx::expect_bitwidth(conv->arg()->type(), "a `%core.conv` source of known width");
700 nat_t w_dst = Idx::expect_bitwidth(conv->type(), "a `%core.conv` target of known width");
701
702 if (w_src == w_dst) return v_src;
703
704 switch (conv.id()) {
705 case core::conv::s: op = w_src < w_dst ? "sext" : "trunc"; break;
706 case core::conv::u: op = w_src < w_dst ? "zext" : "trunc"; break;
707 }
708
709 return bb.assign(name, "{} {} {} to {}", op, t_src, v_src, t_dst);
710 } else if (auto bitcast = Axm::isa<core::bitcast>(def)) {
711 auto dst_type_ptr = Axm::isa<mem::Ptr>(bitcast->type());
712 auto src_type_ptr = Axm::isa<mem::Ptr>(bitcast->arg()->type());
713 auto v_src = emit(bitcast->arg());
714 auto t_src = convert(bitcast->arg()->type());
715 auto t_dst = convert(bitcast->type());
716
717 if (auto lit = Lit::isa(bitcast->arg()); lit && *lit == 0) return "zeroinitializer";
718 // clang-format off
719 if (src_type_ptr && dst_type_ptr) return bb.assign(name, "bitcast {} {} to {}", t_src, v_src, t_dst);
720 if (src_type_ptr) return bb.assign(name, "ptrtoint {} {} to {}", t_src, v_src, t_dst);
721 if (dst_type_ptr) return bb.assign(name, "inttoptr {} {} to {}", t_src, v_src, t_dst);
722 // clang-format on
723
724 auto size2width = [&](const Def* type) {
725 if (type->isa<Nat>()) return 64_n;
726 if (Idx::isa(type)) return Idx::expect_bitwidth(type, "a statically-sized index type");
727 return 0_n;
728 };
729
730 auto src_size = size2width(bitcast->arg()->type());
731 auto dst_size = size2width(bitcast->type());
732
733 op = "bitcast";
734 if (src_size && dst_size) {
735 if (src_size == dst_size) return v_src;
736 op = (src_size < dst_size) ? "zext" : "trunc";
737 }
738 return bb.assign(name, "{} {} {} to {}", op, t_src, v_src, t_dst);
739 }
740 return std::nullopt;
741}

743std::optional<std::string> Emitter::emit_mem(BB& bb, const std::string& name, const Def* def) {
744 if (auto lea = Axm::isa<mem::lea>(def)) {
745 auto [ptr, i] = lea->args<2>();
746 auto pointee = Axm::expect<mem::Ptr>(ptr->type(), "a `%mem.Ptr`")->arg(0);
747 auto v_ptr = emit(ptr);
748 auto t_pointee = convert(pointee);
749 auto t_ptr = convert(ptr->type());
750 if (pointee->isa<Sigma>())
751 return bb.assign(name, "getelementptr inbounds {}, {} {}, i64 0, i32 {}", t_pointee, t_ptr, v_ptr,
752 Lit::expect(i, "a struct-field index"));
753
754 if (!pointee->isa<Arr>()) fe::throwf(MIM_LL_BE "`%mem.lea` on a pointer to a non-aggregate `{}`", pointee);
755 auto [v_i, t_i] = emit_gep_index(bb, name, i);
756
757 return bb.assign(name, "getelementptr inbounds {}, {} {}, i64 0, {} {}", t_pointee, t_ptr, v_ptr, t_i, v_i);
758 } else if (auto malloc = Axm::isa<mem::malloc>(def)) {
759 auto address_space = malloc->decurry()->arg(1);
760 declare("i8* @malloc(i64)");
761
762 emit_unsafe(malloc->arg(0));
763 auto size = emit(malloc->arg(1));
764 auto ptr_t = convert(Axm::expect<mem::Ptr>(def->proj(1)->type(), "a `%mem.Ptr`"));
765 auto i8ptr = bb.assign(name + "i8", "call i8* @malloc(i64 {})", size);
766 std::string i8ptr_t = "i8*";
767 if (Lit::expect(address_space, "an address space") != 0) {
768 i8ptr_t = std::format("i8 addrspace({})*", address_space);
769 i8ptr = bb.assign(name + "i8conv", "addrspacecast i8* {} to {}", i8ptr, i8ptr_t);
770 }
771 return bb.assign(name, "bitcast {} {} to {}", i8ptr_t, i8ptr, ptr_t);
772 } else if (auto free = Axm::isa<mem::free>(def)) {
773 auto address_space = free->decurry()->arg(1);
774 declare("void @free(i8*)");
775 emit_unsafe(free->arg(0));
776 auto ptr = emit(free->arg(1));
777 auto ptr_t = convert(Axm::expect<mem::Ptr>(free->arg(1)->type(), "a `%mem.Ptr`"));
778
779 auto i8ptr = bb.assign(name + "i8", "bitcast {} {} to i8 addrspace({})*", ptr_t, ptr, address_space);
780 if (Lit::expect(address_space, "an address space") != 0)
781 i8ptr = bb.assign(name + "i8conv", "addrspacecast i8 addrspace({})* {} to i8*", address_space, i8ptr);
782 bb.tail("call void @free(i8* {})", i8ptr);
783 return std::string();
784 } else if (auto load = Axm::isa<mem::load>(def)) {
785 emit_unsafe(load->arg(0));
786 auto v_ptr = emit(load->arg(1));
787 auto t_ptr = convert(load->arg(1)->type());
788 auto t_pointee = convert(Axm::expect<mem::Ptr>(load->arg(1)->type(), "a `%mem.Ptr`")->arg(0), false);
789 return bb.assign(name, "load {}, {} {}", t_pointee, t_ptr, v_ptr);
790 } else if (auto store = Axm::isa<mem::store>(def)) {
791 emit_unsafe(store->arg(0));
792 auto v_ptr = emit(store->arg(1));
793 auto v_val = emit(store->arg(2));
794 auto t_ptr = convert(store->arg(1)->type());
795 auto t_val = convert(store->arg(2)->type(), false);
796 std::print(bb.body().emplace_back(), "store {} {}, {} {}", t_val, v_val, t_ptr, v_ptr);
797 return std::string();
798 } else if (auto q = Axm::isa<clos::alloc_jmpbuf>(def)) {
799 // The size of a `jmp_buf` is platform/libc-dependent, so it is computed by a C runtime
800 // wrapper (`rt/mim_rt.c`) rather than hard-coded here; see issue #486.
801 declare_rt("i64 @mim_jmpbuf_size()");
802
803 emit_unsafe(q->arg());
804 auto size = name + ".size";
805 bb.assign(size, "call i64 @mim_jmpbuf_size()");
806 return bb.assign(name, "alloca i8, i64 {}", size);
807 } else if (auto setjmp = Axm::isa<clos::setjmp>(def)) {
808 declare("i32 @_setjmp(i8*) returns_twice");
809
810 auto [mem, jmpbuf] = setjmp->arg()->projs<2>();
811 emit_unsafe(mem);
812 auto v_jb = emit(jmpbuf);
813 return bb.assign(name, "call i32 @_setjmp(i8* {})", v_jb);
814 }
815 return std::nullopt;
816}

818std::optional<std::string> Emitter::emit_math(BB& bb, const std::string& name, const Def* def) {
819 std::string op;
820 if (auto arith = Axm::isa<math::arith>(def)) {
821 auto [mode, ab] = arith->uncurry_args<2>();
822 auto [a, b] = ab->projs<2>([this](auto def) { return emit(def); });
823 auto t = convert(arith->type());
824 auto lmode = static_cast<math::Mode>(Lit::expect(mode, "a `%math.arith` mode"));
825
826 switch (arith.id()) {
827 case math::arith::add: op = "fadd"; break;
828 case math::arith::sub: op = "fsub"; break;
829 case math::arith::mul: op = "fmul"; break;
830 case math::arith::div: op = "fdiv"; break;
831 case math::arith::rem: op = "frem"; break;
832 }
833
834 if (lmode == math::Mode::fast)
835 op += " fast";
836 else {
837 // clang-format off
838 if (fe::has_flag(lmode, math::Mode::nnan )) op += " nnan";
839 if (fe::has_flag(lmode, math::Mode::ninf )) op += " ninf";
840 if (fe::has_flag(lmode, math::Mode::nsz )) op += " nsz";
841 if (fe::has_flag(lmode, math::Mode::arcp )) op += " arcp";
842 if (fe::has_flag(lmode, math::Mode::contract)) op += " contract";
843 if (fe::has_flag(lmode, math::Mode::afn )) op += " afn";
844 if (fe::has_flag(lmode, math::Mode::reassoc )) op += " reassoc";
845 // clang-format on
846 }
847
848 return bb.assign(name, "{} {} {}, {}", op, t, a, b);
849 } else if (auto tri = Axm::isa<math::tri>(def)) {
850 auto a = emit(tri->arg());
851 auto t = convert(tri->type());
852
853 std::string f;
854
855 if (tri.id() == math::tri::sin) {
856 f = std::string("llvm.sin") + detail::llvm_suffix(tri->type());
857 } else if (tri.id() == math::tri::cos) {
858 f = std::string("llvm.cos") + detail::llvm_suffix(tri->type());
859 } else {
860 if (tri.sub() & sub_t(math::tri::a)) f += "a";
861
862 switch (math::tri((fe::to_underlying(tri.id()) & 0x3) | Annex::base<math::tri>())) {
863 case math::tri::sin: f += "sin"; break;
864 case math::tri::cos: f += "cos"; break;
865 case math::tri::tan: f += "tan"; break;
866 case math::tri::ahFF: fe::throwf(MIM_LL_BE "axm `{}` is not supposed to occur", def);
867 default: fe::throwf(MIM_LL_BE "unhandled `%math.tri` id in `{}`", def);
868 }
869
870 if (tri.sub() & sub_t(math::tri::h)) f += "h";
871 f += detail::math_suffix(tri->type());
872 }
873
874 declare("{} @{}({})", t, f, t);
875 return bb.assign(name, "tail call {} @{}({} {})", t, f, t, a);
876 } else if (auto extrema = Axm::isa<math::extrema>(def)) {
877 auto [a, b] = extrema->args<2>([this](auto def) { return emit(def); });
878 auto t = convert(extrema->type());
879 std::string f = "llvm.";
880 switch (extrema.id()) {
881 case math::extrema::fmin: f += "minnum"; break;
882 case math::extrema::fmax: f += "maxnum"; break;
883 case math::extrema::ieee754min: f += "minimum"; break;
884 case math::extrema::ieee754max: f += "maximum"; break;
885 }
886 f += detail::llvm_suffix(extrema->type());
887
888 declare("{} @{}({}, {})", t, f, t, t);
889 return bb.assign(name, "tail call {} @{}({} {}, {} {})", t, f, t, a, t, b);
890 } else if (auto pow = Axm::isa<math::pow>(def)) {
891 auto [a, b] = pow->args<2>([this](auto def) { return emit(def); });
892 auto t = convert(pow->type());
893 std::string f = "llvm.pow";
894 f += detail::llvm_suffix(pow->type());
895 declare("{} @{}({}, {})", t, f, t, t);
896 return bb.assign(name, "tail call {} @{}({} {}, {} {})", t, f, t, a, t, b);
897 } else if (auto rt = Axm::isa<math::rt>(def)) {
898 auto a = emit(rt->arg());
899 auto t = convert(rt->type());
900 std::string f;
901 if (rt.id() == math::rt::sq)
902 f = std::string("llvm.sqrt") + detail::llvm_suffix(rt->type());
903 else
904 f = std::string("cbrt") += detail::math_suffix(rt->type());
905 declare("{} @{}({})", t, f, t);
906 return bb.assign(name, "tail call {} @{}({} {})", t, f, t, a);
907 } else if (auto exp = Axm::isa<math::exp>(def)) {
908 auto a = emit(exp->arg());
909 auto t = convert(exp->type());
910 std::string f = "llvm.";
911 f += (exp.sub() & sub_t(math::exp::log)) ? "log" : "exp";
912 f += (exp.sub() & sub_t(math::exp::bin)) ? "2" : (exp.sub() & sub_t(math::exp::dec)) ? "10" : "";
913 f += detail::llvm_suffix(exp->type());
914 // TODO doesn't work for exp10"
915 declare("{} @{}({})", t, f, t);
916 return bb.assign(name, "tail call {} @{}({} {})", t, f, t, a);
917 } else if (auto er = Axm::isa<math::er>(def)) {
918 auto a = emit(er->arg());
919 auto t = convert(er->type());
920 auto f = er.id() == math::er::f ? std::string("erf") : std::string("erfc");
921 f += detail::math_suffix(er->type());
922 declare("{} @{}({})", t, f, t);
923 return bb.assign(name, "tail call {} @{}({} {})", t, f, t, a);
924 } else if (auto gamma = Axm::isa<math::gamma>(def)) {
925 auto a = emit(gamma->arg());
926 auto t = convert(gamma->type());
927 std::string f = gamma.id() == math::gamma::t ? "tgamma" : "lgamma";
928 f += detail::math_suffix(gamma->type());
929 declare("{} @{}({})", t, f, t);
930 return bb.assign(name, "tail call {} @{}({} {})", t, f, t, a);
931 } else if (auto cmp = Axm::isa<math::cmp>(def)) {
932 auto [a, b] = cmp->args<2>([this](auto def) { return emit(def); });
933 auto t = convert(cmp->arg(0)->type());
934 op = "fcmp ";
935
936 switch (cmp.id()) {
937 // clang-format off
938 case math::cmp:: e: op += "oeq"; break;
939 case math::cmp:: l: op += "olt"; break;
940 case math::cmp:: le: op += "ole"; break;
941 case math::cmp:: g: op += "ogt"; break;
942 case math::cmp:: ge: op += "oge"; break;
943 case math::cmp:: ne: op += "one"; break;
944 case math::cmp:: o: op += "ord"; break;
945 case math::cmp:: u: op += "uno"; break;
946 case math::cmp:: ue: op += "ueq"; break;
947 case math::cmp:: ul: op += "ult"; break;
948 case math::cmp::ule: op += "ule"; break;
949 case math::cmp:: ug: op += "ugt"; break;
950 case math::cmp::uge: op += "uge"; break;
951 case math::cmp::une: op += "une"; break;
952 // clang-format on
953 default: fe::throwf(MIM_LL_BE "unhandled `%math.cmp` id in `{}`", def);
954 }
955
956 return bb.assign(name, "{} {} {}, {}", op, t, a, b);
957 } else if (auto is_finite = Axm::isa<math::is_finite>(def)) {
958 // https://llvm.org/docs/LangRef.html#llvm-is-fpclass-intrinsic
959 // declare i1 @llvm.is.fpclass(<fptype> <op>, i32 <test>)
960 auto a = emit(is_finite->arg());
961 auto at = convert(is_finite->arg()->type());
962 auto t = convert(is_finite->type());
963
964 auto s = detail::llvm_suffix(is_finite->arg()->type());
965 auto f = "llvm.is.fpclass";
966 declare("{} @{}{}({}, i32)", t, f, s, at);
967 return bb.assign(name, "tail call {} @{}{}({} {}, i32 504)", t, f, s, at, a);
968 } else if (auto conv = Axm::isa<math::conv>(def)) {
969 auto v_src = emit(conv->arg());
970 auto t_src = convert(conv->arg()->type());
971 auto t_dst = convert(conv->type());
972
973 auto s_src = math::isa_f(conv->arg()->type());
974 auto s_dst = math::isa_f(conv->type());
975
976 switch (conv.id()) {

1009std::optional<std::string> Emitter::emit_vec(BB& bb, const std::string& name, const Def* def) {
1010 if (auto v = Axm::isa<ll::vec>(def)) {
1011 // `%ll.vec` annotates a loop's exit condition; as a value it is the identity — the
1012 // metadata it requests is attached at the branches into the loop's header.
1013 return emit(v->arg());
1014 } else if (auto zip = Axm::isa<vecp::zip>(def)) {
1015 auto ni_n = zip->decurry()->decurry()->decurry()->arg();
1016 auto nat_ni = Lit::expect(ni_n->proj(2, 0), "the `%vec.zip` inputs count");
1017 auto nat_n = Lit::expect(ni_n->proj(2, 1), "the `%vec.zip` lane count");
1018 auto f = zip->decurry()->arg();
1019 auto inputs = zip->arg();
1020 auto t_in = convert(inputs->proj(nat_ni, 0)->type());
1021 auto t_out = convert(def->type()); // <n x T>
1022
1023 std::string op;
1024 std::string prev;
1025
1026 if (auto nat_op = Axm::isa<core::nat, 1>(f)) {
1027 switch (nat_op.id()) {
1028 case core::nat::add: op = "add nuw nsw"; break;
1029 case core::nat::sub: {
1030 // nat subtraction saturates at 0: cap per-lane when v2 > v1
1031 auto v1 = emit(inputs->proj(nat_ni, 0));
1032 auto v2 = emit(inputs->proj(nat_ni, 1));
1033 auto ugt = bb.assign(name + ".ugt", "icmp ugt {} {}, {}", t_in, v2, v1);
1034 auto raw = bb.assign(name + ".raw", "sub {} {}, {}", t_in, v1, v2);
1035 return prev = bb.assign(name, "select <{} x i1> {}, {} zeroinitializer, {} {}", nat_n, ugt, t_out,
1036 t_out, raw);
1037 }
1038 case core::nat::mul: op = "mul nuw nsw"; break;
1039 case core::nat::div: op = "udiv"; break;
1040 case core::nat::mod: op = "urem"; break;
1041 }
1042 } else if (auto arith_op = Axm::isa<math::arith, 1>(f)) {
1043 auto lmode = static_cast<math::Mode>(
1044 Lit::expect(f->expect<App>("a zipped `%math.arith`")->arg(), "a `%math.arith` mode"));
1045 switch (arith_op.id()) {
1046 case math::arith::add: op = "fadd"; break;
1047 case math::arith::sub: op = "fsub"; break;
1048 case math::arith::mul: op = "fmul"; break;
1049 case math::arith::div: op = "fdiv"; break;
1050 case math::arith::rem: op = "frem"; break;
1051 }
1052
1053 if (lmode == math::Mode::fast)
1054 op += " fast";
1055 else {
1056 if (fe::has_flag(lmode, math::Mode::nnan)) op += " nnan";
1057 if (fe::has_flag(lmode, math::Mode::ninf)) op += " ninf";
1058 if (fe::has_flag(lmode, math::Mode::nsz)) op += " nsz";
1059 if (fe::has_flag(lmode, math::Mode::arcp)) op += " arcp";
1060 if (fe::has_flag(lmode, math::Mode::contract)) op += " contract";
1061 if (fe::has_flag(lmode, math::Mode::afn)) op += " afn";
1062 if (fe::has_flag(lmode, math::Mode::reassoc)) op += " reassoc";
1063 }
1064 } else if (auto ncmp_op = Axm::isa<core::ncmp, 1>(f)) {
1065 op = "icmp ";
1066 switch (ncmp_op.id()) {
1067 case core::ncmp::e: op += "eq"; break;
1068 case core::ncmp::ne: op += "ne"; break;
1069 case core::ncmp::g: op += "ugt"; break;
1070 case core::ncmp::ge: op += "uge"; break;
1071 case core::ncmp::l: op += "ult"; break;
1072 case core::ncmp::le: op += "ule"; break;
1073 default: fe::throwf(MIM_LL_BE "unhandled zipped `%core.ncmp` id in `{}`", def);
1074 }
1075 } else if (auto icmp_op = Axm::isa<core::icmp, 1>(f)) {
1076 op = "icmp ";
1077 switch (icmp_op.id()) {
1078 case core::icmp::e: op += "eq"; break;
1079 case core::icmp::ne: op += "ne"; break;
1080 case core::icmp::sg: op += "sgt"; break;
1081 case core::icmp::sge: op += "sge"; break;
1082 case core::icmp::sl: op += "slt"; break;
1083 case core::icmp::sle: op += "sle"; break;
1084 case core::icmp::ug: op += "ugt"; break;
1085 case core::icmp::uge: op += "uge"; break;
1086 case core::icmp::ul: op += "ult"; break;
1087 case core::icmp::ule: op += "ule"; break;
1088 default: fe::throwf(MIM_LL_BE "unhandled zipped `%core.icmp` id in `{}`", def);
1089 }
1090 } else if (auto mcmp_op = Axm::isa<math::cmp, 1>(f)) {
1091 op = "fcmp ";
1092 switch (mcmp_op.id()) {
1093 case math::cmp::e: op += "oeq"; break;
1094 case math::cmp::l: op += "olt"; break;
1095 case math::cmp::le: op += "ole"; break;
1096 case math::cmp::g: op += "ogt"; break;
1097 case math::cmp::ge: op += "oge"; break;
1098 case math::cmp::ne: op += "one"; break;
1099 case math::cmp::o: op += "ord"; break;
1100 case math::cmp::u: op += "uno"; break;
1101 case math::cmp::ue: op += "ueq"; break;
1102 case math::cmp::ul: op += "ult"; break;
1103 case math::cmp::ule: op += "ule"; break;
1104 case math::cmp::ug: op += "ugt"; break;
1105 case math::cmp::uge: op += "uge"; break;
1106 case math::cmp::une: op += "une"; break;
1107 default: fe::throwf(MIM_LL_BE "unhandled zipped `%math.cmp` id in `{}`", def);
1108 }
1109 } else {
1110 fe::throwf(MIM_LL_BE "unhandled `%vec.zip` operation `{}`", f);
1111 }
1112
1113 auto v1 = emit(inputs->proj(nat_ni, 0));
1114 auto v2 = emit(inputs->proj(nat_ni, 1));
1115 prev = bb.assign(name, "{} {} {}, {}", op, t_in, v1, v2);
1116 return prev;
1117 }
1118 return std::nullopt;
1119}
