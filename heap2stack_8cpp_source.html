26bool Heap2Stack::analyze() {
27 if (max_ == 0) return false;
28 if (RWPhase::analyze()) return true;
29
30 auto nest = std::optional<const Nest>();
31 for (const auto& [lam, chain] : escape_.chains()) {
32 auto allocs = fe::Vector<const App*>();
33 for (auto op : chain.ops) {
34 auto alloc = Axm::isa<mem::alloc>(op);
35 if (!alloc || escape_.is_top(op)) continue;
36 auto [pointee, addr_space] = alloc->decurry()->args<2>();
37 if (Lit::isa(addr_space) == 0 && num_elems(pointee)) allocs.emplace_back(op);
38 }
39 if (allocs.empty()) continue;
40 if (!lam->is_closed()) {
41 if (!nest) nest.emplace(old_world());
42 if (in_loop((*nest)[lam])) continue;
43 }
44 converted_.insert(allocs.begin(), allocs.end());
45 slots_[chain.tail] = std::move(allocs);
46 }
47
48 log().d("heap2stack: slots on {} chains", slots_.size());
49 return false;
50}

52std::optional<nat_t> Heap2Stack::num_elems(const Def* type) const {
53 if (type->isa_mut()) return {};
54
55 if (auto sigma = type->isa<Sigma>()) {
56 auto res = nat_t(0);
57 for (auto op : sigma->ops()) {
58 auto n = num_elems(op);
59 if (!n || (res += *n) > max_) return {};
60 }
61 return res;
62 }
63
64 if (auto arr = type->isa<Arr>()) {
65 auto res = num_elems(arr->body());
66 auto shape = arr->shape();
67 auto rank = shape.rank();
68 if (!res || !rank) return {};
69 for (nat_t i = 0; i != *rank; ++i) {
70 auto e = Shape::extent(shape[i]);
71 if (!e || (*e != 0 && *res > max_ / *e)) return {};
72 *res *= *e;
73 }
74 return res;
75 }
76
77 return 1;
78}
