12const Def* Lower::read_through(const Def* base, Defs index) {
13 auto& w = new_world();
14
15 const Def* input;
16 const Def* s_in;
17 const Def* s_out;
18 const Def* rank;
19 bool wraps; // a repeat reads `idx mod s_in` per axis, a broadcast only ever reads a size-1 axis at 0
20
21 if (auto rep = Axm::isa<tensor::repeat>(base)) {
22 auto [Tr, in, out] = rep->callee()->as<App>()->uncurry_args<3>();
23 input = rep->arg(), s_in = in, s_out = out, rank = Tr->proj(2, 1), wraps = true;
24 } else if (auto bc = Axm::isa<tensor::broadcast>(base)) {
25 auto [in, out, i] = bc->args<3>();
26 auto [_, r] = bc->callee()->as<App>()->args<2>();
27 input = i, s_in = in, s_out = out, rank = r, wraps = false;
28 } else {
29 return nullptr;
30 }
31
32 auto r = Lit::isa(rank);
33 // An index reaching past the shape op's own axes continues into an array element type - not ours to answer.
34 if (!r || index.size() > *r) return nullptr;
35
36 auto new_index = DefVec(*r);
37 for (u64 d = 0, i = 0; d != *r; ++d) {
38 auto in_d = s_in->proj(*r, d);
39 auto out_d = s_out->proj(*r, d);
40 // A size-1 output axis folds out of the array type, so the chain carries no index for it.
41 auto idx_d = Lit::isa(out_d) == 1 ? w.lit_idx(1, 0) : (i < index.size() ? rewrite(index[i++]) : nullptr);
42 if (!idx_d) return nullptr;
43
44 if (in_d == out_d)
45 new_index[d] = idx_d;
46 else if (Lit::isa(in_d) == 1)
47 new_index[d] = w.lit_idx(1, 0);
48 else if (auto e = Lit::isa(in_d), x = Lit::isa(idx_d); wraps && e && x)
49 new_index[d] = w.lit_idx(*e, *x % *e);
50 else
51 return nullptr;
52 }
53
54 return w.extract(rewrite(input), w.tuple(new_index));
55}

57const Def* Lower::rewrite_imm_Extract(const Extract* extract) {
58 auto base = Axm::peel<tensor::buf>(extract->tuple());

73const Def* Lower::lower_via_impl(const App* app, const Def* impl_annex) {
74 auto& w = new_world();
75
76 // The curry chain, innermost App first — hence re-applied in reverse.
77 auto args = DefVec();
78 for (const App* h = app; h; h = h->callee()->isa<App>())
79 args.emplace_back(rewrite(h->arg()));
80
81 // The `_impl` is a `lam`, so applying it triggers beta-reduction. Each `_impl`
82 // body references the `_impl` variants of its dependencies directly, so the
83 // chain bottoms out at the low-level axioms (`map_reduce`, …) in one go.
84 auto impl = impl_annex;
85 for (auto a : args | std::views::reverse)
86 impl = w.app(impl, a);
87 return impl;
88}
