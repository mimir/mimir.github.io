16const Def* Lower::lower_via_impl(const App* app, const Def* impl_annex) {
17 auto& w = new_world();
18
19 // Walk the curry chain (innermost App outermost in syntax) to collect the args
20 // in the order they were applied.
21 DefVec args;
22 const Def* head = app;
23 while (auto h = head->isa<App>()) {
24 args.push_back(rewrite(h->arg()));
25 head = h->callee();
26 }
27 std::reverse(args.begin(), args.end());
28
29 auto impl = impl_annex;
30 for (auto a : args)
31 impl = w.app(impl, a);
32
33 // The `_impl` is a `lam`, so applying it triggers beta-reduction. Each `_impl`
34 // body references the `_impl` variants of its dependencies directly, so the
35 // chain bottoms out at the low-level axioms (`map_reduce`, …) in one go.
36 return impl;
37}
