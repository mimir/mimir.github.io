7bool TailRecElim::is_tail_rec(Lam* lam) {
8 if (auto i = tail_rec_.find(lam); i != tail_rec_.end()) return i->second;
9
10 auto ret_var = lam->ret_var();
11 if (!ret_var || !lam->is_set() || !lam->is_closed()) return tail_rec_[lam] = false;
12
13 // Scan the (immutable) op graphs of all muts in lam's nest for a call `lam (..., lam's ret var)`.
14 auto nest = Nest(lam);
15 DefSet done;
16 auto found = false;
17 auto visit = [&](auto&& visit, const Def* def) -> void {
18 if (found || def->isa_mut() || !done.emplace(def).second) return;
19 if (auto app = def->isa<App>(); app && app->callee() == lam && app->args().back() == ret_var) {
20 found = true;
21 return;
22 }
23 for (auto op : def->deps())
24 visit(visit, op);
25 };
26 for (auto mut : nest.muts())
27 if (mut->is_set())
28 for (auto op : mut->deps())
29 visit(visit, op);
30
31 return tail_rec_[lam] = found;
32}

34const Def* TailRecElim::rewrite_mut_Lam(Lam* old) {
