46const Def* normalize_slice(const Def*, const Def* c, const Def* arg) {
47 // Identity slice: every axis starts at 0 with step 1 and keeps its full extent (s_out == s_in) -> the input itself.
48 auto [Tr, s_in, params] = c->as<App>()->uncurry_args<3>();
49 auto [start, step, s_out] = params->projs<3>();
50 if (s_out != s_in) return nullptr;
51 auto r = Lit::isa<u64>(Tr->proj(2, 1));

56 if (!st || *st != 0 || !sp || *sp != 1) return nullptr;
57 }

63const Def* normalize_pad(const Def*, const Def* c, const Def* arg) {

88const Def* normalize_fastest_axis(const Def*, const Def*, const Def* arg) {
89 // `tensor.fastest_axis (r, t)` reflects which axis of `t` is the fastest-varying (unit-stride)
90 // axis of the tensor actually read once `fuse_tensor`'s read-through has absorbed a pure
91 // re-indexed read behind `t`: without one, `t`'s own last axis; behind one, found by evaluating
92 // the read's access map on distinct `affine.lit` markers — normalization folds the map's
93 // extracts over the marker tuple, and a last component that does not fold back to a marker
94 // (reshape arithmetic) stays unknown. Unknown answers the sentinel `r`.
95 auto& w = arg->world();
96 auto [r, t] = arg->projs<2>();
97 auto r_l = Lit::isa<u64>(r);
98 if (!r_l || *r_l == 0) return r;
99 auto pr = is_pure_read(t);
100 if (!pr) return w.lit_nat(*r_l - 1);
101 // One level only: a source that is itself absorbed would need the composed analysis.
102 if (is_pure_read(pr->src) || Axm::isa<tensor::broadcast>(pr->src)) return r;
103 auto r_src = Lit::isa<u64>(pr->map->type()->as<Pi>()->codom()->arity());
104 if (!r_src || *r_src == 0) return r;
105 // `r` is not type-coupled to `t`; a mismatched rank must answer unknown, not break the app below.
106 auto r_dom = Lit::isa<u64>(pr->map->type()->as<Pi>()->dom()->arity());
107 if (!r_dom || *r_dom != *r_l) return r;
108 // Markers start at 1: 0 is what a broadcast map's `o#d · 0` folds to, so it must not be one.
109 auto markers = DefVec(*r_l, [&](size_t i) { return w.call<affine::lit>(w.lit_nat(i + 1)); });
110 auto last = w.app(pr->map, w.tuple(markers))->proj(*r_src, *r_src - 1);
111 auto c = Axm::isa<affine::lit>(last);
112 if (!c) return r;
113 auto v = Lit::isa<u64>(c->arg());
114 if (!v || *v < 1 || *v > *r_l) return r;
115 return w.lit_nat(*v - 1);
116}

118const Def* normalize_shape(const Def*, const Def* c, const Def* arg) {
119 // `tensor.shape r arr` is the first `r` axes of `arr`'s (fused) array type.
120 auto& w = c->world();
121 auto r = Lit::isa<u64>(c->as<App>()->arg()); // the explicit rank `r`
122 if (!r) return nullptr;
123 if (*r == 0) return w.tuple();
124
125 auto arr = arg->type()->isa<Arr>();
126 if (!arr) return nullptr;
127 auto shape = arr->shape();
128 auto rank = shape.rank();
129 if (!rank || *rank < *r) return nullptr; // `arr` is not (statically) at least rank `r`
130 return *shape.take(*r);
131}
