56const Def* normalize_transpose(const Def*, const Def* c, const Def* arg) {
57 auto callee = c->as<App>()->decurry();
58 auto perm = lit_perm(callee->arg());
59 if (!perm) return nullptr;
60 auto r = perm->size();
61 if (std::ranges::equal(*perm, std::views::iota(u64(0), u64(r)))) return arg;
62
63 auto app = Axm::isa<tensor::transpose>(arg);
64 if (!app) return nullptr;
65 auto inner = transpose_perm(app);
66 if (!inner || inner->size() != r) return nullptr;
67
68 // Axis `j` of the inner input ends up at `perm#(inner#j)`; an identity composite collapses in the rebuilt app.
69 auto& w = c->world();
70 auto comp = DefVec(r, [&](size_t j) { return w.lit_idx(r, (*perm)[(*inner)[j]]); });
71 return w.app(w.app(w.app(callee->callee(), w.tuple(comp)), app->decurry()->arg()), app->arg());
72}

74const Def* normalize_slice(const Def*, const Def* c, const Def* arg) {
75 // Identity slice: every axis starts at 0 with step 1 and keeps its full extent (s_out == s_in) -> the input itself.
76 auto [Tr, s_in, params] = c->as<App>()->uncurry_args<3>();
77 auto [start, step, s_out] = params->projs<3>();
78 if (s_out != s_in) return nullptr;
79 auto r = Lit::isa<u64>(Tr->proj(2, 1));
80 if (!r) return nullptr;
81 for (u64 d = 0; d != *r; ++d) {
82 auto st = Lit::isa<u64>(start->proj(*r, d));
83 auto sp = Lit::isa<u64>(step->proj(*r, d));
84 if (!st || *st != 0 || !sp || *sp != 1) return nullptr;
85 }
86 return arg;
87}

91const Def* normalize_pad(const Def*, const Def* c, const Def* arg) {
92 // Identity pad: every axis has lo == hi == 0 (so s_out == s_in) -> the input itself (the fill value is irrelevant).
93 auto [Tr, s_in, params] = c->as<App>()->uncurry_args<3>();
94 auto [mode, lo, hi] = params->projs<3>();
95 auto r = Lit::isa<u64>(Tr->proj(2, 1));
96 if (!r) return nullptr;
97 for (u64 d = 0; d != *r; ++d) {
98 auto l = Lit::isa<u64>(lo->proj(*r, d));
99 auto h = Lit::isa<u64>(hi->proj(*r, d));
100 if (!l || *l != 0 || !h || *h != 0) return nullptr;
101 }
102 return arg->proj(2, 0); // input (arg = (input, value))
103}

116const Def* normalize_fastest_axis(const Def*, const Def*, const Def* arg) {
117 // `tensor.fastest_axis (r, t)` reflects which axis of `t` is the fastest-varying (unit-stride)
118 // axis of the tensor actually read once `fuse_tensor`'s read-through has absorbed a pure
119 // re-indexed read behind `t`: without one, `t`'s own last axis; behind one, found by evaluating
120 // the read's access map on distinct `affine.lit` markers — normalization folds the map's
121 // extracts over the marker tuple, and a last component that does not fold back to a marker
122 // (reshape arithmetic) stays unknown. Unknown answers the sentinel `r`.
123 auto& w = arg->world();
124 auto [r, t] = arg->projs<2>();
125 auto r_l = Lit::isa<u64>(r);
126 if (!r_l || *r_l == 0) return r;
127 auto pr = is_pure_read(t);
128 if (!pr) return w.lit_nat(*r_l - 1);
129 // One level only: a source that is itself absorbed would need the composed analysis.
130 if (is_pure_read(pr->src) || Axm::isa<tensor::broadcast>(pr->src)) return r;
131 auto r_src = Lit::isa<u64>(pr->map->type()->as<Pi>()->codom()->arity());
132 if (!r_src || *r_src == 0) return r;
133 // `r` is not type-coupled to `t`; a mismatched rank must answer unknown, not break the app below.
134 auto r_dom = Lit::isa<u64>(pr->map->type()->as<Pi>()->dom()->arity());
135 if (!r_dom || *r_dom != *r_l) return r;
136 // Markers start at 1: 0 is what a broadcast map's `o#d · 0` folds to, so it must not be one.
137 auto markers = DefVec(*r_l, [&](size_t i) { return w.call<affine::lit>(w.lit_nat(i + 1)); });
138 auto last = w.app(pr->map, w.tuple(markers))->proj(*r_src, *r_src - 1);
139 auto c = Axm::isa<affine::lit>(last);
140 if (!c) return r;
141 auto v = Lit::isa<u64>(c->arg());
142 if (!v || *v < 1 || *v > *r_l) return r;
143 return w.lit_nat(*v - 1);
144}

149const Def* normalize_vec_width(const Def*, const Def*, const Def* arg) {

153 if (!v) return arg;

163 return arg;
164 }

168const Def* normalize_shape(const Def*, const Def* c, const Def* arg) {
169 // `tensor.shape r arr` is the first `r` axes of `arr`'s (fused) array type.
170 auto& w = c->world();
171 auto r = Lit::isa<u64>(c->as<App>()->arg()); // the explicit rank `r`
172 if (!r) return nullptr;
173 if (*r == 0) return w.tuple();
174
175 auto arr = arg->type()->isa<Arr>();
176 if (!arr) return nullptr;
177 auto shape = arr->shape();
178 auto rank = shape.rank();
179 if (!rank || *rank < *r) return nullptr; // `arr` is not (statically) at least rank `r`
180 return *shape.take(*r);
181}
