7bool FreeLocals::analyze() {
8 if (RWPhase::analyze()) return true;
9
10 auto shared = fe::Vector<std::pair<Lam*, const Def*>>();
11 for (const auto& [lam, chain] : escape_.chains()) {
12 for (size_t i = 0, n = chain.ops.size(); i != n; ++i) {
13 auto alloc = chain.ops[i];
14 if (!EscapeAnalysis::allocates(alloc) || escape_.is_top(alloc) || escape_.freed(alloc)) continue;
15 if (escape_.confined(alloc)) {
16 auto last = fe::lookup(chain.last, alloc);
17 auto j = (last ? *last : i) + 1;
18 frees_[j < n ? chain.ops[j] : chain.tail].emplace_back(alloc);
19 } else {
20 shared.emplace_back(lam, alloc);
21 }
22 }
23 }
24
25 if (!shared.empty()) {
26 auto homes = LamSet();
27 for (auto [lam, _] : shared)
28 homes.emplace(lam);
29 // A home nested in another must hang below it, not below the virtual root.
30 auto roots = fe::Vector<Def*>();
31 for (auto lam : homes)
32 if (std::ranges::none_of(homes, [lam](Lam* home) { return home->nests(lam); })) roots.emplace_back(lam);
33
34 const auto nest = Nest(old_world(), roots);
35 mentions(nest.root());
36 for (auto [lam, alloc] : shared)
37 place(nest[lam], nest[lam], alloc);
38 }
39
40 log().d("free_locals: frees before {} ops, on {} edges", frees_.size(), edges_.size());
41 return false;
42}

44const DefSet& FreeLocals::mentions(const Nest::Node* node) {
45 auto res = DefSet();
46 if (auto lam = node->mut() ? node->mut()->isa_mut<Lam>() : nullptr)
47 if (auto i = escape_.chains().find(lam); i != escape_.chains().end())
48 for (auto [alloc, _] : i->second.last)
49 res.emplace(alloc);

57const DefSet& FreeLocals::reach(const Nest::Node* node) {
58 if (auto i = reach_.find(node); i != reach_.end()) return i->second;
59
60 auto res = DefSet();
61 auto todo = fe::BFSWorklist<GIDSet<const Nest::Node*>>();
62 todo.push(node->sibl_deps().nodes());
63 while (!todo.empty()) {
64 auto s = todo.pop();
65 const auto& sub = mentions_.at(s);
66 res.insert(sub.begin(), sub.end());
67 todo.push(s->sibl_deps().nodes());
68 }
69 return reach_[node] = std::move(res);

72void FreeLocals::place(const Nest::Node* home, const Nest::Node* node, const Def* alloc) {
73 for (auto child : node->children().nodes())
74 place(home, child, alloc);
75
76 auto lam = node->mut()->isa_mut<Lam>();
77 if (!lam || escape_.escaping(lam) || (node != home && !live(home, node, alloc))) return;
78 auto i = escape_.chains().find(lam);
79 if (i == escape_.chains().end()) return;
80
81 const auto& chain = i->second;
82 auto deads = DefVec();
83 for (auto target : chain.targets)
84 if (dead(home, target, alloc)) deads.emplace_back(target);
85
86 if (deads.size() == chain.targets.size())
87 frees_[chain.tail].emplace_back(alloc);
88 else
89 for (auto target : deads)
90 edges_[chain.tail][target].emplace_back(alloc);
91}

93bool FreeLocals::live(const Nest::Node* home, const Nest::Node* node, const Def* alloc) {
94 if (mentions_.at(node).contains(alloc)) return true;
95 for (auto x = node; x != home; x = x->inest())
96 if (reach(x).contains(alloc)) return true;
97 return false;

100bool FreeLocals::dead(const Nest::Node* home, const Def* target, const Def* alloc) {

107const Def* FreeLocals::rewrite_imm_App(const App* app) {
108 auto allocs = fe::lookup(frees_, app);
109 auto edges = fe::lookup(edges_, app);
110 if (!allocs && !edges) return RWPhase::rewrite_imm_App(app);
111
112 auto& w = new_world();
113 auto mem = allocs ? mem_def(app->arg()) : nullptr;
114 if (allocs && !mem) log().d("free_locals: `{}` threads no memory, leaking {} allocations", app, allocs->size());
115 auto new_arg = mem ? thread(app->arg(), mem, *allocs) : rewrite(app->arg());
116
117 if (!edges) return w.app(rewrite(app->callee()), new_arg);
118 auto args = DefVec(app->num_targs(), [&](size_t i) {
119 auto arg = app->targ(i);
120 return Pi::isa_cn(arg->type()) ? rewrite_target(arg, *edges) : new_arg->proj(app->num_targs(), i);
121 });
122 return w.app(rewrite_target(app->callee(), *edges), w.tuple(args));
123}

138const Def* FreeLocals::rewrite_target(const Def* target, const DefMap<DefVec>& edges) {
139 auto& w = new_world();
140 if (auto allocs = fe::lookup(edges, target)) return trampoline(rewrite(target), *allocs);
141 if (auto tuple = target->isa<Tuple>())
142 return w.tuple(DefVec(tuple->num_ops(), [&](size_t i) { return rewrite_target(tuple->op(i), edges); }));
143 if (auto sw = Switch(target)) return w.extract(rewrite_target(sw.tuple(), edges), rewrite(sw.index()));
144 return rewrite(target);
145}

147const Def* FreeLocals::trampoline(const Def* cont, const DefVec& allocs) {
148 auto& w = new_world();
149 auto lam = w.mut_con(Pi::isa_cn(cont->type())->dom())->set("free_then");
150 auto args = DefVec(lam->vars());
151 auto mem = std::ranges::find_if(args, [](const Def* arg) { return isa_mem(arg); });
152 if (mem == args.end()) {
153 log().d("free_locals: `{}` takes no memory, leaking {} allocations", cont, allocs.size());
154 return cont;
155 }
156 *mem = free(*mem, allocs);
157 lam->app(false, cont, args);
158 return lam;
159}
