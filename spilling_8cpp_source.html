164void Spiller::cfg() {
165 rpo_pos_.resize(n_, Inf);
166 idom_.resize(n_, Inf);
167 depth_.resize(n_, 0);
168 loop_.resize(n_);
169 ret_cont_.resize(n_, false);
170 branch_target_.resize(n_, false);
171 done_.resize(n_, false);
172
173 auto post = fe::Vector<size_t>();
174 auto seen = fe::Vector<bool>(n_, false);
175 auto stack = fe::Vector<std::pair<size_t, size_t>>{{0, 0}};
176 seen[0] = true;
177 while (!stack.empty()) {
178 auto& [b, i] = stack.back();
179 if (i < block(b).succs.size()) {
180 auto s = block(b).succs[i++];
181 if (!seen[s]) seen[s] = true, stack.emplace_back(s, 0);
182 } else {
183 post.emplace_back(b);
184 stack.pop_back();
185 }
186 }
187 rpo_.assign(post.rbegin(), post.rend());
188 for (size_t i = 0; i != rpo_.size(); ++i)
189 rpo_pos_[rpo_[i]] = i;
190
191 // Cooper, Harvey, Kennedy: A Simple, Fast Dominance Algorithm.
192 auto intersect = [&](size_t a, size_t b) {
193 while (a != b) {
194 while (rpo_pos_[a] > rpo_pos_[b])
195 a = idom_[a];
196 while (rpo_pos_[b] > rpo_pos_[a])
197 b = idom_[b];
198 }
199 return a;
200 };
201 idom_[0] = 0;
202 for (bool todo = true; todo;) {
203 todo = false;
204 for (auto b : rpo_ | std::views::drop(1)) {
205 auto dom = Inf;
206 for (auto p : block(b).preds)
207 if (idom_[p] != Inf) dom = dom == Inf ? p : intersect(p, dom);
208 if (idom_[b] != dom) idom_[b] = dom, todo = true;
209 }
210 }
211
212 auto dominates = [&](size_t a, size_t b) {
213 for (;; b = idom_[b]) {
214 if (a == b) return true;
215 if (b == 0) return false;
216 }
217 };
218 auto header = fe::Vector<size_t>(n_, Inf);
219 for (size_t b = 0; b != n_; ++b) {
220 for (auto h : block(b).succs) {
221 if (!dominates(h, b)) continue;
222 auto& body = loop_[h];
223 auto in = [&](size_t x) { return header[x] == h; };
224 if (!in(h)) header[h] = h, body.emplace_back(h);
225 for (auto todo = fe::Vector<size_t>{b}; !todo.empty();) {
226 auto x = todo.back();
227 todo.pop_back();
228 if (in(x)) continue;
229 header[x] = h;
230 body.emplace_back(x);
231 for (auto p : block(x).preds)
232 todo.emplace_back(p);
233 }
234 }
235 }
236 for (const auto& body : loop_)
237 for (auto x : body)
238 ++depth_[x];
239
240 for (size_t b = 0; b != n_; ++b) {
241 auto kd = kind(b);
242 if (kd == Term::Call)
243 for (auto a : term(b)->args())
244 if (auto lam = Lam::isa_mut_basicblock(a); lam && live_.block(lam)) ret_cont_[index(lam)] = true;
245 if (kd == Term::Branch)
246 for (auto s : block(b).succs)
247 branch_target_[s] = true;
248 }
249}

251void Spiller::next_uses() {
252 uses_.resize(n_);
253 nu_out_.resize(n_);
254 for (size_t b = 0; b != n_; ++b) {
255 for (size_t i = 0; i != size(b); ++i)
256 for (auto v : members(live_.uses(block(b).insns[i])))
257 uses_[b][v].emplace_back(i);
258 for (auto v : members(live_.term_uses(block(b))))
259 uses_[b][v].emplace_back(size(b));
260 }
261
262 for (bool todo = true; todo;) {
263 todo = false;
264 for (auto b : rpo_ | std::views::reverse) {
265 for (auto v : block(b).live_out) {
266 if (!member(v)) continue;
267 auto d = Inf;
268 for (auto s : block(b).succs)
269 if (block(s).live_in.contains(v))
270 d = std::min(d, add(next_use(s, v, 0), depth_[s] < depth_[b] ? Loop_Exit : 0));
271 auto [i, ins] = nu_out_[b].emplace(v, d);
272 if (!ins && d < i->second) i->second = d, todo = true;
273 if (ins && d != Inf) todo = true;
274 }
275 }
276 }

301void Spiller::entry(size_t b) {
302 auto lam = block(b).lam;
303 auto n = lam->num_vars();
304 auto params = DefVec();
305 for (size_t i = 0; i != n; ++i)
306 if (auto x = lam->var(n, i); member(x)) params.emplace_back(x);
307
308 auto take = DefVec();
309 auto by_use = [&](DefVec& vs) {
310 std::ranges::sort(vs, {}, [&](const Def* v) { return std::pair(next_use(b, v, 0), v->gid()); });
311 };
312 auto fill = [&](DefVec vs, size_t cap) {
313 by_use(vs);
314 for (auto v : vs)
315 if (take.size() < cap && next_use(b, v, 0) != Inf && !std::ranges::contains(take, v)) take.emplace_back(v);
316 };
317
318 auto vars = lam->vars();
319 auto cands = DefVec();
320 for (auto v : block(b).live_in)
321 if (member(v)) cands.emplace_back(v);
322
323 // These parameters cannot become memory φs; a dead one leaves its register right away.
324 auto fixed = b == 0 || ret_cont_[b] || branch_target_[b];
325 for (auto x : params)
326 if (fixed || next_use(b, x, 0) == Inf) take.emplace_back(x);
327 if (take.size() > k_)
328 lam->blame("{} backend: `{}` takes more than {} register arguments", target_.name(), lam, k_).bail();
329 std::erase_if(params, [&](const Def* x) { return std::ranges::contains(take, x); });
330 std::ranges::copy(params, std::back_inserter(cands));
331
332 const auto& preds = block(b).preds;
333 if (std::ranges::all_of(preds, [&](size_t p) { return done_[p]; })) {
334 auto avail = DefMap<size_t>();
335 for (auto p : preds) {
336 for (auto v : cands) {
337 auto i = size_t(std::ranges::find(vars, v) - vars.begin());
338 auto a = i < n ? arg(p, b, i) : v;
339 if (!a || Lit::isa(a) || w_exit_[p].contains(a)) ++avail[v];
340 }
341 }
342 auto all = DefVec(), some = DefVec();
343 for (auto v : cands)
344 if (auto i = avail.find(v); i != avail.end()) (i->second == preds.size() ? all : some).emplace_back(v);
345 fill(all, k_);
346 fill(some, k_);
347 } else {
348 auto in_loop = DefVec(), through = DefVec();
349 for (auto v : cands)
350 (next_use(b, v, 0) < Loop_Exit ? in_loop : through).emplace_back(v);
351 fill(in_loop, k_);
352 auto pressure = size_t(0);
353 for (auto x : loop_[b])
354 pressure = std::max(pressure, block(x).max_live[cls_]);
355 auto busy = pressure > through.size() ? pressure - through.size() : 0;
356 fill(through, std::min(k_, take.size() + (k_ > busy ? k_ - busy : 0)));
357 }
358
359 w_entry_[b] = DefSet(take.begin(), take.end());
360 for (auto x : params)
361 if (!w_entry_[b].contains(x)) mphi_.emplace(x);
362}

364void Spiller::walk(size_t b) {
365 entry(b);
366 auto w = w_entry_[b];
367 auto n = size(b);
368
369 auto use = [&](const DefVec& us, size_t pos, size_t more, const Def* at) {
370 if (us.size() + more > k_)
371 at->blame("{} backend: `{}` needs more than {} registers", target_.name(), at, k_).bail();
372 for (auto v : us)
373 if (w.emplace(v).second) reload(b, pos, v);
374 limit(b, w, k_ - more, pos);
375 };
376
377 for (size_t i = 0; i != n; ++i) {
378 auto insn = block(b).insns[i];
379 auto us = members(live_.uses(insn));
380 auto c = target_.constraint(insn);
381 // Fixed operands, results, and clobbers of an instruction are general-purpose registers.
382 if (cls_ != Liveness::Gpr) c.fixed.clear(), c.clobbers.clear();
383 if (c.tied && !member(c.ops[*c.tied])) c.tied.reset();
384 auto live = [&](const Def* v) { return next_use(b, v, i + 1) != Inf; };
385 auto cp = DefVec();
386 for (auto [j, _] : copies(c, live))
387 if (live(c.ops[j])) cp.emplace_back(c.ops[j]);
388 // If the copies of live operands do not fit, these operands leave their registers at the instruction instead.
389 auto evict = us.size() + c.clobbers.size() + cp.size() > k_;
390 use(us, i, c.clobbers.size() + (evict ? 0 : cp.size()), insn);
391 if (evict)
392 for (auto v : cp)
393 w.erase(v);
394 auto ds = members(live_.defs(insn));
395 if (target_.is_sched_cc(insn)) std::erase_if(ds, [&](const Def* d) { return !live(d); });
396 if (ds.size() > k_)
397 insn->blame("{} backend: `{}` defines more than {} registers", target_.name(), insn, k_).bail();
398 limit(b, w, k_ - ds.size(), i + 1);
399 w.insert(ds.begin(), ds.end());
400 }
401
402 auto t = kind(b);
403 if (t != Term::Jump) use(members(live_.term_uses(block(b))), n, 0, term(b));
404 if (t == Term::Call) limit(b, w, k_ > caller_saved() ? k_ - caller_saved() : 0, n + 1);
405 w_exit_[b] = std::move(w);
406 done_[b] = true;
407}

409void Spiller::couple() {
410 for (size_t b = 0; b != n_; ++b) {
411 auto lam = block(b).lam;
412 auto n = lam->num_vars();
413 for (auto p : block(b).preds) {
414 auto end = size(p);
415 auto need = [&](const Def* v) {
416 if (w_exit_[p].contains(v)) return;
417 if (block(p).succs.size() != 1)
418 lam->blame("{} backend: critical edge from `{}` to `{}`", target_.name(), block(p).lam, lam).bail();
419 reload(p, end, v);
420 };
421
422 for (auto v : w_entry_[b])
423 if (block(b).live_in.contains(v)) need(v);
424
425 for (size_t i = 0; i != n; ++i) {
426 auto x = lam->var(n, i);
427 auto a = arg(p, b, i);
428 if (!a || !member(x) || Lit::isa(a)) continue;
429 if (!w_entry_[b].contains(x)) {
430 if (!mphi_.contains(a)) spill_.emplace(a);
431 } else if (next_use(b, x, 0) != Inf) {
432 need(a);
433 }
434 }
435 }
436 }

467size_t Spiller::read_entry(const Def* v, size_t b) {
468 if (auto ver = fe::lookup(entry_[b], v)) return *ver;
469 if (auto ver = fe::lookup(def_ver_, v); ver && &live_.def_block(v) == &block(b)) return *ver;
470
471 const auto& preds = block(b).preds;
472 if (preds.empty()) v->blame("{} backend: `{}` is not in a register at the entry", target_.name(), v).bail();
473 if (preds.size() == 1) return entry_[b][v] = read_exit(v, preds.front());
474
475 // Braun et al., Simple and Efficient Construction of Static Single Assignment Form, CC 2013.
476 auto phi = new_ver();
477 phi_[phi].b = b;
478 phi_[phi].value = v;
479 entry_[b][v] = phi;
480 for (auto p : preds) {
481 auto op = read_exit(v, p);
482 phi_[phi].ops.emplace_back(op);
483 phi_[op].users.emplace_back(phi);
484 }
485 return entry_[b][v] = try_remove(phi);
486}

488size_t Spiller::try_remove(size_t phi) {
489 auto same = Inf;
490 for (auto op : phi_[phi].ops) {
491 op = find(op);
492 if (op == same || op == phi) continue;
493 if (same != Inf) return phi;
494 same = op;
495 }
496 assert(same != Inf);
497 parent_[phi] = same;
498 auto users = std::move(phi_[phi].users);
499 for (auto u : users)
500 phi_[same].users.emplace_back(u);
501 for (auto u : users)
502 if (u != phi && find(u) == u) try_remove(u);
503 return same;
504}

524void Spiller::build() {
525 vers_.resize(parent_.size());
526 for (size_t b = 1; b != n_; ++b) {
527 auto lam = block(b).lam;
528 auto n = lam->num_vars();
529 auto mems = std::ranges::count_if(lam->vars(), [&](const Def* x) { return mphi_.contains(x); });
530 if (mems == 0 && phis_[b].empty()) continue;
531 if (ret_cont_[b])
532 lam->blame("{} backend: return continuation `{}` would need new parameters", target_.name(), lam).bail();
533
534 auto dom = DefVec();
535 for (size_t i = 0; i != n; ++i) {
536 auto x = lam->var(n, i);
537 dom.emplace_back(mphi_.contains(x) ? Target::spill_type(x->type()) : x->type());
538 }
539 for (auto phi : phis_[b])
540 dom.emplace_back(phi_[phi].value->type());
541 auto new_lam = world_.mut_con(dom);
542 new_lam->set(lam->dbg());
543 auto m = dom.size();
544 for (size_t i = 0; i != n; ++i)
545 new_lam->var(m, i)->set(lam->var(n, i)->sym());
546 for (size_t j = 0; j != phis_[b].size(); ++j)
547 new_lam->var(m, n + j)->set(phi_[phis_[b][j]].value->sym());
548 new_lams_[lam] = new_lam;
549 }
550
551 for (auto b : rpo_)
552 build(b);
553}

555void Spiller::build(size_t b) {
556 auto old = block(b).lam;
557 auto new_lam = fe::lookup(new_lams_, old);
558 if (!new_lam) new_lam = old;
559 auto n = old->num_vars();
560 auto m = new_lam->num_vars();
561 auto cur = Def2Def();
562 m_ = nullptr;
563
564 for (size_t i = 0; i != n; ++i) {
565 auto x = old->var(n, i);
566 auto y = new_lam->var(m, i);
567 if (target_.is_mem(x) && !m_) m_ = y;
568 if (mphi_.contains(x))
569 stack_[x] = y;
570 else
571 cur[x] = y;
572 if (auto ver = fe::lookup(def_ver_, x)) vers_[*ver] = y;
573 }
574 for (size_t j = 0; j != phis_[b].size(); ++j)
575 vers_[phis_[b][j]] = new_lam->var(m, n + j);
576 for (const auto& [v, ver] : entry_[b])
577 cur[v] = vers_[find(ver)];
578 // Another class' live-in was already rebuilt together with its own block.
579 for (auto v : block(b).live_in)
580 if (target_.cls(v) != cls_)
581 if (auto nv = fe::lookup(new_defs_, v)) cur[v] = nv;
582 for (auto x : old->vars())
583 if (spill_.contains(x)) stack_[x] = spill(cur[x]);
584
585 auto subst = [&](const Def* def) { return Subst(target_, world_, cur, m_, &new_lams_).rewrite(def); };
586 auto reloads = [&](size_t pos) {
587 for (auto [v, ver] : reloads_[b][pos]) {
588 auto app = root(v);
589 cur[v] = vers_[ver] = is_remat(v) ? reroot(v, app->proj(2, 1), sched(Target::sched_insn(app), app->arg()))
590 : sched(Target::reload(v->type()), slot(v));
591 }
592 };
593
594 for (size_t i = 0, e = size(b); i != e; ++i) {
595 reloads(i);
596 auto insn = block(b).insns[i];
597 auto new_app = Subst(target_, world_, cur, m_, &new_lams_).Rewriter::rewrite(insn);
598 m_ = target_.is_mem(new_app) ? new_app : new_app->proj(2, 0);
599 cur[insn] = new_app;
600 new_defs_[insn] = new_app;
601 for (auto d : live_.defs(insn)) {
602 cur[d] = reroot(d, insn, new_app);
603 new_defs_[d] = cur[d];
604 if (auto ver = fe::lookup(def_ver_, d)) vers_[*ver] = cur[d];
605 if (spill_.contains(d)) stack_[d] = spill(cur[d]);
606 }
607 }
608 reloads(size(b));
609
610 auto t = term(b);
611 auto kd = kind(b);
612 auto body = static_cast<const Def*>(nullptr);
613 if (kd == Term::Jump) {
614 auto target = t->callee()->as_mut<Lam>();
615 auto s = index(target);
616 auto k = target->num_vars();
617 auto args = DefVec();
618 for (size_t i = 0; i != k; ++i) {
619 auto x = target->var(k, i);
620 auto a = t->arg(k, i);
621 if (!mphi_.contains(x)) {
622 // A value this block does not define belongs to a dead parameter, which may receive anything.
623 auto dead = member(a) && !Lit::isa(a) && !cur.contains(a);
624 args.emplace_back(dead ? world_.lit(x->type(), 0) : subst(a));
625 } else if (Lit::isa(a)) {
626 args.emplace_back(spill(a));
627 } else {
628 args.emplace_back(slot(a));
629 }
630 }
631 auto p = size_t(std::ranges::find(block(s).preds, b) - block(s).preds.begin());
632 for (auto phi : phis_[s])
633 args.emplace_back(vers_[find(phi_[phi].ops[p])]);
634 // m_ may have moved on through the spills of literals.
635 for (auto& a : args)
636 if (target_.is_mem(a)) a = m_;
637 body = world_.app(subst(target), world_.tuple(args));
638 } else {
639 if (kd == Term::Branch)
640 for (auto target : t->callee()->as<Extract>()->tuple()->projs())
641 if (auto lam = target->isa_mut<Lam>(); lam && new_lams_.contains(lam))
642 lam->blame("{} backend: branch target `{}` would need new parameters", target_.name(), lam).bail();
643 body = Subst(target_, world_, cur, m_, &new_lams_).rewrite_jump(t);
644 }
645
646 auto filter = old->filter();
647 if (new_lam == old)
648 old->unset()->set(subst(filter), body);
649 else
650 new_lam->set(filter->is_closed() ? filter : world_.lit_ff(), body);
651}

653/// A hash-consed branch tuple may share one target among several branches; each branch then gets its own wrapper.
654void split_critical_edges(Lam* fun) {
655 auto& w = fun->world();
656 auto preds = LamMap<size_t>();
657 auto blocks = fe::Vector<Lam*>{fun};
658 auto seen = LamSet{fun};
659 for (size_t i = 0; i != blocks.size(); ++i)
660 if (auto term = blocks[i]->body()->isa<App>())
661 for_each_succ(term, [&](Lam* bb) {
662 ++preds[bb];
663 if (seen.emplace(bb).second) blocks.emplace_back(bb);
664 });
665
666 for (auto lam : blocks) {
667 auto term = lam->body()->isa<App>();
668 auto ex = term ? term->callee()->isa<Extract>() : nullptr;
669 if (!ex) continue;
670 auto split = false;
671 auto targets = ex->tuple()->projs();
672 for (auto& target : targets) {
673 auto bb = Lam::isa_mut_basicblock(target);
674 if (!bb || preds[bb] < 2) continue;
675 split = true;
676 auto wrapper = w.mut_con(bb->dom());
677 wrapper->set(bb->dbg());
678 target = wrapper->set(w.lit_ff(), w.app(bb, wrapper->var()));
679 w.log().d("split critical edge from `{}` to `{}` via `{}`", lam, bb, wrapper);
680 }
681 if (split) {
682 auto filter = lam->filter();
683 lam->unset()->set(filter, w.app(w.extract(w.tuple(targets), ex->index()), term->arg()));
684 }
685 }
686}
