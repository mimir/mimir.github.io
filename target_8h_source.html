13/// What an instruction on the chain prescribes about the registers of its operands; registers are hardware encodings.
14struct Constraint {
15 DefVec ops; ///< See Target::operands.
16 fe::Vector<std::pair<size_t, size_t>> fixed; ///< Non-literal operands that have to be in a given register.
17 fe::Vector<std::pair<size_t, size_t>> results; ///< Results, see Target::result, that land in a given register.
18 std::optional<size_t> tied; ///< The operand whose register the (only) result takes.
19 fe::Vector<size_t> clobbers; ///< Registers written while the operands still occupy theirs.
20};
21
22/// The fixed operands with their registers, and the tied one with no register if it is @p live_after.
23inline fe::Vector<std::pair<size_t, std::optional<size_t>>> copies(const Constraint& c, auto live_after) {
24 auto res = fe::Vector<std::pair<size_t, std::optional<size_t>>>();
25 for (auto [j, r] : c.fixed)
26 res.emplace_back(j, r);
27 if (c.tied && live_after(c.ops[*c.tied])) res.emplace_back(*c.tied, std::nullopt);
28 return res;
29}
