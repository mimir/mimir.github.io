19bool AddMem::analyze() {
20 auto queue = fe::BFSWorklist<DefSet>();
21 auto pins = fe::BFSWorklist<DefSet>();
22 for (auto mut : old_world().externals().muts())
23 queue.push(mut);
24
25 while (!queue.empty()) {
26 auto def = queue.pop();
27
28 // These continuations receive the current memory.
29 if (auto app = def->isa<App>();
30 app && app->axm() && !Axm::isa<mem::fresh>(app) && !Axm::isa<mem::slot>(app) && !Axm::isa<mem::mslot>(app))
31 for (auto arg : app->arg()->projs())
32 if (auto lam = arg->isa_mut<Lam>()) pins.push(lam);
33
34 for (auto d : def->deps())
35 queue.push(d);
36 }
37
38 while (!pins.empty()) {
39 auto def = pins.pop();
40 if (auto lam = def->isa_mut<Lam>()) pins_.emplace(lam);
41 for (auto d : def->deps())
42 pins.push(d);
43 }
44
45 return false;
46}

48const Def* AddMem::rewrite(const Def* old_def) {
49 if (pinned_) {
50 // The memo holds the mem-extended variant of a type.
51 if (auto pi = old_def->isa_imm<Pi>()) return rewrite_imm_Pi(pi);
52 if (auto sigma = old_def->isa_imm<Sigma>()) return rewrite_imm_Sigma(sigma);
53 if (auto arr = old_def->isa_imm<Arr>()) return rewrite_imm_Seq(arr);
54 } else if (curr_mem_) {
55 // Hash-consed equal tuples splice different memories.
56 if (auto tuple = old_def->isa<Tuple>(); tuple && std::ranges::any_of(tuple->ops(), is_mem))
57 return rewrite_imm_Tuple(tuple);
58 auto new_def = Rewriter::rewrite(old_def); // may advance curr_mem_
59 auto splice = [this](const Def* def) { return is_mem(def) && def->type() == curr_mem_->type(); };
60 if (is_mem(old_def)) return splice(new_def) ? curr_mem_ : new_def;
61
62 // An η-reduced result of a memory operation carries its memory along.
63 if (auto type = old_def->type(); type && type->isa<Sigma>() && has_mem_ty(type)) {
64 auto n = new_def->num_projs();
65 return new_world().tuple(DefVec(n, [&](size_t i) {
66 auto proj = new_def->proj(n, i);
67 return splice(proj) ? curr_mem_ : proj;
68 }));
69 }
70 return new_def;
71 }
72 return Rewriter::rewrite(old_def);
73}

75const Def* AddMem::rewrite_imm_Pi(const Pi* pi) {
76 auto new_pi = Rewriter::rewrite_imm_Pi(pi)->as<Pi>();
77 if (pinned_ || !Pi::isa_cn(pi) || has_mem(pi)) return new_pi;
78
79 auto& w = new_world();
80 auto mem = w.call<mem::M>(0);
81
82 // Prepending shifts the indices a dependent domain refers to (see issue #177).
83 if (auto [sigma, old_var] = new_pi->dom()->isa_binder<Sigma>(); sigma) {
84 auto n = sigma->num_ops();
85 auto new_sigma = w.mut_sigma(sigma->type(), n + 1);
86 new_sigma->set(0, mem);
87 for (size_t i = 0; i != n; ++i) {
88 auto shift = w.tuple(DefVec(n, [&](size_t j) { return j < i ? new_sigma->var(n + 1, j + 1) : mem; }));
89 new_sigma->set(i + 1, VarRewriter(old_var, shift).rewrite(sigma->op(i)));
90 }
91 return w.cn(new_sigma);
92 }
93
94 return w.cn(Def::cat(mem, new_pi->doms()));

119 // What hangs off the memory of one function is out of scope in another.
120 auto scopes = std::move(scopes_);

122 scopes_.emplace_back(std::move(scopes.front()));
123 push();
124 bind_new(new_lam->has_var());

127 auto scheduler = Scheduler(nest);
128 auto outer = std::exchange(scheduler_, &scheduler);
129 thread(old_lam, new_lam);

138Lam* AddMem::thread(Lam* old_lam, Lam* new_lam) {
139 auto outer = active_;
140 if (!frames_.empty()) frames_[active_].mem = curr_mem_;
141 frames_.emplace_back(old_lam, nullptr);
142 active_ = frames_.size() - 1;
143 curr_mem_ = mem_var(new_lam);
144
145 new_lam->set(rewrite(old_lam->filter()), rewrite(old_lam->body()));
146
147 // A nested continuation may have advanced the chain of an outer one.
148 frames_.pop_back();
149 active_ = outer;
150 curr_mem_ = frames_.empty() ? nullptr : frames_[active_].mem;
151 return new_lam;
152}

184const Def* AddMem::thread(const App* app) {
185 auto& w = new_world();
186
187 // The memory effects of callee and argument precede the call.
188 auto new_callee = rewrite(app->callee());
189 auto new_arg = rewrite(app->arg());
190 auto mem = curr_mem_;
191
192 if (auto old_pi = app->callee()->type()->isa<Pi>())
193 if (auto new_pi = new_callee->type()->isa<Pi>(); new_pi && new_pi->num_doms() == old_pi->num_doms() + 1)
194 new_arg = w.tuple(Def::cat(mem, new_arg->projs(old_pi->num_doms())));

197 if (auto m = mem_def(new_app); m && m->type() == mem->type()) curr_mem_ = m;

201const Def* AddMem::rewrite_imm_Tuple(const Tuple* tuple) {
202 if (!curr_mem_) return Rewriter::rewrite_imm_Tuple(tuple);
203
204 // Values advance the memory that the memory operands splice, and continuations run after both.
205 auto rank = [](const Def* op) {
206 if (is_mem(op)) return 1;
207 return op->type() && Pi::isa_cn(op->type()) ? 2 : 0;
208 };
209
210 auto n = tuple->num_ops();
211 auto new_ops = DefVec(n);
212 for (int r = 0; r != 3; ++r)
213 for (size_t i = 0; i != n; ++i)
214 if (rank(tuple->op(i)) == r) new_ops[i] = rewrite(tuple->op(i));
215
216 // A continuation may have advanced the chain after the memory operands were rewritten.
217 for (size_t i = 0; i != n; ++i)
218 if (rank(tuple->op(i)) == 1 && new_ops[i]->type() == curr_mem_->type()) new_ops[i] = curr_mem_;
219 return new_world().tuple(rewrite(tuple->type()), new_ops);
220}
