13void AddMem::start() {
14 // Collect the lams whose ABI is pinned: everything (transitively) reachable from an axm-app argument
15 // (combiners, affine index mappings, initial accumulators of `%matrix.map_reduce_aff`, …).
16 unique_queue<DefSet> roots;
17 for (auto mut : old_world().externals().muts())
18 if (mut) roots.push(mut);
19
20 unique_queue<DefSet> pinned;
21 while (!roots.empty()) {
22 auto def = roots.pop();
23 if (auto app = def->isa<App>(); app && app->axm())
24 for (auto arg : app->arg()->projs())
25 if (auto lam = arg->isa_mut<Lam>()) pinned.push(lam);
26 for (auto op : def->ops())
27 if (op) roots.push(op);
28 }
29 while (!pinned.empty()) {
30 auto def = pinned.pop();
31 if (auto lam = def->isa_mut<Lam>()) preserved_.emplace(lam);
32 for (auto op : def->ops())
33 if (op) pinned.push(op);
34 }
35
36 RWPhase::start();
37}

39const Def* AddMem::rewrite(const Def* old_def) {
40 // Rewrite every memory operand to the current memory - after threading the operand's producers, which
41 // advances curr_mem_ along the way. Placeholders (`⊥`/`⊤ : %mem.M 0`) are thereby spliced into the chain.
42 if (curr_mem_ && !preserving_ && !is_bootstrapping() && !old_def->isa_mut() && isa_mem(old_def)) {
43 Rewriter::rewrite(old_def);
44 return curr_mem_;
45 }
46 return Rewriter::rewrite(old_def);
47}

49const Def* AddMem::rewrite_imm_Pi(const Pi* pi) {
50 auto new_pi = Rewriter::rewrite_imm_Pi(pi)->as<Pi>();
51 if (is_bootstrapping() || preserving_) return new_pi;
52
53 // Only continuations are mem-extended; a pi that already threads memory is left alone.
54 if (Pi::isa_cn(pi) && !has_leading_mem(pi)) {
55 auto& w = new_world();
56 auto mem = w.call<mem::M>(0);
57 auto dom = new_pi->dom();
58
59 // A dependent domain refers to its own Var.
60 // So prepending a leading mem shifts every component's index by one.
61 // Rebuild the domain as a fresh mutable Sigma and remap the old domain-Var to the shifted components of the new
62 // one. Otherwise the dependent references dangle (see issue #177).
63 if (auto sigma = dom->isa_mut<Sigma>(); sigma && sigma->has_var()) {
64 auto n = sigma->num_ops();
65 auto old_var = sigma->has_var();
66 auto new_sigma = w.mut_sigma(sigma->type(), n + 1);
67 new_sigma->set(0, mem);
68 // Component `i` may only refer to earlier components, whose (index-shifted) new Vars are already
69 // set - so build the substitution per component; padding slots `≥ i` are never extracted.
70 for (size_t i = 0; i != n; ++i) {
71 auto shift = w.tuple(DefVec(n, [&](size_t j) { return j < i ? new_sigma->var(n + 1, j + 1) : mem; }));
72 auto rw = VarRewriter(old_var, shift);
73 new_sigma->set(i + 1, rw.rewrite(sigma->op(i)));
74 }
75 return w.cn(new_sigma);
76 }
77
78 auto new_dom = DefVec();
79 new_dom.emplace_back(mem);
80 for (size_t i = 0, e = new_pi->num_doms(); i != e; ++i)
81 new_dom.emplace_back(new_pi->dom(i));
82 return w.cn(new_dom);
83 }
84 return new_pi;
85}

87const Def* AddMem::rewrite_mut_Lam(Lam* old_lam) {
88 if (is_bootstrapping() || preserving_) return Rewriter::rewrite_mut_Lam(old_lam);
89
90 // Pinned ABI (an axm-app argument and everything below it): rewrite verbatim - no memory threaded or added.
91 if (preserved_.contains(old_lam)) {
92 auto save = std::exchange(preserving_, true);
93 auto new_it = Rewriter::rewrite_mut_Lam(old_lam);
94 preserving_ = save;
95 return new_it;
96 }
97
98 auto new_lam = new_world().mut_lam(rewrite(old_lam->type())->as<Pi>())->set(old_lam->dbg());
99 map(old_lam, new_lam);
100
101 // Map the parameters, accounting for a possibly inserted leading mem var.
102 if (auto n = old_lam->num_vars(); n != 0) {
103 auto offset = new_lam->num_doms() - old_lam->num_doms(); // 1 iff we prepended a mem var
104 for (size_t i = 0; i != n; ++i)
105 map(old_lam->var(i), new_lam->var(i + offset)->set(old_lam->var(i)->dbg()));
106 // A use of the whole parameter tuple is reconstructed from the new (shifted) components.
107 if (n > 1)
108 map(old_lam->var(), new_world().tuple(DefVec(n, [&](size_t i) { return new_lam->var(i + offset); })));
109 }
110
111 if (!old_lam->is_set()) return new_lam;
112
113 // The body's current memory is this lam's (leading or grouped) mem parameter - or none for direct-style fns.
114 auto save = std::exchange(curr_mem_, mem::mem_var(new_lam));
115 new_lam->set(rewrite(old_lam->filter()), rewrite(old_lam->body()));
116 curr_mem_ = save;
117 return new_lam;
118}

120const Def* AddMem::rewrite_imm_App(const App* app) {
121 if (is_bootstrapping() || preserving_ || !curr_mem_) return Rewriter::rewrite_imm_App(app);
122
123 auto& w = new_world();
124 // Rewrite the argument before the callee (as the base Rewriter does). This threads the current memory
125 // through the argument's memory effects first; and because operands are rewritten before their users, a
126 // shared memory operation is anchored in the scope that consumes its result mem (its own scope) rather
127 // than one that merely reuses its non-mem result. curr_mem_ then holds the memory *after* the argument.
128 auto new_arg = rewrite(app->arg());
129 auto mem = curr_mem_;
130 auto new_callee = rewrite(app->callee());
131
132 auto old_pi = app->callee()->type()->isa<Pi>();
133 auto new_pi = new_callee->type()->isa<Pi>();
134 if (old_pi && new_pi && new_pi->num_doms() == old_pi->num_doms() + 1) {
135 // The callee gained a leading mem parameter: splice the current memory in front of the arguments.
136 auto n = old_pi->num_doms();
137 auto args = DefVec(n + 1);
138 args[0] = mem;
139 for (size_t i = 0; i != n; ++i)
140 args[i + 1] = new_arg->proj(n, i);
141 new_arg = w.tuple(args);
142 }
143
144 auto new_app = w.app(new_callee, new_arg);
145 advance_mem(new_app);
146 return new_app;
147}

149const Def* AddMem::rewrite_imm_Tuple(const Tuple* tuple) {
150 if (is_bootstrapping() || preserving_ || !curr_mem_) return Rewriter::rewrite_imm_Tuple(tuple);
151
152 // The current memory must be threaded through the operands in the right order, because a memory operand
153 // (which resolves to the *current* memory) is positioned freely relative to the operands that establish
154 // the real ordering. Rewrite in three groups:
155 // 1. plain values first - e.g. the `buf` of a buffer op's `(⊥ : %mem.M 0, buf)` argument, whose memory
156 // effects must precede the memory operand;
157 // 2. memory operands next - now resolving to the up-to-date current memory;
158 // 3. continuation values last - their bodies run later, so a shared memory operation they capture must
159 // already have been anchored (threaded) in this scope by the memory operands above.
160 auto rank = [](const Def* op) { return isa_mem(op) ? 1 : (op->type() && Pi::isa_cn(op->type()) ? 2 : 0); };
161
162 auto& w = new_world();
163 auto n = tuple->num_ops();
164 auto new_ops = DefVec(n);
165 for (int r = 0; r != 3; ++r)
166 for (size_t i = 0; i != n; ++i)
167 if (rank(tuple->op(i)) == r) new_ops[i] = rewrite(tuple->op(i));
168 return w.tuple(rewrite(tuple->type()), new_ops);
169}
