38void Liveness::chain(size_t b) {
39 auto lam = blocks_[b].lam;
40 auto term = lam->body()->isa<App>();
41 if (!term) lam->blame("{} backend: `{}` does not end in a jump", target_->name(), lam).bail();
42
43 const Def* end = nullptr;
44 for (auto arg : term->args())
45 if (target_->is_mem(arg)) end = arg;
46 if (!end) lam->blame("{} backend: `{}` has no `mem.M` chain", target_->name(), lam).bail();
47
48 auto insns = fe::Vector<const App*>();
49 for (auto m = end; !(m->isa<Extract>() && m->as<Extract>()->tuple() == lam->var()) && m != lam->var();) {
50 auto node = m->isa<Extract>() ? m->as<Extract>()->tuple() : m;
51 auto app = node->isa<App>();
52 auto prev = app ? target_->prev_mem(app) : nullptr;
53 if (!prev) m->blame("{} backend: unexpected memory `{}` in `{}`", target_->name(), m, lam).bail();
54 insns.emplace_back(app);
55 m = prev;
56 }
57 std::ranges::reverse(insns);
58 for (size_t i = 0; i != insns.size(); ++i)
59 insn2block_.emplace(insns[i], std::pair(b, i));
60 blocks_[b].insns = std::move(insns);
61
62 for_each_succ(term, [&](Lam* bb) {
63 auto s = add(bb);
64 blocks_[b].succs.emplace_back(s);
65 blocks_[s].preds.emplace_back(b);
66 });
67}

95void Liveness::values(Block& block) {
96 auto run = [this](auto... defs) {
97 auto res = DefVec();
98 auto done = DefSet();
99 (values(defs, res, done), ...);
100 return res;
101 };
102
103 block.params = run(block.lam->var());
104 for (auto insn : block.insns) {
105 block.defs.emplace_back(run(insn));
106 block.uses.emplace_back(run(insn->callee(), insn->arg()));

109 auto ex = term->callee()->isa<Extract>();
110 if (auto cc = ex ? target_->isa_sched_cc(ex->index()) : nullptr)
111 block.term_uses = run(cc->arg(), ex->tuple(), term->arg());

124void Liveness::live_in(size_t b, const Def* value) {
125 auto def = size_t(&def_block(value) - blocks_.data());
126 auto stack = fe::Vector<size_t>{b};
127 while (!stack.empty()) {
128 auto i = stack.back();
129 stack.pop_back();
130 if (!blocks_[i].live_in.emplace(value).second) continue;
131 for (auto p : blocks_[i].preds)
132 if (blocks_[p].live_out.emplace(value).second && p != def) stack.emplace_back(p);
133 }

136void Liveness::sweep(Block& block) {
137 auto live = block.live_out;
138 auto count = Pressure{};
139 for (auto value : live)
140 ++count[target_->cls(value)];
141
142 auto gen = [&](const DefVec& values) {
143 for (auto value : values)
144 if (live.emplace(value).second) ++count[target_->cls(value)];
145 for (size_t c = 0; c != Num_Classes; ++c)
146 block.max_live[c] = std::max(block.max_live[c], count[c]);
147 };
148 auto kill = [&](const DefVec& values) {
149 for (auto value : values)
150 if (live.erase(value)) --count[target_->cls(value)];
151 };
152
153 gen(block.term_uses);
154 for (size_t i = block.insns.size(); i-- != 0;) {
155 const auto& ds = block.defs[i];
156 // A dead result occupies a register nonetheless, unless it is a condition that only a branch consumes.
157 if (!target_->is_sched_cc(block.insns[i])) gen(ds);
158 kill(ds);
159 gen(block.uses[i]);
160 }
161 gen(block.params);
162}

166/// The block that @p block should fall through to; @p ret is the return continuation of its function.
167Lam* fall_through(const Target& target, const Liveness::Block& block, const Def* ret) {
168 auto term = block.lam->body()->as<App>();
169 auto callee = term->callee();
170 if (callee == ret) return nullptr;
171 if (Lam::isa_mut_returning(callee)) {
172 auto k = term->arg(term->num_args(), term->num_args() - 1);
173 return k == ret ? nullptr : Lam::isa_mut_basicblock(k);
174 }
175 if (auto ex = callee->isa<Extract>()) {
176 if (ex->tuple()->num_projs() != 2) return nullptr;
177 auto [e, t] = ex->tuple()->projs<2>([](const Def* d) { return Lam::isa_mut_basicblock(d); });
178 if (!e || !t) return nullptr;
179 return takes_values(target, t) ? t : e;
180 }
181 return Lam::isa_mut_basicblock(callee);
182}

190fe::Vector<const Liveness::Block*> layout(const Liveness& live) {
191 auto ret = live.blocks().front().lam->ret_var();
192 auto order = fe::Vector<const Liveness::Block*>();
193 auto placed = LamSet();
194 for (auto queue = std::deque<Lam*>{live.blocks().front().lam}; !queue.empty(); queue.pop_front()) {
195 for (auto lam = queue.front(); lam && placed.emplace(lam).second;) {
196 auto b = live.block(lam);
197 order.emplace_back(b);
198 for (auto s : b->succs)
199 queue.emplace_back(live.blocks()[s].lam);
200 lam = fall_through(live.target(), *b, ret);
201 if (lam && !live.block(lam)) lam = nullptr;
202 }
203 }
204 return order;
205}

216const Def* Assign::rewrite(const Def* def) {
217 auto& w = world();
218 if (def->isa_mut() || def->is_closed()) return def;
219 if (live_.target().is_in(def)) {
220 auto app = def->as<App>();
221 return w.app(rewrite(app->callee()), raw(app->arg()));
222 }
223 if (live_.local(def)) {
224 if (auto r = fe::lookup(regs_, def)) return live_.target().in(cls_, *r, raw(def));
225 // A tuple of values, e.g. the whole variable of a block or an instruction, is used through its projections.
226 if (auto projs = def->projs(); projs.size() > 1 && std::ranges::any_of(projs, [&](const Def* d) {
227 return regs_.contains(d) || d->num_projs() > 1;
228 })) {
229 auto ops = DefVec();
230 for (auto d : projs)
231 ops.emplace_back(rewrite(d));
232 return w.tuple(ops);
233 }
234 return raw(def);
235 }
236 return Rewriter::rewrite(def);
237}
