36void Liveness::chain(size_t b) {
37 auto lam = blocks_[b].lam;
38 auto term = lam->body()->isa<App>();
39 if (!term) lam->blame(MIM_X86_BE "`{}` does not end in a jump", lam).bail();
40
41 const Def* end = nullptr;
42 for (auto arg : term->args())
43 if (is_mem(arg)) end = arg;
44 if (!end) lam->blame(MIM_X86_BE "`{}` has no `mem.M` chain", lam).bail();
45
46 auto insns = fe::Vector<const App*>();
47 for (auto m = end; !(m->isa<Extract>() && m->as<Extract>()->tuple() == lam->var()) && m != lam->var();) {
48 auto node = m->isa<Extract>() ? m->as<Extract>()->tuple() : m;
49 auto app = node->isa<App>();
50 auto prev = app ? prev_mem(app) : nullptr;
51 if (!prev) m->blame(MIM_X86_BE "unexpected memory `{}` in `{}`", m, lam).bail();
52 insns.emplace_back(app);
53 m = prev;
54 }
55 std::ranges::reverse(insns);
56 for (size_t i = 0; i != insns.size(); ++i)
57 insn2block_.emplace(insns[i], std::pair(b, i));
58 blocks_[b].insns = std::move(insns);
59
60 for_each_succ(term, [&](Lam* bb) {
61 auto s = add(bb);
62 blocks_[b].succs.emplace_back(s);
63 blocks_[s].preds.emplace_back(b);
64 });
65}

127void Liveness::live_in(size_t b, const Def* value) {
128 auto def = size_t(&def_block(value) - blocks_.data());
129 auto stack = fe::Vector<size_t>{b};

139void Liveness::sweep(Block& block) {
140 auto live = block.live_out;
141 auto count = Pressure{};
142 for (auto value : live)
143 ++count[cls(value)];
144
145 auto gen = [&](const DefVec& values) {
146 for (auto value : values)
147 if (live.emplace(value).second) ++count[cls(value)];
148 for (size_t c = 0; c != Num_Classes; ++c)
149 block.max_live[c] = std::max(block.max_live[c], count[c]);
150 };
151 auto kill = [&](const DefVec& values) {
152 for (auto value : values)
153 if (live.erase(value)) --count[cls(value)];
154 };
155
156 gen(block.term_uses);
157 for (size_t i = block.insns.size(); i-- != 0;) {
158 const auto& ds = block.defs[i];
159 // A dead result occupies a register nonetheless, unless it is a condition that only a branch consumes.
160 if (!is_sched_cc(block.insns[i])) gen(ds);
161 kill(ds);
162 gen(block.uses[i]);
163 }
164 gen(block.params);
165}
