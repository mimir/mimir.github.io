109 assert(rows_ == rhs.rows() && cols_ == rhs.cols());

111 for (size_t r = 0; r < rows_; ++r)

118 assert(rows_ == src.rows() && cols_ == src.cols());

125 /// Erases the rows in `rows`, which must be sorted and distinct.
126 /// The `&` ensures erasing an rvalue cannot compact the storage yet drop the shrunken dimensions.
127 constexpr void erase_rows(fe::View<size_t> rows) & requires(!std::is_const_v<T>) {
128 if (rows.empty()) return;
129 assert(std::ranges::is_sorted(rows) && "erase_rows: rows not sorted");
130 assert(std::ranges::adjacent_find(rows) == rows.end() && "erase_rows: duplicate row");
131 assert(rows.back() < rows_ && "erase_rows: row out of range");
132 if (row_major()) {
133 size_t dst = 0;
134 size_t src = 0;
135 for (const size_t row : rows) {
136 if (dst != src) std::copy_n(data_ + src * cols_, (row - src) * cols_, data_ + dst * cols_);
137 dst += row - src;
138 src = row + 1;
139 }
140 if (dst != src) std::copy_n(data_ + src * cols_, (rows_ - src) * cols_, data_ + dst * cols_);
141 dst += rows_ - src;
142 *this = BasicMatrix(data_, dst, cols_, true);
143 } else {
144 erase_storage_cols(rows);
145 *this = BasicMatrix(data_, rows_ - rows.size(), cols_, false);
146 }
147 }
148
149 /// Erases the columns in `cols`, which must be sorted and distinct.
150 constexpr void erase_cols(fe::View<size_t> cols) & requires(!std::is_const_v<T>) {
151 *this = transposed();
152 erase_rows(cols);
153 *this = transposed();
154 }

156private:
157 constexpr BasicMatrix(T* data, size_t rows, size_t cols, bool row_major)
158 : data_(data)
159 , rows_(rows)
160 , cols_(cols)
161 , row_stride_(row_major || rows <= 1 || cols <= 1 ? cols : 1)
162 , col_stride_(row_major || rows <= 1 || cols <= 1 ? 1 : rows) {}
163
164 /// Erases the storage columns in `cols`, which are positions within each of the `cols_` rows of `rows_` entries.
165 constexpr void erase_storage_cols(fe::View<size_t> cols) {
166 T* out = data_;
167 for (size_t r = 0; r < cols_; ++r) {
168 const T* const base = data_ + r * rows_;
169 const T* in = base;
170 for (const auto col : cols) {
171 out = std::copy(in, base + col, out);
172 in = base + col + 1;
173 }
174 out = std::copy(in, base + rows_, out);
175 }
176 }
177
178 T* data_ = nullptr;
179 size_t rows_ = 0;
180 size_t cols_ = 0;
181 size_t row_stride_ = 0;
182 size_t col_stride_ = 1;
183};
