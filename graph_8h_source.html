30public:

34 constexpr Cost(Rep value)

36
37 [[nodiscard]] static constexpr Cost inf() { return std::numeric_limits<Rep>::max(); }
38 [[nodiscard]] static constexpr Cost min() { return std::numeric_limits<Rep>::min(); }
39
40 [[nodiscard]] constexpr Rep value() const { return value_; }
41 [[nodiscard]] constexpr bool is_inf() const { return value_ == std::numeric_limits<Rep>::max(); }
42 constexpr explicit operator Rep() const { return value_; }
43
44 friend constexpr auto operator<=>(Cost, Cost) = default;

46 /// Any sign; saturates and Cost::inf absorbs.
47 friend constexpr Cost operator+(Cost lhs, Cost rhs) {
48 if (lhs.is_inf() || rhs.is_inf()) return inf();
49 const auto sum = int64_t{lhs.value_} + rhs.value_;
50 return static_cast<Rep>(std::clamp<int64_t>(sum, min().value_, inf().value_));
51 }

115 assert(rows_ == rhs.cols() && cols_ == rhs.rows());
116 assert((size() <= 1 || data_ != rhs.data()) && "add_transposed: rhs aliases this");
117 for (size_t r = 0; r < rows_; ++r)

122 /// Replaces every entry with the transpose of `src`.
123 constexpr void transpose_from(BasicMatrix<const Cost> src) requires std::same_as<T, Cost> {
124 assert(rows_ == src.cols() && cols_ == src.rows());
125 assert((size() <= 1 || data_ != src.data()) && "transpose_from: src aliases this");
126 for (size_t r = 0; r < rows_; ++r)
127 for (size_t c = 0; c < cols_; ++c)

129 }

131 // `&` so that erasing on an rvalue cannot compact the storage yet drop the shrunken dimensions.
132 constexpr void erase_row(size_t row) & requires std::same_as<T, Cost> {
133 assert(row < rows_);
134 std::copy_n(data_ + (row + 1) * cols_, (rows_ - row - 1) * cols_, data_ + row * cols_);
135 --rows_;
136 }
137
138 constexpr void erase_col(size_t col) & requires std::same_as<T, Cost> {
139 assert(col < cols_);
140 T* out = data_;
141 for (size_t r = 0; r < rows_; ++r) {
142 const Cost* const in = data_ + r * cols_;
143 out = std::copy(in, in + col, out);
144 out = std::copy(in + col + 1, in + cols_, out);
145 }
146 --cols_;
147 }

149private:
150 T* data_ = nullptr;
151 size_t rows_ = 0;
152 size_t cols_ = 0;

158/// A PBQP problem graph.
159///
160/// Every node carries a cost vector with one entry per option. Every edge
161/// carries a cost matrix indexed by the options of its endpoints:
162///
163/// matrix[src_option, dst_option]
164///
165/// The total cost of an assignment `x` is
166///
167/// sum_n c_n(x_n) + sum_{mn} M_mn(x_m, x_n)
168///
169/// A node holds at most one edge to any other node, so parallel edges have to be merged by summing their matrices
170/// first. A Graph only grows; ids are handed out consecutively from 0.
