18/// Drops, from the (unfolded) index `idx`, the components of size-1 dimensions of `shape`
19/// (`%buffer.Buf` normalizes size-1 axes away, mirroring the folding of array types).
20const Def* fold_index(const Def* shape, const Def* idx) {
21 auto& w = shape->world();
22 auto r = shape->num_projs();
23 DefVec out;
24 bool dropped = false;
25 for (size_t i = 0; i != r; ++i) {
26 if (auto l = Lit::isa<u64>(shape->proj(r, i)); l && *l == 1)
27 dropped = true;
28 else
29 out.push_back(idx->proj(r, i));
30 }
31 // Without dropped axes the tuple below would just eta-reduce back to `idx` — but only after
32 // World::tuple's pack normalization has alpha-compared the projections, which walks `idx`'s whole
33 // (mem-threaded, Var-dependent) coordinate chain per element pair — exponentially. Return `idx` directly.
34 if (!dropped) return idx;
35 return w.tuple(out);
36}

38/// Builds a counting `affine.For` loop body carrying `acc` (a `{mem, …}` tuple).
39std::pair<Lam*, const Def*> counting_for(const Def* bound, const Def* acc, const Def* exit, Sym name) {
40 auto& w = bound->world();
41 auto acc_ty = acc->type();
42 auto body = w.mut_con({/* iter */ w.type_i64(), /* acc */ acc_ty, /* return */ w.cn(acc_ty)})->set(name);
43 auto for_loop = w.call<affine::For>(body, exit, Defs{w.lit_i64(0), bound, w.lit_i64(1), acc});
44 return {body, for_loop};
45}

47/// Pointwise scaffold shared by `pad`/`concat`: a `[mem, ins] → [mem, Buf]` fun (spliced via
48/// `%cps.cps2ds`) that allocates the output buffer, loops over `s_out` carrying `{mem, buf}`, and writes
49/// `compute`'s element at the (identity) output coordinates.
50/// `compute(iters, ins, mem)` receives the raw i64 loop counters, the fun's inputs var, and the current
51/// mem; it returns `(mem', element)`.
52template<class Compute>
53const Def* build_pointwise(World& w,
54 const Def* result_ty, // [%mem.M 0, %buffer.Buf (r, s_out, T)]
55 const Def* op_mem,
56 const Def* op_ins,
57 const Def* s_out,
58 u64 rn,
59 const std::string& name,
60 Compute&& compute) {
61 auto mem_ty = w.call<mem::M>(0);
62 auto fun = w.mut_fun(w.sigma({mem_ty, op_ins->type()}), result_ty)->set(name);
63 auto call = w.app(cps::op_cps2ds_dep(fun), w.tuple({op_mem, op_ins}));
64 auto [fun_mem, ins] = fun->var(0_n)->projs<2>();
65 auto cont = fun->var(1);
66
67 auto [obr, obs, obT] = Axm::isa<buffer::Buf>(result_ty->proj(1))->args<3>();
68 auto [a_mem, out_buf] = buffer::op_alloc(obr, obs, obT, fun_mem)->projs<2>();
69 const Def* acc = w.tuple({a_mem, out_buf});
70 auto current_mut = fun;
71
72 DefVec iters; // raw i64 loop counters
73 iters.reserve(rn);
74 for (u64 d = 0; d < rn; ++d) {
75 auto bound = w.call<core::bitcast>(w.type_i64(), s_out->proj(rn, d));
76 auto [body, for_call] = counting_for(bound, acc, cont, w.sym(name + "_" + std::to_string(d)));
77 auto [iter, new_acc, yield] = body->vars<3>();
78 cont = yield;
79 iters.push_back(iter);
80 acc = new_acc;
81 current_mut->set(true, for_call);
82 current_mut = body;
83 }
84 auto [loop_mem, loop_buf] = acc->projs<2>();
85
86 std::pair<const Def*, const Def*> el = compute(iters, ins, loop_mem);
87 auto [el_mem, element] = el;
88
89 DefVec wcoords(rn);
90 for (u64 d = 0; d < rn; ++d)
91 wcoords[d] = w.call(core::conv::u, s_out->proj(rn, d), iters[d]);
92 auto [wr_mem, wr_buf]
93 = buffer::op_write(obr, obs, obT, el_mem, loop_buf, fold_index(s_out, w.tuple(wcoords)), element)->projs<2>();
94 current_mut->app(true, cont, w.tuple({wr_mem, loop_buf}));
95 return call;
96}

110const Def* LowerAff::lower_buffer_constant(const App* app) {
111 // `%buffer.constant (r, s, T) (mem, val)` fills every element with `val`. Emit a fill loop (via the same
112 // pointwise scaffold as pad/concat) so the store never materializes as one giant literal array. A
113 // non-literal rank has no static loop nest, so leave it for `%buffer.lower_ptr`'s monolithic fallback.
114 auto [r, s, T] = app->callee()->as<App>()->args<3>();
115 auto s_out = rewrite(s);
116 auto rn = Lit::isa<u64>(rewrite(r));
117 if (!rn) return RWPhase::rewrite_imm_App(app);
118
119 auto& w = new_world();
120 auto [mem, val] = rewrite(app->arg())->projs<2>();
121 auto result_ty = rewrite(app->type()); // [%mem.M 0, %buffer.Buf (r, s, T)]
122 // `compute` ignores the loop counters and writes the (loop-invariant) scalar `ins` everywhere.
123 return build_pointwise(w, result_ty, mem, val, s_out, *rn, "constant_fill",
124 [](const DefVec&, const Def* ins, const Def* m) -> std::pair<const Def*, const Def*> {
125 return {m, ins};
126 });

154 // Builds `%affine.map @(m, n) @(sin, sout) f idxs mem`. The map is mem-threaded (its divisions consume mem),
155 // and this phase threads real memory, so the caller passes the current mem and receives `(mem', coords)`.

157 const Def* mem) {

204 // Reduction loops; accumulator `{mem, element}`.

207 DefVec red_iters;
208 red_iters.reserve(rr);

219 }
220 auto [red_mem, element_acc] = acc->projs<2>();
221
222 DefVec iters_v = out_iters;
223 iters_v.insert(iters_v.end(), red_iters.begin(), red_iters.end());
224 auto iters = w.tuple(iters_v);
225
226 // Read one element from each input at its affine coordinates, threading memory.
227 auto cur = red_mem;
228 DefVec input_elements(nis_nat);
229 for (u64 i = 0; i < nis_nat; ++i) {
230 auto in_buf = new_inputs->proj(nis_nat, i);

239 }
240
241 // The combiner is mem-threaded (`Fn [mem, To, «nis; Tis»] → [mem, To]`): call it directly, its result feeds `cont`.
242 current_mut->app(true, comb, w.tuple({w.tuple({cur, element_acc, w.tuple(input_elements)}), cont}));
243
244 return call;

293const Def* LowerAff::lower_pad(const App* app) {
294 auto& w = new_world();
295 auto c = rewrite(app->callee())->as<App>();
296
297 // callee: pad {T, r} [s_in, s_out, mode, lo, hi]. The shapes are the logical ones; buffer reads and
298 // writes fold size-1 axes (the `Buf` handles are normalized), while the loops cover all logical dims.
299 auto [Tr, params] = c->uncurry_args<2>();
300 auto [s_in, s_out, mode, lo, hi] = params->projs<5>();
301 auto [op_mem, input, value] = rewrite(app->arg())->projs<3>();
302 auto result_ty = rewrite(app->type()); // [%mem.M 0, %buffer.Buf (r, s_out, T)]
303
304 auto r_l = Lit::isa<u64>(Tr->proj(2, 1));
305 auto mode_l = Lit::isa<u64>(mode);
306 if (!r_l || !mode_l) {
307 WLOG("{} doesn't have a lowering-time known rank/mode", app);
308 return RWPhase::rewrite_imm_App(app);
309 }
310 auto rn = *r_l;
311 auto mode_nat = *mode_l;
312 auto i64 = w.type_i64();
313
314 // select(cond, t, f) == `(f, t)#cond` (cf. %core.select); cond : Bool.
315 auto sel = [&](const Def* cond, const Def* t, const Def* f) { return w.extract(w.tuple({f, t}), cond); };
316
317 auto compute = [&](const DefVec& iters, const Def* ins, const Def* mem) -> std::pair<const Def*, const Def*> {
318 auto [in_buf, fill] = ins->projs<2>();
319 auto [ibr, ibs, ibT] = Axm::isa<buffer::Buf>(in_buf->type())->args<3>();
320 DefVec clamped(rn); // per-axis read index, kept in range, as `Idx (s_in#d)`
321 DefVec valid; // per-axis in-bounds flag (constant mode only)
322 for (u64 d = 0; d < rn; ++d) {
323 auto lo_d = w.call<core::bitcast>(i64, lo->proj(rn, d));
324 auto sin_d = w.call<core::bitcast>(i64, s_in->proj(rn, d));
325 auto in_d = w.call(core::wrap::sub, core::Mode::none, Defs{iters[d], lo_d}); // o#d − lo#d
326 const Def* idx_i64;
327 if (mode_nat == 0) { // constant: a single unsigned `<` covers both bounds (underflow wraps high)
328 auto v_d = w.call(core::icmp::ul, w.tuple({in_d, sin_d}));
329 valid.push_back(v_d);
330 idx_i64 = sel(v_d, in_d, w.lit_i64(0));
331 } else { // replicate: clamp the read to the nearest edge [0, s_in#d − 1]
332 auto sin_m1 = w.call(core::wrap::sub, core::Mode::none, Defs{sin_d, w.lit_i64(1)});
333 idx_i64 = w.call(core::extrema::smax,
334 w.tuple({w.lit_i64(0), w.call(core::extrema::smin, w.tuple({in_d, sin_m1}))}));
335 }
336 clamped[d] = w.call(core::conv::u, s_in->proj(rn, d), idx_i64);
337 }
338 auto [rd_mem, elem] = buffer::op_read(ibr, ibs, ibT, mem, in_buf, fold_index(s_in, w.tuple(clamped)))->projs<2>();
339 if (mode_nat != 0) return {rd_mem, elem}; // replicate: always a (clamped) read
340 auto all_valid = valid.empty() ? w.lit_tt() : valid[0];
341 for (u64 d = 1; d < valid.size(); ++d)
342 all_valid = w.call(core::bit2::and_, w.lit_nat(2), w.tuple({all_valid, valid[d]}));
343 return {rd_mem, sel(all_valid, elem, fill)}; // constant: fill out-of-region cells with `value`
344 };
345
346 return build_pointwise(w, result_ty, op_mem, w.tuple({input, value}), s_out, rn, "pad", compute);
347}

349const Def* LowerAff::lower_concat(const App* app) {
350 auto& w = new_world();
351 auto c = rewrite(app->callee())->as<App>();
352
353 // callee: concat {T, nis, r} [ax] {Sis} [s_out]. The shapes are the logical ones; buffer reads and
354 // writes fold size-1 axes (the `Buf` handles are normalized), while the loops cover all logical dims.
355 auto [TnisR, ax, Sis, s_out] = c->uncurry_args<4>();
356 auto [T, nis, r] = TnisR->projs<3>();
357 auto [op_mem, op_is] = rewrite(app->arg())->projs<2>();
358 auto result_ty = rewrite(app->type()); // [%mem.M 0, %buffer.Buf (r, s_out, T)]
359
360 auto nis_l = Lit::isa<u64>(nis);
361 auto r_l = Lit::isa<u64>(r);
362 auto ax_l = Lit::isa<u64>(ax);
363 if (!nis_l || !r_l || !ax_l) {
364 WLOG("{} doesn't have lowering-time known nis/rank/axis", app);
365 return RWPhase::rewrite_imm_App(app);
366 }
367 auto nisn = *nis_l, rn = *r_l, axn = *ax_l;
368
369 // Prefix offsets along `ax`: off#i = Σ_{j<i} Sis#i#ax (literal extents required).
370 DefVec off(nisn);
371 Vector<u64> ext(nisn);
372 u64 acc_off = 0;
373 for (u64 i = 0; i < nisn; ++i) {
374 off[i] = w.lit_i64(acc_off);
375 auto ei = Lit::isa<u64>(Sis->proj(nisn, i)->proj(rn, axn));
376 if (!ei) {
377 WLOG("{} input {} has a non-literal extent along the concat axis", app, i);
378 return RWPhase::rewrite_imm_App(app);
379 }
380 ext[i] = *ei;
381 acc_off += *ei;
382 }
383
384 auto sel = [&](const Def* cond, const Def* t, const Def* f) { return w.extract(w.tuple({f, t}), cond); };
385
386 auto compute = [&](const DefVec& iters, const Def* ins, const Def* mem) -> std::pair<const Def*, const Def*> {
387 auto o_ax = iters[axn];
388 const Def* cur = mem;
389 // Read input `i` at `iters`, but with the `ax` coordinate shifted by off#i and clamped into input `i`.
390 auto read_i = [&](u64 i) -> const Def* {
391 auto in_buf = ins->proj(nisn, i);
392 auto [ibr, ibs, ibT] = Axm::isa<buffer::Buf>(in_buf->type())->args<3>();
393 auto Sis_i = Sis->proj(nisn, i);
394 auto e_i_m1 = w.lit_i64(ext[i] - 1);
395 auto loc = w.call(core::wrap::sub, core::Mode::none, Defs{o_ax, off[i]});
396 auto clamp = w.call(core::extrema::smax,
397 w.tuple({w.lit_i64(0), w.call(core::extrema::smin, w.tuple({loc, e_i_m1}))}));
398 DefVec coords(rn);
399 for (u64 d = 0; d < rn; ++d) {
400 auto idx_i64 = (d == axn) ? clamp : iters[d];
401 coords[d] = w.call(core::conv::u, Sis_i->proj(rn, d), idx_i64);
402 }
403 auto [rd_mem, rd_val] = buffer::op_read(ibr, ibs, ibT, cur, in_buf, fold_index(Sis_i, w.tuple(coords)))->projs<2>();
404 cur = rd_mem;
405 return rd_val;
406 };
407 // Select chain: the highest `i` with off#i ≤ o_ax owns the cell (offsets increase, later wins).
408 auto result = read_i(0);
409 for (u64 i = 1; i < nisn; ++i) {
410 auto cond = w.call(core::icmp::uge, w.tuple({o_ax, off[i]}));
411 result = sel(cond, read_i(i), result);
412 }
413 return {cur, result};
414 };
415
416 return build_pointwise(w, result_ty, op_mem, op_is, s_out, rn, "concat", compute);
417}
