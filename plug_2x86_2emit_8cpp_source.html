81/// Width of the register holding a value of @p type; a spill slot is as wide as its value.
82nat_t width(const Def* type) {
83 if (auto w = math::isa_f(type)) return *w;
84 if (auto spill = Axm::isa<mach::Spill>(type)) return width(spill->arg());
85 return std::bit_ceil(std::max<nat_t>(Idx::expect_bitwidth(type, "an integer"), 8));

154std::string Emitter::imm(const Def* lit, nat_t w) const {
155 auto l = *Lit::isa(lit);
156 auto s = std::bit_cast<int64_t>(l << (64 - w)) >> (64 - w);
157 if (s < INT32_MIN || s > INT32_MAX)
158 lit->blame("{} backend: literal `{}` does not fit an immediate", backend(), lit).bail();
159 return std::format("${}", s);

167std::string Emitter::addr(const Def* a) const {
168 auto ix = Axm::isa<x86::index>(a);
169 auto [disp, base] = Axm::as<x86::addr>(ix ? ix->decurry()->arg() : a)->args<2>();
170 auto d = std::bit_cast<int32_t>(u32(nat(disp)));
171 if (auto l = Lit::isa(base); l && *l != 0) base->blame("{} backend: literal base `{}`", backend(), base).bail();
172 auto has_base = !Lit::isa(base);
173 if (!has_base && !ix) return std::to_string(d);
174
175 auto res = d != 0 ? std::to_string(d) : ""s;
176 res += "(";
177 if (has_base) res += name(use_reg(base), 64);
178 if (ix) {
179 auto [index, scale] = ix->args<2>();
180 res += std::format(",{},{}", name(use_reg(index), 64), 1 << nat(scale));
181 }
182 return res + ")";
183}

204void Emitter::mov_imm(Reg dst, const Def* lit) {
205 auto w = width(lit->type());
206 auto l = *Lit::isa(lit);
207 if (dst.cls == Target::Fpr) {
208 if (w != 32 && w != 64) lit->blame("{} backend: `math.F` of {} bits is not supported", backend(), w).bail();
209 return line("mov{} {}, {}", fsz(w), pool(l, w == 64 ? 0 : l), xmm(dst.num));
210 }
211 // `xorl` clobbers the flags, so only use it while none are pending.
212 if (l == 0 && flags_clobbered()) return line("xorl {}, {}", name(dst.num, 32), name(dst.num, 32));
213 if (w < 64) return line("movl ${}, {}", l, name(dst.num, 32));
214 auto s = std::bit_cast<int64_t>(l);
215 if (s < INT32_MIN || s > INT32_MAX)
216 line("movabsq ${}, {}", s, name(dst.num, 64));
217 else
218 line("movq ${}, {}", s, name(dst.num, 64));
219}

221void Emitter::insn(const Def* res, const Def* f, const Def* x) {
222 auto axm = std::get<0>(Axm::get(f));
223 auto id = axm->base();
224 auto keep = id == Annex::Base<mov> || id == Annex::Base<lea> || id == Annex::Base<movzx> || id == Annex::Base<movsx>
225 || id == Annex::Base<cc> || id == Annex::Base<cmovcc> || id == Annex::Base<fmov> || id == Annex::Base<fadd>
226 || id == Annex::Base<fsub> || id == Annex::Base<fmul> || id == Annex::Base<fdiv>
227 || id == Annex::Base<fsqrt> || id == Annex::Base<fabs> || id == Annex::Base<i2f> || id == Annex::Base<f2i>
228 || id == Annex::Base<f2u> || id == Annex::Base<f2f>;
229 if (!keep) clobber_flags();
230
231 auto binary = [&](std::string_view mnemonic) {
232 auto [a, b] = x->projs<2>();
233 auto w = width(res->type());
234 auto dst = def_reg(res);
235 assert(use_reg(a) == dst);
236 line("{}{} {}, {}", mnemonic, suffix(w), src(b, w), name(dst, w));
237 };
238 auto shift = [&](std::string_view mnemonic) {
239 auto [a, b] = x->projs<2>();
240 auto w = width(res->type());
241 auto dst = def_reg(res);
242 assert(use_reg(a) == dst);
243 if (auto l = Lit::isa(b))
244 line("{}{} ${}, {}", mnemonic, suffix(w), *l % w, name(dst, w));
245 else
246 line("{}{} %cl, {}", mnemonic, suffix(w), name(dst, w));
247 };
248 auto divide = [&](bool sign) {
249 auto [a, b] = x->projs<2>();
250 auto w = width(a->type());
251 assert(use_reg(a) == encoding(reg::rax));
252 if (!sign)
253 line("xorl %edx, %edx");
254 else
255 line("{}", w == 64 ? "cqto" : w == 32 ? "cltd" : "cwtd");
256 line("{}div{} {}", sign ? "i" : "", suffix(w), name(use_reg(b), w));
257 };
258 auto extend = [&](char kind) {
259 auto ws = width(x->type());
260 auto wd = width(res->type());
261 auto rs = use_reg(x);
262 auto rd = def_reg(res);
263 if (ws < wd && !(kind == 'z' && ws == 32))
264 line("mov{}{}{} {}, {}", kind, suffix(ws), suffix(wd), name(rs, ws), name(rd, wd));
265 else if (ws < wd)
266 line("movl {}, {}", name(rs, 32), name(rd, 32));
267 else if (rs != rd)
268 line("movq {}, {}", name(rs, 64), name(rd, 64));
269 };
270
271 // clang-format off
272 if (auto mn = alu(id); !mn.empty()) binary(mn);
273 else if (id == Annex::Base<div>) divide(false);
274 else if (id == Annex::Base<idiv>) divide(true);
275 else if (id == Annex::Base<shl>) shift("shl");
276 else if (id == Annex::Base<shr>) shift("shr");
277 else if (id == Annex::Base<sar>) shift("sar");
278 else if (id == Annex::Base<movzx>) extend('z');
279 else if (id == Annex::Base<movsx>) extend('s');
280 // clang-format on
281 else if (id == Annex::Base<mov>) {
282 if (!Lit::isa(x)) x->blame("{} backend: `x86.mov` of `{}`, which is no literal", backend(), x).bail();
283 mov_imm(Reg{def_reg(res), Target::Gpr}, x);
284 } else if (id == Annex::Base<lea>) {
285 line("leaq {}, {}", addr(x), name(def_reg(res), 64));
286 } else if (id == Annex::Base<trunc>) {
287 auto w = width(res->type());
288 auto rs = use_reg(x);
289 auto rd = def_reg(res);
290 auto n = *Idx::isa_lit(res->type());
291 auto full = w == 64 ? 0 : 1_n << w;
292 if (n != 0 && !std::has_single_bit(n)) {
293 // Nothing reserves rax, rcx, and rdx for the division, so save them around it.
294 auto save = fe::Vector<size_t>();
295 for (auto r : {encoding(reg::rax), encoding(reg::rcx), encoding(reg::rdx)})
296 if (r != rd) save.emplace_back(r);
297 for (auto r : save)
298 line("pushq {}", name(r, 64));
299 auto wx = width(x->type());
300 auto ws = std::max<nat_t>(wx, 32);
301 if (wx < 32)
302 line("movz{}l {}, %eax", suffix(wx), name(rs, wx));
303 else
304 line("mov{} {}, {}", suffix(ws), name(rs, ws), name(encoding(reg::rax), ws));
305 line("xorl %edx, %edx");
306 line("mov{} ${}, {}", suffix(ws), n, name(encoding(reg::rcx), ws));
307 line("div{} {}", suffix(ws), name(encoding(reg::rcx), ws));
308 if (rd != encoding(reg::rdx)) line("movq %rdx, {}", name(rd, 64));
309 for (auto r : save | std::views::reverse)
310 line("popq {}", name(r, 64));
311 } else {
312 if (rs != rd) line("movq {}, {}", name(rs, 64), name(rd, 64));
313 if (std::has_single_bit(n) && n != full) line("and{} ${}, {}", suffix(w), n - 1, name(rd, w));
314 }
315 } else if (id == Annex::Base<cmp> || id == Annex::Base<test>) {
316 auto [a, b] = x->projs<2>();
317 auto w = width(a->type());
318 line("{}{} {}, {}", id == Annex::Base<cmp> ? "cmp" : "test", suffix(w), src(b, w), name(use_reg(a), w));
319 set_flags(res);
320 } else if (id == Annex::Base<cc>) {
321 use_flags(x);
322 // A condition that only a branch consumes has no register.
323 if (auto r = isa_reg(res)) line("set{} {}", ccs[axm->sub()], name(*r, 8));
324 } else if (id == Annex::Base<cmovcc>) {
325 auto [fl, a, b] = x->projs<3>();
326 auto w = width(res->type());
327 auto dst = def_reg(res);
328 assert(use_reg(a) == dst);
329 use_flags(fl);
330 line("cmov{} {}, {}", ccs[axm->sub()], name(use_reg(b), w), name(dst, w));
331 } else if (id == Annex::Base<fadd> || id == Annex::Base<fsub> || id == Annex::Base<fmul>
332 || id == Annex::Base<fdiv>) {
333 auto [a, b] = x->projs<2>();
334 auto w = width(res->type());
335 auto dst = def_reg(res);
336 assert(use_reg(a) == dst);
337 line("{}{} {}, {}", farith(id), fsz(w), xmm(use_reg(b)), xmm(dst));
338 } else if (id == Annex::Base<fsqrt> || id == Annex::Base<i2f> || id == Annex::Base<f2i> || id == Annex::Base<f2u>
339 || id == Annex::Base<f2f>) {
340 auto ws = width(x->type());
341 auto r = use_reg(x);
342 unary(id, math::isa_f(x->type()) ? xmm(r) : std::string(name(r, ws)), ws, res);
343 } else if (id == Annex::Base<fabs>) {
344 auto w = width(res->type());
345 auto dst = def_reg(res);
346 assert(use_reg(x) == dst);
347 line("and{} {}, {}", w == 64 ? "pd" : "ps", abs_mask(w), xmm(dst));
348 } else if (id == Annex::Base<fmov>) {
349 if (!Lit::isa(x)) x->blame("{} backend: `x86.fmov` of `{}`, which is no literal", backend(), x).bail();
350 mov_imm(Reg{def_reg(res), Target::Fpr}, x);
351 } else if (id == Annex::Base<fcmp>) {
352 auto [a, b] = x->projs<2>();
353 ucomi(axm->sub(), width(a->type()), xmm(use_reg(b)), use_reg(a), def_reg(res));
354 } else if (id == Annex::Base<fsel>) {
355 auto [c, a, b] = x->projs<3>();
356 auto dst = def_reg(res);
357 assert(use_reg(a) == dst);
358 auto l = local("fsel");
359 line("testb {}, {}", name(use_reg(c), 8), name(use_reg(c), 8));
360 line("je {}", l);
361 line("movaps {}, {}", xmm(use_reg(b)), xmm(dst));
362 label_here(l);
363 } else if (id == Annex::Base<u2f>) {
364 auto ws = width(x->type());
365 auto wr = width(res->type());
366 if (ws == 64) f->blame("{} backend: `x86.u2f` from `i64` is not supported yet", backend()).bail();
367 auto src = name(use_reg(x), 32);
368 auto dst = xmm(def_reg(res));
369 auto l = local("u2f");
370 // Exact in `f64`, so an `f32` result rounds only once.
371 line("cvtsi2sdl {}, {}", src, dst);
372 line("testl {}, {}", src, src);
373 line("jns {}", l);
374 line("addsd {}, {}", two32(), dst);
375 label_here(l);
376 if (wr == 32) line("cvtsd2ss {}, {}", dst, dst);
377 } else {
378 f->blame("{} backend: cannot emit `{}`", backend(), axm).bail();
379 }
380}

382void Emitter::ucomi(size_t sub, nat_t w, std::string_view src, size_t a, size_t dst) {
383 auto [cc, unordered] = fccs[sub];
384 if (cc.empty()) return line("movl ${}, {}", sub == Fcmp_T ? 1 : 0, name(dst, 32));
385 line("ucomi{} {}, {}", fsz(w), src, xmm(a));
386 line("set{} {}", cc, name(dst, 8));
387 if (!unordered) return;
388 auto l = local("fcmp");
389 line("jnp {}", l);
390 line("movl ${}, {}", *unordered ? 1 : 0, name(dst, 32));
391 label_here(l);
392}

394void Emitter::unary(flags_t id, std::string_view src, nat_t ws, const Def* res) {
395 auto wr = width(res->type());
396 auto dst = def_reg(res);
397 if (id == Annex::Base<fsqrt>) return line("sqrt{} {}, {}", fsz(wr), src, xmm(dst));
398 if (id == Annex::Base<i2f>) return line("cvtsi2{}{} {}, {}", fsz(wr), ws == 64 ? "q" : "l", src, xmm(dst));
399 if (id == Annex::Base<f2i>) return line("cvtt{}2si{} {}, {}", fsz(ws), wr == 64 ? "q" : "l", src, name(dst, wr));
400 if (id == Annex::Base<f2f>) return line("cvt{}2{} {}, {}", fsz(ws), fsz(wr), src, xmm(dst));
401 if (wr == 64) res->blame("{} backend: `x86.f2u` to `i64` is not supported yet", backend()).bail();
402 line("cvtt{}2siq {}, {}", fsz(ws), src, name(dst, 64));
403 line("movl {}, {}", name(dst, 32), name(dst, 32));
404}

406void Emitter::step(const App* app) {
407 if (Axm::isa<mach::sched>(app)) return insn(app->proj(2, 1), Target::sched_insn(app), app->arg());
408
409 if (!Axm::isa<load>(app) && !Axm::isa<store>(app) && !Axm::isa<fload>(app) && !Axm::isa<fstore>(app))
410 clobber_flags();
411
412 if (auto op = Axm::isa<rm>(app)) {
413 auto [_, a, p] = app->args<3>();
414 auto w = width(a->type());
415 auto dst = def_reg(app->proj(2, 1));
416 auto insn = op->decurry()->arg()->as<App>()->axm();
417 if (insn->base() == Annex::Base<fcmp>) return ucomi(insn->sub(), w, addr(p), use_reg(a), dst);
418 assert(use_reg(a) == dst);
419 if (math::isa_f(a->type()))
420 line("{}{} {}, {}", farith(insn->base()), fsz(w), addr(p), xmm(dst));
421 else
422 line("{}{} {}, {}", alu(insn->flags()), suffix(w), addr(p), name(dst, w));
423 return;
424 }
425
426 if (Axm::isa<rm1>(app)) {
427 auto [_, p] = app->args<2>();
428 auto f = app->callee()->as<App>()->arg();
429 return unary(insn::op(app), addr(p), width(f->type()->as<Pi>()->dom()), app->proj(2, 1));
430 }
431
432 if (auto op = Axm::isa<mr>(app)) {
433 auto [_, p, b] = app->args<3>();
434 auto w = width(b->type());
435 line("{}{} {}, {}", alu(op->decurry()->arg()->as<App>()->axm()->flags()), suffix(w), src(b, w), addr(p));
436 return;
437 }
438
439 if (Axm::isa<load>(app)) {
440 auto [_, p] = app->args<2>();
441 auto res = app->proj(2, 1);
442 auto w = width(res->type());
443 line("mov{} {}, {}", suffix(w), addr(p), name(def_reg(res), w));
444 return;
445 }
446
447 if (Axm::isa<fload>(app)) {
448 auto [_, p] = app->args<2>();
449 auto res = app->proj(2, 1);
450 auto w = width(res->type());
451 line("mov{} {}, {}", fsz(w), addr(p), xmm(def_reg(res)));
452 return;
453 }
454
455 if (Axm::isa<fstore>(app)) {
456 auto [_, p, v] = app->args<3>();
457 auto w = width(v->type());
458 line("mov{} {}, {}", fsz(w), xmm(use_reg(v)), addr(p));
459 return;
460 }
461
462 auto [_, p, v] = app->args<3>();
463 auto w = width(v->type());

467void Emitter::spill(Src src, const Def* type, size_t i) {
468 if (src.reg.cls != Target::Fpr) {
469 auto s = src.imm ? imm(src.imm, 64) : std::string(name(src.reg.num, 64));
470 return line("movq {}, {}", s, slot(i));
471 }
472 auto w = width(type);
473 if (src.imm) {
474 auto l = *Lit::isa(src.imm);
475 line("movl ${}, {}", u32(l), slot(i));
476 if (w == 64) line("movl ${}, {}", u32(l >> 32), slot(i, 4));
477 } else {
478 line("mov{} {}, {}", fsz(w), xmm(src.reg.num), slot(i));
479 }
480}

482void Emitter::reload(Reg dst, const Def* type, size_t i) {
483 if (dst.cls == Target::Fpr)
484 line("mov{} {}, {}", fsz(width(type)), slot(i), xmm(dst.num));
485 else
486 line("movq {}, {}", slot(i), name(dst.num, 64));
487}

489void Emitter::epilogue() {
490 if (saved(Target::Gpr).empty()) {
491 line("leave");
492 } else {
493 line("leaq -{}(%rbp), %rsp", 8 * saved(Target::Gpr).size());
494 for (auto r : saved(Target::Gpr) | std::views::reverse)
495 line("popq {}", name(r, 64));
496 line("popq %rbp");
497 }
498 line("ret");
499}
