81/// Width of the register holding a value of @p type; a spill slot is as wide as its value.
82nat_t width(const Def* type) {

84 if (auto spill = Axm::isa<Spill>(type)) type = type->world().type_idx(spill->arg());

213void Emitter::collect(const Def* def, DefSet& done) {
214 if (def->isa_mut() || def->is_closed() || is_mem(def) || !done.emplace(def).second) return;
215 if (Axm::isa<x86::in>(def) || Axm::isa<x86::fin>(def)) {
216 regs_.emplace(strip(def), in_reg(def));
217 return;

219 for (auto op : def->ops())

240std::string Emitter::imm(const Def* lit, nat_t w) const {
241 auto l = *Lit::isa(lit);
242 auto s = std::bit_cast<int64_t>(l << (64 - w)) >> (64 - w);
243 if (s < INT32_MIN || s > INT32_MAX) lit->blame(MIM_X86_BE "literal `{}` does not fit an immediate", lit).bail();
244 return std::format("${}", s);

252std::string Emitter::addr(const Def* a) const {
253 auto ix = Axm::isa<x86::index>(a);
254 auto [disp, base] = Axm::as<x86::addr>(ix ? ix->decurry()->arg() : a)->args<2>();
255 auto d = std::bit_cast<int32_t>(u32(nat(disp)));
256 if (auto l = Lit::isa(base); l && *l != 0) base->blame(MIM_X86_BE "literal base `{}`", base).bail();
257 auto has_base = !Lit::isa(base);
258 if (!has_base && !ix) return std::to_string(d);
259
260 auto res = d != 0 ? std::to_string(d) : ""s;
261 res += "(";
262 if (has_base) res += name(use_reg(base), 64);
263 if (ix) {
264 auto [index, scale] = ix->args<2>();
265 res += std::format(",{},{}", name(use_reg(index), 64), 1 << nat(scale));
266 }
267 return res + ")";
268}

285void Emitter::mov_imm(Reg dst, const Def* lit) {
286 auto w = width(lit->type());
287 auto l = *Lit::isa(lit);
288 if (dst.cls == Liveness::Fpr) {
289 if (w != 32 && w != 64) lit->blame(MIM_X86_BE "`math.F` of {} bits is not supported", w).bail();
290 return line("mov{} {}, {}", fsz(w), pool(l, w == 64 ? 0 : l), xmm(dst.num));
291 }
292 // `xorl` clobbers the flags, so only use it while none are pending.
293 if (l == 0 && flags_ == nullptr) return line("xorl {}, {}", name(dst.num, 32), name(dst.num, 32));
294 if (w < 64) return line("movl ${}, {}", l, name(dst.num, 32));
295 auto s = std::bit_cast<int64_t>(l);
296 if (s < INT32_MIN || s > INT32_MAX)
297 line("movabsq ${}, {}", s, name(dst.num, 64));
298 else
299 line("movq ${}, {}", s, name(dst.num, 64));
300}

311void Emitter::insn(const Def* res, const Def* f, const Def* x) {
312 auto axm = std::get<0>(Axm::get(f));
313 auto id = axm->base();
314 auto keep = id == Annex::Base<mov> || id == Annex::Base<x86::spill> || id == Annex::Base<reload>
315 || id == Annex::Base<lea> || id == Annex::Base<movzx> || id == Annex::Base<movsx> || id == Annex::Base<cc>
316 || id == Annex::Base<cmovcc> || id == Annex::Base<fmov> || id == Annex::Base<fspill>
317 || id == Annex::Base<freload> || id == Annex::Base<fadd> || id == Annex::Base<fsub>
318 || id == Annex::Base<fmul> || id == Annex::Base<fdiv> || id == Annex::Base<fsqrt>
319 || id == Annex::Base<fabs> || id == Annex::Base<i2f> || id == Annex::Base<f2i> || id == Annex::Base<f2u>
320 || id == Annex::Base<f2f>;
321 if (!keep) flags_ = nullptr;
322
323 auto binary = [&](std::string_view mnemonic) {
324 auto [a, b] = x->projs<2>();
325 auto w = width(res->type());
326 auto dst = def_reg(res);
327 assert(use_reg(a) == dst);
328 line("{}{} {}, {}", mnemonic, suffix(w), src(b, w), name(dst, w));
329 };
330 auto shift = [&](std::string_view mnemonic) {
331 auto [a, b] = x->projs<2>();
332 auto w = width(res->type());
333 auto dst = def_reg(res);
334 assert(use_reg(a) == dst);
335 if (auto l = Lit::isa(b))
336 line("{}{} ${}, {}", mnemonic, suffix(w), *l % w, name(dst, w));
337 else
338 line("{}{} %cl, {}", mnemonic, suffix(w), name(dst, w));
339 };
340 auto divide = [&](bool sign) {
341 auto [a, b] = x->projs<2>();
342 auto w = width(a->type());
343 assert(use_reg(a) == encoding(reg::rax));
344 if (!sign)
345 line("xorl %edx, %edx");
346 else
347 line("{}", w == 64 ? "cqto" : w == 32 ? "cltd" : "cwtd");
348 line("{}div{} {}", sign ? "i" : "", suffix(w), name(use_reg(b), w));
349 };
350 auto extend = [&](char kind) {
351 auto ws = width(x->type());
352 auto wd = width(res->type());
353 auto rs = use_reg(x);
354 auto rd = def_reg(res);
355 if (ws < wd && !(kind == 'z' && ws == 32))
356 line("mov{}{}{} {}, {}", kind, suffix(ws), suffix(wd), name(rs, ws), name(rd, wd));
357 else if (ws < wd)
358 line("movl {}, {}", name(rs, 32), name(rd, 32));
359 else if (rs != rd)
360 line("movq {}, {}", name(rs, 64), name(rd, 64));
361 };
362
363 // clang-format off
364 if (auto mn = alu(id); !mn.empty()) binary(mn);
365 else if (id == Annex::Base<div>) divide(false);
366 else if (id == Annex::Base<idiv>) divide(true);
367 else if (id == Annex::Base<shl>) shift("shl");
368 else if (id == Annex::Base<shr>) shift("shr");
369 else if (id == Annex::Base<sar>) shift("sar");
370 else if (id == Annex::Base<movzx>) extend('z');
371 else if (id == Annex::Base<movsx>) extend('s');
372 // clang-format on
373 else if (id == Annex::Base<mov>) {
374 if (!Lit::isa(x)) x->blame(MIM_X86_BE "`x86.mov` of `{}`, which is no literal", x).bail();
375 mov_imm(Reg{def_reg(res), Liveness::Gpr}, x);
376 } else if (id == Annex::Base<x86::spill>) {
377 if (Lit::isa(x))
378 line("movq {}, {}", imm(x, 64), slot(slot_of(res)));
379 else
380 line("movq {}, {}", name(use_reg(x), 64), slot(slot_of(res)));
381 } else if (id == Annex::Base<reload>) {
382 line("movq {}, {}", slot(slot_of(x)), name(def_reg(res), 64));
383 } else if (id == Annex::Base<lea>) {
384 line("leaq {}, {}", addr(x), name(def_reg(res), 64));
385 } else if (id == Annex::Base<trunc>) {
386 auto w = width(res->type());
387 auto rs = use_reg(x);
388 auto rd = def_reg(res);
389 auto n = *Idx::isa_lit(res->type());
390 auto full = w == 64 ? 0 : 1_n << w;
391 if (n != 0 && !std::has_single_bit(n)) {
392 // Nothing reserves rax, rcx, and rdx for the division, so save them around it.
393 auto save = fe::Vector<size_t>();
394 for (auto r : {encoding(reg::rax), encoding(reg::rcx), encoding(reg::rdx)})
395 if (r != rd) save.emplace_back(r);
396 for (auto r : save)
397 line("pushq {}", name(r, 64));
398 auto wx = width(x->type());
399 auto ws = std::max<nat_t>(wx, 32);
400 if (wx < 32)
401 line("movz{}l {}, %eax", suffix(wx), name(rs, wx));
402 else
403 line("mov{} {}, {}", suffix(ws), name(rs, ws), name(encoding(reg::rax), ws));
404 line("xorl %edx, %edx");
405 line("mov{} ${}, {}", suffix(ws), n, name(encoding(reg::rcx), ws));
406 line("div{} {}", suffix(ws), name(encoding(reg::rcx), ws));
407 if (rd != encoding(reg::rdx)) line("movq %rdx, {}", name(rd, 64));
408 for (auto r : save | std::views::reverse)
409 line("popq {}", name(r, 64));
410 } else {
411 if (rs != rd) line("movq {}, {}", name(rs, 64), name(rd, 64));
412 if (std::has_single_bit(n) && n != full) line("and{} ${}, {}", suffix(w), n - 1, name(rd, w));
413 }
414 } else if (id == Annex::Base<cmp> || id == Annex::Base<test>) {
415 auto [a, b] = x->projs<2>();
416 auto w = width(a->type());
417 line("{}{} {}, {}", id == Annex::Base<cmp> ? "cmp" : "test", suffix(w), src(b, w), name(use_reg(a), w));
418 flags_ = res;
419 } else if (id == Annex::Base<cc>) {
420 use_flags(x);
421 // A condition that only a branch consumes has no register.
422 if (auto r = fe::lookup(regs_, res)) line("set{} {}", ccs[axm->sub()], name(*r, 8));
423 } else if (id == Annex::Base<cmovcc>) {
424 auto [fl, a, b] = x->projs<3>();
425 auto w = width(res->type());
426 auto dst = def_reg(res);
427 assert(use_reg(a) == dst);
428 use_flags(fl);
429 line("cmov{} {}, {}", ccs[axm->sub()], name(use_reg(b), w), name(dst, w));
430 } else if (id == Annex::Base<fadd> || id == Annex::Base<fsub> || id == Annex::Base<fmul>
431 || id == Annex::Base<fdiv>) {
432 auto [a, b] = x->projs<2>();
433 auto w = width(res->type());
434 auto dst = def_reg(res);
435 assert(use_reg(a) == dst);
436 line("{}{} {}, {}", farith(id), fsz(w), xmm(use_reg(b)), xmm(dst));
437 } else if (id == Annex::Base<fsqrt> || id == Annex::Base<i2f> || id == Annex::Base<f2i> || id == Annex::Base<f2u>
438 || id == Annex::Base<f2f>) {
439 auto ws = width(x->type());
440 auto r = use_reg(x);
441 unary(id, math::isa_f(x->type()) ? xmm(r) : std::string(name(r, ws)), ws, res);
442 } else if (id == Annex::Base<fabs>) {
443 auto w = width(res->type());
444 auto dst = def_reg(res);
445 assert(use_reg(x) == dst);
446 line("and{} {}, {}", w == 64 ? "pd" : "ps", abs_mask(w), xmm(dst));
447 } else if (id == Annex::Base<fmov>) {
448 if (!Lit::isa(x)) x->blame(MIM_X86_BE "`x86.fmov` of `{}`, which is no literal", x).bail();
449 mov_imm(Reg{def_reg(res), Liveness::Fpr}, x);
450 } else if (id == Annex::Base<fspill>) {
451 auto w = width(x->type());
452 auto s = slot_of(res);
453 if (auto l = Lit::isa(x)) {
454 line("movl ${}, {}", u32(*l), slot(s));
455 if (w == 64) line("movl ${}, {}", u32(*l >> 32), slot(s, 4));
456 } else {
457 line("mov{} {}, {}", fsz(w), xmm(use_reg(x)), slot(s));
458 }
459 } else if (id == Annex::Base<freload>) {
460 auto w = width(res->type());
461 line("mov{} {}, {}", fsz(w), slot(slot_of(x)), xmm(def_reg(res)));
462 } else if (id == Annex::Base<fcmp>) {
463 auto [a, b] = x->projs<2>();
464 ucomi(axm->sub(), width(a->type()), xmm(use_reg(b)), use_reg(a), def_reg(res));
465 } else if (id == Annex::Base<fsel>) {
466 auto [c, a, b] = x->projs<3>();
467 auto dst = def_reg(res);
468 assert(use_reg(a) == dst);
469 auto l = local("fsel");
470 line("testb {}, {}", name(use_reg(c), 8), name(use_reg(c), 8));
471 line("je {}", l);
472 line("movaps {}, {}", xmm(use_reg(b)), xmm(dst));
473 std::println(body_, "{}:", l);
474 } else if (id == Annex::Base<u2f>) {
475 auto ws = width(x->type());
476 auto wr = width(res->type());
477 if (ws == 64) f->blame(MIM_X86_BE "`x86.u2f` from `i64` is not supported yet").bail();
478 auto src = name(use_reg(x), 32);
479 auto dst = xmm(def_reg(res));
480 auto l = local("u2f");
481 // Exact in `f64`, so an `f32` result rounds only once.
482 line("cvtsi2sdl {}, {}", src, dst);
483 line("testl {}, {}", src, src);
484 line("jns {}", l);

505void Emitter::unary(flags_t id, std::string_view src, nat_t ws, const Def* res) {
506 auto wr = width(res->type());

508 if (id == Annex::Base<fsqrt>) return line("sqrt{} {}, {}", fsz(wr), src, xmm(dst));

513 line("cvtt{}2siq {}, {}", fsz(ws), src, name(dst, 64));
514 line("movl {}, {}", name(dst, 32), name(dst, 32));

517void Emitter::step(const App* app) {
518 if (Axm::isa<sched>(app)) return insn(app->proj(2, 1), sched_insn(app), app->arg());
519
520 if (Axm::isa<x86::slot>(app)) {
521 num_slots_ += (nat(app->callee()->as<App>()->arg()) + 7) / 8;
522 line("leaq {}, {}", slot(num_slots_ - 1), name(def_reg(app->proj(2, 1)), 64));
523 return;
524 }
525
526 if (!Axm::isa<load>(app) && !Axm::isa<store>(app) && !Axm::isa<fload>(app) && !Axm::isa<fstore>(app))
527 flags_ = nullptr;
528
529 if (auto op = Axm::isa<rm>(app)) {
530 auto [_, a, p] = app->args<3>();
531 auto w = width(a->type());
532 auto dst = def_reg(app->proj(2, 1));
533 auto insn = op->decurry()->arg()->as<App>()->axm();
534 if (insn->base() == Annex::Base<fcmp>) return ucomi(insn->sub(), w, addr(p), use_reg(a), dst);
535 assert(use_reg(a) == dst);
536 if (math::isa_f(a->type()))
537 line("{}{} {}, {}", farith(insn->base()), fsz(w), addr(p), xmm(dst));
538 else
539 line("{}{} {}, {}", alu(insn->flags()), suffix(w), addr(p), name(dst, w));
540 return;
541 }
542
543 if (Axm::isa<rm1>(app)) {
544 auto [_, p] = app->args<2>();
545 auto f = app->callee()->as<App>()->arg();
546 return unary(insn::op(app), addr(p), width(f->type()->as<Pi>()->dom()), app->proj(2, 1));
547 }
548
549 if (auto op = Axm::isa<mr>(app)) {
550 auto [_, p, b] = app->args<3>();
551 auto w = width(b->type());
552 line("{}{} {}, {}", alu(op->decurry()->arg()->as<App>()->axm()->flags()), suffix(w), src(b, w), addr(p));
553 return;
554 }
555
556 if (Axm::isa<load>(app)) {
557 auto [_, p] = app->args<2>();
558 auto res = app->proj(2, 1);
559 auto w = width(res->type());
560 line("mov{} {}, {}", suffix(w), addr(p), name(def_reg(res), w));
561 return;
562 }
563
564 if (Axm::isa<fload>(app)) {
565 auto [_, p] = app->args<2>();
566 auto res = app->proj(2, 1);
567 auto w = width(res->type());
568 line("mov{} {}, {}", fsz(w), addr(p), xmm(def_reg(res)));
569 return;
570 }
571
572 if (Axm::isa<fstore>(app)) {
573 auto [_, p, v] = app->args<3>();
574 auto w = width(v->type());
575 line("mov{} {}, {}", fsz(w), xmm(use_reg(v)), addr(p));
576 return;
577 }
578
579 auto [_, p, v] = app->args<3>();
580 auto w = width(v->type());
581 line("mov{} {}, {}", suffix(w), src(v, w), addr(p));
582}

584/// Sequentializes the parallel copies @p moves between registers and @p slot_moves between spill slots.
585void Emitter::parallel(Moves moves, fe::Vector<std::pair<size_t, size_t>> slot_moves) {
586 std::erase_if(slot_moves, [](auto m) { return m.first == m.second; });
587 for (auto [_, from] : slot_moves)
588 line("pushq {}", slot(from));
589 for (auto [to, _] : slot_moves | std::views::reverse)
590 line("popq {}", slot(to));
591
592 auto imms = Moves();
593 for (auto& m : moves)
594 if (m.second.imm) imms.emplace_back(m);
595 std::erase_if(moves, [](const auto& m) { return m.second.imm || m.first == m.second.reg; });
596
597 while (!moves.empty()) {
598 auto free = std::ranges::find_if(moves, [&](const auto& m) {
599 return std::ranges::none_of(moves, [&](const auto& n) { return n.second.reg == m.first; });
600 });
601 if (free != moves.end()) {
602 move(free->first, free->second.reg);
603 moves.erase(free);
604 continue;
605 }
606 // Only cycles are left.
607 auto [d, s] = moves.back();
608 moves.pop_back();
609 swap(s.reg, d);
610 for (auto& [dst, from] : moves)
611 if (from.reg == d)
612 from.reg = s.reg;
613 else if (from.reg == s.reg)
614 from.reg = d;
615 std::erase_if(moves, [](const auto& m) { return m.first == m.second.reg; });
616 }
617
618 for (auto [dst, s] : imms)
619 mov_imm(dst, s.imm);
620}

622void Emitter::copy(Lam* target, const Def* arg) {
623 auto moves = Moves();
624 auto slot_moves = fe::Vector<std::pair<size_t, size_t>>();
625 auto n = target->num_vars();
626 for (size_t i = 0; i != n; ++i) {
627 auto x = target->var(n, i);
628 auto a = arg->proj(n, i);
629 if (is_mem(x)) continue;
630 if (Axm::isa<Spill>(x->type()) || Axm::isa<FSpill>(x->type()))
631 slot_moves.emplace_back(slot_of(x), slot_of(a));
632 else if (auto r = fe::lookup(regs_, x))
633 moves.emplace_back(Reg{*r, Liveness::cls(x)}, source(a));
634 }
635 parallel(std::move(moves), std::move(slot_moves));
636}

654void Emitter::call(Lam* callee, const Def* arg) {
655 auto n = arg->num_projs();
656 auto moves = Moves();
657 auto a = size_t(0), f = size_t(0);
658 for (size_t i = 0; i + 1 < n; ++i) {
659 if (auto x = arg->proj(n, i); !is_mem(x)) {
660 auto cls = Liveness::cls(strip(x));
661 auto& c = cls == Liveness::Gpr ? a : f;
662 if (c == num_arg_regs(cls)) arg->blame(MIM_X86_BE "too many arguments are not supported yet").bail();
663 moves.emplace_back(Reg{arg_reg(cls, c++), cls}, source(x));
664 }
665 }
666 parallel(std::move(moves), {});
667 // A variadic callee reads the number of vector registers used from `al`.
668 line("movl ${}, %eax", f);
669 line("call {}", symbol(callee));
670
671 auto ret = arg->proj(n, n - 1);
672 if (ret == ret_) return epilogue();
673
674 auto k = Lam::isa_mut_basicblock(ret);
675 auto m = k->num_vars();
676 auto back = Moves();
677 for (size_t i = 0; i != m; ++i)
678 if (auto x = k->var(m, i); !is_mem(x)) {
679 auto cls = Liveness::cls(x);
680 if (auto r = fe::lookup(regs_, x)) back.emplace_back(Reg{*r, cls}, Src{.reg = {ret_reg(cls), cls}});
681 }
682 parallel(std::move(back), {});
683 jump(k);
684}

686/// The block that @p b should fall through to.
687Lam* Emitter::tail(const Liveness::Block& b) const {
688 auto term = b.lam->body()->as<App>();
689 auto callee = term->callee();
690 if (callee == ret_) return nullptr;
691 if (Lam::isa_mut_returning(callee)) {
692 auto ret = term->arg(term->num_args(), term->num_args() - 1);
693 return ret == ret_ ? nullptr : Lam::isa_mut_basicblock(ret);
694 }
695 if (auto ex = callee->isa<Extract>()) {
696 if (ex->tuple()->num_projs() != 2) return nullptr;
697 auto [e, t] = ex->tuple()->projs<2>([](const Def* d) { return Lam::isa_mut_basicblock(d); });
698 if (!e || !t) return nullptr;
699 return takes_values(t) ? t : e;
700 }
701 return Lam::isa_mut_basicblock(callee);
702}

704void Emitter::block(const Liveness::Block& b) {
705 for (auto insn : b.insns)
706 step(insn);
707
708 auto term = b.lam->body()->as<App>();
709 auto arg = term->arg();
710 auto callee = term->callee();
711 if (callee == ret_) {
712 auto moves = Moves();
713 for (auto a : term->args()) {
714 if (is_mem(a)) continue;
715 auto cls = Liveness::cls(strip(a));
716 moves.emplace_back(Reg{ret_reg(cls), cls}, source(a));
717 }
718 if (moves.size() > 1) b.lam->blame(MIM_X86_BE "`{}` returns more than one value", b.lam).bail();
719 parallel(std::move(moves), {});
720 epilogue();
721 } else if (auto f = Lam::isa_mut_returning(callee)) {
722 call(f, arg);
723 } else if (auto ex = callee->isa<Extract>()) {
724 auto targets = ex->tuple();
725 if (targets->num_projs() != 2) b.lam->blame(MIM_X86_BE "only two-way branches are supported yet").bail();
726 auto [e, t] = targets->projs<2>([](const Def* d) { return Lam::isa_mut_basicblock(d); });
727 if (!e || !t) b.lam->blame(MIM_X86_BE "`{}` branches to something other than blocks", b.lam).bail();
728 auto c = ex->index();
729 auto trampoline = takes_values(t);
730 auto t_lbl = trampoline ? label(b.lam) + ".t" : label(t);
731 if (auto app = isa_sched_cc(c)) {
732 use_flags(app->arg());
733 line("j{} {}", ccs[Axm::as<cc, 1>(sched_insn(app)).sub()], t_lbl);
734 } else {
735 auto r = name(use_reg(c), 8);
736 line("testb {}, {}", r, r);
737 line("jne {}", t_lbl);
738 }
739 copy(e, arg);
740 if (trampoline) {
741 line("jmp {}", label(e));
742 std::println(body_, "{}:", t_lbl);
743 copy(t, arg);
744 jump(t);
745 } else {
746 jump(e);
747 }
748 } else if (auto bb = Lam::isa_mut_basicblock(callee)) {
749 copy(bb, arg);
750 jump(bb);
751 } else {
752 b.lam->blame(MIM_X86_BE "cannot emit the jump to `{}`", callee).bail();
753 }
754}

756void Emitter::emit(bool global) {
757 if (!ret_) fun_->blame(MIM_X86_BE "`{}` does not end in a return continuation", fun_).bail();
758
759 auto done = DefSet();
760 for (const auto& b : live_.blocks()) {
761 for (auto insn : b.insns) {
762 collect(insn->callee(), done);
763 collect(insn->arg(), done);
764 for (auto [j, r] : constraint(insn).results)
765 regs_.emplace(result(insn, j), *Lit::isa(r));
766 }
767 collect(b.lam->body(), done);
768 }
769 auto used = std::array<bool, Num_Regs>{};
770 for (auto [def, r] : regs_)
771 if (Liveness::cls(def) == Liveness::Gpr) used[r] = true;
772 for (size_t r = 0; r != Num_Regs; ++r)
773 if (!Caller_Saved[r] && !reserved(r) && used[r]) saved_.emplace_back(r);
774
775 auto order = fe::Vector<const Liveness::Block*>();
776 auto placed = LamSet();
777 for (auto queue = std::deque<Lam*>{fun_}; !queue.empty();) {
778 for (auto lam = queue.front(); lam && placed.emplace(lam).second;) {
779 auto b = live_.block(lam);
780 order.emplace_back(b);
781 for (auto s : b->succs)
782 queue.emplace_back(live_.blocks()[s].lam);
783 lam = tail(*b);
784 if (lam && !live_.block(lam)) lam = nullptr;
785 }
786 queue.pop_front();
787 }
788
789 auto n = fun_->num_vars();
790 auto moves = Moves();
791 for (size_t i = 0, a = 0, f = 0; i + 1 < n; ++i) {
792 if (auto x = fun_->var(n, i); !is_mem(x)) {
793 auto cls = Liveness::cls(x);
794 auto& c = cls == Liveness::Gpr ? a : f;
795 if (c == num_arg_regs(cls)) fun_->blame(MIM_X86_BE "too many arguments are not supported yet").bail();
796 if (auto r = fe::lookup(regs_, x)) moves.emplace_back(Reg{*r, cls}, Src{.reg = {arg_reg(cls, c), cls}});
797 ++c;
798 }
799 }
800 parallel(std::move(moves), {});
801
802 for (size_t i = 0; i != order.size(); ++i) {
803 auto b = order[i];
804 next_ = i + 1 != order.size() ? order[i + 1]->lam : nullptr;
805 if (i != 0) std::println(body_, "{}:", label(b->lam));
806 flags_ = nullptr;
807 block(*b);
808 }
809
810 auto sym = symbol(fun_);
811 auto pushed = 8 * saved_.size();
812 auto frame = fe::pad(8 * num_slots_ + pushed, 16) - pushed;
813 if (global) std::println(os_, "\t.globl {}", sym);
814 std::println(os_, "\t.type {}, @function", sym);
815 std::println(os_, "{}:", sym);
816 std::println(os_, "\tpushq %rbp");
817 std::println(os_, "\tmovq %rsp, %rbp");
818 for (auto r : saved_)
819 std::println(os_, "\tpushq {}", name(r, 64));
820 if (frame != 0) std::println(os_, "\tsubq ${}, %rsp", frame);
821 os_ << body_.view();
822 std::println(os_, "\t.size {}, .-{}", sym, sym);
823}

829 if (auto o = arg_value(args(), "o", "output")) path = *o;

864}
