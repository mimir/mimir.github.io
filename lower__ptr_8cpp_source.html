14/// Successively offsets `ptr` by each component of the index `tuple`, peeling one array dimension per `%mem.lea`.
15const Def* op_lea_tuple(const Def* ptr, const Def* tuple) {
16 auto n = tuple->num_projs();
17 auto element = ptr;
18 for (size_t i = 0; i < n; ++i)
19 element = mem::op_lea(element, tuple->proj(n, i));

21}

23/// Builds the nested array type `«s; T»` from a shape tuple `s` and element type `T`.

33/// Builds the nested `pack` `‹s; val›` replicating `val` across every element.
34const Def* pack_tuple(const Def* s, const Def* val) {
