46 static constexpr std::string_view tag2str(Tag tag) {
47 switch (tag) {
48 case Tag::Nil: return "<nil>";
49 case Tag::EoF: return "<end of file>";
50 case Tag::M_id: return "<key>";
51 case Tag::L_str: return "<string>";
52 case Tag::D_brckt_l: return "[";
53 case Tag::D_brckt_r: return "]";
54 case Tag::T_assign: return "=";
55 case Tag::T_comma: return ",";
56 }
57 std::unreachable();
58 }

154Tok Lexer::lex() {
155 while (true) {
156 start();
157
158 if (accept(utf8::EoF)) return {loc_, Tok::Tag::EoF};
159 if (accept(utf8::isspace)) continue;
160 if (recover_utf8()) continue;
161
162 if (accept('[')) return {loc_, Tok::Tag::D_brckt_l};
163 if (accept(']')) return {loc_, Tok::Tag::D_brckt_r};
164 if (accept(',')) return {loc_, Tok::Tag::T_comma};
165 if (accept('=')) return {loc_, Tok::Tag::T_assign};
166
167 if (accept('#')) {
168 accept_while_none_of(char8_t('\r'), char8_t('\n'));
169 continue;
170 }
171
172 if (accept('"')) return lex_str('"');
173 if (accept('\'')) return lex_str('\'');
174
175 if (accept(is_key_char)) {
176 accept_while(is_key_char);
177 return {loc_, Tok::Tag::M_id, std::string(view())};
178 }
179
180 recover_char();
181 }
182}

184Tok Lexer::lex_str(char32_t quote) {
185 std::string res;
186
187 while (true) {
188 if (accept(quote)) return {loc_, Tok::Tag::L_str, std::move(res)};
189
190 if (eol()) {
191 error().e(loc_, "unterminated string literal");
192 return {loc_, Tok::Tag::L_str, std::move(res)};
193 }
194
195 if (quote == '"' && accept('\\')) {
196 if (eol()) continue;
197 auto backslash = loc_.end.off - 1;
198 if (auto u = unescape(next())) {
199 res += *u;
200 } else {
201 auto escaped = buf_.substr(backslash + 1, loc_.end.off - backslash - 1);
202 error().e(Loc(src_, Pos(backslash), loc_.end), "invalid escape character `\\{}`", escaped);
203 res += escaped;
204 }
205 } else {
206 auto begin = loc_.end.off;
207 next();
208 res += buf_.substr(begin, loc_.end.off - begin);
209 }
210 }
211}

226Manifest Parser::parse() {
227 Manifest manifest{};
228 std::array<bool, lists.size()> seen{};
229
230 while (!ahead().isa(Tok::Tag::EoF)) {
231 auto key = expect(Tok::Tag::M_id, "manifest entry");
232 if (!key) {
233 lex();
234 continue;
235 }
236 auto entry = std::ranges::find(lists, key.str(), &List_Entry::key);
237 if (entry == lists.end()) {
238 auto keys = std::ranges::fold_left(
239 std::views::drop(lists, 1), fe::format_cite("`{}`", lists.begin()->key),
240 [](fe::Cited acc, const auto& e) { return fe::format_cite("{}, `{}`", acc, e.key); });
241 error().e(key.loc(), "unknown manifest key `{}`", key.str()).n("expected one of {}", keys);
242 } else {
243 auto i = entry - lists.begin();
244 if (seen[i]) error().e(key.loc(), "duplicate manifest key `{}`", key.str());
245 seen[i] = true;
246 }
247
248 expect(Tok::Tag::T_assign, "manifest entry");
249
250 auto brckt = expect(Tok::Tag::D_brckt_l, "value of key `{}`", key);
251 if (!brckt) {
252 lex();
253 continue;
254 }
255 auto _ = anchor(brckt, Tok::Tag::D_brckt_r);
256
257 for (bool first = true; !ahead().isa(Tok::Tag::EoF) && !ahead().isa(Tok::Tag::D_brckt_r);) {
258 if (!first) {
259 expect(Tok::Tag::T_comma, "value of key `{}`", key);
260 if (ahead().isa(Tok::Tag::D_brckt_r)) break;
261 }
262 first = false;
263
264 auto val = expect(Tok::Tag::L_str, "value of key `{}`", key);
265 if (!val) {
266 lex();
267 continue;
268 }
269 if (entry == lists.end()) continue;
270
271 if (key.str() == Plugin_Arg_Key && !val.str().contains(':'))
272 error().e(val.loc(), "expected `<plugin>:<arg>`, got `{}`", val.str());
273
274 auto& dst = std::invoke(entry->list, &manifest);
275 dst.emplace_back(std::move(val.str()));
276 }
277
278 expect(Tok::Tag::D_brckt_r, "value of key `{}`", key);
279 }
280
281 return manifest;
282}

288Manifest load_manifest(const fs::path& file) {
289 std::error_code ec{};
290 if (!fs::is_regular_file(file, ec))
291 fe::throwf<std::invalid_argument>("cannot read config file `{}`", file.string());
292
293 fe::Driver driver;
294 auto src = driver.src().add(file).first;
295 if (!src) fe::throwf<std::invalid_argument>("cannot read config file `{}`", file.string());
296
297 manifest::Parser parser{driver, *src};
298 Manifest manifest = parser.parse();
299 try {
300 driver.error().ack();
301 } catch (const fe::Error::Bail& bail) { throw std::invalid_argument(bail.what()); }
302
303 // Relative `*-path` entries anchor to the manifest file's directory.
304 // `paths` and `p` are references, so assigning to `p` rewrites the manifest entry in place.
305 auto resolve = [base = file.parent_path()](fe::Vector<std::string>& paths) {
306 for (auto& p : paths)
307 if (auto fp = fs::path(p); !fp.is_absolute()) p = (base / fp).lexically_normal().string();
308 };
309 resolve(manifest.plugin_paths);
310 resolve(manifest.import_paths);
311 resolve(manifest.prefix_paths);
312 return manifest;
313}
