52/// Additionally to the derivation, the pullback is registered and the maps are initialized.
53const Def* Eval::derive_(const Def* def) {
54 auto lam = def->as_mut<Lam>(); // TODO check if mutable
55 DLOG("Derive lambda: {}", def);
56 auto deriv_ty = autodiff_type_fun_pi(lam->type());
57 auto deriv = world().mut_lam(deriv_ty)->set(lam->sym().str() + "_deriv");
58
59 // We first pre-register the derivatives.
60 // This knowledge is needed for recursion.
61 // (Alternatively, we could also use projections out the variables instead of pre-partial-pullback
62 // initialization.)
63 derived[lam] = deriv;
64
65 auto [arg_ty, ret_pi] = lam->type()->doms<2>();
66 auto deriv_all_args = deriv->var();
67 const Def* deriv_arg = deriv->var(0uz)->set("arg");
68
69 // We generate the shadow pullbacks dynamically to save work and avoid code duplication.
70 // Only the toplevel pullback for arguments and return continuation is special cased.
71
72 // TODO: check identity: could use identity tangent(arg_ty) = tangent(augment(arg_ty)) with deriv_arg->type() =
73 // augment(arg_ty) We give the argument the identity pullback.
74 auto arg_id_pb = id_pullback(arg_ty);
75 partial_pullback[deriv_arg] = arg_id_pb;
76 // The return continuation has to formally exist but should never be directly accessed.
77 auto ret_var = deriv->var(1);
78 auto ret_pb = zero_pullback(lam->var(1)->type(), arg_ty);
79 partial_pullback[ret_var] = ret_pb;
80
81 shadow_pullback[deriv_all_args] = world().tuple({arg_id_pb, ret_pb});
82 DLOG("pullback for argument {} : {} is {} : {}", deriv_arg, deriv_arg->type(), arg_id_pb, arg_id_pb->type());
83 DLOG("args shadow pb is {} : {}", shadow_pullback[deriv_all_args], shadow_pullback[deriv_all_args]->type());
84
85 // We pre-register the augment replacements.
86 // The function and its variables are replaced by their new derived versions.
87 // TODO: maybe leave out function call (duplication with derived)
88 augmented[def] = deriv;
89 DLOG("Associate {} with {}", def, deriv);
90 DLOG(" {} : {}", lam, lam->type());
91 DLOG(" {} : {}", deriv, deriv->type());
92 augmented[lam->var()] = deriv->var();
93 DLOG("Associate vars {} with {}", lam->var(), deriv->var());
94
95 // already contains the correct application of
96 // deriv->ret_var() by specification
97 // f : cn[R] has a partial derivative (exception to closed rule)
98 // f': cn[R, cn[R, cn[A]]]
99 // this is needed for continuations (without closure conversion)
100 // but also essentially for the return continuation
101
102 // Here a reminder of types:
103 // The expression `e: B` has the implicit function `e_fun: A -> B`
104 // The partial pullback is then `e*: B* -> A*`
105 // The derivatived version is `e': B' × (B* -> A*)` which is an application of `e'_fun: A' -> B' × (B* -> A*)`
106 auto new_body = augment(lam->body(), lam, deriv);
107 deriv->set(true, new_body);
108
109 return deriv;
110}

129const Def* Eval::augment_lam(Lam* lam, Lam* f, Lam* f_diff) {
130 // TODO: we need partial pullbacks for tuples (higher-order / ret-cont application)
131 // also for higher-order args, ret_cont (at another point)
132 // the pullback is not important but formally required by tuple rule
133 if (augmented.count(lam)) {
134 // We already know the function:
135 // * recursion
136 // * higher order arguments
137 // * new encounter of previous function
138 DLOG("already augmented {} : {} to {} : {}", lam, lam->type(), augmented[lam], augmented[lam]->type());
139 return augmented[lam];
140 }
141 // TODO: better fix (another pass as analysis?)
142 // TODO: handle open functions
143 if (Lam::isa_basicblock(lam) || lam->sym().view().contains("ret") || lam->sym().view().contains("_cont")) {
144 // A open continuation behaves the same as return:
145 // ```
146 // cont: Cn[X]
147 // cont': Cn[X,Cn[X,A]]
148 // ```
149 // There is dependency on the closed function context.
150 // (All derivatives are with respect to the arguments of a closed function.)
151
152 DLOG("found an open continuation {} : {}", lam, lam->type());
153 auto cont_dom = lam->type()->dom(); // not only 0 but all
154 auto pb_ty = pullback_type(cont_dom, f->dom(2, 0));
155 auto aug_dom = autodiff_type_fun(cont_dom);
156 DLOG("augmented domain {}", aug_dom);
157 DLOG("pb type is {}", pb_ty);
158 auto aug_lam = world().mut_con({aug_dom, pb_ty})->set("aug_"s + lam->sym().str());
159 auto aug_var = aug_lam->var((nat_t)0);
160 augmented[lam->var()] = aug_var;
161 augmented[lam] = aug_lam; // TODO: only one of these two
162 derived[lam] = aug_lam;
163 auto pb = aug_lam->var(1);
164 partial_pullback[aug_var] = pb;
165 // We are still in same closed function.
166 auto new_body = augment(lam->body(), f, f_diff);
167 // TODO we also need to rewrite the filter
168 aug_lam->set(lam->filter(), new_body);
169
170 auto lam_pb = zero_pullback(lam->type(), f->dom(2, 0));
171 partial_pullback[aug_lam] = lam_pb;
172 DLOG("augmented {} : {}", lam, lam->type());
173 DLOG("to {} : {}", aug_lam, aug_lam->type());
174 DLOG("ppb for lam cont: {}", lam_pb);
175
176 return aug_lam;
177 }
178 DLOG("found a closed function call {} : {}", lam, lam->type());
179 // Some general function in the program needs to be differentiated.
180 // The old pass emitted a new `%autodiff.ad` application here and relied on the PassMan to revisit it;
181 // as a Phase we derive eagerly instead (derive() pre-registers itself, so recursion terminates).
182 auto aug_lam = derive(lam);
183 // TODO: directly more association here? => partly inline op_autodiff
184 DLOG("augmented function is {} : {}", aug_lam, aug_lam->type());
185 return aug_lam;
186}

188const Def* Eval::augment_extract(const Extract* ext, Lam* f, Lam* f_diff) {
189 auto tuple = ext->tuple();
190 auto index = ext->index();
191
192 auto aug_tuple = augment(tuple, f, f_diff);
193 auto aug_index = augment(index, f, f_diff);
194
195 const Def* pb;
196 DLOG("tuple was: {} : {}", tuple, tuple->type());
197 DLOG("aug tuple: {} : {}", aug_tuple, aug_tuple->type());
198 if (shadow_pullback.count(aug_tuple)) {
199 auto shadow_tuple_pb = shadow_pullback[aug_tuple];
200 DLOG("Shadow pullback: {} : {}", shadow_tuple_pb, shadow_tuple_pb->type());
201 pb = world().extract(shadow_tuple_pb, aug_index);
202 } else {
203 // ```
204 // e:T, b:B
205 // b = e#i
206 // b* = \lambda (s:B). e* (insert s at i in (zero T))
207 // ```
208 assert(partial_pullback.count(aug_tuple));
209 auto tuple_pb = partial_pullback[aug_tuple];
210 auto pb_ty = pullback_type(ext->type(), f->dom(2, 0));
211 auto pb_fun = world().mut_lam(pb_ty)->set("extract_pb");
212 DLOG("Pullback: {} : {}", pb_fun, pb_fun->type());
213 auto pb_tangent = pb_fun->var(0uz)->set("s");
214 auto tuple_tan = world().insert(world().call<zero>(aug_tuple->type()), aug_index, pb_tangent)->set("tup_s");
215 pb_fun->app(true, tuple_pb, {tuple_tan, pb_fun->var(1) /* ret_var but make sure to select correct one */});
216 pb = pb_fun;
217 }
218
219 auto aug_ext = world().extract(aug_tuple, aug_index);
220 partial_pullback[aug_ext] = pb;
221
222 return aug_ext;

226 // TODO: should use ops instead?

228 auto aug_tup = world().tuple(aug_ops);
229
230 auto pbs = DefVec(Defs(aug_ops), [&](const Def* op) { return partial_pullback[op]; });
231 DLOG("tuple pbs {}", fe::Join(pbs));
232 // shadow pb = tuple of pbs
233 auto shadow_pb = world().tuple(pbs);
234 shadow_pullback[aug_tup] = shadow_pb;

301const Def* Eval::augment_app(const App* app, Lam* f, Lam* f_diff) {
302 auto callee = app->callee();
303 auto arg = app->arg();
304
305 auto aug_arg = augment(arg, f, f_diff);
306 auto aug_callee = augment(callee, f, f_diff);
307
308 DLOG("augmented argument <{}> {} : {}", aug_arg->unique_name(), aug_arg, aug_arg->type());
309 DLOG("augmented callee <{}> {} : {}", aug_callee->unique_name(), aug_callee, aug_callee->type());
310 // TODO: move down to if(!is_cont(callee))
311 if (!Pi::isa_cn(callee->type()) && Pi::isa_cn(aug_callee->type())) {
312 aug_callee = cps::op_cps2ds_dep(aug_callee);
313 DLOG("wrapped augmented callee: <{}> {} : {}", aug_callee->unique_name(), aug_callee, aug_callee->type());
314 }
315
316 // nested (inner application)
317 if (app->type()->isa<Pi>()) {
318 DLOG("Nested application callee: {} : {}", aug_callee, aug_callee->type());
319 DLOG("Nested application arg: {} : {}", aug_arg, aug_arg->type());
320 auto aug_app = world().app(aug_callee, aug_arg);
321 DLOG("Nested application result: <{}> {} : {}", aug_app->unique_name(), aug_app, aug_app->type());
322 // We do not add a pullback as the pullback is bundled in the cps call or returned by the ds call
323 return aug_app;
324 }
325
326 // continuation (ret, if, ...)
327 if (Pi::isa_basicblock(callee->type())) {
328 // TODO: check if function (not operator)
329 // The original function is an open function (return cont / continuation) of type `Cn[E]`
330 // The augmented function `aug_callee` looks like a function but is not really a function has the type `Cn[E,
331 // Cn[E, Cn[A]]]`
332
333 // ret(e) => ret'(e, e*)
334
335 DLOG("continuation {} : {} => {} : {}", callee, callee->type(), aug_callee, aug_callee->type());
336
337 auto arg_pb = partial_pullback[aug_arg];
338 auto aug_app = world().app(aug_callee, {aug_arg, arg_pb});
339 DLOG("Augmented application: {} : {}", aug_app, aug_app->type());
340 return aug_app;
341 }
342
343 // ds function
344 if (!Pi::isa_cn(callee->type())) {
345 auto aug_app = world().app(aug_callee, aug_arg);

348 DLOG("ds function: {} : {}", aug_app, aug_app->type());
349 // The calle is ds function (e.g. operator (or its partial application))
350 auto [aug_res, fun_pb] = aug_app->projs<2>();
351 // We compose `fun_pb` with `argument_pb` to get the result pb
352 // TODO: combine case with cps function case
353 auto arg_pb = partial_pullback[aug_arg];
354 assert(arg_pb);
355 // `fun_pb: out_tan -> arg_tan`
356 // `arg_pb: arg_tan -> fun_tan`
357 DLOG("function pullback: {} : {}", fun_pb, fun_pb->type());
358 DLOG("argument pullback: {} : {}", arg_pb, arg_pb->type());
359 auto res_pb = compose_cn(arg_pb, fun_pb);
360 DLOG("result pullback: {} : {}", res_pb, res_pb->type());
361 partial_pullback[aug_res] = res_pb;
362 world().debug_dump();
363 return aug_res;
364 }
365
366 // TODO: dest with a function such that f args != g args
367 {
368 // normal function app
369 // ```
370 // g: cn[E, cn X]
371 // g(args,cont)
372 // g': cn[E, cn[X, cn[X, cn E]]]
373 // g'(aug_args, ____)
374 // ```
375 auto g = callee;
376 // At this point g_deriv might still be "autodiff ... g".
377 auto g_deriv = aug_callee;
378 DLOG("g: {} : {}", g, g->type());
379 DLOG("g': {} : {}", g_deriv, g_deriv->type());
380
381 auto [real_aug_args, aug_cont] = aug_arg->projs<2>();
382 DLOG("real_aug_args: {} : {}", real_aug_args, real_aug_args->type());
383 DLOG("aug_cont: {} : {}", aug_cont, aug_cont->type());
384 auto e_pb = partial_pullback[real_aug_args];
385 DLOG("e_pb (arg_pb): {} : {}", e_pb, e_pb->type());
386
387 // TODO: better debug names
388 auto ret_g_deriv_ty = g_deriv->type()->as<Pi>()->dom(1);
389 DLOG("ret_g_deriv_ty: {} ", ret_g_deriv_ty);

407const Def* Eval::augment_(const Def* def, Lam* f, Lam* f_diff) {
408 // We use macros above to avoid recomputation.
409 // TODO: Alternative:
410 // Use class instances to rewrite inside a function and save such values (f, f_diff, f->dom(2, 0)).
411
412 DLOG("Augment def {} : {}", def, def->type());
413
414 // Applications are continuations, operators, or full functions
415 if (auto app = def->isa<App>()) {
416 auto callee = app->callee();
417 auto arg = app->arg();
418 DLOG("Augment application: app {} with {}", callee, arg);
419 return augment_app(app, f, f_diff);
420 } else if (auto ext = def->isa<Extract>()) {
421 auto tuple = ext->tuple();
422 auto index = ext->index();
423 DLOG("Augment extract: {} #[{}]", tuple, index);
424 return augment_extract(ext, f, f_diff);
425 } else if (auto var = def->isa<Var>()) {
426 DLOG("Augment variable: {}", var);
427 return augment_var(var, f, f_diff);
428 } else if (auto lam = def->isa_mut<Lam>()) {
429 DLOG("Augment mut lambda: {}", lam);
430 return augment_lam(lam, f, f_diff);
431 } else if (auto lam = def->isa<Lam>()) {
432 ELOG("Augment lambda: {}", lam);
433 assert(false && "can not handle non-mutable lambdas");
434 } else if (auto lit = def->isa<Lit>()) {
435 DLOG("Augment literal: {}", def);
436 return augment_lit(lit, f, f_diff);
437 } else if (auto tup = def->isa<Tuple>()) {
438 DLOG("Augment tuple: {}", def);
439 return augment_tuple(tup, f, f_diff);
440 } else if (auto pack = def->isa<Pack>()) {
441 // TODO: handle mut packs (dependencies in the pack) (=> see paper about vectors)
442 auto arity = pack->arity(); // TODO: arity vs shape
443 auto body = pack->body();
444 DLOG("Augment pack: {} : {} with {}", arity, arity->type(), body);
445 return augment_pack(pack, f, f_diff);
446 } else if (auto ax = def->isa<Axm>()) {
447 // TODO: move concrete handling to own function / file / directory (file per plugin)
448 DLOG("Augment axm: {} : {}", ax, ax->type());
449 DLOG("axm curry: {}", ax->curry());
450 DLOG("axm flags: {}", ax->flags());

452 find_and_replace(diff_name, ".", "_");
453 find_and_replace(diff_name, "%", "");
454 diff_name = "internal_diff_" + diff_name;

456 DLOG("axm function name: {}", diff_name);
457
458 // Look the derivative up in the old world: the new world's externals are still being populated while this
459 // phase runs, so an internal_diff_* function may not have been carried over yet.
460 auto old_diff_fun = old_world().externals()[old_world().sym(diff_name)];
461 if (!old_diff_fun) {
462 ELOG("derivation not found: {}", diff_name);
463 auto expected_type = autodiff_type_fun(ax->type());
464 ELOG("expected: {} : {}", diff_name, expected_type);
465 assert(false && "unhandled axm");
466 }
467 // TODO: why does this cause a depth error?
468 return rewrite(old_diff_fun);
469 }
470
471 // TODO: handle Pi for axm app
472 // TODO: remaining (lambda, axm)
473
474 ELOG("did not expect to augment: {} : {}", def, def->type());
475 ELOG("node: {}", def->node_name());
476 assert(false && "augment not implemented on this def");
477 fe::unreachable();
478}
