32 /// An annex name is global: it lives in one flat table, independent of the module structure.
33 static bool is_anx(Sym sym) { return sym && sym[0] == '%'; }
34
35 void push() { scopes_.emplace_back(); }
36 void push(Scope& scope) { scopes_.emplace_back(scope); }
37
38 void pop() {
39 assert(scopes_.size() > barrier_);
40 scopes_.pop_back();
41 }
42
43 /// A file must not see the scope of whoever imports it, so its Scope becomes the new lookup floor.
44 size_t push_barrier(Scope& scope) {
45 push(scope);
46 return std::exchange(barrier_, scopes_.size() - 1);
47 }
48
49 void pop_barrier(size_t old) {
50 barrier_ = old;
51 pop();
52 }
53
54 const Decl* find(Dbg dbg, bool quiet = false) {
55 if (dbg.is_anon()) return nullptr;
56
57 if (is_anx(dbg.sym())) {
58 if (auto decl = fe::lookup(anx_, dbg.sym())) return decl;
59 } else {
60 for (auto& frame : scopes_ | std::views::drop(barrier_) | std::views::reverse)
61 if (auto decl = fe::lookup(frame.scope(), dbg.sym())) return decl;
62 }
63
64 if (!quiet) {
65 error().e(dbg.loc(), "identifier `{}` not found", dbg.sym());
66 bind(dbg, dummy()); // put into scope to prevent further errors
67 }
68 return nullptr;
69 }
70
71 void bind(Dbg dbg, const Decl* decl, bool rebind = false, bool quiet = false) {
72 if (dbg.is_anon()) return;
73
74 auto& scope = is_anx(dbg.sym()) ? anx_ : top();
75 if (rebind) {
76 scope[dbg.sym()] = decl;
77 } else if (auto [i, ins] = scope.try_emplace(dbg.sym(), decl); !ins) {
78 auto prev = i->second;
79 if (!quiet && !prev->isa<DummyDecl>()) // if prev stems from an error - don't complain
80 error().e(dbg.loc(), "redeclaration of `{}`", dbg).n(prev->dbg().loc(), "previous declaration here");
81 }
82 }

164void Path::bind(Scopes& s, bool quiet) const {
165 decl_ = s.find(front(), quiet);
166 auto prev = front();
167
168 for (const auto& dbg : dbgs() | std::views::drop(1)) {
169 if (!decl_) return;
170 auto scope = decl_->scope();
171 auto member = scope ? fe::lookup(*scope, dbg.sym()) : nullptr;
172 if (!member) {
173 if (!quiet) {
174 if (scope)
175 s.error().e(dbg.loc(), "`{}` has no member `{}`", prev.sym(), dbg.sym());
176 else
177 s.error().e(dbg.loc(), "`{}` is not a module", prev.sym());
178 }
179 decl_ = nullptr;
180 return;
181 }
182 decl_ = member;
183 prev = dbg;
184 }

321void AxmDecl::bind(Scopes& s) const {
322 type()->bind(s);
323
324 annex_ = s.ast().name2annex(dbg(), nullptr);
325
326 if (annex_ && annex_->fresh) {
327 annex_->normalizer = normalizer();
328 annex_->pi = type()->isa<PiExpr>() || type()->isa<ArrowExpr>();
329 } else {
330 auto pi = type()->isa<PiExpr>() || type()->isa<ArrowExpr>();
331 if (annex_ && pi ^ *annex_->pi)
332 s.error().e(dbg().loc(), "all declarations of annex `{}` must be function types if one of them is",
333 dbg().sym());
334
335 if (annex_ && annex_->normalizer.sym() != normalizer().sym()) {
336 auto l = normalizer().loc() ? normalizer().loc() : loc().anew_end();
337 auto& err = s.error().e(l, "normalizer mismatch for axm `{}`", dbg());
338 if (auto norm = annex_->normalizer)
339 err.n(norm.loc(), "previous normalizer `{}` declared here", norm);
340 else
341 err.n("initially no normalizer was specified");
342 }
343 }
344
345 if (num_subs() == 0) {
346 s.bind(dbg(), this);
347 } else {
348 if (auto old = s.find(dbg(), true)) {
349 if (auto old_ax = old->isa<AxmDecl>()) {
350 if (old_ax->num_subs() == 0) {
351 s.error()
352 .e(dbg().loc(), "axm `{}` was declared without subs and cannot be redeclared with subs", dbg())
353 .n(old_ax->dbg().loc(), "previous declaration here");
354 }
355 }
356 }
357
358 if (annex_) {
359 offset_ = annex_->subs.size();
360 for (const auto& aliases : subs())
361 for (const auto& alias : aliases)
362 alias->bind(s, this);
363
364 for (auto& sub : subs()) {
365 auto& aliases = annex_->subs.emplace_back(std::deque<Sym>());
366 for (const auto& alias : sub)

369 }
