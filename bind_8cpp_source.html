62 const Decl* find(Dbg dbg, bool quiet = false) {
63 if (dbg.is_anon()) return nullptr;
64
65 for (auto& frame : scopes_ | std::views::drop(barrier_) | std::views::reverse)
66 if (auto decl = fe::lookup(frame.scope(), dbg.sym())) return decl;
67
68 if (!quiet) {
69 auto& diag = error().e(dbg.loc(), "identifier `{}` not found", dbg.sym());
70 // An infix operator only exists as whatever the user bound its escaped name to.
71 if (dbg.sym().view().starts_with('`'))
72 diag.n("an infix operator means whatever you bind its escaped name to");
73 bind(dbg, dummy()); // put into scope to prevent further errors
74 }
75 return nullptr;
76 }

78 /// Diagnostic-only: is a module named @p sym reachable, shadowed by whatever `find` would actually return?
79 /// Only `ModDecl`/`Import` ever yield a non-null Decl::scope, so this never confuses a value for a module.
80 const Decl* find_shadowed_module(Sym sym) {
81 for (auto& frame : scopes_ | std::views::drop(barrier_) | std::views::reverse)
82 if (auto decl = fe::lookup(frame.scope(), sym); decl && decl->scope()) return decl;
83 return nullptr;
84 }

86 void bind(Dbg dbg, const Decl* decl, bool rebind = false, bool quiet = false) {
87 if (dbg.is_anon()) return;
88
89 auto& scope = top();
90 if (rebind) {
91 scope[dbg.sym()] = decl;
92 } else if (auto [i, ins] = scope.try_emplace(dbg.sym(), decl); !ins) {
93 auto prev = i->second;
94 if (!quiet && !prev->isa<DummyDecl>()) // if prev stems from an error - don't complain
95 error().e(dbg.loc(), "redeclaration of `{}`", dbg).n(prev->dbg().loc(), "previous declaration here");
96 } else if (!quiet && !decl->scope()) {
97 if (auto mod = find_shadowed_module(dbg.sym()))
98 error()
99 .w(dbg.loc(), "`{}` shadows a module of the same name", dbg)
100 .n(mod->dbg().loc(), "module declared here; a later `{}.member` would fail to resolve it",
101 dbg.sym());
102 }
103 }

173void Path::bind(Scopes& s, bool quiet) const {
174 decl_ = s.find(front(), quiet);
175 auto prev = front();
176
177 for (const auto& dbg : dbgs() | std::views::drop(1)) {
178 if (!decl_) return;
179 auto scope = decl_->scope();
180 auto member = scope ? fe::lookup(*scope, dbg.sym()) : nullptr;
181 if (!member) {
182 if (!quiet) {
183 if (scope) {
184 s.error().e(dbg.loc(), "`{}` has no member `{}`", prev.sym(), dbg.sym());
185 } else {
186 auto& err = s.error().e(prev.loc(), "`{}` is not a module", prev.sym());
187 if (auto mod = s.find_shadowed_module(prev.sym()))
188 err.n(mod->dbg().loc(), "a module `{}` exists here but is shadowed by the `{}` in scope",
189 prev.sym(), prev.sym());
190 }
191 }
192 decl_ = nullptr;
193 return;
194 }
195 decl_ = member;
196 prev = dbg;
197
198 // A dotted path always crosses into decl_'s enclosing mod from outside: `priv` blocks it.
199 if (decl_->vis() == Vis::Priv) {
200 if (!quiet) s.error().e(dbg.loc(), "`{}` is private to its enclosing `mod`", dbg.sym());
201 decl_ = nullptr;
202 return;
203 }
204 }

314AnnexInfo* AST::name2annex(Scopes& s, Dbg dbg, sub_t* sub_id) {
315 if (!dbg) return nullptr;
316
317 auto depth = s.mod_depth();
318 if (depth > 1) {
319 error().e(dbg.loc(), "`{}` sits {} `mod` levels deep; an `anx` declaration may nest at most one", dbg, depth);
320 return nullptr;
321 }
322
323 auto plugin_s = dbg.loc().src ? sym(fs::path(dbg.loc().src->path()).stem().string()) : sym_error();
324 auto tag_s = depth == 0 ? dbg.sym() : s.enclosing_mod();
325 Sym sub_s = depth == 0 ? Sym() : dbg.sym();
326
327 auto& sym2annex = plugin2sym2annex_[plugin_s];
328 auto tag_id = sym2annex.size();
329
330 if (plugin_s == sym_error()) error().e(dbg.loc(), "plugin name `{}` is reserved", dbg);
331 if (tag_id > std::numeric_limits<tag_t>::max())
332 error().e(dbg.loc(), "exceeded maximum number of annexes in current plugin");
333
334 if (!Annex::mangle(plugin_s)) {
335 error().e(dbg.loc(), "invalid annex name `{}`", dbg);
336 plugin_s = sym_error();
337 }
338
339 auto [i, fresh] = sym2annex.try_emplace(tag_s, AnnexInfo{plugin_s, tag_s, (tag_t)tag_id});
340 auto annex = &i->second;
341
342 if (sub_s) {
343 if (sub_id) {
344 *sub_id = annex->subs.size();
345 auto& aliases = annex->subs.emplace_back();
346 aliases.emplace_back(sub_s);
347 } else {
348 error().e(dbg.loc(), "annex `{}` must not have a subtag", dbg);
349 }
350 }
351
352 if (!fresh) annex->fresh = false;
353 return annex;
354}

390void AliasDecl::bind(Scopes& s) const {
391 path()->bind(s);
392 s.bind(dbg(), this);
393
394 auto target = path()->decl();
395 if (!target) return;
396
397 if (target->isa<AliasDecl>()) {
398 s.error().e(loc(), "`{}` aliases `{}`, which is itself an alias; alias chains are not supported", dbg(),
399 path()->back());
400 return;
401 }
402
403 std::tie(annex_, sub_) = target->annex_sub();
404 if (!annex_) {
405 s.error().e(loc(), "`{}` must alias a compiler-exposed (`anx`) declaration", dbg());
406 return;
407 }
408

410 // named after the target itself, so this alias has a slot to share.
411 if (sub_ >= annex_->subs.size()) {
412 assert(sub_ == annex_->subs.size());
413 annex_->subs.emplace_back(std::deque<Sym>{target->dbg().sym()});
