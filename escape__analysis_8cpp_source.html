18const Def* EscapeAnalysis::rewrite_imm(const Def* def) {
19 auto res = mim::Analysis::rewrite_imm(def);
20 if (def->isa<Var>() || !def->type()) return res;
21
22 // A value whose type cannot hold a handle (a loaded scalar, say) cannot alias one.
23 if (!may_hold(def->type())) {
24 // Unless it is computed from one, as a bitcast would.
25 if (auto app = res->isa<App>(); app && !Pi::isa_cn(app->callee()->type()) && !mem_def(app->arg()))
26 escape(app->arg());
27 return res;
28 }
29
30 auto carried = DefVec();
31 if (auto app = res->isa<App>()) {
32 if (allocates(app)) {
33 carried.emplace_back(app);
34 } else {
35 join(carried, app->arg());
36 if (!app->callee()->isa_mut<Lam>()) join(carried, app->callee());
37 }
38 } else if (Var::isa_tproj(res).first || res->isa_mut()) {
39 join(carried, res);
40 } else {
41 for (auto op : res->ops())
42 join(carried, op);
43 }
44 dedup(carried);
45
46 if (carried.empty()) return res;
47 if (res != def) aliases_[res] = carried;
48 aliases_[def] = std::move(carried);
49 return res;
50}

52void EscapeAnalysis::leave() {
53 auto lam = curr_mut<Lam>();
54 if (!lam || !lam->is_set()) return;
55
56 auto tail = lam->body()->isa<App>();
57 auto ops = tail ? walk(lam, tail) : std::nullopt;
58 if (!ops) {
59 if (!tail) return escape(lam->body());
60 escape(tail->callee());
61 escape(tail->arg());
62 return;
63 }
64
65 auto escaping = escaping_.contains(lam);
66 auto& chain = chains_[lam] = Chain{tail, std::move(*ops), {}, {}};
67 auto own = DefSet(chain.ops.begin(), chain.ops.end());
68 for (size_t i = 0, n = chain.ops.size(); i != n; ++i) {
69 auto op = chain.ops[i];
70 auto carried = DefVec();
71 join(carried, op->callee());
72 join(carried, op->arg());
73 dedup(carried);
74 for (auto d : carried) {
75 if (auto k = d->isa_mut<Lam>()) {
76 escape(k);
77 continue;
78 }
79 chain.last[d] = i;
80 if (escaping)
81 pin(d);
82 else if (!own.contains(d))
83 shared_.emplace(d);
84 }
85
86 // A pointer stored into memory may be loaded again anywhere.
87 if (auto store = Axm::isa<mem::store>(op)) {
88 escape(store->arg(3, 2));
89 } else if (!Axm::isa<mem::load>(op)) {
90 escape(op->callee());
91 escape(op->arg());
92 }
93 }
94
95 if (auto known = jumps(tail->callee()); !known.empty()) {
96 for (auto k : known)
97 jump(k, tail, chain.targets);
98 } else {
99 control(tail->callee(), chain.targets);
100 for (auto arg : tail->targs())
101 pass(rewrite(arg), chain.targets);
102 }
103
104 if (escaping)
105 for (auto target : chain.targets)
106 if (auto k = target->isa_mut<Lam>()) escape(k);
107}

109bool EscapeAnalysis::may_hold(const Def* type) const {
110 if (type->isa_mut() || type->isa<Pi>() || Axm::isa<mem::Ptr>(type)) return true;
111 if (auto sigma = type->isa<Sigma>())

117std::optional<fe::Vector<const App*>> EscapeAnalysis::walk(Lam* lam, const App* tail) {
118 auto ops = fe::Vector<const App*>();
119 auto cur = mem_def(tail->arg());
120 if (!cur) return ops;
121 auto var = mem_var(lam);
122 if (!var) return {};
123 while (cur != var) {
124 auto ex = cur->isa<Extract>();
125 auto op = ex ? ex->tuple()->isa<App>() : cur->isa<App>();
126 if (!op || !(cur = mem_def(op->arg()))) return {};
127 ops.emplace_back(op);
128 }
129 std::ranges::reverse(ops);
130 return ops;

133void EscapeAnalysis::join(DefVec& carried, const Def* def) const {
134 if (auto lam = def->isa_mut<Lam>()) {
135 if (!lam->is_closed()) carried.emplace_back(lam);
136 } else if (auto [var, i] = Var::isa_tproj(def); var && i) {
137 if (auto val = lattice(def); val && val != def) join(carried, val);
138 } else if (var) {
139 if (auto lam = var->binder()->isa_mut<Lam>())
140 for (auto tvar : lam->tvars())
141 if (auto val = lattice(tvar); val && val != tvar) join(carried, val);
142 } else if (auto i = aliases_.find(def); i != aliases_.end()) {
143 carried.insert(carried.end(), i->second.begin(), i->second.end());
144 }

175void EscapeAnalysis::jump(Lam* k, const App* tail, DefVec& targets) {
176 targets.emplace_back(k);

187void EscapeAnalysis::propagate(Lam* k, const Def* var, const Def* val) {
188 if (val == var) return;
189 auto cur = lattice(var);
190 if (cur == var) return escape(val);
191 auto outside = val->in_scope(k) && !val->free_vars().contains(k->has_var());
192 if (outside && !cur) {
193 lattice(var, val);
194 return;
195 }
196 if (outside && cur == val) return;
197 escape(val);
198 if (cur) escape(cur);
199 pin(var);
200}

203 if (def->is_closed()) return;

211 control(app->callee(), targets);
