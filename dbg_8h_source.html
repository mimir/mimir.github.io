23/// Renders Def%s with their plain Def::sym instead of Def::unique_name while alive.
24/// A gid is noise in a diagnostic about the user's source - but it is also the only thing that tells two
25/// same-named Def%s apart, so PlainNames::clashed reports when a message has to be rendered again with gids.
26/// The state lives in Driver::names, so two Driver%s formatting at once never share it.

234/// Streams @p str, rendering the `` `code` `` spans our diagnostics cite in color.
235/// Color and backticks are alternative ways of setting a citation apart, so the backticks are dropped
236/// when coloring and kept verbatim otherwise.
237inline std::ostream& stream_code(std::ostream& os, std::string_view str) {
238 if (!fe::term::use_color(os)) return os << str;
239
240 for (size_t i = 0, e = str.size(); i != e;) {
241 auto l = str.find('`', i);
242 if (l == std::string_view::npos) return os << str.substr(i);
243 auto r = str.find('`', l + 1);
244 if (r == std::string_view::npos) return os << str.substr(i); // unpaired: not a citation
245 os << str.substr(i, l - i) << fe::term::FG::Cyan << str.substr(l + 1, r - l - 1) << fe::term::FG::Reset;
246 i = r + 1;
247 }
248 return os;
249}
