96const Def* Legalize::rewrite_imm_App(const App* app) {
97 auto& w = new_world();
98 if (Axm::isa<core::Ptr>(app)) return w.type_i64();
99
100 if (auto lea = Axm::isa<mem::lea>(app)) {
101 auto [p, i] = lea->args<2>();
102 auto pointee = mem::pointee(p);
103 auto l = Lit::isa(i);
104 const Def* off = nullptr;
105 if (!pointee->isa<Arr>()) {
106 if (!l) i->blame("index `{}` into `{}` must be a literal", i, pointee).bail();
107 off = w.lit_i64(field_offset(pointee, *l));
108 } else if (auto size = trait(core::trait::size, mem::pointee(lea)); l) {
109 off = w.lit_i64(*l * size);
110 } else {
111 off = core::zext(w, index(i));
112 if (size != 1) off = w.call(core::wrap::mul, core::Mode::none, Defs{off, w.lit_i64(size)});
113 }
114 return w.call(core::wrap::add, core::Mode::none, Defs{rewrite(p), off});
115 }
116
117 if (auto load = Axm::isa<mem::load>(app)) {
118 auto [T, a] = load->decurry()->args<2>();
119 auto [m, p] = load->args<2>();
120 check_addr_space(app, a);
121 return w.call<mach::load>(rewrite(T), Defs{rewrite(m), rewrite(p)});
122 }
123
124 if (auto store = Axm::isa<mem::store>(app)) {
125 auto [m, p, x] = store->args<3>();
126 check_addr_space(app, store->decurry()->arg(2, 1));
127 return w.call<mach::store>(Defs{rewrite(m), rewrite(p), rewrite(x)});
128 }
129
130 // A view of an array changes only the pointee.
131 if (auto assume = Axm::isa<mem::assume>(app)) return rewrite(assume->arg(2, 0));
132
133 if (auto mslot = Axm::isa<mem::mslot>(app)) {
134 auto [msz, k] = mslot->args<2>();
135 auto [m, size] = msz->projs<2>();
136 check_addr_space(app, mslot->decurry()->arg(2, 1));
137 auto n = Lit::isa(size);
138 if (!n) app->blame("native backend: `mem.mslot` needs a literal size").bail();
139 return w.app(rewrite(k), w.call<slot>(w.lit_nat(*n), rewrite(m)));
140 }
141
142 if (auto malloc = Axm::isa<mem::malloc>(app)) {
143 auto [m, n] = malloc->args<2>();
144 auto size = val(n);
145 if (size->type() != w.type_i64()) n->blame("native backend: size `{}` of `mem.malloc` is no value", n).bail();
146 return call_c(malloc_, "malloc", app, {rewrite(m), size});
147 }
148
149 if (auto free = Axm::isa<mem::free>(app)) {
150 auto [m, p] = free->args<2>();
151 return call_c(free_, "free", app, {rewrite(m), rewrite(p)});
152 }
153
154 if (auto nat = Axm::isa<core::nat>(app); nat && is_value(nat->arg())) {
155 auto [a, b] = nat->args<2>([this](const Def* def) { return val(def); });
156 auto div = [&](core::div op) {
157 auto m = w.bot(w.call<core::M>(w.lit_nat_0()));
158 return w.call(op, Defs{m, w.tuple({a, b})})->proj(2, 1);
159 };
160 switch (nat.id()) {
161 case core::nat::add: return w.call(core::wrap::add, core::Mode::none, Defs{a, b});
162 case core::nat::mul: return w.call(core::wrap::mul, core::Mode::none, Defs{a, b});
163 case core::nat::sub: {
164 auto diff = w.call(core::wrap::sub, core::Mode::none, Defs{a, b});
165 return w.extract(w.tuple({diff, w.lit_i64(0)}), w.call(core::icmp::ul, Defs{a, b}));
166 }
167 case core::nat::div: return div(core::div::udiv);
168 case core::nat::rem: return div(core::div::urem);
169 }
170 }
171
172 if (auto ncmp = Axm::isa<core::ncmp>(app); ncmp && is_value(ncmp->arg())) {
173 auto [a, b] = ncmp->args<2>([this](const Def* def) { return val(def); });
174 return w.call(unsigned_cmp(ncmp.id()), Defs{a, b});
175 }
176
177 if (auto idx = Axm::isa<core::idx>(app); idx && is_value(idx->arg()))
178 return w.call(core::conv::u, rewrite(Idx::isa(idx->type())), val(idx->arg()));
179
180 if (auto i2n = Axm::isa<core::idx2nat>(app)) return core::zext(w, rewrite(i2n->arg()));
181
182 if (auto cast = Axm::isa<core::bitcast>(app)) {
183 auto type = rewrite(cast->type());
184 auto x = val(cast->arg());
185 return x->type() == type ? x : w.call<core::bitcast>(type, x);
186 }
187
188 return RWPhase::rewrite_imm_App(app);
189}
