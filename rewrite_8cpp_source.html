236const Def* Rewriter::rewrite_mut_Seq(Seq* seq) {
237 if (!seq->is_set()) return rewrite_stub(seq, world().mut_seq(seq->is_intro(), rewrite(seq->type())));

239 auto new_shape = rewrite(*seq->shape())->zonk();

291const Def* VarRewriter::rewrite(const Def* old_def) {
292 // Only Vars, mutables and non-ground immutables are ever mapped, so a ground immutable needs no lookup.
293 if (!old_def->isa_mut() && old_def->local_vars().empty() && old_def->local_muts().empty()) return old_def;

311const Def* Zonker::lookup(const Def* old_def) {
312 // Holes only ever get set, so a cached result goes stale once a Hole within it got set since.
313 // A self-map is a fixpoint, though: e.g. `binder#i` keeps a set Hole in its type until the binder is rewired.
314 if (auto new_def = fe::lookup(old2new(), old_def);
315 new_def && (new_def == old_def || new_def->isa_mut() || !new_def->needs_zonk()))
316 return new_def;
317 return nullptr;
318}

329const Def* Zonker::rewire_mut(Def* mut) {
330 map(mut, mut);
331
332 auto old_type = mut->type();
333 auto old_ops = DefVec(mut->ops().begin(), mut->ops().end());
334

336
