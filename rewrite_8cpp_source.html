240const Def* Rewriter::rewrite_mut_Seq(Seq* seq) {
241 if (seq->is_immutabilizable()) return rewrite_imm_Seq(seq);
242
243 if (!seq->is_set()) {
244 auto new_seq = seq->as_mut<Seq>()->stub(world(), rewrite(seq->type()));
245 return map(seq, new_seq);
246 }
247
248 auto new_arity = rewrite(seq->arity())->zonk();
249 auto l = Lit::isa(new_arity);
250 if (l && *l == 0) return world().prod(seq->is_intro());
251
252 if (auto var = seq->has_var(); var && l && *l <= world().flags().scalarize_threshold) {
253 auto new_ops = absl::FixedArray<const Def*>(*l);
254 for (size_t i = 0, e = *l; i != e; ++i) {
255 push();
256 map(var, world().lit_idx(e, i));
257 new_ops[i] = rewrite(seq->body());
258 pop();
259 }
260 return map(seq, world().prod(seq->is_intro(), new_ops));
261 }
262
263 if (!seq->has_var()) return map(seq, world().seq(seq->is_intro(), new_arity, rewrite(seq->body())));
264 return rewrite_stub(seq->as_mut(), world().mut_seq(seq->is_intro(), rewrite(seq->type())));
265}

313const Def* Zonker::lookup(const Def* old_def) {
314 for (auto& old2new : old2news_ | std::views::reverse) {
315 const Def* repr;
316 auto path = DefVec();
317 while (true) {
318 repr = get(old_def);
319
320 if (repr == nullptr) break;
321
322 path.emplace_back(repr);
323 if (repr == old_def) break; // explicit self-map
324
325 old_def = repr;
326 }
327
328 if (path.empty()) continue;
329
330 // path compression: flatten all visited nodes
331 for (auto def : path)
332 old2new[def] = repr;
333
334 return repr;
335 }
336
337 return nullptr;
338}

340const Def* Zonker::rewrite(const Def* def) {
341 if (auto hole = def->isa_mut<Hole>()) {
342 auto [last, op] = hole->find();
343 def = op ? op : last;

349const Def* Zonker::rewire_mut(Def* mut) {
350 map(mut, mut);
351
352 auto old_type = mut->type();
353 auto old_ops = absl::FixedArray<const Def*>(mut->ops().begin(), mut->ops().end());
354
355 mut->unset()->set_type(rewrite(old_type));
356
357 for (size_t i = 0, e = mut->num_ops(); i != e; ++i)
358 mut->set(i, rewrite(old_ops[i]));
359
360 if (auto new_imm = mut->immutabilize()) return map(mut, new_imm);
361
362 return mut;
363}
