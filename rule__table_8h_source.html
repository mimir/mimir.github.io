21/// What a rule wants of the term a meta variable matches; a Param is bound but no operand.
22/// A Lit is a Param that must be a literal.
23/// Reg is a general-purpose register, FReg a floating-point one, and Base/Index an address without/with an index.
24/// Addr is either address form, and Scaled an index with its scale.
25/// Imm is a literal that fits the backend's immediates, Sh a literal shift count, which the machine masks.
26/// RegImm and RegSh are a Reg, or an Imm and a Sh for a literal that is one.
27enum class Want : u8 {
28#define X(w) w,
29 MIM_NATIVE_RULE_WANTS(X)
30#undef X
31};

33#define MIM_NATIVE_RULE_TYS(X) X(Reg) X(FReg) X(Base) X(Index) X(Scaled) X(Lit)

135/// Does a value of non-terminal @p ty fill a hole of type @p want as it is?
136constexpr bool fills(Ty ty, Want want) {
137 switch (want) {
138 case Want::Reg:
139 case Want::RegImm:
140 case Want::RegSh: return ty == Ty::Reg;
141 case Want::FReg: return ty == Ty::FReg;
142 case Want::Base: return ty == Ty::Base;
143 case Want::Index: return ty == Ty::Index;
144 case Want::Addr: return ty == Ty::Base || ty == Ty::Index;
145 case Want::Scaled: return ty == Ty::Scaled;
146 default: return false;
147 }
148}

150/// The register hole that a value of non-terminal @p ty fills.
151constexpr Want reg_want(Ty ty) { return ty == Ty::FReg ? Want::FReg : Want::Reg; }

153/// Is a value of non-terminal @p ty in the same register class as a hole of type @p want?
154/// Conversions never cross between general-purpose and floating-point registers.
155constexpr bool same_class(Ty ty, Want want) {
156 if (want == Want::FReg) return ty == Ty::FReg;
157 return ty != Ty::FReg;
158}

160/// The cheapest chains of conversions between two non-terminals: their costs and each one's first conversion.
161struct Chains {
162 std::array<std::array<u32, Num_Tys>, Num_Tys> cost{};
163 std::array<std::array<u32, Num_Tys>, Num_Tys> first{};
164 /// The non-terminals of the backend: Reg, Lit, and those its conversions mention.
165 std::array<bool, Num_Tys> used{};
166
167 /// The closure of @p conversions, a range of Conversion-like rules with `from`, `to`, and `cost`.
168 constexpr Chains(const auto& conversions) {
169 for (size_t i = 0; i != Num_Tys; ++i)
170 for (size_t j = 0; j != Num_Tys; ++j)
171 cost[i][j] = i == j ? 0 : Inf, first[i][j] = Inf;
172 used[size_t(Ty::Reg)] = used[size_t(Ty::Lit)] = true;
173 u32 n = 0;
174 for (const auto& c : conversions) {
175 used[size_t(c.from)] = used[size_t(c.to)] = true;
176 if (auto& d = cost[size_t(c.from)][size_t(c.to)]; u32(c.cost) < d)
177 d = u32(c.cost), first[size_t(c.from)][size_t(c.to)] = n;
178 ++n;
179 }
180 for (size_t k = 0; k != Num_Tys; ++k)
181 for (size_t i = 0; i != Num_Tys; ++i)
182 for (size_t j = 0; j != Num_Tys; ++j)
183 if (auto c = add(cost[i][k], cost[k][j]); c < cost[i][j]) cost[i][j] = c, first[i][j] = first[i][k];
184 }
185
186 /// The non-terminal that @p from converts to most cheaply for a hole of type @p want, and the cost.
187 constexpr std::pair<Ty, u32> target(Ty from, Want want) const {
188 auto res = std::pair(from, Inf);
189 for (size_t t = 0; t != Num_Tys; ++t)
190 if (auto c = cost[size_t(from)][t]; fills(Ty(t), want) && c < res.second) res = {Ty(t), c};
191 return res;
192 }
193
194 /// Cost of converting a replacement or operand of non-terminal @p from into a hole of type @p want.
195 constexpr u32 conv(Ty from, Want want) const {
196 if (is_literal(want)) return from == Ty::Lit ? 0 : Inf;
197 return target(from, want).second;
198 }
199
200 /// Does every non-terminal of @p w's register class convert into a hole of type @p w, as an atomic rule's holes
201 /// must?
202 constexpr bool total(Want w) const {
203 for (size_t t = 0; t != Num_Tys; ++t)
204 if (used[t] && same_class(Ty(t), w) && conv(Ty(t), w) == Inf) return false;
205 return true;
206 }
207
208 /// Does some non-terminal of the register class of a rule convert into a hole of type @p w?
209 constexpr bool reachable(Want w) const {
210 for (size_t t = 0; t != Num_Tys; ++t)
211 if (Ty(t) != Ty::Lit && same_class(Ty(t), w) && conv(Ty(t), w) != Inf) return true;
212 return false;
213 }
214};
