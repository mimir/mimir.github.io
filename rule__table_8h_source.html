20/// What a rule wants of the term a meta variable matches; a Param is bound but no operand.
21/// A Lit is a Param that must be a literal.
22/// Reg is a general-purpose register, FReg a floating-point one, and Base/Index an `x86.Base`/`x86.Index`.
23/// Addr is an `x86.Addr i` for any `i`, and Scaled an index with its scale.
24/// Imm is a literal that fits a sign-extended `i32` immediate, Sh a literal shift count, which x86 masks.
25/// RegImm and RegSh are a Reg, or an Imm and a Sh for a literal that is one.
26enum class Want : u8 {
27#define X(w) w,
28 MIM_X86_RULE_WANTS(X)
29#undef X
30};

131/// Does a value of non-terminal @p ty fill a hole of type @p want as it is?
132constexpr bool fills(Ty ty, Want want) {
133 switch (want) {
134 case Want::Reg:
135 case Want::RegImm:
136 case Want::RegSh: return ty == Ty::Reg;
137 case Want::FReg: return ty == Ty::FReg;
138 case Want::Base: return ty == Ty::Base;
139 case Want::Index: return ty == Ty::Index;
140 case Want::Addr: return ty == Ty::Base || ty == Ty::Index;
141 case Want::Scaled: return ty == Ty::Scaled;
142 default: return false;
143 }
144}

146/// The register hole that a value of non-terminal @p ty fills.

149/// Is a value of non-terminal @p ty in the same register class as a hole of type @p want?
150/// Conversions never cross between general-purpose and floating-point registers.
151constexpr bool same_class(Ty ty, Want want) {
152 if (want == Want::FReg) return ty == Ty::FReg;

156/// The cheapest chains of conversions between two non-terminals: their costs and each one's first conversion.
157struct Chains {
158 std::array<std::array<u32, Num_Tys>, Num_Tys> cost{};
159 std::array<std::array<u32, Num_Tys>, Num_Tys> first{};
160
161 /// The closure of @p conversions, a range of Conversion-like rules with `from`, `to`, and `cost`.
162 constexpr Chains(const auto& conversions) {
163 for (size_t i = 0; i != Num_Tys; ++i)
164 for (size_t j = 0; j != Num_Tys; ++j)
165 cost[i][j] = i == j ? 0 : Inf, first[i][j] = Inf;
166 u32 n = 0;
167 for (const auto& c : conversions) {
168 if (auto& d = cost[size_t(c.from)][size_t(c.to)]; u32(c.cost) < d)
169 d = u32(c.cost), first[size_t(c.from)][size_t(c.to)] = n;
170 ++n;
171 }
172 for (size_t k = 0; k != Num_Tys; ++k)
173 for (size_t i = 0; i != Num_Tys; ++i)
174 for (size_t j = 0; j != Num_Tys; ++j)
175 if (auto c = add(cost[i][k], cost[k][j]); c < cost[i][j]) cost[i][j] = c, first[i][j] = first[i][k];
176 }
177
178 /// The non-terminal that @p from converts to most cheaply for a hole of type @p want, and the cost.
179 constexpr std::pair<Ty, u32> target(Ty from, Want want) const {
180 auto res = std::pair(from, Inf);
181 for (size_t t = 0; t != Num_Tys; ++t)
182 if (auto c = cost[size_t(from)][t]; fills(Ty(t), want) && c < res.second) res = {Ty(t), c};
183 return res;
184 }
185
186 /// Cost of converting a replacement or operand of non-terminal @p from into a hole of type @p want.
187 constexpr u32 conv(Ty from, Want want) const {
188 if (is_literal(want)) return from == Ty::Lit ? 0 : Inf;
189 return target(from, want).second;
190 }
191
192 /// Does every non-terminal of @p w's register class convert into a hole of type @p w, as an atomic rule's holes
193 /// must?
194 constexpr bool total(Want w) const {
195 for (size_t t = 0; t != Num_Tys; ++t)
196 if (same_class(Ty(t), w) && conv(Ty(t), w) == Inf) return false;
197 return true;
198 }
199
200 /// Does some non-terminal of the register class of a rule convert into a hole of type @p w?
201 constexpr bool reachable(Want w) const {
202 for (size_t t = 0; t != Num_Tys; ++t)
203 if (Ty(t) != Ty::Lit && same_class(Ty(t), w) && conv(Ty(t), w) != Inf) return true;
204 return false;
205 }
206};
