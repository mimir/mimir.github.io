7/// A term-level, direct-style function that we can and want to convert.

9static bool convertible(Lam* lam) {
10 if (!lam->is_set() || lam->is_external() || lam->is_annex() || Lam::isa_cn(lam)) return false;
11 auto codom = lam->codom();
12 return !codom->isa<Type>() && !codom->isa<Pi>();

33const Def* Conv::convert(Lam* old_lam) {
34 auto& w = new_world();
35 auto old_pi = old_lam->type();
36 auto new_dom = rewrite(old_pi->dom());
37
38 Lam* new_lam;
39 if (auto old_var = old_pi->has_var()) {
40 // dependent codom: bind it against a fresh mutable Sigma [a: A, Cn B[old_var → a]]
41 auto sigma = w.mut_sigma(2);
42 sigma->var();
43 scoped_ = w.vars().insert(scoped_, sigma->has_var());
44 push();
45 map(old_var, sigma->var(2, 0));
46 auto new_codom = rewrite(old_pi->codom());
47 pop();
48 sigma->set(0, new_dom);
49 sigma->set(1, w.cn(new_codom));
50 new_lam = w.mut_con(sigma);
51 } else {
52 new_lam = w.mut_fun(new_dom, rewrite(old_pi->codom()));
53 }
54 new_lam->set(old_lam->dbg_key())->debug_suffix("_cps");
55
56 auto [param, ret] = new_lam->vars<2>();
57 map(old_lam->var(), param);
58 auto wrapper = map(old_lam, op_cps2ds_dep(new_lam));
59
60 DLOG("convert {}: {} ↝ {}: {}", old_lam, old_pi, new_lam, new_lam->type());
61
62 auto scope = Scope(*this, true);
63 auto new_filter = rewrite(old_lam->filter());
64 auto tail = w.app(ret, rewrite(old_lam->body()));
65 new_lam->set(new_filter, wire(scope.base(), tail));
66
67 return wrapper;
68}

93const Def* Conv::wire(size_t base, const Def* body) {
