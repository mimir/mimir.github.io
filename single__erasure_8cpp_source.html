25size_t SingleErasure::remap(const Sieve& keep, const Def* index) {
26 auto i = Lit::isa(index);
27 if (!i)
28 index->blame("index is not a literal")
29 .n("dropping an information-free component shifts the indices of its successors")
30 .bail();
31
32 auto j = keep[*i];
33 assert(j != Sieve::Gone && "an Extract/Insert of a dropped component is erased before it is remapped");
34 return j;
35}
