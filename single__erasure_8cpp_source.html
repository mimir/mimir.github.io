9/// Collects @p def%'s immutable subtree into @p set; stops at mutables.
10static void collect(DefSet& set, const Def* def) {
11 if (!set.emplace(def).second) return;
12 if (def->isa_mut()) return;
13 for (auto d : def->deps())
14 collect(set, d);
15}

17const Def* SingleErasure::Analysis::rewrite(const Def* old) {
18 // Visit the subtree *before* inspecting: pin() seeds `def ↦ def` into the rewriter map, which would
19 // short-circuit the traversal into `old` and skip its subtree for the rest of the round.
20 auto res = mim::Analysis::rewrite(old);
21
22 if (auto app = old->isa<App>()) {
23 // Rebuilding an Axm application re-derives its shapes from the Axm's generic type instead of rewriting
24 // them (`Idx n_groups` becomes `Idx 1` all over again), so whatever its signature dictates must keep its
25 // arity. Subtrees merely *substituted* in via earlier (type) arguments impose no shape, though - they
26 // rewrite consistently with the rest of the World: seed the walk's visited set with them.
27 if (app->uncurry_callee()->isa<Axm>()) {
28 auto skips = DefSet();
29 for (auto d = old; auto a = d->isa<App>(); d = a->callee())
30 collect(skips, a->arg());
31 pin_tree(app->callee_type()->dom(), skips);
32 pin_tree(app->type(), skips);
33 }
34
35 // Assignability is alpha-equivalence, so an App may connect a dom and an arg whose types are *distinct*
36 // defs - a dependent dom and its instance (`[n: Nat, «n; *»]` vs `[Nat, []]`). Dropping decides per def,
37 // which would tear such an edge apart: the instance loses what the dependent dom still expects.
38 if (auto dom = app->callee_type()->dom(); dom != app->arg()->type()) {
39 auto visited = DefSet();
40 pin_tree(dom, visited);
41 pin_tree(app->arg()->type(), visited);
42 }
43 }
44
45 return res;
46}

48void SingleErasure::Analysis::pin_tree(const Def* def, DefSet& visited) {
49 if (!visited.emplace(def).second) return;
50 if (def->isa<Sigma>() || def->isa<Arr>()) pin(def);
51 for (auto d : def->deps())
52 pin_tree(d, visited);
53}

59bool SingleErasure::is_gone(const Def* type) {
60 if (type->isa<Single>()) return true;
61 if (Idx::isa_lit(type) == 1) return true;
62 if (analysis_.pinned(type)) return false;
63 // An immutable aggregate is information-free iff all of its components are; this also covers `[]`.
64 if (auto sigma = type->isa_imm<Sigma>())
65 return std::ranges::all_of(sigma->ops(), [this](const Def* op) { return is_gone(op); });
66 if (auto arr = type->isa_imm<Arr>()) return is_gone(arr->body());
67 return false;
68}

70Sieve SingleErasure::sieve(const Sigma* sigma) {
71 // A foreign declaration has no components to inspect; a pinned one must keep its arity.
72 if (!sigma->is_set() || analysis_.pinned(sigma)) return Sieve(sigma->num_ops());
73 return Sieve(sigma->ops(), [this](const Def* op) { return !is_gone(op); });
74}

87size_t SingleErasure::remap(const Sieve& keep, const Def* index) {
88 auto i = Lit::isa(index);
89 if (!i)
90 index->blame("index is not a literal")
91 .n("dropping an information-free component shifts the indices of its successors")
92 .bail();
93
94 auto j = keep[*i];
95 assert(j != Sieve::Gone && "an Extract/Insert of a dropped component is erased before it is remapped");
96 return j;

127const Def* SingleErasure::rewrite_imm_Tuple(const Tuple* tuple) {
128 if (is_gone(tuple->type())) return inhabitant(tuple->type());

130 // Drive the drop by the *type*: a dependent component may reduce to a gone type while its Sigma op does not.

167const Def* SingleErasure::rewrite_mut_Seq(Seq* seq) {
168 // A mutable Seq has no immutable type to ask, so inspect its body directly.
169 if (seq->is_set() && !analysis_.pinned(seq->is_intro() ? seq->type() : seq)
170 && is_gone(seq->is_intro() ? seq->body()->type() : seq->body()))
