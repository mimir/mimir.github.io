19/// The curried meta arguments of a `tensor.map_reduce_post` or `btensor.map_reduce_post` callee (same layout).
20struct MapRedPost {
21 const Def *nis_nps, *meta, *shapes, *in_tys, *comb_init, *map_out, *maps_all;
22 const Def *nis, *nps, *To, *Tp, *Ro, *Rn, *TSched, *So, *Sr, *sched;
23 const Def *Tis, *Ris, *Sis, *Tps, *Rps, *Sps, *comb, *init, *post, *maps, *post_maps;

25 static MapRedPost of(const App* callee) {
26 auto [nis_nps, meta, shapes, in_tys, comb_init, map_out, maps_all] = callee->uncurry_args<7>();
27 auto [nis, nps] = nis_nps->projs<2>();
28 auto [To, Tp, Ro, Rn, TSched] = meta->projs<5>();
29 auto [So, Sr, sched] = shapes->projs<3>();
30 auto [Tis, Ris, Sis, Tps, Rps, Sps] = in_tys->projs<6>();
31 auto [comb, init, post] = comb_init->projs<3>();
32 auto [maps, post_maps] = maps_all->projs<2>();
33 return {nis_nps, meta, shapes, in_tys, comb_init, map_out, maps_all, nis, nps, To,
34 Tp, Ro, Rn, TSched, So, Sr, sched, Tis, Ris, Sis,
35 Tps, Rps, Sps, comb, init, post, maps, post_maps};
36 }

40 u64 nloops() const { return ro + rr; }

43 /// The literal input counts and loop ranks; `std::nullopt` unless all are literal and `Rn >= Ro`.
44 std::optional<Ranks> ranks() const {
45 auto nis_l = Lit::isa<u64>(nis);
46 auto nps_l = Lit::isa<u64>(nps);
47 auto ro_l = Lit::isa<u64>(Ro);
48 auto rn_l = Lit::isa<u64>(Rn);
49 if (!nis_l || !nps_l || !ro_l || !rn_l || *rn_l < *ro_l) return {};
50 return Ranks{*nis_l, *nps_l, *ro_l, *rn_l - *ro_l};
51 }
52};

54/// `affine.map @(m, n) @(s_in, s_out) f idxs as mem` with the address space `as` of @p mem ↦ `[mem', coords]`.
55inline const Def* op_affine_map(const Def* f,
56 const Def* m,
57 const Def* n,
58 const Def* s_in,
59 const Def* s_out,
60 const Def* idxs,
61 const Def* mem) {
62 auto& w = mem->world();
63 auto a = w.app(w.annex<affine::map>(), Defs{m, n});
64 a = w.app(a, Defs{s_in, s_out});
65 a = w.app(a, f);
66 a = w.app(a, idxs);
67 a = w.app(a, mem->type()->as<App>()->arg());
68 return w.app(a, mem);
69}

71/// Calls the combiner/epilogue @p f from @p caller with @p args and the continuation @p ret.
72/// Scalarize may have flattened @p f's domain `Cn [[args], Cn ret]` to `Cn [args…, Cn ret]`; the call matches either.
73inline void app_cps(Lam* caller, const Def* f, DefVec args, const Def* ret) {
74 auto& w = caller->world();
75 if (f->type()->as<Pi>()->dom()->num_projs() == args.size() + 1) {
76 args.emplace_back(ret);
77 caller->app(true, f, w.tuple(args));
78 } else {
79 caller->app(true, f, w.tuple({w.tuple(args), ret}));
80 }
81}

83/// A fresh `fun: Fn arg->type() → codom` named @p name, spliced into direct style as `cps.cps2ds_dep fun arg`.
84/// @returns `{fun, call}`.
85inline std::pair<Lam*, const Def*> ds_splice(const Def* arg, const Def* codom, std::string name) {
86 auto& w = arg->world();

109/// The read coordinates `o#d − lo#d` of a `pad` into `Idx s_in#d` - clamped to the nearest edge if @p replicate, or
110/// else zeroed out of range - together with the all-axes in-range flag (`nullptr` if @p replicate).
111inline std::pair<DefVec, const Def*> pad_coords(Defs iters, const Def* lo, const Def* s_in, bool replicate) {
112 auto& w = lo->world();
113 auto i64 = w.type_i64();
114 auto rn = iters.size();
115 DefVec coords(rn);
116 DefVec valid;
117 for (u64 d = 0; d < rn; ++d) {
118 auto lo_d = w.call<core::bitcast>(i64, lo->proj(rn, d));
119 auto sin_d = w.call<core::bitcast>(i64, s_in->proj(rn, d));
120 auto in_d = w.call(core::wrap::sub, core::Mode::none, Defs{iters[d], lo_d});
121 const Def* idx_i64;
122 if (replicate) {
123 idx_i64 = clamp(in_d, w.call(core::wrap::sub, core::Mode::none, Defs{sin_d, w.lit_i64(1)}));
124 } else { // a single unsigned `<` covers both bounds since underflow wraps high
125 auto v_d = w.call(core::icmp::ul, w.tuple({in_d, sin_d}));
126 valid.push_back(v_d);
127 idx_i64 = w.select(v_d, in_d, w.lit_i64(0));
128 }
129 coords[d] = w.call(core::conv::u, s_in->proj(rn, d), idx_i64);
130 }
131 if (replicate) return {coords, nullptr};
132 auto all_valid = valid.empty() ? w.lit_tt() : valid[0];
133 for (u64 d = 1; d < valid.size(); ++d)
134 all_valid = w.call(core::bit2::and_, w.lit_nat(2), w.tuple({all_valid, valid[d]}));
135 return {coords, all_valid};
136}

138/// The literal extents of the `nis` input shapes @p Sis along the concat axis @p ax; `std::nullopt` if one is not
139/// literal.
140inline std::optional<fe::Vector<u64>> concat_extents(const Def* Sis, u64 nis, u64 r, u64 ax) {
141 auto exts = fe::Vector<u64>(nis);
142 for (u64 i = 0; i < nis; ++i)
143 if (auto e = Lit::isa<u64>(Sis->proj(nis, i)->proj(r, ax)))
144 exts[i] = *e;

150/// The concat cell at the i64 axis coordinate @p o_ax: `read(i, loc)` of the input `i` owning it - the highest one
151/// whose prefix offset is `≤ o_ax` - at the axis coordinate `loc` clamped into that input.
152inline const Def* concat_select(const Def* o_ax, const fe::Vector<u64>& exts, auto read) {
153 auto& w = o_ax->world();
154 auto off = u64(0);
155 auto read_i = [&](u64 i) {
156 auto loc = w.call(core::wrap::sub, core::Mode::none, Defs{o_ax, w.lit_i64(off)});
157 return read(i, clamp(loc, w.lit_i64(exts[i] - 1)));
158 };
159 auto result = read_i(0);
160 for (u64 i = 1; i < exts.size(); ++i) {
161 off += exts[i - 1];
162 auto cond = w.call(core::icmp::uge, w.tuple({o_ax, w.lit_i64(off)}));
163 result = w.select(cond, read_i(i), result);
164 }
165 return result;
166}

168/// A `[mem, dst, ins] → [mem, Buf]` fun spliced via `cps.cps2ds_dep` that loops over @p s_out carrying
169/// `[mem, dst]` and writes `compute`'s element at the loop's (identity) output coordinates.
170/// `compute(iters, ins, mem)` receives the raw i64 loop counters, the fun's inputs var, and the current mem; it
171/// returns `{mem', elem}`.
172inline const Def* build_pointwise(const Def* result_ty, // [mem.M 0, buffer.Buf (r, s_out, T)]
173 const Def* mem,
174 const Def* dst,
175 const Def* ins,
176 const Def* s_out,
177 u64 rn,
178 const std::string& name,
179 auto compute) {
180 auto& w = mem->world();
181 auto [fun, call] = ds_splice(w.tuple({mem, dst, ins}), result_ty, name);
182 auto [fun_mem, out_buf, fun_ins] = fun->var(2, 0)->projs<3>();
183 const Def* cont = fun->var(2, 1);
184 const Def* acc = w.tuple({fun_mem, out_buf});
185 auto cur = fun;
186 auto iters = affine::counting_nest(cur, cont, acc, s_out->projs(rn), name);
187 auto [loop_mem, loop_buf] = acc->projs<2>();
188
189 std::pair<const Def*, const Def*> el = compute(iters, fun_ins, loop_mem);
190 auto [el_mem, elem] = el;
191 auto wr_mem = buffer::op_write(el_mem, loop_buf, folded_idx(s_out, iters), elem)->proj(2, 0);
192 cur->app(true, cont, w.tuple({wr_mem, loop_buf}));
193 return call;
194}
