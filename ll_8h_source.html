24// Lessons learned:
25// * **Always** follow all ops - even if you actually want to ignore one.
26// Otherwise, you might end up with an incorrect schedule.
27// This was the case for an Extract of type Mem.
28// While we want to ignore the value obtained from that, since there is no Mem value in LLVM,
29// we still want to **first** recursively emit code for its operands and **then** ignore the Extract itself.
30// * i1 has a different meaning in LLVM then in Mim:
31// * Mim: {0, 1} = i1
32// * LLVM: {0, -1} = i1
33// This is a problem when, e.g., using an index of type i1 as LLVM thinks like this:
34// getelementptr ..., i1 1 == getelementptr .., i1 -1

261inline static std::optional<std::pair<nat_t, const Def*>> is_simd(const Def* type) {
262 if (auto arr = type->isa<Arr>()) {
263 if (auto l = Lit::isa(arr->arity())) {
264 if (arr->body()->isa<Nat>() || Idx::isa(arr->body()) || Axm::isa<math::F>(arr->body()))
265 return std::pair{*l, arr->body()};
266 }
267 }
268 return {};
269}

271inline static std::optional<std::pair<nat_t, const Def*>> is_simd_aggregate(Defs types) {
272 if (std::ranges::all_of(types, [&](auto i) { return i == types[0]; })) {
273 if (types[0]->isa<Nat>() || Idx::isa(types[0]) || Axm::isa<math::F>(types[0]))
274 return std::pair{types.size(), types[0]};
275 }
276
277 return {};

280inline static const Def* find_common_simd_src(const App* app) {
281 const Def* common_src = nullptr;
282 size_t lane = 0;
283 for (auto arg : app->args()) {
284 if (Axm::isa<mem::M>(arg->type())) continue;
285 auto extract = arg->isa<Extract>();
286 if (!extract || !is_simd(extract->tuple()->type())) return nullptr;
287 // Only devectorized args - lane i in position i - may forward the whole vector;
288 // anything else (e.g. a Select with a non-literal index) must stay scalar.
289 if (auto index = Lit::isa(extract->index()); !index || *index != lane++) return nullptr;
290 if (!common_src)
291 common_src = extract->tuple();
292 else if (common_src != extract->tuple())
293 return nullptr;
294 }
295 if (common_src) {
296 auto simd = is_simd(common_src->type());
297 if (!simd || simd->first != lane) return nullptr;
298 }
299 return common_src;
300}

346inline bool Emitter::load_rt_module(std::string_view filename) {
347 for (const auto& dir : world().driver().search_paths()) {
348 auto path = dir / "rt" / std::string(filename);
349 std::error_code ec;
350 if (!std::filesystem::is_regular_file(path, ec) || ec) continue;
351 if (auto ifs = std::ifstream(path)) {
352 world().DLOG("ll backend: loaded runtime module `{}`", path.string());
353 rt_module_.assign(std::istreambuf_iterator<char>(ifs), std::istreambuf_iterator<char>());
354 return true;
355 }
356 }
357 return false;

374inline std::string Emitter::prepare() {
375 auto internal = root()->is_external() ? "" : "internal ";
376 auto ret_t = convert_ret_pi(root()->type()->ret_pi());
377 std::print(func_impls_, "define {} {} {}(", internal, ret_t, id(root()));
378
379 auto vars = root()->vars();
380 for (auto sep = ""; auto var : vars.view().rsubspan(1)) {
381 if (Axm::isa<mem::M>(var->type())) continue;
382 if (auto sigma = var->type()->isa<Sigma>(); sigma && sigma->num_ops() == 0) continue;
383 if (auto arr = var->type()->isa<Arr>())
384 if (is_simd(arr->body())) convert(arr->body()); // pre-add input vector to cache
385 auto name = id(var);
386 locals_[var] = name;
387 std::print(func_impls_, "{}{} {}", sep, convert(var->type()), name);
388 sep = ", ";
389 }
390
391 std::print(func_impls_, ") {{\n");
392 return root()->unique_name();
393}
