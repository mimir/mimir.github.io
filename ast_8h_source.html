29/// Visibility tier of a ValDecl.
30/// `Priv` is only visible inside its enclosing `mod`; `Pub` is visible via a path/`use` from outside.
31/// Orthogonal to this, a ValDecl may independently be `extern` (Decl::is_extern) and/or `anx` (Decl::is_anx);
32/// either one nudges the default visibility to `Pub` unless `priv` is given explicitly.
33enum class Vis { Priv, Pub };
34
35/// Raw, unvalidated combination of `priv`/`pub`/`extern`/`anx` modifiers written before a declaration.
36/// Parser::parse_modifiers only rejects a modifier being repeated (`priv priv`, `extern extern`, ...);
37/// whether a given combination makes sense for the decl that follows is up to that decl's own parser.
38struct Mods {
39 std::optional<Vis> vis = {};
40 bool is_extern = false;
41 bool is_anx = false;
42
43 /// `extern`/`anx` nudge the default visibility to `Pub` unless `priv` is given explicitly.
44 Vis default_vis() const { return (is_extern || is_anx) ? Vis::Pub : Vis::Priv; }
45 Vis resolved_vis() const { return vis.value_or(default_vis()); }
46};
