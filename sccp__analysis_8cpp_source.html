11const Def* SCCP::Analysis::propagate(const Def* var, const Def* def) {
12 // `⊥ ⊔ x` is `x`, but unusable if lam nests it.
13 if (lam_of(var)->nests(def)) return pin(var), var;
14
15 auto cur = lattice(var);
16 if (!cur) { // ⊥ ⊔ def = def; lattice(var, def) invalidates, as it inserts a fresh non-⊤ fact
17 lattice(var, def);
18 log().d("propagate {} → {}", var, def);
19 return def;
20 }
21
22 if (def->isa<Bot>() || cur == def || cur == var) return cur; // cur ⊔ ⊥ = cur ⊔ cur = cur; ⊤ stays ⊤
23
24 if (cur->isa<Bot>()) { // ⊥ ⊔ def = def; lattice(var, def) invalidates, as it overwrites cur
25 lattice(var, def);
26 return def;
27 }
28
29 return pin(var), var; // two different values join to ⊤; lattice(var, var) therein invalidates, as it overwrites cur
30}
