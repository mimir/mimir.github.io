39/// @name Plugin Argument Lookup
40/// Picks the `-X <plugin>:<arg>` strings of Driver::args / Phase::args apart.
41/// Each helper matches any of @p keys - `arg_value(args(), "o", "output")` - and the last occurrence wins.
42///@{
43namespace detail {
44/// `<key>` ↦ `""`, `<key>=<value>` ↦ `<value>`, anything else ↦ `std::nullopt`.
45inline std::optional<std::string_view> arg_split(std::string_view arg, std::string_view key) {
46 if (!arg.starts_with(key)) return {};
47 auto val = arg.substr(key.size());
48 if (val.empty()) return val;
49 if (val.front() == '=') return val.substr(1);
50 return {};
51}
52} // namespace detail
53
54/// Value of `<key>=<value>`; `std::nullopt` if none of @p keys carries one.
55template<class... Keys>
56std::optional<std::string_view> arg_value(fe::View<std::string> args, Keys... keys) {
57 std::optional<std::string_view> res;
58 for (std::string_view arg : args)
59 for (std::string_view key : {std::string_view(keys)...})
60 if (auto val = detail::arg_split(arg, key); val && !val->empty()) res = val;
61 return res;
62}

64/// An @p on key ↦ `true`, an @p off key ↦ `false`; `std::nullopt` if neither occurs.
65inline std::optional<bool> arg_bool(fe::View<std::string> args,
66 std::initializer_list<std::string_view> on,
67 std::initializer_list<std::string_view> off) {
68 std::optional<bool> res;
69 for (std::string_view arg : args) {
70 for (auto key : on)
71 if (arg == key) res = true;
72 for (auto key : off)
73 if (arg == key) res = false;
74 }
75 return res;
76}

94 /// Compares major/minor/suffix, ignores hash.
95 constexpr auto operator<=>(const Version& other) const noexcept {
96 auto cmp = std::tie(major, minor) <=> std::tie(other.major, other.minor);
97 if (cmp != 0) return cmp;

102 /// Compares major/minor/suffix, ignores hash.

104 return major == other.major && minor == other.minor && std::strcmp(suffix, other.suffix) == 0;
105 }

108 return os << v.major << '.' << v.minor << v.suffix << " (" << v.hash << ")";

148 /// @name Mangling Plugin Name
149 ///@{
150 static constexpr size_t Max_Plugin_Size = 8;
151 static constexpr plugin_t Global_Plugin = 0xffff'ffff'ffff'0000_u64;
152
153 /// Mangles @p s into a dense 48-bit representation.
154 /// The layout is as follows:
155 /// ```
156 /// |---7--||---6--||---5--||---4--||---3--||---2--||---1--||---0--|
157 /// 7654321076543210765432107654321076543210765432107654321076543210
158 /// Char67Char66Char65Char64Char63Char62Char61Char60|---reserved---|
159 /// ```
160 /// The `reserved` part is used for the Axm::tag and the Axm::sub.
161 /// Each `Char6x` is 6-bit wide and hence a plugin name has at most Axm::Max_Plugin_Size = 8 chars.
162 /// It uses this encoding:
163 /// | `Char6` | ASCII |
164 /// |---------|---------|
165 /// | 1: | `_` |
166 /// | 2-27: | `a`-`z` |
167 /// | 28-53: | `A`-`Z` |
168 /// | 54-63: | `0`-`9` |
169 /// The 0 is special and marks the end of the name if the name has less than 8 chars.
170 /// @returns `std::nullopt` if encoding is not possible.
171 static std::optional<plugin_t> mangle(std::string_view plugin);
172
173 /// Reverts an Axm::mangle%d @p plugin back to its name; never longer than Annex::Max_Plugin_Size.
174 /// Ignores lower 16-bit of @p plugin.
175 static std::string demangle(plugin_t plugin);
176
177 static std::tuple<Sym, Sym, Sym> split(Driver&, Sym);
178 ///@}
179
180 /// @name Annex Name
181 /// @anchor annex_name
182 /// Anatomy of an Annex name:
183 /// ```
184 /// %plugin.tag.sub
185 /// | 48 | 8 | 8 | <-- Number of bits per field.
186 /// ```
187 /// * Def::name() retrieves the full name as Sym.
188 /// * Def::flags() retrieves the full name as Axm::mangle%d 64-bit integer.
189 ///@{
190 /// Yields the `plugin` part of the name as integer.
191 /// It consists of 48 relevant bits that are returned in the highest 6 bytes of a 64-bit integer.
192 static constexpr plugin_t flags2plugin(flags_t f) { return f & Global_Plugin; }
193
194 /// Yields the `tag` part of the name as integer.
195 static constexpr tag_t flags2tag(flags_t f) { return tag_t((f & 0x0000'0000'0000'ff00_u64) >> 8_u64); }
196
197 /// Yields the `sub` part of the name as integer.
198 static constexpr sub_t flags2sub(flags_t f) { return sub_t(f & 0x0000'0000'0000'00ff_u64); }
199
200 /// Includes Axm::plugin() and Axm::tag() but **not** Axm::sub.
201 static constexpr flags_t flags2base(flags_t f) { return f & ~0xff_u64; }
202

204 static constexpr flags_t flags(plugin_t p, tag_t t, sub_t s = 0) { return p | (flags_t(t) << 8_u64) | flags_t(s); }
205 ///@}
