56const Def* SEO::Analysis::sccp_join(const Def* var, const Def* def) {
57 DLOG("sccp_join({}, {})", var, def);
58
59 // Pin %mem.M-typed vars to top: mem must stay threaded through every lam,
60 // as later stages (clos conversion, ll backend) rely on each lam having its own mem var.
61 if (Axm::isa<mem::M>(var->type())) return pin_top(var);
62
63 // `⊥ ⊔ x` is `x`, but unusable if lam nests it.
64 if (!def->isa<Proxy>() && lam_of(var)->nests(def)) {
65 DLOG("cannot propagate {} -> {}: out of scope", var, def);
66 return pin_top(var);
67 }
68
69 auto cur = lattice(var);
70 if (!cur) { // ⊥ ⊔ def = def; update() invalidates, as it inserts a fresh fact
71 update(var, def);
72 DLOG("propagate: {} -> {}", var, def);
73 return def;
74 }
75
76 if (def->isa<Bot>() || cur == def || cur == var || Proxy::isa<Proxy_GVN>(cur) || Proxy::isa<Proxy_SCCP_Top>(cur))
77 return cur;
78
79 DLOG("cannot propagate {} -> {}, trying GVN", var, def);
80 // update() invalidates, as it overwrites cur in both cases.
81 if (cur->isa<Bot>()) { // Bot ⊔ def = def
82 update(var, def);
83 return def;
84 }
85 auto top = mk_top(var); // we reached top for propagate; the Proxy_Top marks this to bundle for GVN
86 update(var, top);
87 return top;
88}

108void SEO::Analysis::gvn_bundle(Defs vars, Defs abstr_args, Span<const Def*> abstr_vars) {
109 auto n_all = vars.size();
110 for (size_t i = 0; i != n_all; ++i) {
111 if (!Proxy::isa<Proxy_SCCP_Top>(abstr_vars[i])) continue;
112
113 auto bundle_vars = DefVec();
114 bundle_vars.emplace_back(vars[i]);
115
116 for (size_t j = i + 1; j != n_all; ++j)
117 if (Proxy::isa<Proxy_SCCP_Top>(abstr_vars[j]) && abstr_args[j] == abstr_args[i])
118 bundle_vars.emplace_back(vars[j]);
119
120 if (bundle_vars.size() == 1) {
121 abstr_vars[i] = pin_top(vars[i]);
122 } else {
123 auto bundle = mk_bundle(vars[i], bundle_vars);
124
125 for (auto p : bundle->ops()) {
126 auto j = idx_of(vars, p);
127 update(vars[j], abstr_vars[j] = bundle);
128 }
129
130 DLOG("bundle: {}", bundle);
131 }
132 }

135void SEO::Analysis::gvn_split(Defs vars, Span<const Def*> abstr_args, Span<const Def*> abstr_vars) {
136 // E.g.: Say we started with `{a, b, c, d, e}` as a single bundle for all tvars of `lam`.
137 // Now, we see `lam (x, y, x, y, z)`. Then we have to build:
138 // a -> {a, c}
139 // b -> {b, d}
140 // c -> {a, c}
141 // d -> {b, d}
142 // e -> e (top)
143 for (size_t i = 0, n = vars.size(); i != n; ++i) {
144 if (auto proxy = Proxy::isa<Proxy_GVN>(abstr_vars[i])) {
145 auto num = proxy->num_ops();
146 auto split_vars = DefVec();
147
148 for (auto p : proxy->ops()) {
149 auto j = idx_of(vars, p);
150 if (p == vars[j] && abstr_args[i] == abstr_args[j]) split_vars.emplace_back(vars[j]);
151 }
152
153 auto new_num = split_vars.size();
154 if (new_num == 1) {
155 abstr_vars[i] = pin_top(vars[i]);
156 DLOG("single: {}", vars[i]);
157 } else if (new_num != num) {
158 auto new_proxy = mk_bundle(abstr_args[i], split_vars);
159 DLOG("split: {}", new_proxy);
160
161 for (auto p : new_proxy->ops()) {
162 auto j = idx_of(vars, p);
163 if (p == vars[j]) update(vars[j], abstr_vars[j] = new_proxy);
164 }
165 }
166 // if new_num == num: do nothing
167 }
168 }
169}

203const Def* SEO::Analysis::rewrite_imm_App(const App* app) {
204 if (auto slot = Axm::isa<mem::slot>(app)) {
205 if (!is_top(slot)) {
206 auto [mem, id] = slot->args<2>();
207 auto [_, ptr] = slot->projs<2>();
208 auto abstr_mem = rewrite(mem);
209 auto abstr_id = rewrite(id);
210 auto sloxy = world().proxy(ptr->type(), {curr_mut(), abstr_id}, Proxy_Slot);
211 sloxy2slot_[sloxy] = slot;
212 slots_.emplace(ptr);
213 DLOG("slot {} -> sloxy {}", ptr, sloxy);
214 set(ptr, sloxy);
215 return world().tuple({abstr_mem, sloxy});
216 }
217 } else if (auto store = Axm::isa<mem::store>(app)) {
218 auto [mem, ptr, val] = store->args<3>();
219 auto abstr_mem = rewrite(mem);
220 auto abstr_ptr = rewrite(ptr);
221 auto abstr_val = rewrite(val);
222
223 if (auto sloxy = Proxy::isa<Proxy_Slot>(abstr_ptr)) {
224 sloxy2val(sloxy, abstr_val);
225 DLOG("store: {} <- {}", sloxy, abstr_val);
226 return abstr_mem;
227 }
228 DLOG("store w/ unknown ptr: {} <- {}", abstr_ptr, abstr_val);
229 } else if (auto load = Axm::isa<mem::load>(app)) {
230 auto [T, as] = load->decurry()->args<2>();
231 auto [mem, ptr] = load->args<2>();
232 auto [_, val] = load->projs<2>();
233 auto abstr_mem = rewrite(mem);
234 auto abstr_ptr = rewrite(ptr);
235
236 if (auto sloxy = Proxy::isa<Proxy_Slot>(abstr_ptr)) {
237 if (auto abstr_val = sloxy2val(sloxy)) {
238 DLOG("load: {} -> {}", sloxy, abstr_val);
239 set(val, abstr_val);
240 return world().tuple({abstr_mem, abstr_val});
241 }
242 DLOG("load w/ unknown value: {}", sloxy);
243 return world().tuple({abstr_mem, world().bot(T)});
244 }
245
246 DLOG("load w/ unknown ptr: {}", abstr_ptr);
247 return set(load, load);
248 } else {
249 auto abstr_callee = rewrite(app->callee());
250 auto abstr_arg = rewrite(app->arg());
251 auto known = abstr_callee->isa_mut<Lam>();
252 if (isa_optimizable(known)) {
253 DefVec all_vars;
254 DefVec all_abstr_args;
255
256 // propagate vars
257 for (size_t i = 0; i != app->num_targs(); ++i) {
258 all_vars.emplace_back(known->tvar(i));
259 all_abstr_args.emplace_back(abstr_arg->tproj(i));
260 }
261
262 propagate_phis(known, all_vars, all_abstr_args);
263
264 auto all_abstr_vars = sccp(all_vars, all_abstr_args);
265 gvn_bundle(all_vars, all_abstr_args, all_abstr_vars);
266 gvn_split(all_vars, all_abstr_args, all_abstr_vars);
267
268 set(known->var(), world().tuple(all_abstr_vars.span().subspan(0, app->num_targs())));
269
270 for (size_t i = app->num_targs(), e = all_vars.size(); i != e; ++i)
271 set(all_vars[i], all_abstr_vars[i]);
272
273 return world().app(known, all_abstr_args.span().subspan(0, app->num_targs()));
274 }
275
276 auto phi_vars = DefVec();
277 auto phi_abstr_args = DefVec();
278 auto all_muts = world().muts().merge(abstr_callee->local_muts(), abstr_arg->local_muts());
279 for (auto mut : all_muts) {
280 if (auto lam = mut->isa<Lam>()) {
281 if (lam == known || lam->is_closed()) continue;
282 DLOG("unknown edge: {} -> {}", curr_mut(), lam);
283 propagate_phis(lam, phi_vars, phi_abstr_args);
284 }
285 }
286
287 for (size_t i = 0, e = phi_vars.size(); i != e; ++i)
288 sccp_join(phi_vars[i], phi_abstr_args[i]);
289 }
290
291 return Super::rewrite_imm_App(app);
292}

304void SEO::Analysis::analyze(const Def* def) {
305 if (def->isa<Var>()) return;
306 if (auto [_, ins] = visited_.emplace(def); !ins) return;
307 if (auto l = lookup(def)) def = l;
308
309 if (auto proxy = def->isa<Proxy>()) {
310 if (proxy->tag() == Proxy_Slot) {
311 auto slot = sloxy2slot_[proxy];
312 assert(slot);
313 auto [_, ptr] = slot->projs<2>();
314 set(slot, slot);
315 set(ptr, ptr);
316 DLOG("sloxy {} survived; setting slot to top: {}", def, slot);
317 invalidate();
318 }
319 return; // never walk a proxy's deps (would drag in meta info)
320 }
321
322 // A Lam escapes (and hence its vars must go to top) iff it is reached as a *value*.
323 if (auto app = def->isa<App>()) {
324 if (auto lam = app->callee()->isa_mut<Lam>(); isa_optimizable(lam)) {
325 // lam is applied here, not escaped: traverse its body without seeding its vars to top
326 analyze(app->type());
327 analyze(app->arg());
328 for (auto d : lam->deps())
329 analyze(d);
330 return;
331 }
332 } else if (auto [lam, var] = def->isa_binder<Lam>(); lam) {
333 DLOG("lam {} escapes", lam);
334 escaped_.emplace(lam);
335 for (auto v : var->tprojs())
336 if (auto old = update(v, v); old && old != v) DLOG("top: {}", v);
337 }
338
339 for (auto d : def->deps())
340 analyze(d);
341}

348static bool keep(const Def* old_var, const Def* abstr) {
349 if (!abstr) return true; // no info -> keep
350 if (old_var == abstr) return true; // top
351 if (Proxy::isa<Proxy_SCCP_Top>(abstr)) return true; // pending ⊤: nothing was propagated -> keep
352 if (auto proxy = Proxy::isa<Proxy_GVN>(abstr))
353 // TODO: if old_var is a phi, only keep it if all the entries in the bundle are phis. if not, prefer the first
354 // non-phi
355 return proxy->op(0) == old_var; // first in GVN bundle
356 else
357 return false;
358}

360const Def* SEO::rewrite_imm_App(const App* old_app) {
361 if (auto slot = Axm::isa<mem::slot>(old_app)) {
362 auto [mem, id] = slot->args<2>();
363 auto [_, ptr] = slot->projs<2>();
364 if (abstracted(ptr)) {
365 auto new_mem = rewrite(mem);
366 auto new_ptr = new_world().bot(rewrite(ptr->type())); // we hopefully proved that no one uses it
367 return new_world().tuple({new_mem, new_ptr});
368 }
369 } else if (auto store = Axm::isa<mem::store>(old_app)) {
370 auto [mem, ptr, val] = store->args<3>();
371 if (abstracted(ptr)) return rewrite(mem);
372 } else if (auto load = Axm::isa<mem::load>(old_app)) {
373 auto [res_mem, res_val] = load->projs<2>();
374 auto [mem, ptr] = load->args<2>();
375 DLOG("rewriting a load from {} ({})", ptr, old_app);
376 if (auto abstr_val = abstracted(res_val)) {
377 DLOG("rewriting a load from {}, we know that it's {}", ptr, abstr_val);
378 auto new_mem = rewrite(mem);
379 return new_world().tuple({new_mem, rewrite(abstr_val)});
380 }
381 } else {
382 auto old_lam = old_app->callee()->isa_mut<Lam>();
383 if (!old_lam) {
384 // The callee may fold to a rebuilt lam in the new world only, e.g. a branch
385 // `(f, t)#cond` whose cond becomes constant after GVN merged vars.
386 if (auto new_lam = rewrite(old_app->callee())->isa_mut<Lam>())
387 if (auto i = lam_new2old_.find(new_lam); i != lam_new2old_.end()) old_lam = i->second;
388 }
389
390 if (old_lam) {
391 DLOG("in {}, found app of {}", curr_mut(), old_lam);
392
393 auto& phis = phis_of(old_lam);
394 if (needs_seo(phis, old_lam)) {
395 auto new_lam = build_lam(phis, old_lam);
396 auto new_args = build_args(phis, old_lam, old_app);
397 return map(old_app, new_world().app(new_lam, new_args));
398 }
399 }
400 }
401
402 return Super::rewrite_imm_App(old_app);
403}

405const Def* SEO::rewrite_mut_Lam(Lam* old_lam) {
406 // A lam that gets a new signature must never be rebuilt generically:
407 // otherwise two new versions of old_lam exist and their (hash-consed, cached) body defs
408 // reference whichever version was rewritten first - leaving free vars in the other one.
409 if (auto& phis = phis_of(old_lam); needs_seo(phis, old_lam)) return build_lam(phis, old_lam);
410 return Super::rewrite_mut_Lam(old_lam);

427bool SEO::needs_seo(View<Phi> phis, Lam* old_lam) {
428 // An escaped lam is used as a value somewhere; its signature must stay as is.
429 if (analysis_.escaped().contains(old_lam)) return false;
430
431 if (abstracted(old_lam->var())) return true;
432
433 for (auto [sloxy, phi, val] : phis)
434 if (keep(phi, val)) return true;
435
436 return false;

439Lam* SEO::build_lam(View<Phi> phis, Lam* old_lam) {
440 if (auto i = lam_old2new_.find(old_lam); i != lam_old2new_.end()) return i->second;
441
442 DLOG("building a new lam for {}", old_lam);
443 invalidate();
444 size_t num_old = old_lam->num_tvars();
445
446 // build new dom
447 auto keeps = absl::FixedArray<bool>(num_old);
448 auto new_doms = DefVec();
449 for (size_t i = 0; i != num_old; ++i) {
450 auto old_var = old_lam->var(num_old, i);
451 keeps[i] = keep(old_var, lattice(old_var));
452 if (keeps[i]) new_doms.emplace_back(rewrite(old_lam->dom(num_old, i)));
453 }
454
455 for (auto [sloxy, phi, val] : phis)
456 if (keep(phi, val)) new_doms.emplace_back(rewrite(phi->type()));
457
458 size_t num_new_vars = new_doms.size();
459
460 // build new lam
461 auto var_map = absl::FixedArray<const Def*>(num_old);
462 auto new_lam = new_world().mut_lam(new_doms, rewrite(old_lam->codom()))->set(old_lam->dbg());
463 lam_old2new_[old_lam] = new_lam;
464 lam_new2old_[new_lam] = old_lam;
465
466 // Map all *kept* vars/phis before rewriting any propagated value below:
467 // that rewrite may recursively re-enter old_lam (via apps of it in other muts) and
468 // must be able to resolve the kept projections - otherwise the rewriter falls back
469 // to a second, generic stub of old_lam whose cached body defs poison ours.
470 size_t j = 0;
471
472 for (size_t i = 0; i != num_old; ++i) {
473 if (keeps[i]) {
474 auto old_var = old_lam->var(num_old, i);
475 auto v = new_lam->var(num_new_vars, j++)->set(old_var->dbg());
476 var_map[i] = map(old_var, v);
477 if (auto abstr = lattice(old_var))
478 if (auto bundle = Proxy::isa<Proxy_GVN>(abstr)) map(bundle, v); // GVN bundle
479 }
480 }
481
482 for (auto [sloxy, phi, val] : phis) {
483 if (keep(phi, val)) {
484 auto v = new_lam->var(num_new_vars, j++);
485 DLOG("mapping phi {} to {}", phi, v);
486 map(phi, v);
487 if (val != phi) map(val, v); // phi is part of a GVN bundle
488 }
489 }
490
491 // now resolve the dropped vars/phis to their propagated values
492 for (size_t i = 0; i != num_old; ++i)
493 if (!keeps[i]) var_map[i] = rewrite(lattice(old_lam->var(num_old, i))); // SCCP propagate
494
495 for (auto [sloxy, phi, val] : phis)
496 if (!keep(phi, val)) {
497 DLOG("mapping phi {} to its propagated value {}", phi, val);
498 map(phi, rewrite(val));
499 }
500
501 map(old_lam->var(), var_map);
502 {
503 auto _ = enter(old_lam);
504 auto new_filter = rewrite(old_lam->filter());
505 auto new_body = rewrite(old_lam->body());
506 new_lam->set(new_filter, new_body);
507 }
508
509 return new_lam;
510}

512DefVec SEO::build_args(View<Phi> phis, Lam* old_lam, const App* old_app) {
513 size_t num_old = old_lam->num_tvars();
514 auto new_args = DefVec();
515
516 for (size_t i = 0; i != num_old; ++i) {
517 auto old_var = old_lam->var(num_old, i);
518 auto abstr = lattice(old_var);
519 if (keep(old_var, abstr)) new_args.emplace_back(rewrite(old_app->targ(i)));
520 }
521
522 DLOG("wiring up phi arguments");
523 for (auto [sloxy, phi, val] : phis)
524 if (keep(phi, val)) {
525 auto arg = analysis_.lam2sloxy2val(curr_mut<Lam>(), sloxy);
526 assert(arg);
527 new_args.emplace_back(rewrite(arg));
528 }
529
530 return new_args;
