27/// The symbolic extents of one `dims[i] · dims[s + 1] · dims[j + 1]` cost term, sorted by Def::gid and
28/// padded with `nullptr`; literal extents fold into the coefficient instead.
29using Mono = std::array<const Def*, 3>;
30
31/// A chain cost as a polynomial in the symbolic extents.
32class Poly {
33public:
34 void add(Mono mono, u64 coeff) {
35 for (auto& [m, c] : terms_)
36 if (m == mono) {
37 c += coeff;
38 return;
39 }
40 terms_.emplace_back(mono, coeff);
41 }
42
43 void add(const Poly& other) {
44 for (const auto& [m, c] : other.terms_)
45 add(m, c);
46 }
47
48 u64 coeff(Mono mono) const {
49 for (const auto& [m, c] : terms_)
50 if (m == mono) return c;
51 return 0;
52 }
53
54 /// Is `this` at most @p other under *every* instantiation of the symbolic extents?
55 /// Extents are non-negative, so a coefficient-wise `≤` is sufficient - but not necessary, which is
56 /// what makes this order partial.
57 bool dominates(const Poly& other) const {
58 for (const auto& [m, c] : terms_)
59 if (c > other.coeff(m)) return false;
60 return true;
61 }
62
63 std::string str() const {
64 if (terms_.empty()) return "0";
65 auto s = std::string();
66 for (const auto& [mono, coeff] : terms_) {
67 if (!s.empty()) s += " + ";
68 s += std::format("{}", coeff);
69 for (auto d : mono)
70 if (d) s += std::format("·{}", d);
71 }
72 return s;
73 }
74
75private:
76 fe::Vector<std::pair<Mono, u64>> terms_;
77};

79/// The cost of one product `«x, y» · «y, z»`.
80Poly mul_cost(const Def* x, const Def* y, const Def* z) {
81 auto coeff = 1_u64;
82 auto syms = DefVec();
83
84 for (auto d : {x, y, z})
85 if (auto l = Lit::isa<u64>(d))
86 coeff *= *l;
87 else
88 syms.emplace_back(d);
89 std::ranges::sort(syms, [](const Def* a, const Def* b) { return a->gid() < b->gid(); });
90
91 auto mono = Mono{};
92 std::ranges::copy(syms, mono.begin());
93
94 auto poly = Poly();
95 poly.add(mono, coeff);
96 return poly;
97}

99Poly cost_of(fe::View<Split> splits, Defs dims) {
100 auto poly = Poly();
101 for (auto [i, s, j] : splits)
102 poly.add(mul_cost(dims[i], dims[s + 1], dims[j + 1]));
103 return poly;
104}

106/// Every bracketing of `lo … hi`.
107fe::Vector<Splits> bracketings(u64 lo, u64 hi) {
108 if (lo == hi) return {Splits()};
109
110 auto res = fe::Vector<Splits>();
111 for (auto s = lo; s != hi; ++s)
112 for (const auto& l : bracketings(lo, s))
113 for (const auto& r : bracketings(s + 1, hi)) {
114 auto b = l;
115 b.append_range(r);
116 b.emplace_back(Split{lo, s, hi});
117 res.emplace_back(std::move(b));
118 }
119 return res;
120}

122/// Drops every bracketing that another one provably beats; equal costs keep the first.
123/// A single survivor is hence the optimum under *every* instantiation of the symbolic extents.
124/// Several survivors need not each win for some instantiation - domination is only sufficient for `≤`.
125fe::Vector<Splits> pareto(fe::View<Splits> cands, Defs dims) {
126 auto keep = fe::Vector<Splits>();
127 auto costs = fe::Vector<Poly>();
128
129 for (const auto& cand : cands) {
130 auto cost = cost_of(cand, dims);
131 if (std::ranges::any_of(costs, [&](const Poly& k) { return k.dominates(cost); })) continue;
132 for (auto i = costs.size(); i-- != 0;)
133 if (cost.dominates(costs[i])) keep.erase(keep.begin() + i), costs.erase(costs.begin() + i);
134 keep.emplace_back(cand);
135 costs.emplace_back(std::move(cost));
136 }
137
138 return keep;
139}

148/// Matrix-chain order: matrix `i` of the chain has shape `dims[i] × dims[i + 1]`.
149/// @returns the split table - `split[i * n + j]` is the last matrix of the left factor of the cheapest
150/// parenthesization of `i … j` - together with that parenthesization's cost, or nothing at all if some
151/// subchain has no candidate that provably beats all the others.
152std::optional<std::pair<fe::Vector<u64>, Poly>> matrix_chain_order(Defs dims) {
153 auto n = dims.size() - 1;
154 auto cost = fe::Vector<Poly>(n * n);
155 auto split = fe::Vector<u64>(n * n, 0);
156
157 for (auto len = 2_u64; len <= n; ++len) {
158 for (auto i = 0_u64; i + len <= n; ++i) {
159 auto j = i + len - 1;
160 auto cands = fe::Vector<Poly>();
161 for (auto s = i; s != j; ++s) {
162 auto c = mul_cost(dims[i], dims[s + 1], dims[j + 1]);
163 c.add(cost[i * n + s]);
164 c.add(cost[(s + 1) * n + j]);
165 cands.emplace_back(std::move(c));
166 }
167
168 auto best = std::ranges::find_if(cands, [&](const Poly& a) {
169 return std::ranges::all_of(cands, [&](const Poly& b) { return a.dominates(b); });
170 });
171 if (best == cands.end()) return {};
172
173 split[i * n + j] = i + (best - cands.begin());
174 cost[i * n + j] = std::move(*best);
175 }
176 }
177
178 return std::pair{std::move(split), std::move(cost[n - 1])};
179}

216void Reassoc::flatten(const Def* def, const Def* ring, const Def* rows, DefVec& mats, DefVec& dims, Splits& orig) {
217 auto lo = mats.size();
218
219 if (auto i = consumers_.find(def); i != consumers_.end() && i->second == 1)
220 if (auto link = isa_link(def, ring)) {
221 auto [t1, t2] = link->app->args<2>();
222 flatten(t1, ring, link->m, mats, dims, orig);
223 auto mid = mats.size();
224 flatten(t2, ring, link->k, mats, dims, orig);
225 orig.emplace_back(Split{lo, mid - 1, mats.size() - 1});
226 return;

233const Def* Reassoc::build(const Def* head, Defs mats, Defs dims, fe::View<u64> split, u64 i, u64 j) {

244const Def* Reassoc::cost_expr(Defs dims, const Splits& splits) {
245 auto& w = new_world();
246 const Def* sum = nullptr;
247
248 for (auto [i, s, j] : splits) {
249 auto p = w.app(w.annex(core::nat::mul), {rewrite(dims[i]), rewrite(dims[s + 1])});
250 p = w.app(w.annex(core::nat::mul), {p, rewrite(dims[j + 1])});
251 sum = sum ? w.app(w.annex(core::nat::add), {sum, p}) : p;
252 }
253
254 return sum;

257const Def* Reassoc::dispatch(const Def* head, const Def* res_ty, Defs mats, Defs dims, fe::View<Splits> cands) {
258 auto& w = new_world();
259 auto n = mats.size();
260 auto pi = w.pi(w.sigma(), res_ty);
261
262 // Each bracketing goes behind a thunk so that only the selected one runs. The filter is `tt`, so once
263 // the comparison folds - a caller that knows the extents, `%compile.lam_spec` - the winner inlines and
264 // the losers become unreachable.
265 const Def* best = nullptr;
266 const Def* best_cost = nullptr;
267 for (const auto& cand : cands) {
268 auto thunk = w.mut_lam(pi)->set(true, build(head, mats, dims, split_table(cand, n), 0, n - 1));
269 auto cost = cost_expr(dims, cand);
270
271 if (!best) {
272 best = thunk, best_cost = cost;
273 } else {
274 auto cheaper = w.app(w.annex(core::ncmp::l), {cost, best_cost});
275 best = w.extract(w.tuple({best, (const Def*)thunk}), cheaper);
276 best_cost = w.extract(w.tuple({best_cost, cost}), cheaper);
277 }
278 }
279
280 return w.app(best, w.tuple());
281}

283const Def* Reassoc::reassoc(const App* app) {
284 auto head = app->callee()->as<App>()->callee()->as<App>();
285 auto link = isa_link(app, head->arg());
286 if (!link) return nullptr;
287
288 auto [t1, t2] = app->args<2>();
289 auto mats = DefVec();
290 auto dims = DefVec();
291 auto orig = Splits();
292 flatten(t1, head->arg(), link->m, mats, dims, orig);
293 auto mid = mats.size();
294 flatten(t2, head->arg(), link->k, mats, dims, orig);
295 dims.emplace_back(link->l);
296 orig.emplace_back(Split{0_u64, mid - 1, mats.size() - 1});
297
298 // Two matrices admit only one parenthesization.
299 auto n = mats.size();
300 if (n < 3) return nullptr;
301
302 auto orig_cost = cost_of(orig, dims);
303
304 if (n <= max_dispatch_) {
305 auto cands = pareto(bracketings(0, n - 1), dims);
306 if (cands.size() != 1) {
307 log().d("dispatch chain {} over {} bracketings, written as {}", fe::Join(dims, "×"), cands.size(),
308 orig_cost.str());
309 return dispatch(rewrite(head), rewrite(app->type()), mats, dims, cands);
310 }
311
312 auto cost = cost_of(cands.front(), dims);
313 if (orig_cost.dominates(cost)) return nullptr;
314 log().d("reassociate chain {}: {} → {} multiplications", fe::Join(dims, "×"), orig_cost.str(), cost.str());
315 return build(rewrite(head), mats, dims, split_table(cands.front(), n), 0, n - 1);
316 }
317
318 auto order = matrix_chain_order(dims);
319 if (!order) return nullptr;
320
321 auto& [split, cost] = *order;
322 if (orig_cost.dominates(cost)) return nullptr;
323
324 log().d("reassociate chain {}: {} → {} multiplications", fe::Join(dims, "×"), orig_cost.str(), cost.str());
325 return build(rewrite(head), mats, dims, split, 0, n - 1);
326}
