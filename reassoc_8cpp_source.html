19/// Matrix-chain order: matrix `i` of the chain has shape `dims[i] × dims[i + 1]`.
20/// @returns the split table - `split[i * n + j]` is the last matrix of the left factor of the cheapest
21/// parenthesization of `i … j` - together with that parenthesization's cost.
22std::pair<Row, u64> matrix_chain_order(fe::View<u64> dims) {
23 auto n = dims.size() - 1;
24 auto cost = Row(n * n, 0);
25 auto split = Row(n * n, 0);
26
27 for (auto len = 2_u64; len <= n; ++len) {
28 for (auto i = 0_u64; i + len <= n; ++i) {
29 auto j = i + len - 1;
30 cost[i * n + j] = u64(-1);
31 for (auto s = i; s != j; ++s) {
32 auto c = cost[i * n + s] + cost[(s + 1) * n + j] + dims[i] * dims[s + 1] * dims[j + 1];
33 if (c < cost[i * n + j]) cost[i * n + j] = c, split[i * n + j] = s;
34 }
35 }
36 }
37
38 return {std::move(split), cost[n - 1]};
39}

53void Reassoc::start() {
54 // The old world does not track uses, so count consumers up front: flatten() may only pull a
55 // product apart where doing so cannot leave it materialized for another consumer as well.
56 auto wl = fe::BFSWorklist<DefSet>();
57 for (auto root : old_world().roots())
58 wl.push(root);
59
60 while (!wl.empty()) {

73std::optional<Reassoc::Link> Reassoc::isa_link(const Def* def, const Def* ring) const {
74 auto app = Axm::isa<tensor::product_2d>(def);

82 if (!m || !k || !l) return {};

86void Reassoc::flatten(const Def* def, const Def* ring, u64 rows, DefVec& mats, Row& dims, u64& cost) {
87 if (auto i = consumers_.find(def); i != consumers_.end() && i->second == 1)
88 if (auto link = isa_link(def, ring)) {
89 auto [t1, t2] = link->app->args<2>();
90 flatten(t1, ring, link->m, mats, dims, cost);
91 flatten(t2, ring, link->k, mats, dims, cost);
92 cost += link->m * link->k * link->l;

100const Def* Reassoc::build(const Def* head, Defs mats, fe::View<u64> dims, fe::View<u64> split, u64 i, u64 j) {
101 if (i == j) return rewrite(mats[i]);
102
103 auto& w = new_world();

111const Def* Reassoc::reassoc(const App* app) {
112 auto head = app->callee()->as<App>()->callee()->as<App>();
113 auto link = isa_link(app, head->arg());
114 if (!link) return nullptr;
115
116 auto [t1, t2] = app->args<2>();
117 auto mats = DefVec();
118 auto dims = Row();
119 auto orig = link->m * link->k * link->l;
120 flatten(t1, head->arg(), link->m, mats, dims, orig);
121 flatten(t2, head->arg(), link->k, mats, dims, orig);
122 dims.emplace_back(link->l);
123
124 // Two matrices admit only one parenthesization.
125 if (mats.size() < 3) return nullptr;
126
127 auto [split, cost] = matrix_chain_order(dims);
128 if (cost >= orig) return nullptr;
129
130 log().d("reassociate chain {}: {} → {} multiplications", fe::Join(dims, "×"), orig, cost);
131 return build(rewrite(head), mats, dims, split, 0, mats.size() - 1);
132}
