41/// `(t#is)#js` as *one* index into @p t; `nullptr` unless @p t's own shape has room for both ranks.
42/// The Extract and the Insert rule are duals and share this check.
43Shape fused_index(const Def* t, Shape is, Shape js) {
44 auto arr = t->unfold_type()->isa<Arr>();
45 if (!arr) return {};
46 auto ri = is.rank();
47 auto rj = js.rank();
48 auto rt = arr->shape().rank();

63/// Checks the multi-dimensional @p index against @p d and yields the type it peels to.
64const Def* nary_elem_type(const Def* d, Shape index) {
65 auto type = d->unfold_type();
66 auto arr = type->isa<Arr>();
67 if (!arr) index->blame("multi-dimensional index expects an array but got `{}`", type).bail();
68
69 auto shape = arr->shape();
70 auto ri = index.rank();
71 auto rt = shape.rank();
72 // A *partial* index peels only its own axes and yields a sub-array.
73 auto fit = ri && rt ? *ri <= *rt : Checker::alpha<Checker::Check>(index->arity(), shape->arity());
74 if (!fit)
75 index->blame("index of rank `{}` does not fit shape `{}` of rank `{}`", index->arity(), *shape, shape->arity())
76 .bail();
77
78 // Only the axes World::extract folds away reach World::extract1's check, so the ones staying fused
79 // need it here. A fused shape never binds the Arr's own var, so its extents compare as they stand.
80 if (ri && rt)
81 for (nat_t i = 0; i != *ri; ++i)
82 if (auto size = Idx::isa(index[i]->unfold_type()); size && !Checker::alpha<Checker::Check>(shape[i], size))
83 index[i]->blame("index does not fit within arity `{}`", shape[i]).bail();
84
85 return d->world().peel(arr, *index);
86}

229template<int sort>
230const Def* World::umax(Defs ops_) {
231 DefVec ops;
232 ops.reserve(ops_.size());
233 for (auto op : ops_) {
234 op = op->zonk();
235
236 // Peel off as many layers as the sort of the incoming ops demands to arrive at a Univ level:
237 // a Univ level is already there, a Kind is a `Type lvl`, a Type needs one unfold, a term two.
238 if constexpr (sort >= UMax::Type) op = op->unfold_type();
239 if constexpr (sort == UMax::Term) op = op->unfold_type();
240 if constexpr (sort >= UMax::Kind) {
241 if (auto type = op->isa<Type>())
242 op = type->level();
243 else
244 op->blame("operand must be a `Type` of some universe level").bail();
245 }
246
247 flatten_umax(ops, op);
248 }
249
250 level_t lvl = 0;
251 DefVec res;
252 res.reserve(ops.size());
253 for (auto op : ops) {
254 if (!op->isa_type<Univ>())
255 op->blame("operand of a universe max must be of type `Univ` but is of type `{}`", type_of(op)).bail();
256
257 if (auto l = Lit::isa(op))
258 lvl = std::max(lvl, *l);
259 else
260 res.emplace_back(op);
261 }
262
263 const Def* l = lit_univ(lvl);
264 if (res.empty()) return sort == UMax::Univ ? l : type(l);
265 if (lvl > 0) res.emplace_back(l);
266
267 sort_unique(res);
268 const Def* umax = unify<UMax>(*this, res);
269 return sort == UMax::Univ ? umax : type(umax);
270}

295const Def* World::app(const Def* callee, const Def* arg) {
296 callee = callee->zonk();
297 arg = arg->zonk();
298
299 auto pi = callee->isa_type<Pi>();
300 if (!pi)
301 callee->blame("callee is not of function type")
302 .n("callee `{}` has type `{}`", callee, type_of(callee))
303 .n(callee->loc(), "callee `{}` declared here", callee)
304 .bail();
305
306 auto new_arg = Checker::assignable(pi->dom(), arg);
307 if (!new_arg)
308 arg->blame("argument is not assignable to callee's domain")
309 .n("expected `{}`, got `{}`", pi->dom(), type_of(arg))
310 .n(callee->loc(), "callee `{}` declared here", callee)
311 .bail();
312
313 // re-zonk after assignable check above - we might have inferred new stuff
314 arg = new_arg->zonk();
315 callee = callee->zonk();
316 pi = callee->isa_type<Pi>();
317
318 // always β-reduce non-recursive, non-parametric lambdas
319 if (auto imm = callee->isa_imm<Lam>()) return imm->body();
320
321 if (auto lam = callee->isa_mut<Lam>(); lam && lam->is_set()) {
322 auto var = lam->has_var();
323
324 // Applying a Lam to its own Var is the identity substitution, so it resolves to the body.
325 // This unfolds a self-application / fixed-point reference.
326 if (var && arg == var) return lam->body();
327
328 // β-reduce or partially evaluate a set, mutable Lam.
329 if (lam->filter() != lit_ff()) {
330 if (!var) {
331 if (lam->filter() == lit_tt()) return lam->body();
332 } else if (auto i = move_.substs.find({var, arg}); i != move_.substs.end()) {
333 // Reuse the cached reduct if its filter held.
334 auto [filter, body] = i->second->ops<2>();
335 if (filter == lit_tt()) return body;
336 } else {
337 // Evaluate the filter; if it holds, reduce the body and cache the reduct.
338 auto rw = VarRewriter(var, arg);
339 auto filter = rw.rewrite(lam->filter());
340 if (filter == lit_tt()) {
341 log().d("partially evaluate {} ({})", lam, arg);
342 auto body = rw.rewrite(lam->body());
343 cache_reduct(var, arg, {filter, body});
344 return body;
345 }
346 }
347 }
348 }
349
350 auto type = pi->reduce(arg)->zonk();
351 callee = callee->zonk();
352 auto [axm, curry, trip] = Axm::next(callee);
353
354 if (axm)
355 if (auto normalizer = axm->normalizer(); Normalize && normalizer && curry == 0)
356 if (auto norm = normalizer(type, callee, arg)) return norm;
357
358 return raw_app(axm, curry, trip, type, callee, arg);
359}

430const Def* World::extract(const Def* d, const Def* index_) {
431 if (!d || !index_) return nullptr; // can happen if frozen
432 d = d->zonk();
433
434 auto index = check_index(index_->zonk());
435 if (index.is_dim()) return extract1(d, *index);
436
437 auto r = index.rank();
438 if (!r) return extract_fused(d, index); // a dynamic rank has no axes to fold one by one
439 if (*r == 0) return d; // the index folded away entirely, i.e. `d` is its own only element
440
441 // Fold from the left as far as `d` allows; whatever is left stays *fused*, mirroring `«a, b; T»`.
442 for (nat_t i = 0; i != *r; ++i) {
443 auto c = index[i];
444 auto next = extract1(d, c);
445 if (auto ex = next->isa<Extract>(); ex && ex->tuple() == d && ex->index() == c)
446 return extract_fused(d, index.drop(i));
447 d = next;
448 }
449
450 return d;
451}

474 if (!index) return nullptr; // can happen if frozen

487const Def* World::fuse(Seq* seq) {
488 auto outer = seq->shape();
489 auto inner = seq->body() ? seq->body()->zonk()->isa<Seq>() : nullptr;
490 auto var = seq->has_var();
491 // A ragged nest - one whose inner extents depend on the outer index - has no fused shape to spell them in.
492 if (!inner || inner->node() != seq->node() || (var && inner->shape()->has_free_var(var))) return seq->zonk_mut();
493
494 auto ro = outer.rank();
495 auto shape = outer + inner->shape();
496 if (!ro || !shape) return seq->zonk_mut();
497
498 // The fused index supplies both levels: its leading `ro` axes replace the outer one, the rest the inner.
499 return mut_shaped(seq, shape, inner->body()->unfold_type(), [&](Seq* res) {
500 auto v = Shape(res->var());
501 return seq->reduce(*v.take(*ro))->as<Seq>()->reduce(*v.drop(*ro));
502 });

514/// Builds `d#index` for a multi-dimensional @p index that does not fold any further.
515const Def* World::extract_fused(const Def* d, Shape index) {
516 if (index.is_dim()) return extract1(d, *index);
517
518 // An index reaching past `d`'s own shape continues through the element type - `«2; [Nat, Bool]»` is rank 1,
519 // yet `t#(i, j)` is still `t#i#j`. Split it at the boundary, so each Extract matches the shape it indexes.
520 auto ri = index.rank();
521 auto rt = own_rank(d->unfold_type());
522 if (ri && rt && *ri > *rt && !is_frozen()) return extract(extract_fused(d, index.take(*rt)), *index.drop(*rt));
523
524 // `d#is ← val` then `#is` -> `val`; cf. the same rule in World::extract1
525 if (auto insert = d->isa<Insert>(); insert && insert->index() == *index) return insert->value();
526 return unify<Extract>(nary_elem_type(d, index), d, *index);
527}

545const Def* World::extract1(const Def* d, const Def* index) {
546 auto index_ty = index->unfold_type();
547 auto size = Idx::isa(index_ty);
548 auto lidx = Lit::isa(index);
549 auto type = d->unfold_type();
550
551 if (is_folded_axis(type, index, size)) return d;
552
553 if (auto pack = d->isa<Pack>()) {
554 if (pack->shape().is_fused()) return peel(pack, index); // one axis off a fused Pack yields a sub-Pack
555 if (pack->has_var())
556 return pack->reduce(index);
557 else
558 return pack->body();
559 }
560
561 // extract(insert(x, index, val), index) -> val
562 if (auto insert = d->isa<Insert>()) {
563 if (index == insert->index()) return insert->value();
564 }
565
566 // `(t#is)#j` -> `t#(is, j)`: the maximal-rank Extract into one fused Arr is the normal form.
567 if (auto ex = is_frozen() ? nullptr : d->isa<Extract>())
568 if (auto is = fused_index(ex->tuple(), ex->index(), index)) return extract_fused(ex->tuple(), is);
569
570 if (lidx) {
571 // A Hole only tuplefies into a known number of components; a dynamic rank leaves a symbolic Extract.
572 if (auto hole = d->isa_mut<Hole>())
573 if (auto n = Idx::isa_lit(index_ty)) d = hole->tuplefy(*n);
574 if (auto tuple = d->isa<Tuple>()) return tuple->op(*lidx);
575
576 // extract(insert(x, j, val), i) -> extract(x, i) where i != j (guaranteed by rule above)
577 if (auto insert = d->isa<Insert>()) {
578 if (insert->index()->isa<Lit>()) return extract(insert->tuple(), index);
579 }
580
581 if (auto sigma = type->isa<Sigma>()) {
582 if (auto var = sigma->has_var()) {
583 if (d == var) return unify<Extract>(sigma->op(*lidx), d, index); // `var -> var` is the identity
584 // Frozen, only an already cached reduct can be replayed - rewriting would create nodes.
585 if (is_frozen()) {
586 auto t = cached_reduct(var, d, *lidx);
587 return t ? unify<Extract>(t, d, index) : nullptr;
588 }
589 return unify<Extract>(reduce(var, d, *lidx), d, index);
590 }
591
592 return unify<Extract>(sigma->op(*lidx), d, index);
593 }
594 }
595
596 const Def* elem_t;
597 if (auto arr = type->isa<Arr>()) {
598 elem_t = peel(arr, index);
599 } else {
600 auto sigma = type->as<Sigma>();
601 elem_t = nullptr;
602 // «(a_0, ..., a_{n-1})#index; body» is more precise than the join if all ops are Arrs of the same body.
603 if (sigma->isa_imm()) {
604 const Def* body = nullptr;
605 auto extents = DefVec();
606 for (auto op : sigma->ops()) {
607 auto op_arr = op->zonk()->isa<Arr>();
608 if (!op_arr || (body && op_arr->body()->zonk() != body)) {
609 extents.clear();
610 break;
611 }
612 body = op_arr->body()->zonk();
613 extents.emplace_back(*op_arr->shape());
614 }
615 if (!extents.empty()) elem_t = this->arr(extract(tuple(extents), index), body);
616 }
617 if (!elem_t) elem_t = join(sigma->ops());
618 }
619
620 if (index->isa<Top>()) {
621 if (auto hole = Hole::isa_unset(d)) {
622 auto elem_hole = mut_hole(elem_t);
623 hole->set(pack(size, elem_hole));
624 return elem_hole;
625 }
626 }
627
628 assert(d);
629 return unify<Extract>(elem_t, d, index);
630}

632const Def* World::insert(const Def* d, const Def* index_, const Def* val) {
633 d = d->zonk();
634 val = val->zonk();
635
636 auto type = d->unfold_type();
637 auto index = check_index(index_->zonk());
638
639 // `d#(i, j, k) ← val` stays *fused* unless the outermost write rebuilds an aggregate on the spot.
640 if (!index.is_dim()) {
641 auto r = index.rank();
642 // The index folded away entirely, so the write replaces all of `d`.
643 if (r && *r == 0) return assign_or_bail(type, val);
644
645 // A write deeper than `d`'s own shape reads down to the boundary and writes the levels back out;
646 // cf. World::extract_fused. So does one whose outermost write rebuilds an aggregate.
647 auto rt = own_rank(type);
648 auto cut = r && rt && *r > *rt ? rt
649 : r && insert_rebuilds(d, index[0], flags().scalarize_threshold) ? std::optional<nat_t>(1)
650 : std::nullopt;
651 if (cut) {
652 auto head = *index.take(*cut);
653 return insert(d, head, insert(extract(d, head), *index.drop(*cut), val));
654 }
655
656 auto new_val = assign_or_bail(nary_elem_type(d, index), val);
657
658 // `d#is ← (d#is)#js ← val` -> `d#(is, js) ← val`; cf. the scalar path below, which reaches this
659 // through World::extract's per-axis loop - the fused path has none, so it needs the rule itself.
660 if (auto inner = is_frozen() ? nullptr : new_val->isa<Insert>())
661 if (auto ex = inner->tuple()->isa<Extract>(); ex && ex->tuple() == d && ex->index() == *index)
662 if (auto is = fused_index(d, *index, inner->index())) return insert(d, *is, inner->value());
663
664 // `d#is ← d#is` -> `d` and `(d#is ← y)#is ← val` -> `d#is ← val`; cf. the scalar path below
665 if (auto ex = new_val->isa<Extract>(); ex && ex->tuple() == d && ex->index() == *index) return d;
666 if (auto insert = d->isa<Insert>(); insert && insert->index() == *index) d = insert->tuple();
667 return unify<Insert>(d, *index, new_val);
668 }
669
670 auto size = Idx::isa(index->unfold_type());
671 auto lidx = Lit::isa(*index);
672
673 if (is_folded_axis(type, *index, size)) return assign_or_bail(type, val); // the write replaces all of `d`
674 // A Sigma needs a literal index to name the component; an Arr peels to an element type at any index.
675 if (auto arr = type->isa<Arr>()) {
676 if (auto elem = peel(arr, *index)) val = assign_or_bail(elem, val);
677 } else if (lidx) {
678 val = assign_or_bail(type->proj(Lit::as(size), *lidx), val);
679 }
680 // The only `Idx 1` left is a mutable 1-tuple; d could be mut - that's why the tuple ctor is needed.
681 if (Lit::isa(size) == 1) return tuple(d, {val});
682
683 // insert((a, b, c, d), 2, x) -> (a, b, x, d)
684 if (auto t = d->isa<Tuple>(); t && lidx) {
685 auto new_ops = DefVec(t->ops().begin(), t->ops().end());
686 new_ops[*lidx] = val;
687 return tuple(type, new_ops);
688 }
689
690 // insert(‹4; x›, 2, y) -> (x, x, y, x)
691 if (auto a = pack_unrolls(d, *index, flags().scalarize_threshold)) {
692 auto new_ops = DefVec(*a, d->as<Pack>()->body());
693 new_ops[*lidx] = val;
694 return tuple(type, new_ops);
695 }
696
697 // `d#i ← (d#i)#js ← val` -> `d#(i, js) ← val`: the dual of the Extract fusion in World::extract1, and what
698 // keeps the read-modify-write chain a *single* write. The aggregate-rebuilding rules above win over it.
699 if (auto inner = is_frozen() ? nullptr : val->isa<Insert>())
700 if (auto ex = inner->tuple()->isa<Extract>(); ex && ex->tuple() == d && ex->index() == *index)
701 if (auto is = fused_index(d, *index, inner->index())) return insert(d, *is, inner->value());
702
703 // insert(d, index, d#index) -> d
704 if (auto ex = val->isa<Extract>())
705 if (ex->tuple() == d && ex->index() == *index) return d;
706
707 // insert(insert(x, index, y), index, val) -> insert(x, index, val)
708 if (auto insert = d->isa<Insert>()) {
709 if (insert->index() == *index) d = insert->tuple();
710 }
711
712 return unify<Insert>(d, *index, val);
713}

715const Def* World::seq(bool is_pack, Shape shape, const Def* body) {
716 shape = shape.zonk();
717 body = body->zonk();
718
719 auto shape_ty = shape->unfold_type();
720 if (!Shape::isa_extents(shape_ty)) shape->blame("expected a shape but got a value of type `{}`", shape_ty).bail();
721
722 // `«1; T»` ≡ `T`, mirroring `[T]` ≡ `T`, so a literal size-1 axis folds out of the shape.
723 shape = shape.fold();
724 auto r = shape.rank();
725 if (r) {
726 if (*r == 0) return body;
727 // A literal-`0` extent empties everything below it: `«2, 0, 3; T»` is `«2; []»`, not `[]`.
728 for (nat_t i = 0; i != *r; ++i)
729 if (Shape::extent(shape[i]) == 0) return seq(is_pack, shape.take(i), unit(is_pack));
730 }
731
732 // `«a; «b; T»»` ≡ `«a, b; T»` - the very compression that already makes `«3; T»` out of `[T, T, T]`,
733 // one level up. A *mutable* body binds an index the fused shape could not express, so it stops fusion.
734 if (auto inner = body->isa_imm<Seq>(); inner && inner->is_intro() == is_pack)
735 if (auto fused = shape + inner->shape()) return seq(is_pack, fused, inner->body());
736
737 if (is_pack) return unify<Pack>(arr(shape, body->unfold_type()), *shape, body);
738 return unify<Arr>(body->unfold_type(), *shape, body);
739}

842const Def* World::match(Defs ops_) {
843 auto ops = Def::zonk(ops_);
844 if (ops.size() == 1) return ops.front();
845
846 auto scrutinee = ops.front();
847 auto arms = ops.span().subspan(1);
848
849 for (auto arm : arms)
850 if (!arm->isa_type<Pi>())
851 arm->blame("arm of a test expression does not have a function type but has type `{}`", type_of(arm)).bail();
852
853 auto scrut_t = scrutinee->unfold_type();
854 if (!scrut_t) scrutinee->blame("scrutinee of a test expression has no type").bail();
855 auto join = scrut_t->isa<Join>();
856
857 // A scrutinee that is not a union is the degenerate one-case union; cf. `d#i` for `i: Idx 1`.
858 auto cases = Match::cases(scrutinee);
859
860 // The first accepting arm handles a case, any later one is dead; arms are *not* sorted, their order is the
861 // tie-break.
862 auto sel = DefVec(cases.size());
863 for (size_t i = 0, e = cases.size(); i != e; ++i) {
864 auto c = cases[i];
865 auto j = std::ranges::find_if(arms, [c](const Def* arm) { return Match::accepts(arm, c); });
866 // Hole%s are resolved only once no arm matches outright, so a mere search never commits to one.
867 if (j == arms.end())
868 j = std::ranges::find_if(arms, [c](const Def* arm) { return Match::accepts(arm, c, true); });
869 if (j == arms.end()) scrutinee->blame("test expression has no arm for union case `{}`", c).bail();
870 sel[i] = *j;
871 }
872
873 // Keep the arms actually selected, in the order written; a dead arm is dropped.
874 auto res = DefVec();
875 const Def* type = nullptr;
876 res.emplace_back(scrutinee);
877 for (auto arm : arms) {
878 if (!std::ranges::contains(sel, arm)) continue;
879 auto codom = arm->isa_type<Pi>()->codom();
880 type = type ? this->join({type, codom}) : codom;
881 res.emplace_back(arm);
882 }
883
884 // An arm covering several cases expects the wider union, so narrow values are injected back into it.
885 auto dispatch
886 = [this](const Def* arm, const Def* value) { return app(arm, inj(arm->isa_type<Pi>()->dom(), value)); };
887
888 if (!join) return dispatch(sel[0], scrutinee);
889
890 // A constructor fixes the active union case. Dispatch before the Match can
891 // escape into later lowering phases, where the payload representation may
892 // already have changed (for example, a tensor may have become a buffer).
893 if (auto inj = scrutinee->isa<Inj>()) {
894 auto value = inj->value();
895 for (size_t i = 0, e = cases.size(); i != e; ++i)
896 if (Checker::alpha<Checker::Check>(value->unfold_type(), cases[i])) return dispatch(sel[i], value);
897 scrutinee->blame("injected value type `{}` is not a case of union type `{}`", type_of(value), join).bail();
898 }
899
900 return unify<Match>(type, res);
901}

940Defs World::reduce(const Var* var, const Def* arg) {
941 auto mut = var->binder();
942 auto off = mut->reduction_offset();
943 auto n = mut->num_ops() - off;
944 if (var == arg) return {mut->ops().begin() + off, n}; // `[var -> var]` is the identity

947 auto rw = VarRewriter(var, arg); // one rewriter for all slots: they share their sub-rewrites
948 for (size_t i = 0; i != n; ++i) {

950 if (slot) continue;
951 assert(slot != Filling && "op requires its own reduction");
952 slot = Filling;
953 slot = rw.rewrite(mut->op(i + off));
954 }
955
956 return reduct->ops();
