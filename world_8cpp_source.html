21bool is_shape(const Def* s) {
22 if (s->isa<Nat>()) return true;
23 if (auto arr = s->isa<Arr>()) return arr->body()->zonk()->isa<Nat>();
24 if (auto sig = s->isa_imm<Sigma>())
25 return std::ranges::all_of(sig->ops(), [](const Def* op) { return op->isa<Nat>(); });
26
27 return false;
28}

30/// Is @p def an `Idx` - or an aggregate of `Idx`%s, i.e. a multi-dimensional index?
31bool isa_indices(const Def* def) {
32 if (!def) return false; // Univ has no type
33 if (Idx::isa(def)) return true;
34 if (auto sigma = def->isa<Sigma>()) return std::ranges::all_of(sigma->ops(), [](auto op) { return Idx::isa(op); });
35 if (auto arr = def->isa<Arr>()) return Idx::isa(arr->body());
36 return false;
37}

39/// Sorts by gid and drops duplicates; Def%s are hash-consed, so pointer identity *is* structural identity.
40void sort_unique(DefVec& defs) {
41 std::ranges::sort(defs, GIDLt<const Def*>());
42 defs.erase(std::unique(defs.begin(), defs.end()), defs.end());

163template<int sort>
164const Def* World::umax(Defs ops_) {
165 DefVec ops;
166 ops.reserve(ops_.size());
167 for (auto op : ops_) {
168 op = op->zonk();
169
170 // Peel off as many layers as the sort of the incoming ops demands to arrive at a Univ level:
171 // a Univ level is already there, a Kind is a `Type lvl`, a Type needs one unfold, a term two.
172 if constexpr (sort >= UMax::Type) op = op->unfold_type();
173 if constexpr (sort == UMax::Term) op = op->unfold_type();
174 if constexpr (sort >= UMax::Kind) {
175 if (auto type = op->isa<Type>())
176 op = type->level();
177 else
178 op->blame("operand `{}` must be a `Type` of some universe level", op).bail();
179 }
180
181 flatten_umax(ops, op);
182 }
183
184 level_t lvl = 0;
185 DefVec res;
186 res.reserve(ops.size());
187 for (auto op : ops) {
188 if (!op->isa_type<Univ>())
189 op->blame("operand `{}` of a universe max must be of type `Univ` but is of type `{}`", op, type_of(op))
190 .bail();
191
192 if (auto l = Lit::isa(op))
193 lvl = std::max(lvl, *l);
194 else
195 res.emplace_back(op);
196 }
197
198 const Def* l = lit_univ(lvl);
199 if (res.empty()) return sort == UMax::Univ ? l : type(l);
200 if (lvl > 0) res.emplace_back(l);
201
202 sort_unique(res);
203 const Def* umax = unify<UMax>(*this, res);
204 return sort == UMax::Univ ? umax : type(umax);
205}

229template<bool Normalize>
230const Def* World::app(const Def* callee, const Def* arg) {
231 callee = callee->zonk();
232 arg = arg->zonk();
233
234 auto pi = callee->isa_type<Pi>();
235 if (!pi)
236 callee->blame("callee is not of function type")
237 .n("callee `{}` has type `{}`", callee, type_of(callee))
238 .n(callee->loc(), "callee `{}` declared here", callee)
239 .bail();
240
241 auto new_arg = Checker::assignable(pi->dom(), arg);
242 if (!new_arg)
243 arg->blame("argument is not assignable to callee's domain")
244 .n("expected `{}`, got `{}`", pi->dom(), type_of(arg))
245 .n(callee->loc(), "callee `{}` declared here", callee)
246 .bail();
247
248 // re-zonk after assignable check above - we might have inferred new stuff
249 arg = new_arg->zonk();
250 callee = callee->zonk();
251 pi = callee->isa_type<Pi>();
252
253 // always β-reduce non-recursive, non-parametric lambdas
254 if (auto imm = callee->isa_imm<Lam>()) return imm->body();
255
256 if (auto lam = callee->isa_mut<Lam>(); lam && lam->is_set()) {
257 auto var = lam->has_var();
258
259 // Applying a Lam to its own Var is the identity substitution, so it resolves to the body.
260 // This unfolds a self-application / fixed-point reference.
261 if (var && arg == var) return lam->body();
262
263 // β-reduce or partially evaluate a set, mutable Lam.
264 if (lam->filter() != lit_ff()) {
265 if (!var) {
266 if (lam->filter() == lit_tt()) return lam->body();
267 } else if (auto i = move_.substs.find({var, arg}); i != move_.substs.end()) {
268 // Reuse the cached reduct if its filter held.
269 auto [filter, body] = i->second->defs<2>();
270 if (filter == lit_tt()) return body;
271 } else {
272 // Evaluate the filter; if it holds, reduce the body and cache the reduct.
273 auto rw = VarRewriter(var, arg);
274 auto filter = rw.rewrite(lam->filter());
275 if (filter == lit_tt()) {
276 log().d("partially evaluate {} ({})", lam, arg);
277 auto body = rw.rewrite(lam->body());
278 cache_reduct(var, arg, {filter, body});
279 return body;
280 }
281 }
282 }
283 }
284
285 auto type = pi->reduce(arg)->zonk();
286 callee = callee->zonk();
287 auto [axm, curry, trip] = Axm::next(callee);
288
289 if (axm)
290 if (auto normalizer = axm->normalizer(); Normalize && normalizer && curry == 0)
291 if (auto norm = normalizer(type, callee, arg)) return norm;
292
293 return raw_app(axm, curry, trip, type, callee, arg);
294}

334const Def* World::tuple(const Def* type, Defs ops_) {
335 // TODO type-check type vs inferred type
336 type = type->zonk();
337 auto ops = Def::zonk(ops_);
338
339 auto n = ops.size();
340 if (!type->isa_mut<Sigma>()) {
341 if (n == 0) return tuple();
342 if (n == 1) return ops[0];
343 if (auto uni = Checker::is_uniform(ops)) return pack(n, uni);
344 }
345
346 // eta rule for tuples: (extract(tup, 0), extract(tup, 1), extract(tup, 2)) -> tup
347 if (auto ex0 = n != 0 ? ops[0]->isa<Extract>() : nullptr) {
348 auto tup = ex0->tuple();
349 bool eta = tup->type() == type;
350 for (size_t i = 0; i != n && eta; ++i) {
351 auto ex = ops[i]->isa<Extract>();
352 auto id = ex ? Lit::isa(ex->index()) : std::nullopt;
353 eta = ex && id && *id == u64(i) && ex->tuple() == tup;
354 }
355
356 if (eta) return tup;
357 }
358
359 return unify<Tuple>(type, ops);
360}

366const Def* World::extract(const Def* d, const Def* index) {
367 if (!d || !index) return nullptr; // can happen if frozen
368 d = d->zonk();
369 index = index->zonk();
370
371 // The scalar case is by far the most common one, so probe it first and only fall back to the aggregate check.
372 auto index_ty = index->unfold_type();
373 auto size = Idx::isa(index_ty);
374 auto lidx = Lit::isa(index);
375 if (!size && !isa_indices(index_ty))
376 index->blame("index `{}` must be of `Idx` type but is of type `{}`", index, type_of(index)).bail();
377
378 if (auto tuple = index->isa<Tuple>()) {
379 for (auto op : tuple->ops())
380 d = extract(d, op);
381 return d;
382 } else if (auto pack = index->isa<Pack>()) {
383 if (auto a = Lit::isa(index->arity())) {
384 for (nat_t i = 0, e = *a; i != e; ++i) {
385 auto idx = pack->has_var() ? pack->reduce(lit_idx(*a, i)) : pack->body();
386 d = extract(d, idx);
387 }
388 return d;
389 }
390 }
391
392 auto type = d->unfold_type();
393
394 if (size) {
395 if (auto l = Lit::isa(size); l && *l == 1) {
396 if (!lidx || *lidx != 0) log().w("index of `Idx 1` is not the literal 0: {}", index);
397 // A *mutable* Sigma may be a 1-tuple and still needs a real Extract; TODO mutable Arr?
398 auto sigma = type->isa_mut<Sigma>();
399 if (!sigma || sigma->num_ops() != 1) return d;
400 }
401 }
402
403 if (size && !Checker::alpha<Checker::Check>(type->arity(), size))
404 index->blame("index `{}` does not fit within arity `{}`", index, type->arity()).bail();
405 // TODO if we have indices we need to check as well that this is compatible with `d`
406
407 if (auto pack = d->isa<Pack>()) {
408 if (pack->has_var())
409 return pack->reduce(index);
410 else
411 return pack->body();
412 }
413
414 // extract(insert(x, index, val), index) -> val
415 if (auto insert = d->isa<Insert>()) {
416 if (index == insert->index()) return insert->value();
417 }
418
419 if (lidx) {
420 if (auto hole = d->isa_mut<Hole>()) d = hole->tuplefy(Idx::as_lit(index_ty));
421 if (auto tuple = d->isa<Tuple>()) return tuple->op(*lidx);
422
423 // extract(insert(x, j, val), i) -> extract(x, i) where i != j (guaranteed by rule above)
424 if (auto insert = d->isa<Insert>()) {
425 if (insert->index()->isa<Lit>()) return extract(insert->tuple(), index);
426 }
427
428 if (auto sigma = type->isa<Sigma>()) {
429 if (auto var = sigma->has_var()) {
430 if (is_frozen()) return nullptr; // if frozen, we don't risk rewriting
431 auto t = VarRewriter(var, d).rewrite(sigma->op(*lidx));
432 return unify<Extract>(t, d, index);
433 }
434
435 return unify<Extract>(sigma->op(*lidx), d, index);
436 }
437 }
438
439 const Def* elem_t;
440 if (auto arr = type->isa<Arr>()) {
441 elem_t = arr->reduce(index);
442 } else {
443 auto sigma = type->as<Sigma>();
444 elem_t = nullptr;
445 // «(a_0, ..., a_{n-1})#index; body» is more precise than the join if all ops are Arrs of the same body.
446 if (sigma->isa_imm()) {
447 const Def* body = nullptr;
448 auto extents = DefVec();
449 for (auto op : sigma->ops()) {
450 auto op_arr = op->zonk()->isa<Arr>();
451 if (!op_arr || (body && op_arr->body()->zonk() != body)) {
452 extents.clear();
453 break;
454 }
455 body = op_arr->body()->zonk();
456 extents.emplace_back(op_arr->arity());
457 }
458 if (!extents.empty()) elem_t = this->arr(extract(tuple(extents), index), body);
459 }
460 if (!elem_t) elem_t = join(sigma->ops());
461 }
462
463 if (index->isa<Top>()) {
464 if (auto hole = Hole::isa_unset(d)) {
465 auto elem_hole = mut_hole(elem_t);
466 hole->set(pack(size, elem_hole));
467 return elem_hole;
468 }
469 }
470
471 assert(d);
472 return unify<Extract>(elem_t, d, index);
473}

475const Def* World::insert(const Def* d, const Def* index, const Def* val) {
476 d = d->zonk();
477 index = index->zonk();
478 val = val->zonk();
479
480 auto type = d->unfold_type();
481 auto size = Idx::isa(index->unfold_type());
482 auto lidx = Lit::isa(index);
483
484 if (!size) index->blame("index `{}` must be of `Idx` type but is of type `{}`", index, type_of(index)).bail();
485
486 if (!Checker::alpha<Checker::Check>(type->arity(), size))
487 index->blame("index `{}` does not fit within arity `{}`", index, type->arity()).bail();
488
489 if (lidx) {
490 auto elem_type = type->proj(*lidx);
491 auto new_val = Checker::assignable(elem_type, val);
492 if (!new_val) {
493 val->blame("value is not assignable to element type")
494 .n("expected `{}`, got `{}`", elem_type, type_of(val))
495 .n("value: `{}`", val)
496 .bail();
497 }
498 val = new_val;
499 }
500
501 if (auto l = Lit::isa(size); l && *l == 1)
502 return tuple(d, {val}); // d could be mut - that's why the tuple ctor is needed
503
504 // insert((a, b, c, d), 2, x) -> (a, b, x, d)
505 if (auto t = d->isa<Tuple>(); t && lidx) {
506 auto new_ops = DefVec(t->ops().begin(), t->ops().end());
507 new_ops[*lidx] = val;
508 return tuple(type, new_ops);
509 }
510
511 // insert(‹4; x›, 2, y) -> (x, x, y, x)
512 if (auto pack = d->isa<Pack>(); pack && lidx) {
513 if (auto a = Lit::isa(pack->arity()); a && *a < flags().scalarize_threshold) {
514 auto new_ops = DefVec(*a, pack->body());
515 new_ops[*lidx] = val;
516 return tuple(type, new_ops);
517 }
518 }
519
520 // insert(insert(x, index, y), index, val) -> insert(x, index, val)
521 if (auto insert = d->isa<Insert>()) {
522 if (insert->index() == index) d = insert->tuple();
523 }
524
525 return unify<Insert>(d, index, val);
526}

528const Def* World::seq(bool is_pack, const Def* arity, const Def* body) {
529 arity = arity->zonk();
530 body = body->zonk();
531
532 auto arity_ty = arity->unfold_type();
533 if (!is_shape(arity_ty)) arity->blame("expected arity but got `{}` of type `{}`", arity, arity_ty).bail();
534
535 if (auto a = Lit::isa(arity)) {
536 if (*a == 0) return unit(is_pack);
537 if (*a == 1) return body;
538 }
539
540 // «(a, b, c); body» -> «a; «(b, c); body»»
541 // e.g. when var, but still has array type
542 if (auto arr_arity = arity_ty->isa<Seq>())
543 if (auto n = Lit::isa(arr_arity->arity())) {
544 auto inner = DefVec(*n - 1, [&](u64 i) { return arity->proj(*n, i + 1); });
545 return seq(is_pack, arity->proj(*n, 0), seq(is_pack, tuple(inner), body));
546 }
547
548 if (is_pack) return unify<Pack>(arr(arity, body->unfold_type()), body);
549 return unify<Arr>(body->unfold_type(), arity, body);
550}

589const Def* World::bound(Defs ops_) {
590 auto ops = DefVec();
591 ops.reserve(ops_.size());
592 for (auto op_ : ops_) {
593 auto op = op_->zonk();
594 if (!op->isa<TExt<!Up>>()) ops.emplace_back(op); // ignore: ext<!Up>

605 if (ops.size() == 1) return ops[0];

629const Def* World::inj(const Def* type, const Def* value) {
630 type = type->zonk();

644const Def* World::match(Defs ops_) {
645 auto ops = Def::zonk(ops_);
646 if (ops.size() == 1) return ops.front();
647
648 auto scrutinee = ops.front();
649 auto arms = ops.span().subspan(1);
650 auto join = scrutinee->isa_type<Join>();
651
652 if (!join)
653 scrutinee
654 ->blame("scrutinee `{}` of a test expression must be of union type but has type `{}`", scrutinee,
655 type_of(scrutinee))
656 .bail();
657
658 if (arms.size() != join->num_ops())
659 scrutinee->blame("test expression has {} arms but union type has {} cases", arms.size(), join->num_ops())
660 .bail();
661
662 for (auto arm : arms)
663 if (!arm->isa_type<Pi>())
664 arm->blame("arm `{}` of test expression does not have a function type but has type `{}`", arm, type_of(arm))
665 .bail();
666
667 std::ranges::sort(arms, GIDLt<const Def*>(), [](const Def* arm) { return arm->isa_type<Pi>()->dom(); });
668
669 const Def* type = nullptr;
670 for (size_t i = 0, e = arms.size(); i != e; ++i) {
671 auto arm = arms[i];
672 auto pi = arm->isa_type<Pi>();
673 if (!Checker::alpha<Checker::Check>(pi->dom(), join->op(i)))
674 arm->blame("domain type `{}` of test-expression arm does not match union case type `{}`", pi->dom(),
675 join->op(i))
676 .bail();
677 type = type ? this->join({type, pi->codom()}) : pi->codom();
678 }
679
680 // A constructor fixes the active union case. Dispatch before the Match can
681 // escape into later lowering phases, where the payload representation may
682 // already have changed (for example, a tensor may have become a buffer).
683 if (auto inj = scrutinee->isa<Inj>()) {
684 for (size_t i = 0, e = arms.size(); i != e; ++i)
685 if (Checker::alpha<Checker::Check>(inj->value()->unfold_type(), join->op(i)))
686 return app(arms[i], inj->value());
687 scrutinee->blame("injected value type `{}` is not a case of union type `{}`", type_of(inj->value()), join)
688 .bail();
689 }
690
691 return unify<Match>(type, ops);
692}

732void World::for_each(bool elide_empty, std::function<void(Def*)> f, bool schedule /* = false */) {
733 fe::BFSWorklist<MutSet> queue;
734 for (auto mut : externals().muts())
735 queue.push(mut);
736
737 auto muts = fe::Vector<Def*>();
738 while (!queue.empty()) {
739 auto mut = queue.pop();
740 if (mut->is_closed() && (!elide_empty || mut->is_set())) muts.emplace_back(mut);
741
742 for (auto op : mut->deps())
743 for (auto local_mut : op->local_muts())
744 queue.push(local_mut);
745 }
746
747 // Schedules the mutables in post-order to ensure that they
748 // are emitted in the correct order of dependencies.
749 if (schedule) {
750 const auto mut_nest = Nest(muts);
751 auto schedule = Scheduler::schedule(mut_nest) | std::views::reverse | std::views::filter([&](Def* mut) {
752 return mut->is_closed() && (!elide_empty || mut->is_set());
753 });
754 for (auto* mut : schedule)
755 f(mut);
756 } else {
757 for (auto* mut : muts)
758 f(mut);
759 }
760}
