19Splits splits(const Liveness& live, const Liveness::Block& block) {
20 const auto& target = live.target();
21 auto res = Splits(block.insns.size());
22 auto after = block.live_out;
23 for (auto v : live.term_uses(block))
24 after.emplace(v);

36 }
