22Splits splits(const Liveness& live, const Liveness::Block& block) {
23 auto res = Splits(block.insns.size());
24 auto after = block.live_out;
25 for (auto v : live.term_uses(block))
26 after.emplace(v);
27
28 for (size_t i = block.insns.size(); i-- != 0;) {
29 auto insn = block.insns[i];
30 auto c = constraint(insn);
31 res[i].pre = copies(c, [&](const Def* v) { return after.contains(v); });
32 for (auto [j, reg] : c.results)
33 if (after.contains(result(insn, j))) res[i].post.emplace_back(j, reg);
34 for (auto d : live.defs(insn))
35 after.erase(d);
36 for (auto u : live.uses(insn))
37 after.emplace(u);
38 }
39 return res;
40}
