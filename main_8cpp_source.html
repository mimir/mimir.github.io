27/// One `--output-*` option: the file name from the command line and the stream to write to.
28class Out {
29public:
30 std::string& name() { return name_; } ///< Bound to the option by fe::Cli.
31
32 /// The stream to write to; `nullptr` if this output was not requested, `std::cout` for `"-"`.
33 /// Opens the file upon first use, so an output no one writes to leaves no file behind.
34 std::ostream* os() {
35 if (name_.empty()) return nullptr;
36 if (name_ == "-") return &std::cout;
37 if (!ofs_.is_open()) {
38 ofs_.open(name_);
39 if (!ofs_) fe::throwf("cannot open output file `{}`", name_);
40 }
41 return &ofs_;
42 }
43
44private:
45 std::string name_;
46 std::ofstream ofs_;
47};

58void emit_help(fe::Cli& cli, Driver& driver, const std::vector<std::string>& plugins, bool md) {
59 for (auto&& plugin : plugins) // a plugin declares its `-X` arguments in its shared library
60 driver.load(plugin);
61
62 if (!driver.known_args().empty()) {
63 cli.section("Plugin Arguments");
64
65 for (const auto& [plugin, args] : driver.known_args()) {
66 auto rows = fe::Cli::Rows();
67 for (const auto& arg : args)
68 rows.emplace_back(arg.syntax, arg.descr);
69 // The Markdown gets an anchor, so that a plugin's own page can link to its table.
70 auto title = md ? std::format("-X {0}:<arg> {{#xarg_{0}}}", plugin) : std::format("-X {}:<arg>", plugin);
71 cli.section(std::move(title), "Argument", std::move(rows));
72 }
73 }
74 if (!driver.known_envs().empty()) {
75 cli.section("Plugin Environment Variables");
76
77 for (const auto& [plugin, envs] : driver.known_envs()) {
78 auto rows = fe::Cli::Rows();
79 for (const auto& env : envs)
80 rows.emplace_back(env.name, env.descr);
81 // The Markdown gets an anchor, so that a plugin's own page can link to its table.
82 auto title = md ? std::format("{0} {{#env_{0}}}", plugin) : std::string(plugin);
83 cli.section(std::move(title), "Variable", std::move(rows));

87 if (md)
88 cli.markdown(std::cout);

102/// Parses `Opts::input` into @p driver's World, optimizes it, and emits whatever `--output-*` asked for.
103int compile(Driver& driver, Opts& opts) {
104 auto& world = driver.world();
105 auto& outs = opts.outs;
106
107 try {
108 auto name = fs::path(opts.input).filename().replace_extension().string();
109 world.set(name);
110
111 auto ast = ast::AST(world);
112 auto parser = ast::Parser(ast);
113 auto mod = parser.import_main(opts.input, opts.plugins, outs[Md].os());
114
115 if (!mod) {
116 ast.error().ack(); // prefer the parser's own diagnostic, if it recorded one
117 fe::throwf("could not read file `{}`", opts.input);
118 }
119
120 if (auto s = outs[AST].os()) {
121 auto tab = fe::Tab::spaces();
122 mod->stream(tab, *s);
123 }
124
125 if (auto h = outs[H].os(), py = outs[PY].os(); h || py) {
126 mod->bind(ast);
127 ast.error().ack();
128 auto plugin = world.sym(name);
129 if (h) ast.bootstrap(plugin, *h);
130 if (py) ast.bootstrap_py(plugin, *py);
131 return EXIT_SUCCESS;
132 }
133
134 mod->compile(ast);
135 optimize(world);
136
137 auto types = opts.sexpr_include_types;
138 if (auto s = outs[Dot].os()) world.dot(*s, opts.dot);
139 if (auto s = outs[Mim].os()) world.dump(*s);
140 if (auto s = outs[NestDot].os()) mim::Nest(world).dot(*s);
141 if (auto s = outs[SExpr].os()) (types ? sexpr::emit_typed : sexpr::emit)(world, *s);
142 if (auto s = outs[Slotted].os()) (types ? sexpr::emit_slotted_typed : sexpr::emit_slotted)(world, *s);
143 if (auto s = outs[Profile].os()) emit_profile(driver, *s);
144 } catch (const Error::Bail& e) {
145 std::cerr << e;
146 return EXIT_FAILURE;
147 }
148
149 return EXIT_SUCCESS;
150}
