34void emit_help(fe::Cli& cli, Driver& driver, const std::vector<std::string>& plugins, bool md) {
35 for (auto&& plugin : plugins) // a plugin declares its `-X` arguments in its shared library
36 driver.load(plugin);
37
38 if (!driver.known_args().empty()) {
39 cli.section(md ? "Plugin Arguments {#xargs}" : "Plugin Arguments");
40
41 for (const auto& [plugin, args] : driver.known_args()) {
42 auto rows = fe::Cli::Rows();
43 for (const auto& arg : args)
44 rows.emplace_back(arg.syntax, arg.descr);
45 // The Markdown links back to the plugin's page and carries the anchor that page links to.
46 auto title = md ? std::format("[{0}](@ref {0}) {{#xarg_{0}}}", plugin) : plugin;
47 cli.section(std::move(title), "Argument", std::move(rows));
48 }
49 }
50 if (!driver.known_envs().empty()) {
51 cli.section(md ? "Plugin Environment Variables {#envs}" : "Plugin Environment Variables");
52
53 for (const auto& [plugin, envs] : driver.known_envs()) {
54 auto rows = fe::Cli::Rows();
55 for (const auto& env : envs)
56 rows.emplace_back(env.name, env.descr);
57 // Ditto.
58 auto title = md ? std::format("[{0}](@ref {0}) {{#env_{0}}}", plugin) : std::string(plugin);
59 cli.section(std::move(title), "Variable", std::move(rows));
60 }
61 }
62
63 if (md)
64 cli.markdown(std::cout);
65 else
66 std::cout << cli;
67}

78/// Parses `Opts::input` into @p driver's World, optimizes it, and emits whatever `--output-*` asked for.
79int compile(Driver& driver, Opts& opts) {
80 auto& world = driver.world();
81 auto& outs = opts.outs;
82
83 try {
84 auto name = fs::path(opts.input).filename().replace_extension().string();
85 world.set(name);
86
87 auto ast = ast::AST(world);
88 auto parser = ast::Parser(ast);
89 auto file = parser.import_main(opts.input, opts.plugins, outs[Md].os());
90
91 if (!file) {
92 ast.error().ack(); // prefer the parser's own diagnostic, if it recorded one
93 fe::throwf("could not read file `{}`", opts.input);
94 }
95
96 if (auto s = outs[AST].os()) {
97 auto tab = fe::Tab::spaces();
98 file->stream(tab, *s);
99
100 // Md is emitted while parsing, so nothing else needs a World; binding still catches scoping errors.
101 auto only_parse = opts.plugin_args.empty()
102 && std::ranges::none_of(std::views::iota(0, int(Num_Emits)),
103 [&](int i) { return i != AST && i != Md && outs[i].os(); });
104 if (only_parse) {
105 file->bind(ast);
106 ast.error().ack();
107 return EXIT_SUCCESS;
108 }
109 }
110
111 if (auto h = outs[H].os(), py = outs[PY].os(); h || py) {
112 file->bind(ast);
113 ast.error().ack();
114 auto plugin = world.sym(name);
115 if (h) ast.bootstrap(plugin, *h);
116 if (py) ast.bootstrap_py(plugin, *py);
117 return EXIT_SUCCESS;
118 }
119
120 file->compile(ast);
121 std::exception_ptr opt_error;
122 if (opts.no_opt)
123 Cleanup(world).run(); // resolves the Hole%s elaboration solved, as optimize would first
124 else
125 try {
126 optimize(world);
127 } catch (...) {
128 opt_error = std::current_exception(); // still emit the world the failing phase saw
129 }
130
131 if (auto s = outs[Dot].os()) world.dot(*s, opts.dot);
132 if (auto s = outs[Mim].os()) world.dump(*s);
133 if (auto s = outs[NestDot].os()) mim::Nest(world).dot(*s, opts.dot);
134 if (auto s = outs[Profile].os()) emit_profile(driver, *s);
135 if (opt_error) std::rethrow_exception(opt_error);
136 } catch (const Error::Bail& e) {
137 std::cerr << e;
138 return EXIT_FAILURE;
139 }
140
141 return EXIT_SUCCESS;
142}
