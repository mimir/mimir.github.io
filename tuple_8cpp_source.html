116std::string tuple2str(const Def* def) {

153const Def* Prod::cat(bool term, nat_t n, nat_t m, const Def* a, const Def* b) {
154 auto& w = a->world();
155 // Two *fully* spliced Seq%s of the same element concatenate without materializing their n + m projections.
156 if (auto sa = a->isa_imm<Seq>(), sb = b->isa_imm<Seq>(); sa && sb && Lit::isa(a->arity()) == n
157 && Lit::isa(b->arity()) == m && sa->is_intro() == term
158 && sb->is_intro() == term && sa->elem() == sb->elem())
159 return w.seq(term, n + m, sa->elem());
160
161 return w.prod(term, cat_projs(n, m, a, b));
162}

164const Def* Prod::cat(bool term, const Def* a, const Def* b) {

170const Def* tuple_of_types(const Def* t) {
171 auto& world = t->world();
172 if (auto sigma = t->isa<Sigma>()) return world.tuple(sigma->ops());
173 if (auto arr = t->isa<Arr>()) {
174 // One entry per *top-level* element, so a fused Arr contributes its sub-arrays, not its elements.
175 if (auto elem = arr->elem()) return world.pack(arr->arity(), elem);
176 }
177 return t;
178}
