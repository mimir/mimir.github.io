24static constexpr unsigned node2dep(Node node, bool mut) {
25 auto dep = mut ? Dep::Mut : Dep::None;
26 // clang-format off
27 switch (node) {
28 case Node::Hole: dep |= Dep::Hole; break;
29 case Node::Proxy: dep |= Dep::Proxy; break;
30 case Node::Var: dep |= Dep::Var | Dep::Mut; break;
31 default: break;
32 }
33 // clang-format on
34 return fe::to_underlying(dep);
35}

247// free_vars() is a union, so any predicate over it distributes over that union: ask f for the local Vars and
248// then for each local mutable's free_vars() - instead of merging them all into one throw-away Set.
249template<class F>
250static bool any_free_vars(const Def* def, F f) {
251 if (auto mut = def->isa_mut()) return f(mut->free_vars());

253 if (f(def->local_vars())) return true;

272Vars Def::free_vars() {
273 if (mark_ == 0) {
274 // fixed-point iteration to recompute free vars:
275 // (run - 1) identifies the previous iteration; so make sure to offset run by 2 for the first iteration
276 auto& w = world();
277 bool cyclic = false;
278 w.next_run();
279 free_vars<true>(w, cyclic, w.next_run());
280
281 for (bool todo = cyclic; todo;) {
282 todo = false;
283 free_vars<false>(w, todo, w.next_run());
284 }
285 }
286
287 return vars_;
288}

290// w is threaded through instead of re-deriving it via world() - which chases the type chain - at every node.
291template<bool init>
292Vars Def::free_vars(World& w, bool& todo, u32 run) {
293 // If init == true : todo flag detects cycle.
294 // If init == false: todo flag keeps track whether sth changed.
295 //
296 // Recursively recompute free vars. If
297 // * mark_ == 0: Invalid - need to recompute.
298 // * mark_ == run - 1: Previous iteration - need to recompute.
299 // * mark_ == run: We are running in cycles within the *current* iteration of our fixed-point loop.
300 // * otherwise: Valid!
301 if (mark_ != 0 && mark_ != run - 1) {
302 if constexpr (init) todo |= mark_ == run;
303 return vars_;
304 }
305
306 mark_ = run;
307
308 auto fvs0 = vars_;
309 auto fvs = fvs0;
310 auto& muts = w.muts();
311 auto& vars = w.vars();
312
313 for (auto op : deps()) {
314 if constexpr (init) fvs = vars.merge(fvs, op->local_vars());
315
316 for (auto mut : op->local_muts()) {
317 if constexpr (init) mut->muts_ = muts.insert(mut->muts_, this); // register "this" as user of local_mut
318 fvs = vars.merge(fvs, mut->free_vars<init>(w, todo, run));
319 }
320 }
321
322 if (auto var = has_var()) fvs = vars.erase(fvs, var); // FV(λx.e) = FV(e) \ {x}
323
324 if constexpr (!init) todo |= fvs0 != fvs;
325
326 return vars_ = fvs;
327}

354 auto var = has_var(); // `this` is fixed across the recursion, and so is its Var
355 auto fvs = mut->free_vars();

428Defs Def::deps() const noexcept {
429 // deps() hands out `[type_, op0, op1, ...]` as one contiguous array by stepping back from ops_ptr()
430 // (which is `this + 1`). That only holds while type_ occupies the *last* 8 bytes of Def, so moving it
431 // - or appending any member after it - would silently corrupt every deps() walk.
432 assert((const void*)(ops_ptr() - 1) == (const void*)&type_
433 && "Def::type_ must stay Def's last member: Def::deps() and Def::ops_ptr() depend on it");
434
435 // Univ, Type, and Var are the only nodes built without a type - and none of them has deps.
436 if (!type_) {
437 assert(isa<Univ>() || isa<Type>() || isa<Var>());
438 return Defs();
439 }
440
441 // Not is_set(): its assertion scans *all* ops and deps() is an inner loop.
442 bool set = num_ops_ == 0 || ops_ptr()[num_ops_ - 1];
443 return Defs(ops_ptr() - 1, (set ? num_ops_ : 0) + 1);
444}

462Def::Cmp Def::cmp(const Def* a, const Def* b) {
463 if (a == b) return Cmp::E;
464
465 if (a->is_mutable() != b->is_mutable()) return a->is_mutable() ? Cmp::G : Cmp::L;
466
467 // clang-format off
468 if (a->node() != b->node() ) return a->node() < b->node() ? Cmp::L : Cmp::G;
469 if (a->num_ops() != b->num_ops()) return a->num_ops() < b->num_ops() ? Cmp::L : Cmp::G;
470 if (a->flags() != b->flags() ) return a->flags() < b->flags() ? Cmp::L : Cmp::G;
471 // clang-format on
472
473 if (a->is_mutable()) return Cmp::U; // two distinct mutables of the same shape are incomparable
474
475 if (auto va = a->isa<Var>()) {
476 auto vb = b->as<Var>();
477 auto ma = va->binder();
478 auto mb = vb->binder();
479 if (ma->is_set() && ma->free_vars().contains(vb)) return Cmp::L;
480 if (mb->is_set() && mb->free_vars().contains(va)) return Cmp::G;
481 return Cmp::U;
482 }
483
484 // heuristic: iterate backwards as index (often a Lit) comes last and will faster find a solution
485 for (size_t i = a->num_ops(); i-- != 0;)
486 if (auto res = cmp(a->op(i), b->op(i)); res == Cmp::L || res == Cmp::G) return res;
487
488 return cmp(a->type(), b->type());
489}

496 return c == Cmp::L ? a->gid() < b->gid() : a->gid() > b->gid();

515const Def* Def::immutabilize() {
516 auto& w = world();
517 switch (node()) {
518 case Node::Pi: return is_immutabilizable() ? w.pi(as<Pi>()->dom(), as<Pi>()->codom()) : nullptr;
519 case Node::Sigma: return is_immutabilizable() ? w.sigma(ops()) : nullptr;
520 case Node::Rule: return nullptr; // TODO should we ever immutabilize Rules?
521 case Node::Arr:
522 case Node::Pack: {
523 auto seq = as<Seq>();
524 auto arr = node() == Node::Arr;
525 if (is_immutabilizable())
526 return arr ? w.arr(seq->arity(), seq->body()) : w.pack(seq->arity(), seq->body());
527 // below the threshold an unrollable Seq becomes an explicit Sigma/Tuple
528 if (auto n = Lit::isa(seq->arity()); n && *n < w.flags().scalarize_threshold) {
529 auto elems = DefVec(*n, [&](size_t i) { return seq->reduce(w.lit_idx(*n, i)); });
530 return arr ? w.sigma(elems) : w.tuple(elems);
531 }
532 return nullptr;
533 }
534 default: return nullptr;
535 }
