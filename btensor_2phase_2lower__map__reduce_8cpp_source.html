28const Def* LowerMapReduce::lower_buffer_lit(const App* app) {
29 // `buffer.lit (r, s, T) (mem, val)` fills every elem with `val`. Emit a fill loop (via the same
30 // pointwise scaffold as pad/concat) so the store never materializes as one giant literal array. A
31 // non-literal rank has no static loop nest, so leave it for `buffer.lower_ptr`'s monolithic fallback.
32 auto [r, s, T] = app->callee()->as<App>()->args<3>();
33 auto s_out = rewrite(s);
34 auto rn = Lit::isa<u64>(rewrite(r));
35 if (!rn) return RWPhase::rewrite_imm_App(app);
36
37 auto [mem, val] = rewrite(app->arg())->projs<2>();
38 auto result_ty = rewrite(app->type()); // [mem.M 0, buffer.Buf (r, s, T)]
39 auto [obr, obs, obT] = Axm::isa<buffer::Buf>(result_ty->proj(2, 1))->args<3>();
40 auto [a_mem, buf] = buffer::op_alloc(obr, obs, obT, mem)->projs<2>();
41 // `compute` ignores the loop counters and writes the (loop-invariant) scalar `ins` everywhere.
42 return build_pointwise(
43 result_ty, a_mem, buf, val, s_out, *rn, "constant_fill",
44 [](const DefVec&, const Def* ins, const Def* m) -> std::pair<const Def*, const Def*> { return {m, ins}; });
45}

47const Def* LowerMapReduce::lower_map_reduce_post(const App* app) {
48 auto& w = new_world();
49 auto mr = MapRedPost::of(rewrite(app->callee())->as<App>());
50 auto ranks = mr.ranks();
51 if (!ranks) {
52 log().w("rank counts (nis/nps/Ro/Rn) of {} are not known at lowering time", app);
53 return RWPhase::rewrite_imm_App(app);
54 }
55
56 auto impl = rewrite(old_world().annex<btensor::map_reduce_post_impl>());
57 impl = w.app(impl, mr.nis_nps);
58 impl = w.app(impl, {mr.To, mr.Tp, mr.Ro, w.lit_nat(ranks->rr)});
59 impl = w.app(impl, mr.shapes);
60 impl = w.app(impl, {mr.Tis, mr.Tps});
61 impl = w.app(impl, {mr.comb, mr.init, mr.post});
62 impl = w.app(impl, mr.map_out);
63
64 auto [mem, dst, inputs, post_inputs] = rewrite(app->arg())->projs<4>();
65 auto [read_inputs_ty, read_post_ty, write_output_ty] = impl->type()->as<Pi>()->dom()->projs<3>();
66
67 // A `buffer.Buf` has its literal size-1 axes folded away, so the indices fold alike.
68 auto reader = [&](const Def* type, const char* name, u64 num, const Def* rank, const Def* bounds, const Def* rs,
69 const Def* ss, const Def* maps, const Def* buffers) {
70 auto fun = w.mut_lam(type->as<Pi>())->set(name);
71 auto [args, ret] = fun->vars<2>();
72 auto [m, indices] = args->projs<2>();
73 auto values = DefVec(num);
74 for (u64 i = 0; i != num; ++i) {
75 auto si = ss->proj(num, i);
76 auto [mapped, coords]
77 = op_affine_map(maps->proj(num, i), rs->proj(num, i), rank, bounds, si, indices, m)->projs<2>();
78 auto [read, value] = buffer::op_read(mapped, buffers->proj(num, i), *Shape(coords).fold(si))->projs<2>();
79 m = read;
80 values[i] = value;
81 }
82 fun->app(true, ret, {m, w.tuple(values)});
83 return fun;
84 };
85 auto read_inputs = reader(read_inputs_ty, "readInputs", ranks->nis, mr.Rn, mr.Sr, mr.Ris, mr.Sis, mr.maps, inputs);
86 auto read_post
87 = reader(read_post_ty, "readPost", ranks->nps, mr.Ro, mr.So, mr.Rps, mr.Sps, mr.post_maps, post_inputs);
88
89 auto write_output = w.mut_lam(write_output_ty->as<Pi>())->set("writeOutput");
90 {
91 auto [args, ret] = write_output->vars<2>();
92 auto [m, output, coords, value] = args->projs<4>();
93 auto stored = buffer::op_write(m, output, *Shape(coords).fold(mr.So), value);
94 write_output->app(true, ret, stored->proj(2, 0));
95 }
96
97 impl = w.app(impl, {read_inputs, read_post, write_output});
98 return w.app(impl, {mem, dst});
99}

101const Def* LowerMapReduce::lower_broadcast(const App* app) {
102 auto& w = new_world();
103 auto callee = app->callee()->as<App>(); // (broadcast {impl}) (s_in, s_out)
104 auto [s_in, s_out] = rewrite(callee->arg())->projs<2>();
105 auto [op_mem, dst, input] = rewrite(app->arg())->projs<3>();
106 auto result_ty = rewrite(app->type()); // [mem.M 0, buffer.Buf (ro, so, T)]
107
108 auto r_l = Lit::isa(s_out->arity());
109 if (!r_l) {
110 log().w("rank of {} is not known at lowering time", app);
111 return RWPhase::rewrite_imm_App(app);
112 }
113
114 // Non-size-1 input dims mirror the matching output index; size-1 dims are dropped from each buffer index.
115 auto compute = [&](const DefVec& iters, const Def* in_buf, const Def* mem) -> std::pair<const Def*, const Def*> {
116 auto idxs = w.tuple(to_idxs(w, s_out->projs(*r_l), iters));
117 auto [rd_mem, rd_val] = buffer::op_read(mem, in_buf, *Shape(idxs).fold(s_in))->projs<2>();
118 return {rd_mem, rd_val};
119 };
120
121 return build_pointwise(result_ty, op_mem, dst, input, s_out, *r_l, "broadcast", compute);
122}

124const Def* LowerMapReduce::lower_pad(const App* app) {
125 auto& w = new_world();
126 auto c = rewrite(app->callee())->as<App>();
127
128 // callee: pad {T, r} [s_in] [mode, lo, hi] [s_out]. The shapes are the logical ones; buffer reads and
129 // writes fold size-1 axes (the `Buf` handles are normalized), while the loops cover all logical dims.
130 auto [Tr, s_in, params, s_out] = c->uncurry_args<4>();
131 auto [mode, lo, hi] = params->projs<3>();
132 auto [op_mem, dst, input, val] = rewrite(app->arg())->projs<4>();
133 auto result_ty = rewrite(app->type()); // [mem.M 0, buffer.Buf (r, s_out, T)]
134
135 auto r_l = Lit::isa<u64>(Tr->proj(2, 1));
136 auto mode_l = Lit::isa<u64>(mode);
137 if (!r_l || !mode_l) {
138 log().w("rank/mode of {} is not known at lowering time", app);
139 return RWPhase::rewrite_imm_App(app);
140 }
141
142 auto compute = [&](const DefVec& iters, const Def* ins, const Def* mem) -> std::pair<const Def*, const Def*> {
143 auto [in_buf, fill] = ins->projs<2>();
144 auto [coords, valid] = pad_coords(iters, lo, s_in, *mode_l != 0);
145 auto [rd_mem, elem] = buffer::op_read(mem, in_buf, *Shape(w, coords).fold(s_in))->projs<2>();
146 return {rd_mem, valid ? w.select(valid, elem, fill) : elem}; // constant: fill out-of-region cells with `val`
147 };
148
149 return build_pointwise(result_ty, op_mem, dst, w.tuple({input, val}), s_out, *r_l, "pad", compute);
150}

152const Def* LowerMapReduce::lower_concat(const App* app) {
153 auto c = rewrite(app->callee())->as<App>();
154
155 // callee: concat {T, nis, r} [ax] {Sis} [s_out]. The shapes are the logical ones; buffer reads and
156 // writes fold size-1 axes (the `Buf` handles are normalized), while the loops cover all logical dims.
157 auto [TnisR, ax, Sis, s_out] = c->uncurry_args<4>();
158 auto [T, nis, r] = TnisR->projs<3>();
159 auto [op_mem, dst, op_is] = rewrite(app->arg())->projs<3>();
160 auto result_ty = rewrite(app->type()); // [mem.M 0, buffer.Buf (r, s_out, T)]
161
162 auto nis_l = Lit::isa<u64>(nis);
163 auto r_l = Lit::isa<u64>(r);
164 auto ax_l = Lit::isa<u64>(ax);
165 if (!nis_l || !r_l || !ax_l) {
166 log().w("nis/rank/axis of {} are not known at lowering time", app);
167 return RWPhase::rewrite_imm_App(app);
168 }
169 auto nisn = *nis_l, rn = *r_l, axn = *ax_l;
170
171 auto exts = concat_extents(Sis, nisn, rn, axn);
172 if (!exts) {
173 log().w("an extent of {} along the concat axis is not known at lowering time", app);
174 return RWPhase::rewrite_imm_App(app);
175 }
176
177 auto compute = [&](const DefVec& iters, const Def* ins, const Def* mem) -> std::pair<const Def*, const Def*> {
178 const Def* cur = mem;
179 auto read = [&](u64 i, const Def* ax_idx) {
180 auto coords = iters;
181 coords[axn] = ax_idx;
182 auto [rd_mem, rd_val]
183 = buffer::op_read(cur, ins->proj(nisn, i), folded_idx(Sis->proj(nisn, i), coords))->projs<2>();
184 cur = rd_mem;
185 return rd_val;
186 };
187 auto result = concat_select(iters[axn], *exts, read);
188 return {cur, result};
189 };
190
191 return build_pointwise(result_ty, op_mem, dst, op_is, s_out, rn, "concat", compute);
192}

222const Def* LowerMapReduce::lower_scatter(const App* app) {
223 auto& w = new_world();
224 auto c = rewrite(app->callee())->as<App>();
225
226 auto [Tr, shapes, dim] = c->uncurry_args<3>();
227 auto [T, r] = Tr->projs<2>();
228 auto [s_src, s_idx, s_updates] = shapes->projs<3>();
229 auto [op_mem, op_dst, input, idx, updates] = rewrite(app->arg())->projs<5>();
230 auto result_ty = rewrite(app->type());
231
232 auto r_l = Lit::isa<u64>(r);
233 auto dim_l = Lit::isa<u64>(dim);
234 if (!r_l || !dim_l) {
235 log().w("{} doesn't have lowering-time known rank/axis", app);
236 return RWPhase::rewrite_imm_App(app);
237 }
238
239 auto [fun, call] = ds_splice(w.tuple({op_mem, op_dst, input, idx, updates}), result_ty, "scatter");
240 auto [fun_mem, out_buf, in_buf, idx_buf, update_buf] = fun->var(2, 0)->projs<5>();
241 const Def* cont = fun->var(2, 1);
242
243 const Def* acc = w.tuple({buffer::op_copy(fun_mem, out_buf, in_buf), out_buf});
244 auto current = fun;
245 auto iters = affine::counting_nest(current, cont, acc, s_idx->projs(*r_l), "scatter");
246 auto [loop_mem, loop_buf] = acc->projs<2>();
247
248 auto [idx_mem, selected] = buffer::op_read(loop_mem, idx_buf, folded_idx(s_idx, iters))->projs<2>();
249 auto [update_mem, update] = buffer::op_read(idx_mem, update_buf, folded_idx(s_updates, iters))->projs<2>();
250 auto dst = iters;
251 dst[*dim_l] = w.call<core::bitcast>(w.type_i64(), selected);
252 auto write_mem = buffer::op_write(update_mem, loop_buf, folded_idx(s_src, dst), update)->proj(2, 0);
253 current->app(true, cont, w.tuple({write_mem, loop_buf}));
254 return call;
255}
