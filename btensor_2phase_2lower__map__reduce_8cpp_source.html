28const Def* LowerMapReduce::lower_buffer_lit(const App* app) {
29 // `buffer.lit (r, s, T) (mem, val)` fills every elem with `val`. Emit a fill loop (via the same
30 // pointwise scaffold as pad/concat) so the store never materializes as one giant literal array. A
31 // non-literal rank has no static loop nest, so leave it for `buffer.lower_ptr`'s monolithic fallback.
32 auto [r, s, T] = app->callee()->as<App>()->args<3>();
33 auto s_out = rewrite(s);
34 auto rn = Lit::isa<u64>(rewrite(r));
35 if (!rn) return RWPhase::rewrite_imm_App(app);
36
37 auto [mem, val] = rewrite(app->arg())->projs<2>();
38 auto result_ty = rewrite(app->type()); // [mem.M 0, buffer.Buf (r, s, T)]
39 auto [obr, obs, obT] = Axm::isa<buffer::Buf>(result_ty->proj(2, 1))->args<3>();
40 auto [a_mem, buf] = buffer::op_alloc(obr, obs, obT, mem)->projs<2>();
41 // `compute` ignores the loop counters and writes the (loop-invariant) scalar `ins` everywhere.
42 return build_pointwise(
43 result_ty, a_mem, buf, val, s_out, *rn, "constant_fill",
44 [](const DefVec&, const Def* ins, const Def* m) -> std::pair<const Def*, const Def*> { return {m, ins}; });
45}

47const Def* LowerMapReduce::lower_map_reduce_post(const App* app) {
48 auto& w = new_world();
49 auto mr = MapRedPost::of(rewrite(app->callee())->as<App>());
50
51 // The final argument is `[mem, dst, is, post_is]`; the result is `[mem, Buf]`.
52 auto [op_mem, op_dst, op_is, op_post_is] = rewrite(app->arg())->projs<4>();
53 auto result_ty = rewrite(app->type()); // [mem.M 0, buffer.Buf (Ro, So, Tp)]
54
55 auto ranks = mr.ranks();
56 if (!ranks) {
57 log().w("rank counts (nis/nps/Ro/Rn) of {} are not known at lowering time", app);
58 return RWPhase::rewrite_imm_App(app);
59 }
60 auto [nis_nat, nps_nat, ro, rr] = *ranks;
61 auto nloops = ranks->nloops();
62 auto n = w.lit_nat(nloops);
63
64 // `[mem, dst, is, post_is] → [mem, Buf]`, spliced via cps.cps2ds and applied to the op's operands.
65 auto [fun, call] = ds_splice(w.tuple({op_mem, op_dst, op_is, op_post_is}), result_ty, "mapRedAff");
66 auto [fun_mem, out_buf, new_inputs, new_post_is] = fun->var(2, 0)->projs<4>();
67 auto cont = fun->var(2, 1);
68
69 // The op's SCHEDULE `sched` is a target-agnostic chooser over a loop-nest builder (canonically
70 // a `tensor.mk_sched` value, selected in the frontend). This is the tensor→btensor boundary,
71 // so bind it HERE to this target's algebra — `btensor.mr_nest` over the output buffer — then
72 // build only the decision-free pieces: the fold step `cell` (read one element per input, call
73 // the combiner) and the write-back `wb` (read the epilogue inputs, run `post`, store) — and
74 // APPLY the nest to them. Unrolling, interchange and the row accumulator are inside the
75 // builder: plain IR, not lowering behavior.
76 auto i32 = w.type_i32();
77
78 auto [obr, obs, obT] = Axm::isa<buffer::Buf>(result_ty->proj(2, 1))->args<3>();
79
80 auto nest_args = w.tuple({mr.Ro, w.lit_nat(rr), mr.Sr, mr.To, result_ty->proj(2, 1)});
81 auto nest = w.app(w.app(mr.sched, w.app(w.annex<btensor::NestT>(), nest_args)),
82 w.app(w.annex<btensor::mr_nest>(), nest_args));
83
84 // The bound nest dictates the exact `cell`/`wb` signatures (its [init, cell, wb] domain) —
85 // building them from the VALUE's own type sidesteps any Arr/Sigma normalization asymmetry.
86 auto sched_dom = nest->type()->as<Pi>()->dom();
87
88 // The nest value's loop-vector components may appear as one «r; I32» value or flattened into r
89 // separate I32 components (normalization decides) — index the domains verbatim either way.
90 auto load_ivs = [&](Lam* l, u64 ndom, u64 pos, u64 cnt) {
91 DefVec out(cnt);
92 if (ndom == pos + cnt + 1) // flattened: cnt I32 scalars, then the continuation
93 for (u64 d = 0; d < cnt; ++d)
94 out[d] = l->var(ndom, pos + d);
95 else
96 for (u64 d = 0; d < cnt; ++d)
97 out[d] = l->var(ndom, pos)->proj(cnt, d);
98 return out;
99 };
100
101 auto sr = mr.Sr->projs(nloops);
102
103 // cell: Cn [mem, To, «ro+rr; I32», Cn [mem, To]] — fold the elements at one loop vector.
104 auto cdom = sched_dom->proj(3, 1)->as<Pi>()->dom();
105 auto cn = cdom->num_projs();
106 auto cell = w.mut_con(cdom)->set("cell");
107 {
108 auto cm = cell->var(cn, 0);
109 auto cacc = cell->var(cn, 1);
110 auto ck = cell->var(cn, cn - 1);
111 auto iters = w.tuple(to_idxs(w, sr, load_ivs(cell, cn, 2, nloops)));
112 auto cur = cm;
113 DefVec input_elems(nis_nat);
114 for (u64 i = 0; i < nis_nat; ++i) {
115 auto sis_i = mr.Sis->proj(nis_nat, i);
116 auto [mc_mem, coords]
117 = op_affine_map(mr.maps->proj(nis_nat, i), mr.Ris->proj(nis_nat, i), n, mr.Sr, sis_i, iters, cur)
118 ->projs<2>();
119 auto [rd_mem, rd_val]
120 = buffer::op_read(mc_mem, new_inputs->proj(nis_nat, i), *Shape(coords).fold(sis_i))->projs<2>();
121 cur = rd_mem;
122 input_elems[i] = rd_val;
123 }
124 app_cps(cell, mr.comb, {cur, cacc, w.tuple(input_elems)}, ck);
125 }
126
127 // wb: Cn [mem, Buf, To, «ro+rr; I32», Cn [mem, Buf]] — epilogue + store for one folded cell,
128 // threading the output buffer as the nest's write-back target. It receives the full loop
129 // vector; only the leading `ro` output coordinates are read (the trailing reduction slots are
130 // exhausted loop values and are replaced by zeros for `map_out`).
131 auto wdom = sched_dom->proj(3, 2)->as<Pi>()->dom();
132 auto wn = wdom->num_projs();
133 auto wb = w.mut_con(wdom)->set("wb");
134 {
135 auto wm = wb->var(wn, 0);
136 auto wu = wb->var(wn, 1);
137 auto wv = wb->var(wn, 2);
138 auto wk = wb->var(wn, wn - 1);
139 auto wovs = load_ivs(wb, wn, 3, nloops);
140 auto wb_iters = to_idxs(w, Defs(sr).subspan(0, ro), Defs(wovs).subspan(0, ro));
141 for (u64 j = 0; j < rr; ++j)
142 wb_iters.emplace_back(w.call(core::conv::u, sr[ro + j], w.lit(i32, 0)));
143 auto [wc_mem, write_coords]
144 = op_affine_map(mr.map_out, mr.Ro, n, mr.Sr, mr.So, w.tuple(wb_iters), wm)->projs<2>();
145
146 auto pcur = wc_mem;
147 DefVec post_elems(nps_nat);
148 for (u64 j = 0; j < nps_nat; ++j) {
149 auto sps_j = mr.Sps->proj(nps_nat, j);
150 auto [pc_mem, pcoords] = op_affine_map(mr.post_maps->proj(nps_nat, j), mr.Rps->proj(nps_nat, j), mr.Ro,
151 mr.So, sps_j, write_coords, pcur)
152 ->projs<2>();
153 auto [prd_mem, p_val]
154 = buffer::op_read(pc_mem, new_post_is->proj(nps_nat, j), *Shape(pcoords).fold(sps_j))->projs<2>();
155 pcur = prd_mem;
156 post_elems[j] = p_val;
157 }
158 auto after_post = mem::mut_con(mr.Tp)->set("afterPost");
159 auto [post_mem, elem_post] = after_post->vars<2>();
160 auto stored = buffer::op_write(obr, obs, obT, post_mem, wu, *Shape(write_coords).fold(mr.So), elem_post);
161 after_post->app(true, wk, stored);
162 app_cps(wb, mr.post, {pcur, wv, w.tuple(post_elems)}, after_post);
163 }
164
165 // Apply the nest; the output buffer is threaded through as the nest's write-back target and
166 // yielded straight to the op's continuation.
167 fun->app(true, w.app(nest, w.tuple({mr.init, cell, wb})), w.tuple({fun_mem, out_buf, cont}));
168
169 return call;
170}

176 auto [op_mem, dst, input] = rewrite(app->arg())->projs<3>();

178

185 // Non-size-1 input dims mirror the matching output index; size-1 dims are dropped from each buffer index.

196 auto& w = new_world();
197 auto c = rewrite(app->callee())->as<App>();
198
199 // callee: pad {T, r} [s_in] [mode, lo, hi] [s_out]. The shapes are the logical ones; buffer reads and
200 // writes fold size-1 axes (the `Buf` handles are normalized), while the loops cover all logical dims.
201 auto [Tr, s_in, params, s_out] = c->uncurry_args<4>();
202 auto [mode, lo, hi] = params->projs<3>();
203 auto [op_mem, dst, input, val] = rewrite(app->arg())->projs<4>();
204 auto result_ty = rewrite(app->type()); // [mem.M 0, buffer.Buf (r, s_out, T)]
205
206 auto r_l = Lit::isa<u64>(Tr->proj(2, 1));
207 auto mode_l = Lit::isa<u64>(mode);
208 if (!r_l || !mode_l) {
209 log().w("rank/mode of {} is not known at lowering time", app);
210 return RWPhase::rewrite_imm_App(app);
211 }
212
213 auto compute = [&](const DefVec& iters, const Def* ins, const Def* mem) -> std::pair<const Def*, const Def*> {
214 auto [in_buf, fill] = ins->projs<2>();
215 auto [coords, valid] = pad_coords(iters, lo, s_in, *mode_l != 0);
216 auto [rd_mem, elem] = buffer::op_read(mem, in_buf, *Shape(w, coords).fold(s_in))->projs<2>();
217 return {rd_mem, valid ? w.select(valid, elem, fill) : elem}; // constant: fill out-of-region cells with `val`
218 };
219
220 return build_pointwise(result_ty, op_mem, dst, w.tuple({input, val}), s_out, *r_l, "pad", compute);

223const Def* LowerMapReduce::lower_concat(const App* app) {
224 auto c = rewrite(app->callee())->as<App>();
225
226 // callee: concat {T, nis, r} [ax] {Sis} [s_out]. The shapes are the logical ones; buffer reads and
227 // writes fold size-1 axes (the `Buf` handles are normalized), while the loops cover all logical dims.
228 auto [TnisR, ax, Sis, s_out] = c->uncurry_args<4>();
229 auto [T, nis, r] = TnisR->projs<3>();
230 auto [op_mem, dst, op_is] = rewrite(app->arg())->projs<3>();
231 auto result_ty = rewrite(app->type()); // [mem.M 0, buffer.Buf (r, s_out, T)]
232
233 auto nis_l = Lit::isa<u64>(nis);
234 auto r_l = Lit::isa<u64>(r);
235 auto ax_l = Lit::isa<u64>(ax);
236 if (!nis_l || !r_l || !ax_l) {
237 log().w("nis/rank/axis of {} are not known at lowering time", app);
238 return RWPhase::rewrite_imm_App(app);
239 }
240 auto nisn = *nis_l, rn = *r_l, axn = *ax_l;
241
242 auto exts = concat_extents(Sis, nisn, rn, axn);
243 if (!exts) {
244 log().w("an extent of {} along the concat axis is not known at lowering time", app);
245 return RWPhase::rewrite_imm_App(app);
246 }
247
248 auto compute = [&](const DefVec& iters, const Def* ins, const Def* mem) -> std::pair<const Def*, const Def*> {
249 const Def* cur = mem;
250 auto read = [&](u64 i, const Def* ax_idx) {
251 auto coords = iters;
252 coords[axn] = ax_idx;
253 auto [rd_mem, rd_val]
254 = buffer::op_read(cur, ins->proj(nisn, i), folded_idx(Sis->proj(nisn, i), coords))->projs<2>();
255 cur = rd_mem;
256 return rd_val;
257 };
258 auto result = concat_select(iters[axn], *exts, read);
259 return {cur, result};
260 };
261
262 return build_pointwise(result_ty, op_mem, dst, op_is, s_out, rn, "concat", compute);
263}
