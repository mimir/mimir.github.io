156void HostEmitter::emit_epilogue(Lam* lam) {
157 auto& bb = lam2bb_[lam];
158
159 // HACK: we partially re-implement the checks in Super::emit_epilogue to catch target-specific applications
160 auto app = lam->body()->as<App>();
161 if (auto ret = isa_targetspecific_intrinsic(bb, app)) {
162 assert(ret.has_value());
163 if (app->callee() == root()->ret_var()) // return
164 assert(false && "Return not implemented in NVPTX backend");
165 else if (auto dispatch = Dispatch(app))
166 assert(false && "Dispatch not implemented in NVPTX backend");
167 else if (app->callee()->isa<Bot>())
168 assert(false && "Bot not implemented in NVPTX backend");
169 else if (auto _ = Lam::isa_mut_basicblock(app->callee())) // ordinary jump
170 assert(false && "Ordinary Jump not implemented in NVPTX backend");
171 else if (Pi::isa_returning(app->callee_type())) // function call
172 bb.tail("br label {}", ret.value());
173 else
174 assert(false && "Unexpected return case in NVPTX backend");
175 } else {
176 Super::emit_epilogue(lam);
177 }
178}

470
471 // HACK: we partially re-implement the checks in Super::emit_epilogue to catch target-specific applications

483 std::print(vars_decls_, "{} = internal addrspace({}) global {} undef\n", v_ptr, a, convert(T));
484 lam2bb_[ret_lam].phis[ptr].emplace_back(v_ptr, id(lam, true));
485 globals_[ptr] = id(ptr);
486 return bb.tail("br label {}", id(ret_lam));

502std::string DeviceEmitter::prepare() {
503 auto is_kern = kernels_.contains(root());
504 if (!is_kern) return Super::prepare();
505 auto kernel = root();
506
507 std::print(func_impls_, "define ptx_kernel {} {}(", convert_ret_pi(kernel->type()->ret_pi()), id(kernel));
508
509 auto [m1, m3, m4, m5, group_id, item_id, smem, arg, ret_lam] = kernel->vars<9>();
510
511 auto arg_name = id(arg);
512 locals_[arg] = arg_name;
513 std::print(func_impls_, "{} {}) {{\n", convert(arg->type()), arg_name);
514
515 auto& bb = lam2bb_[kernel];
516
517 auto register_sreg_idx = [&](const Def* def, std::string_view sreg) {
518 auto name = id(def);
519 auto type = def->type();
520 auto type_name = convert(type);
521 auto opt_idx_lit = Idx::isa_lit(type);
522 if (!opt_idx_lit) error("Type of '{}' must have known index type but has {}", def, type);
523 auto idx_lit = opt_idx_lit.value();
524 locals_[def] = name;
525 declare("i32 @llvm.nvvm.read.ptx.sreg.{}()", sreg);
526 if (type_name == "i0") {
527 locals_[def] = "0";
528 } else if (type_name == "i32") {
529 bb.assign(name, "call i32 @llvm.nvvm.read.ptx.sreg.{}()", sreg);
