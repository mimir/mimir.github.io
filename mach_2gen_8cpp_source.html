63/// Does @p def use @p meta, a projection of the Var @p var or a Var of its own, directly or by @p var as a whole?
64bool uses(const Def* def, const Def* meta, const Var* var, DefSet& done) {
65 if (def == meta) return true;
66 if (!def || !done.emplace(def).second) return false;
67 if (auto ex = meta->isa<Extract>(); def == var && ex && ex->tuple() == var) return true;
68 if (auto ex = def->isa<Extract>(); ex && ex->tuple() == var && Lit::isa(ex->index())) return false;
69 if (auto type = def->type(); type && uses(type, meta, var, done)) return true;
70 for (auto op : def->ops())
71 if (uses(op, meta, var, done)) return true;
72 return false;

291/// Adds the Rule in the body of the implicit family @p lam, possibly under further implicit families.
292/// If a `refly.check` in that Rule or its guard mentions the family parameter, it is instantiated at each value of its
293/// universe that the checks admit, and the guard decides once all are instantiated. Otherwise, the parameter is a meta
294/// variable of the Rule.
295void Table::family(Lam* lam) {
296 if (!seen_.emplace(lam).second) return;
297 auto body = lam->body();
298 while (body->isa_mut<Lam>() && body->type()->as<Pi>()->is_implicit())
299 body = body->as_mut<Lam>()->body();
300 auto rule = body->isa<Rule>();
301 if (!rule) lam->blame("{}: the body of the family `{}` is no rule", gen_.isel(), lam).bail();
302 auto param = lam->has_var();
303 if (!param) return collect(lam->body());
304
305 // A condition that also mentions an inner family parameter is decided once that one is instantiated.
306 auto conds = DefVec();
307 auto done = DefSet();
308 for (auto def : {rule->type()->dom(), rule->type()->codom(), rule->rhs()})
309 checks(def, conds, done);
310 std::erase_if(conds, [&](const Def* c) {
311 auto fvs = c->free_vars();
312 return fvs.size() != 1 || !fvs.contains(param);
313 });
314 if (conds.empty() && !rule->guard()->has_free_var(param)) {
315 if (lam->body() != rule)
316 lam->blame("{}: `{}` of `{}` must be instantiated", gen_.isel(), lam->dom(), lam).bail();
317 return add(rule, param);
318 }
319
320 auto& w = lam->world();
321 auto dom = lam->dom();
322 auto universe = dom == w.type_nat() ? &sizes_ : dom == w.arr(2, w.type_nat()) ? &formats_ : nullptr;
323 if (!universe) lam->blame("{}: no values to instantiate `{}` of `{}` with", gen_.isel(), dom, lam).bail();
324 for (auto val : *universe) {
325 auto ok = true;
326 for (auto cond : conds) {
327 auto c = Lit::isa<bool>(VarRewriter(param, val).rewrite(cond));
328 if (!c) lam->blame("{}: cannot decide `{}` of `{}` at `{}`", gen_.isel(), cond, lam, val).bail();
329 ok &= *c;
330 }
331 if (ok) collect(lam->reduce_body(val));
332 }
333}

335/// Adds @p rule, whose meta variables are preceded by the family parameter @p param, if any.
336void Table::add(const Rule* rule, const Var* param) {
337 if (!seen_.emplace(rule).second) return;
338 auto fvs = rule->free_vars();
339 if (!(fvs.empty() || (param && fvs.size() == 1 && fvs.contains(param))))
340 rule->blame("{}: rule `{}` is not closed", gen_.isel(), rule).bail();
341 if (rule->guard() == rule->world().lit_ff()) return;
342
343 var_ = rule->has_var();
344 metas_.clear();
345 meta_ids_.clear();
346 if (param) metas_.emplace_back(param);
347 if (var_) {
348 auto n = var_->num_projs();
349 for (size_t i = 0; i != n; ++i)
350 metas_.emplace_back(n == 1 ? var_ : var_->proj(n, i));
351 }
352 for (size_t i = 0; i != metas_.size(); ++i)
353 meta_ids_[metas_[i]] = i;
354 marks_.clear();
355 auto lhs = rule->lhs();
356 auto mapped = u32(-1);
357 if (var_ && lhs == var_) {
358 auto from = var_->type();
359 if (!Idx::isa(from) && classify(from) == Want::Reg)
360 rule->blame("{}: the meta variable of conversion `{}` must be a register or a non-terminal", gen_.isel(),
361 rule)
362 .bail();
363 // `lit x` makes it a conversion of a literal.
364 auto lit = isa_marker(gen_.annexes().lit, rule->guard());
365 bool literal = lit && lit->arg() == var_;
366 if (!literal && rule->guard() != rule->world().lit_tt())
367 rule->blame("{}: the guard of conversion `{}` is no `lit` of its meta variable", gen_.isel(), rule).bail();
368 conversions_.emplace_back(Conversion{.name = rule->sym().str(),
369 .rho = std::format("{} ↝ {}", rule->dom(), rule->codom()),
370 .cost = cost(rule->rhs()),
371 .from = literal ? Ty::Lit : nonterminal(rule, from),
372 .to = type(rule),
373 .rhs = node(rule->rhs()),
374 .arity = num_metas()});
375 var_ = nullptr;
376 return;
377 }
378 // `(m, e)`: the memory `m` of an inner node of `e` becomes that of the rhs.
379 if (auto t = lhs->isa<Tuple>(); t && t->num_ops() == 2 && core::isa_mem(t->op(0))) {
380 mapped = node(t->op(0));
381 lhs = t->op(1);
382 }
383 auto [root, _, __] = Axm::get(lhs);
384 if (!root && !lhs->isa<Extract>())
385 rule->blame("{}: the lhs of `{}` must apply an axiom or extract", gen_.isel(), rule).bail();
386
387 // A family that takes the operation names all of its instances alike.
388 auto name = root ? std::format("{}({})", rule->sym().str(), root->sym().str()) : rule->sym().str();
389 rewrites_.emplace_back(Rewrite{.name = std::move(name),
390 .rho = std::format("{} ↝ {}", rule->dom(), rule->codom()),
391 .cost = cost(rule->rhs()),
392 .root = root ? root->flags() : 0,
393 .type = type(rule),
394 .lhs = node(lhs),
395 .rhs = node(rule->rhs()),
396 .mapped = mapped,
397 .tied = tied(rule->rhs()),
398 .metas = metas(rule)});
399 var_ = nullptr;
400}

402u32 Table::tied(const Def* rhs) const {
403 u32 res = 0;
404 auto done = DefSet();
405 auto todo = DefVec{rhs};
406 while (!todo.empty()) {
407 auto def = todo.back();
408 todo.pop_back();
409 if (def->isa_mut() || !done.emplace(def).second) continue;
410 if (auto [axm, curry, _] = Axm::get(def); axm && curry == 0 && axm->plugin() == gen_.backend_id())
411 if (auto i = gen_.tied(def->as<App>())) {
412 auto arg = def->as<App>()->arg();
413 if (auto m = meta(arg->proj(arg->num_projs(), *i))) res |= 1u << *m;
414 }
415 for (auto op : def->ops())
416 todo.emplace_back(op);
417 }
418 return res;
419}

421std::vector<Table::Meta> Table::metas(const Rule* rule) {
422 auto res = std::vector<Meta>();
423 auto n = num_metas();
424 for (size_t i = 0; i != n; ++i) {
425 auto meta = metas_[i];
426 auto type = meta->type();
427 bool param = !uses(rule->rhs(), meta, var_);
428 for (size_t j = 0; j != n && !param; ++j)
429 param = j != i && uses(metas_[j]->type(), meta, var_);
430
431 auto want = Want::Param;
432 if (!param) {
433 auto mark = marks_.find(i);
434 auto marked = [&](Want dflt) { return mark == marks_.end() ? dflt : mark->second; };
435 if (Idx::isa(type)) {
436 want = marked(Want::Reg);
437 } else if (math::isa_f(type)) {
438 want = marked(Want::FReg);
439 } else if (type->isa<Nat>()) {
440 want = Want::Param;
441 } else if (Axm::isa<core::Ptr>(type)) {
442 if (mark == marks_.end())
443 rule->blame("{}: pointer `{}` of `{}` is no `addr` or `reg`", gen_.isel(), meta, rule).bail();
444 want = mark->second;
445 } else if (Axm::isa<core::M>(type) || type->isa<Pi>()) {
446 want = Want::Param;
447 } else if (auto addr = gen_.classify(type)) {
448 want = *addr;
449 } else {
450 rule->blame("{}: meta variable `{}` of `{}` has type `{}`, which is no operand", gen_.isel(), meta,
451 rule, type)
452 .bail();
453 }
454 }
455 res.emplace_back(want, node(type));
456 }
457 guard(rule->guard(), res);
458 return res;
459}

461void Table::guard(const Def* g, std::vector<Meta>& metas) {
462 auto& w = g->world();
463 if (g == w.lit_tt()) return;
464 if (auto conj = Axm::isa(core::bit2::and_, g)) {
465 auto [a, b] = conj->args<2>();
466 guard(a, metas);
467 guard(b, metas);
468 return;
469 }
470
471 auto set = [&](const Def* meta, Want from, Want to) {
472 for (size_t i = 0; i != num_metas(); ++i)
473 if (metas_[i] == meta && metas[i].want == from) {
474 metas[i].want = to;
475 return;
476 }
477 g->blame("{}: guard `{}` must constrain an operand of type `Idx s` or, for `lit`, a `Nat`", gen_.isel(), g)
478 .bail();
479 };
480
481 if (auto imm = isa_marker(gen_.annexes().imm, g))
482 set(imm->arg(), Want::Reg, Want::Imm);
483 else if (auto lit = isa_marker(gen_.annexes().lit, g))
484 set(lit->arg(), Want::Param, Want::Lit);
485 else
486 g->blame("{}: guard `{}` is no conjunction of `imm` and `lit`", gen_.isel(), g).bail();
487}

495u32 Table::node_(const Def* def) {
496 if (auto i = fe::lookup(meta_ids_, def)) return hole(*i);
497 if (var_ && def == var_) {
498 auto ops = std::vector<u32>();
499 for (size_t i = num_metas() - var_->num_projs(); i != num_metas(); ++i)
500 ops.emplace_back(hole(i));
501 return make({Kind::Tuple, 0, std::move(ops)});
502 }
503
504 const auto& a = gen_.annexes();
505 if (auto addr = isa_marker(a.addr, def)) return marked(addr->arg(), Want::Addr);
506 if (auto reg = isa_marker(a.reg, def)) return marked(reg->arg(), Want::Reg);
507 if (auto ri = isa_marker(a.reg_imm, def)) return widened(ri->arg(), Want::RegImm);
508 if (auto rs = isa_marker(a.reg_sh, def)) return widened(rs->arg(), Want::RegSh);
509 if (auto scaled = isa_marker(a.scaled, def)) return marked(scaled->arg(), Want::Scaled);
510 if (auto axm = def->isa<Axm>()) return make({Kind::Axm, axm->flags(), {}}, axm->sym().str());
511 if (auto lit = def->isa<Lit>()) return make({Kind::Lit, lit->get(), {node(lit->type())}});
512 if (def->isa<Nat>()) return make({Kind::Nat, 0, {}});
513 if (def->isa<Idx>()) return make({Kind::Idx, 0, {}});
514 if (auto app = def->isa<App>()) {
515 auto [axm, curry, _] = Axm::get(app);
516 u64 flags = app->callee_type()->is_implicit() ? u64(Implicit) : 0;
517 auto arg = node(app->arg());
518 if (axm && curry == 0 && !app->type()->isa<Type>()) {
519 flags |= Inner;
520 // The arg may also be the meta variables as a whole.
521 if (commutes_.contains(axm->flags()) && nodes_[arg].kind == Kind::Tuple && nodes_[arg].ops.size() == 2)
522 flags |= Commutes;
523 }
524 return make({Kind::App, flags, {node(app->callee()), arg}});
525 }
526 if (auto ex = def->isa<Extract>()) {
527 auto flags = core::isa_mem(ex) ? Mem : 0;
528 return make({Kind::Extract, flags, {node(ex->tuple()), node(ex->index())}});
529 }
530 if (auto pack = def->isa_imm<Pack>()) return make({Kind::Pack, 0, {node(pack->arity()), node(pack->body())}});
531 if (auto arr = def->isa_imm<Arr>()) return make({Kind::Arr, 0, {node(arr->arity()), node(arr->body())}});
532 if (auto pi = def->isa_imm<Pi>()) return make({Kind::Pi, pi->is_implicit(), {node(pi->dom()), node(pi->codom())}});
533 if (def->isa<Bot>()) return make({Kind::Bot, 0, {}});
534
535 auto ops = std::vector<u32>();
536 if (def->isa<Tuple>() || def->isa_imm<Sigma>())
537 for (auto op : def->ops())
538 ops.emplace_back(node(op));
539 if (def->isa<Tuple>()) return make({Kind::Tuple, 0, std::move(ops)});
540 if (def->isa_imm<Sigma>()) return make({Kind::Sigma, 0, std::move(ops)});
541
542 def->blame("{}: cannot put `{}` into a rule table", gen_.isel(), def).bail();
543}

605Shape Code::match(size_t id, u32 lhs, std::span<const u32> types) {
606 types_ = types;
607 std::println(os_,
608 "inline bool match_{}(const Def* def, [[maybe_unused]] u32 swap, [[maybe_unused]] const Def** m, "
609 "[[maybe_unused]] const Def** r) {{",
610 id);
611 walk(lhs);
612 line("return true;");
613 std::println(os_, "}}");
614 std::println(os_);
615
616 // Number the covers in preorder of the Value refs, so a cover's operands follow it.
617 auto children = std::vector<std::vector<u8>>(shape_.covers.size() + 1);
618 for (const auto& r : shape_.refs)
619 if (r.kind == Ref::Value) children[r.owner].emplace_back(r.cover);
620 auto perm = std::vector<u8>(shape_.covers.size(), u8(-1));
621 auto stack = std::vector<u8>(children[0].rbegin(), children[0].rend());
622 u8 next = 0;
623 while (!stack.empty()) {
624 auto c = stack.back();
625 stack.pop_back();
626 if (perm[c] != u8(-1)) continue;
627 perm[c] = next++;
628 stack.insert(stack.end(), children[c + 1].rbegin(), children[c + 1].rend());
629 }
630 if (next != shape_.covers.size()) fe::throwf("{}: an inner node is no operand of its rule", t_.gen_.isel());
631 auto res = std::move(shape_);
632 auto owner = [&](u8 o) { return o == 0 ? o : u8(perm[o - 1] + 1); };
633 for (auto& r : res.refs)
634 r.cover = perm[r.cover], r.owner = owner(r.owner);
635 for (auto& u : res.uses)
636 u.owner = owner(u.owner);
637 auto covers = res.covers;
638 for (size_t k = 0; k != covers.size(); ++k)
639 res.covers[perm[k]] = covers[k];
640 return res;
641}

677void Code::match_inner(u32 n, const std::string& e) {
678 if (auto var = cover_vars_.find(n); var != cover_vars_.end()) {
679 line("if ({} != {}) return false;", e, var->second);
680 return;
681 }
682 auto var = name(e);
683 cover_vars_[n] = var;
684 auto outer = owner_;

686 match_app(n, var);
687 owner_ = outer;

690void Code::match_app(u32 n, const std::string& e, bool spine) {
691 const auto& x = node(n);
692 auto app = fresh('d');
693 line("auto {} = {}->isa<App>();", app, e);
694 line("if (!{}) return false;", app);
695 match(x.ops[0], app + "->callee()", spine);
696 if (in_type_ || !(x.val & Commutes)) return match(x.ops[1], app + "->arg()");
697 auto arg = name(app + "->arg()");
698 auto bit = shape_.swaps.size();
699 auto& t = node(x.ops[1]);
700 shape_.swaps.emplace_back(t.ops[0], t.ops[1]);
701 line("if (Lit::isa({}->arity()) != 2) return false;", arg);
702 match(t.ops[0], std::format("{}->proj(2, swap >> {} & 1)", arg, bit));
703 match(t.ops[1], std::format("{}->proj(2, ~swap >> {} & 1)", arg, bit));
704}

706void Code::match(u32 n, const std::string& e, bool spine) {
707 const auto& x = node(n);
708 switch (x.kind) {
709 case Kind::Hole: {
710 auto i = x.val;
711 if (!in_type_) shape_.uses.emplace_back(rules::Use{owner_, u8(i), false});
712 auto ptr = is_ptr(types_[i]);
713 if (bound_[i]) {
714 line("if (m[{}] != {}) return false;", i, ptr ? std::format("strip_ptr_cast({})", e) : e);
715 return;
716 }
717 bound_[i] = true;
718 line("m[{}] = {};", i, e);
719 auto outer = std::exchange(in_type_, true);
720 match(types_[i], std::format("m[{}]->type()", i));
721 in_type_ = outer;
722 if (ptr) line("m[{0}] = strip_ptr_cast(m[{0}]);", i);
723 return;
724 }
725 case Kind::Axm: {
726 if (spine) return;
727 auto d = name(e);
728 line("if (!{0}->isa<Axm>() || {0}->flags() != {1:#x}) return false; // {2}", d, x.val, comment(n));
729 return;
730 }
731 case Kind::Lit: {
732 auto lit = fresh('d');
733 line("auto {} = {}->isa<Lit>();", lit, e);
734 line("if (!{0} || {0}->get() != {1:#x}) return false;", lit, x.val);
735 auto outer = std::exchange(in_type_, true);
736 match(x.ops[0], lit + "->type()");
737 in_type_ = outer;
738 return;
739 }
740 case Kind::Nat: line("if (!{}->isa<Nat>()) return false;", e); return;
741 case Kind::Idx: line("if (!{}->isa<Idx>()) return false;", e); return;
742 case Kind::App: {
743 if (!inner(n)) return match_app(n, e, spine);
744 auto d = name(e);
745 if (!cover_vars_.contains(n)) ref(n, d, Ref::Value);
746 return match_inner(n, d);
747 }
748 case Kind::Extract: {
749 auto tuple = x.ops[0];
750 if (!inner(tuple)) return match2(n, e, "isa<Extract>()", "tuple", "index");
751 auto ex = fresh('d');
752 line("auto {} = {}->isa<Extract>();", ex, e);
753 line("if (!{}) return false;", ex);
754 ref(tuple, ex, x.val & Mem ? Ref::Mem : Ref::Value);
755 match_inner(tuple, ex + "->tuple()");
756 match(x.ops[1], ex + "->index()");
757 return;
758 }
759 case Kind::Pack: return match2(n, e, "isa_imm<Pack>()", "arity", "body");
760 case Kind::Arr: return match2(n, e, "isa_imm<Arr>()", "arity", "body");
761 case Kind::Pi: return match2(n, e, "isa_imm<Pi>()", "dom", "codom");
762 case Kind::Bot: line("if (!{}->isa<Bot>()) return false;", e); return;
763 case Kind::Sigma: {
764 auto sigma = fresh('d');
765 line("auto {} = {}->isa_imm<Sigma>();", sigma, e);
766 line("if (!{0} || {0}->num_ops() != {1}) return false;", sigma, x.ops.size());
767 for (size_t j = 0; j != x.ops.size(); ++j)
768 match(x.ops[j], std::format("{}->op({})", sigma, j));
769 return;
770 }
771 case Kind::Tuple: {
772 // η: any term of the tuple's arity matches by its projections.
773 auto d = name(e);
774 line("if (Lit::isa({}->arity()) != {}) return false;", d, x.ops.size());
775 for (size_t j = 0; j != x.ops.size(); ++j)
776 match(x.ops[j], std::format("{}->proj({}, {})", d, x.ops.size(), j));
777 return;
778 }
779 }
780 std::unreachable();
781}

793std::string Code::build_(u32 n) {
794 if (auto i = built_.find(n); i != built_.end()) return i->second;
795 const auto& x = node(n);
796 if (x.kind == Kind::Hole) return built_[n] = std::format("h[{}]", x.val);
797
798 // The holes determine the implicit arguments, which may differ from the rule's, e.g. `i` of an `x86.Addr i`.
799 const auto& callee = node(x.kind == Kind::App ? x.ops[0] : n);
800 if (x.kind == Kind::App && callee.kind == Kind::App && callee.val & Implicit) {
801 auto f = build_(callee.ops[0]);
802 auto a = build_(x.ops[1]);
803 auto res = fresh('b');
804 line("auto {} = w.implicit_app({}, {});", res, f, a);
805 return built_[n] = res;
806 }
807
808 auto ops = std::vector<std::string>();
809 for (auto op : x.ops)
810 ops.emplace_back(build_(op));
811 auto res = fresh('b');
812 switch (x.kind) {
813 case Kind::Axm: line("auto {} = w.annex({:#x}); // {}", res, x.val, comment(n)); break;
814 case Kind::Lit: line("auto {} = w.lit({}, {:#x});", res, ops[0], x.val); break;
815 case Kind::Nat: line("auto {} = w.type_nat();", res); break;
816 case Kind::Idx: line("auto {} = w.type_idx();", res); break;
817 case Kind::App: line("auto {} = w.app({}, {});", res, ops[0], ops[1]); break;
818 case Kind::Extract: line("auto {} = w.extract({}, {});", res, ops[0], ops[1]); break;
819 case Kind::Pack: line("auto {} = w.pack({}, {});", res, ops[0], ops[1]); break;
820 case Kind::Arr: line("auto {} = w.arr({}, {});", res, ops[0], ops[1]); break;
821 case Kind::Tuple: line("auto {} = w.tuple({{{}}});", res, fe::Join(ops, ", ")); break;
822 case Kind::Sigma: line("auto {} = w.sigma({{{}}});", res, fe::Join(ops, ", ")); break;
823 case Kind::Pi: line("auto {} = w.pi({}, {}, {});", res, ops[0], ops[1], x.val != 0); break;
824 case Kind::Bot: line("auto {} = w.bot(w.type());", res); break;
825 case Kind::Hole: std::unreachable();
826 }
827 return built_[n] = res;
828}

844void Table::check(const std::vector<Shape>& shapes, const std::vector<size_t>& ids) const {
845 auto chains = rules::Chains(conversions_);
846 auto atomic = ankerl::unordered_dense::set<flags_t>();
847 for (size_t i = 0, e = rewrites_.size(); i != e; ++i) {
848 const auto& r = rewrites_[i];
849 if (chains.conv(r.type, rules::reg_want(r.type)) == rules::Inf)
850 fe::throwf("{}: rule `{}` yields a value that does not convert to a register", gen_.isel(), r.name);
851 for (auto m : r.metas)
852 if (!rules::is_param(m.want) && !rules::is_literal(m.want) && !chains.reachable(m.want))
853 fe::throwf("{}: no rule type converts to a hole of `{}`", gen_.isel(), r.name);
854 auto total = [&](Meta m) { return rules::is_param(m.want) || chains.total(m.want); };
855 if (shapes[ids[i]].covers.empty() && std::ranges::all_of(r.metas, total)) atomic.emplace(r.root);
856 }
857 for (const auto& r : rewrites_)
858 if (!atomic.contains(r.root)) fe::throwf("{}: no atomic rule shares the root of `{}`", gen_.isel(), r.name);
859}

988 std::println(os);
989 std::println(os, "inline constexpr Chains chains{{conversions}};");
990 std::println(
991 os, "inline constexpr Table table{{wants, all_uses, all_refs, matchers, rewrites, roots, conversions, chains,");
