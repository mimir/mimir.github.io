25/// The signed `I64` @p y modulo @p s as a widened `Idx s`.
26const Def* mod_s(World& w, nat_t s, const Def* y) {
27 auto zero = w.lit_i64(0);
28 auto neg = w.call(icmp::sl, Defs{y, zero});
29 auto m = mod_u(w, s, w.select(neg, op64(w, wrap::sub, zero, y), y));
30 auto flip = w.select(w.call(icmp::e, Defs{m, zero}), zero, op64(w, wrap::sub, w.lit_i64(s), m));
31 return narrow(w, s, w.select(neg, flip, m));
32}

34/// The `[M, I64]` result @p r of a signed effectful op with its value reduced modulo @p s.

40const Def* add(World& w, nat_t s, const Def* a, const Def* b) {
41 auto t = op64(w, wrap::add, a, b);
42 auto over = w.call(icmp::uge, Defs{t, w.lit_i64(s)});
43 // Two values below `s` only wrap around `I64` if `s` exceeds `2^63`; `t < a` catches that.
44 if (s > 1_n << 63) over = w.call(bit2::or_, 0_n, Defs{w.call(icmp::ul, Defs{t, a}), over});
45 return narrow(w, s, w.select(over, op64(w, wrap::sub, t, w.lit_i64(s)), t));
46}

48const Def* sub(World& w, nat_t s, const Def* a, const Def* b) {
49 auto d = op64(w, wrap::sub, a, b);
50 return narrow(w, s, w.select(w.call(icmp::ul, Defs{a, b}), op64(w, wrap::add, d, w.lit_i64(s)), d));
51}

56/// The bits of @p x as `I64`, zero-extended if @p x is narrower.
57const Def* bits(World& w, const Def* x) {
58 auto type = x->type();
59 if (!Idx::isa(type) && !type->isa<Nat>())
60 if (auto size = Lit::isa(op(trait::size, type)); size && *size < 8)

68const Def* LegalizeIdx::rewrite_imm_App(const App* app) {
69 auto& w = new_world();
70 auto zx = [&](const Def* d) { return zext(w, rewrite(d)); };
71 auto sx = [&](nat_t s, const Def* d) { return sext(w, s, rewrite(d)); };
72
73 if (auto s = isa_odd_idx(app)) return w.type_idx(widen_idx(*s));
74
75 if (auto op = Axm::isa<conv>(app); op && op.id() == conv::u) {
76 auto x = op->arg();
77 if (auto d = isa_odd_idx(app->type())) {
78 if (auto src = Idx::isa_lit(x->type()); src && *src != 0 && *src <= *d) return narrow(w, *d, rewrite(x));
79 return narrow(w, *d, mod_u(w, *d, zx(x)));
80 }
81 }
82
83 auto s = isa_odd_idx_op(app);
84 if (!s) {
85 // An op that keeps `Idx s` yields its value in range but still has to widen it like everything else.
86 auto res = RWPhase::rewrite_imm_App(app);
87 if (auto type = rewrite(app->type()); type != res->type() && !type->isa<Pi>()) return coerce(type, res);
88 return res;
89 }
90
91 if (auto op = Axm::isa<wrap>(app)) {
92 auto [a, b] = op->args<2>([this](const Def* d) { return rewrite(d); });
93 auto pow = [&] { return narrow(w, *s, mod_u(w, *s, op64(w, wrap::shl, w.lit_i64(1), zext(w, b)))); };
94 switch (op.id()) {
95 case wrap::add: return add(w, *s, zext(w, a), zext(w, b));
96 case wrap::sub: return sub(w, *s, zext(w, a), zext(w, b));
97 case wrap::mul: return mul(w, *s, a, b);
98 case wrap::shl: return mul(w, *s, a, pow());
99 }
100 } else if (Axm::isa<bitcast>(app) || Axm::isa<idx>(app)) {
101 return narrow(w, *s, mod_u(w, *s, bits(w, rewrite(app->arg()))));
102 } else if (auto op = Axm::isa<conv>(app)) {
103 auto x = op->arg();
104 if (auto d = isa_odd_idx(app->type())) {
105 auto src = isa_odd_idx(x->type());
106 return mod_s(w, *d, src ? sx(*src, x) : w.call(conv::s, w.lit_nat_0(), rewrite(x)));
107 }
108 return w.call(conv::s, rewrite(Idx::isa(app->type())), sx(*s, x));
109 } else if (auto op = Axm::isa<icmp>(app)) {
110 auto [a, b] = op->args<2>([&](const Def* x) { return sx(*s, x); });
111 return w.call(op.id(), Defs{a, b});
112 } else if (auto op = Axm::isa<shr>(app)) {
113 auto [a, b] = op->args<2>();
114 return mod_s(w, *s, w.call(shr::a, Defs{sx(*s, a), zx(b)}));
115 } else if (auto op = Axm::isa<extrema>(app)) {
116 auto [a, b] = op->args<2>();
117 return mod_s(w, *s, w.call(op.id(), Defs{sx(*s, a), sx(*s, b)}));
118 } else if (auto op = Axm::isa<abs>(app)) {
119 auto [m, x] = op->args<2>();
120 return mod_s_mem(w, *s, w.call<abs>(Defs{rewrite(m), sx(*s, x)}));
121 } else if (auto op = Axm::isa<div>(app)) {
122 auto [m, xy] = op->args<2>();
123 auto [x, y] = xy->projs<2>();
124 return mod_s_mem(w, *s, w.call(op.id(), Defs{rewrite(m), w.tuple({sx(*s, x), sx(*s, y)})}));
125 }
126 std::unreachable();
127}
