105constexpr u64 idx_mul(u64 size, u64 a, u64 b) {

254constexpr std::optional<u64> idx_ashr(u64 size, u64 a, u64 b) {

286 // Pre: a, b already in range for Idx size.
287
288 if constexpr (std::is_same_v<Id, wrap>) {
289 if constexpr (id == wrap::add) {
290 if (nuw && idx_add_nuw(size, a, b)) return {};
291 if (nsw && idx_add_nsw(size, a, b)) return {};
292 return idx_add(size, a, b);
293
294 } else if constexpr (id == wrap::sub) {
295 if (nuw && idx_sub_nuw(size, a, b)) return {};
296 if (nsw && idx_sub_nsw(size, a, b)) return {};
297 return idx_sub(size, a, b);
298
299 } else if constexpr (id == wrap::mul) {
300 if (nuw && idx_mul_nuw(size, a, b)) return {};
301 if (nsw && idx_mul_nsw(size, a, b)) return {};
302 return idx_mul(size, a, b);
303
304 } else if constexpr (id == wrap::shl) {
305 return idx_shl(size, a, b, nsw, nuw);
306
307 } else {
308 static_assert(false, "missing wrap subtag");
309 }
310
311 } else if constexpr (std::is_same_v<Id, shr>) {
312 if constexpr (id == shr::a)
313 return idx_ashr(size, a, b);
314 else if constexpr (id == shr::l)
315 return idx_lshr(size, a, b);
316 else
317 static_assert(false, "missing shr subtag");
318
319 } else if constexpr (std::is_same_v<Id, div>) {
320 if constexpr (id == div::udiv) {
321 return idx_udiv(size, a, b);
322
323 } else if constexpr (id == div::urem) {
324 return idx_urem(size, a, b);
325
326 } else if constexpr (id == div::sdiv) {
327 if (idx_sdivrem_ub(size, a, b)) return {};
328 return idx_sdiv(size, a, b);
329
330 } else if constexpr (id == div::srem) {
331 if (idx_sdivrem_ub(size, a, b)) return {};
332 return idx_srem(size, a, b);
333
334 } else {
335 static_assert(false, "missing div subtag");
336 }
337
338 } else if constexpr (std::is_same_v<Id, icmp>) {
339 return u64(fold_icmp_idx<id>(size, a, b));
340
341 } else if constexpr (std::is_same_v<Id, extrema>) {
342 if constexpr (id == extrema::sm)
343 return std::min(a, b);
344
345 else if constexpr (id == extrema::sM)
346 return std::max(a, b);
347
348 else if constexpr (id == extrema::Sm)
349 return idx_slt(size, a, b) ? a : b;
350
351 else if constexpr (id == extrema::SM)
352 return idx_sgt(size, a, b) ? a : b;
353
354 else
355 static_assert(false, "missing extrema subtag");
356
357 } else {
358 static_assert(false, "missing tag");
359 }

362template<class Id, Id id>
363const Def* fold(World& world, const Def* type, const Def*& a, const Def*& b, const Def* mode = {}) {
364 if (a->isa<Bot>() || b->isa<Bot>()) return world.bot(type);
365
366 if (auto la = Lit::isa(a)) {
367 if (auto lb = Lit::isa(b)) {
368 assert(a->type() == b->type());
369
370 auto size = Lit::as(Idx::isa(a->type()));
371
372 bool nsw = false, nuw = false;
373 if constexpr (std::is_same_v<Id, wrap>) {
374 auto m = mode ? static_cast<Mode>(Lit::as(mode)) : Mode::none;
375 nsw = fe::has_flag(m, Mode::nsw);
376 nuw = fe::has_flag(m, Mode::nuw);
377 }
378
379 if (size == 1) {
380 if constexpr (std::is_same_v<Id, div>) {
381 if (*lb == 0) return world.bot(type);
382 }
383 if constexpr (std::is_same_v<Id, icmp>)
384 return world.lit(type, u64(fold_icmp_idx<id>(1, 0, 0)));
385 else
386 return world.lit(type, 0);
387 }
388
389 auto res = fold_idx<Id, id>(size, *la, *lb, nsw, nuw);
390 return res ? world.lit(type, *res) : world.bot(type);
391 }
392 }
393
394 if (::mim::is_commutative(id) && Def::greater(a, b)) std::swap(a, b);
395 return nullptr;
396}

399const Def* fold(World& world, const Def* type, const Def*& a) {
400 if (a->isa<Bot>()) return world.bot(type);
401
402 if (auto la = Lit::isa(a)) {
403 auto size = Lit::as(Idx::isa(a->type()));
404
405 if constexpr (std::is_same_v<Id, abs>) {
406 auto x = idx_sext(size, *la);

414 return world.lit(type, y);
415 } else {
416 static_assert(false, "missing tag");
417 }
418 }
419
420 return nullptr;

423/// Reassociates @p a and @p b according to following rules.
424/// We use the following naming convention while literals are prefixed with an `l`:
425/// ```
426/// a op b
427/// (x op y) op (z op w)
428///
429/// (1) la op (lz op w) -> (la op lz) op w
430/// (2) (lx op y) op (lz op w) -> (lx op lz) op (y op w)
431/// (3) a op (lz op w) -> lz op (a op w)
432/// (4) (lx op y) op b -> lx op (y op b)
433/// ```
434template<class Id>
435const Def* reassociate(Id id, World& world, [[maybe_unused]] const App* ab, const Def* a, const Def* b) {
436 if (!is_associative(id)) return nullptr;
437
438 auto xy = Axm::isa<Id>(id, a);
439 auto zw = Axm::isa<Id>(id, b);
440 auto la = a->isa<Lit>();
441 auto [x, y] = xy ? xy->template args<2>() : std::array<const Def*, 2>{nullptr, nullptr};
442 auto [z, w] = zw ? zw->template args<2>() : std::array<const Def*, 2>{nullptr, nullptr};
443 auto lx = Lit::isa(x);
444 auto lz = Lit::isa(z);
445
446 // if we reassociate, we have to forget about nsw/nuw
447 auto make_op = [&world, id](const Def* a, const Def* b) { return world.call(id, Mode::none, Defs{a, b}); };
448
449 if (la && lz) return make_op(make_op(a, z), w); // (1)
450 if (lx && lz) return make_op(make_op(x, z), make_op(y, w)); // (2)
451 if (lz) return make_op(z, make_op(a, w)); // (3)
452 if (lx) return make_op(x, make_op(y, b)); // (4)
453 return nullptr;
454}

456template<class Id>
457const Def* merge_cmps(std::array<std::array<u64, 2>, 2> tab, const Def* a, const Def* b) {
458 static_assert(sizeof(sub_t) == 1, "if this ever changes, please adjust the logic below");
459 static constexpr size_t num_bits = std::bit_width(Annex::num<Id>() - 1_u64);
460
461 auto& world = a->world();
462 auto a_cmp = Axm::isa<Id>(a);
463 auto b_cmp = Axm::isa<Id>(b);
464
465 if (a_cmp && b_cmp && a_cmp->arg() == b_cmp->arg()) {
466 // push sub bits of a_cmp and b_cmp through truth table
467 sub_t res = 0;
468 sub_t a_sub = a_cmp.sub();
469 sub_t b_sub = b_cmp.sub();
470 for (size_t i = 0; i != num_bits; ++i, res >>= 1, a_sub >>= 1, b_sub >>= 1)
471 res |= tab[a_sub & 1][b_sub & 1] << 7_u8;
472 res >>= (7_u8 - u8(num_bits));
473
474 if constexpr (std::is_same_v<Id, math::cmp>)
475 return world.call(math::cmp(res), /*mode*/ a_cmp->decurry()->arg(), a_cmp->arg());
476 else
477 return world.call(icmp(Annex::base<icmp>() | res), a_cmp->arg());
478 }
479
480 return nullptr;
481}

485template<nat id>
486const Def* normalize_nat(const Def* type, const Def* callee, const Def* arg) {
487 auto& world = type->world();
488 auto [a, b] = arg->projs<2>();
489 if (is_commutative(id) && Def::greater(a, b)) std::swap(a, b);
490 auto la = Lit::isa(a);
491 auto lb = Lit::isa(b);
492
493 if (la) {
494 if (lb) {
495 switch (id) {
496 case nat::add: return world.lit_nat(*la + *lb);
497 case nat::sub: return *la < *lb ? world.lit_nat_0() : world.lit_nat(*la - *lb);
498 case nat::mul: return world.lit_nat(*la * *lb);
499 case nat::div: return *lb == 0 ? world.lit_nat_0() : world.lit_nat(*la / *lb);
500 case nat::mod: return *lb == 0 ? a : world.lit_nat(*la % *lb);
501 }
502 }
503
504 if (*la == 0) {
505 switch (id) {
506 case nat::add: return b;
507 case nat::sub: return a; // 0 - b = 0
508 case nat::mul: return a; // 0 * b = 0
509 case nat::div: return world.lit_nat_0(); // 0 / b = 0
510 case nat::mod: return world.lit_nat_0(); // 0 % b = 0
511 }
512 }
513
514 if (*la == 1 && id == nat::mul) return b; // 1 * b = b
515 }
516
517 if (lb) {
518 if (*lb == 0) {
519 switch (id) {
520 case nat::sub: return a; // a - 0 = a
521 case nat::div: return world.lit_nat_0(); // a / 0 = 0
522 case nat::mod: return a; // a % 0 = a
523 default: break;
524 }
525 }
526 if (*lb == 1) {
527 switch (id) {
528 case nat::div: return a; // a / 1 = a
529 case nat::mod: return world.lit_nat_0(); // a % 1 = 0
530 default: break;
531 }
532 }
533 }
534
535 if (a == b) {
536 switch (id) {
537 case nat::add: return world.call(nat::mul, Defs{world.lit_nat(2), a}); // a + a = 2 * a
538 case nat::sub: return world.lit_nat(0); // a - a = 0
539 case nat::mul: break;
540 case nat::div: break; // 0 / 0 = 0, so we cannot fold a / a = 1 symbolically
541 case nat::mod: return world.lit_nat_0(); // a % a = 0 (even for a = 0, since 0 % 0 = 0)
542 }
543 }
544
545 return world.raw_app(type, callee, {a, b});
546}

644template<bit2 id>
645const Def* normalize_bit2(const Def* type, const Def* c, const Def* arg) {
646 auto& world = type->world();
647 auto callee = c->as<App>();
648 auto [a, b] = arg->projs<2>();
649 auto s = callee->decurry()->arg();
650 auto ls = Lit::isa(s);
651 // TODO cope with wrap around
652
653 if (is_commutative(id) && Def::greater(a, b)) std::swap(a, b);
654
655 auto tab = make_truth_table(id);
656 if (auto res = merge_cmps<icmp>(tab, a, b)) return res;
657 if (auto res = merge_cmps<math::cmp>(tab, a, b)) return res;
658
659 auto la = Lit::isa(a);
660 auto lb = Lit::isa(b);
661
662 // clang-format off
663 switch (id) {
664 case bit2:: f: return world.lit(type, 0);
665 case bit2:: t: if (ls) return world.lit(type, *ls-1_u64); break;
666 case bit2:: fst: return a;
667 case bit2:: snd: return b;
668 case bit2:: nfst: return world.call(bit1::neg, s, a);
669 case bit2:: nsnd: return world.call(bit1::neg, s, b);
670 case bit2:: ciff: return world.call(bit2:: iff, s, Defs{b, a});
671 case bit2::nciff: return world.call(bit2::niff, s, Defs{b, a});
672 default: break;
673 }
674
675 if (la && lb && ls) {
676 switch (id) {
677 case bit2::and_: return world.lit_idx (*ls, *la & *lb);
678 case bit2:: or_: return world.lit_idx (*ls, *la | *lb);
679 case bit2::xor_: return world.lit_idx (*ls, *la ^ *lb);
680 case bit2::nand: return world.lit_idx_mod(*ls, ~(*la & *lb));
681 case bit2:: nor: return world.lit_idx_mod(*ls, ~(*la | *lb));
682 case bit2::nxor: return world.lit_idx_mod(*ls, ~(*la ^ *lb));
683 case bit2:: iff: return world.lit_idx_mod(*ls, ~ *la | *lb);
684 case bit2::niff: return world.lit_idx (*ls, *la & ~*lb);
685 default: fe::unreachable();
686 }
687 }
688
689 // TODO rewrite using bit2
690 auto unary = [&](bool x, bool y, const Def* a) -> const Def* {
691 if (!x && !y) return world.lit(type, 0);
692 if ( x && y) return ls ? world.lit(type, *ls-1_u64) : nullptr;
693 if (!x && y) return a;
694 if ( x && !y && id != bit2::xor_) return world.call(bit1::neg, s, a);
695 return nullptr;
696 };
697 // clang-format on
698
699 if (is_commutative(id) && a == b) {
700 if (auto res = unary(tab[0][0], tab[1][1], a)) return res;
701 }
702
703 if (la) {
704 if (*la == 0) {
705 if (auto res = unary(tab[0][0], tab[0][1], b)) return res;
706 } else if (ls && *la == *ls - 1_u64) {
707 if (auto res = unary(tab[1][0], tab[1][1], b)) return res;
708 }
709 }
710
711 if (lb) {
712 if (*lb == 0) {
713 if (auto res = unary(tab[0][0], tab[1][0], a)) return res;
714 } else if (ls && *lb == *ls - 1_u64) {
715 if (auto res = unary(tab[0][1], tab[1][1], a)) return res;
716 }
717 }
718
719 if (auto res = reassociate<bit2>(id, world, callee, a, b)) return res;
720
721 return world.raw_app(type, callee, {a, b});
722}

775template<wrap id>
776const Def* normalize_wrap(const Def* type, const Def* c, const Def* arg) {
777 auto& world = type->world();
778 auto callee = c->as<App>();
779 auto [a, b] = arg->projs<2>();
780 auto mode = callee->arg();
781 auto s = Idx::isa(a->type());
782 auto ls = Lit::isa(s);
783 auto width = ls.transform(idx_shift_width);
784
785 if (auto result = fold<wrap, id>(world, type, a, b, mode)) return result;
786
787 // clang-format off
788 if (auto la = Lit::isa(a)) {
789 if (*la == 0) {
790 switch (id) {
791 case wrap::add: return b; // 0 + b -> b
792 case wrap::sub: break;
793 case wrap::mul: return a; // 0 * b -> 0
794 case wrap::shl: return a; // 0 << b -> 0
795 }
796 } else if (*la == 1) {
797 switch (id) {
798 case wrap::add: break;
799 case wrap::sub: break;
800 case wrap::mul: return b; // 1 * b -> b
801 case wrap::shl: break;
802 }
803 }
804 }
805
806 if (auto lb = Lit::isa(b)) {
807 if (*lb == 0) {
808 switch (id) {
809 case wrap::sub: return a; // a - 0 -> a
810 case wrap::shl: return a; // a >> 0 -> a
811 default: fe::unreachable();
812 // add, mul are commutative, the literal has been normalized to the left
813 }
814 }
815
816 if (auto lm = Lit::isa(mode); lm && ls && *lm == 0 && id == wrap::sub)
817 return world.call(wrap::add, mode, Defs{a, world.lit_idx_mod(*ls, ~*lb + 1_u64)}); // a - lb -> a + (~lb + 1)
818 else if (id == wrap::shl && width && *lb >= *width)
819 return world.bot(type);
820 }
821
822 if (a == b) {
823 switch (id) {
824 case wrap::add: return world.call(wrap::mul, mode, Defs{world.lit(type, 2), a}); // a + a -> 2 * a
825 case wrap::sub: return world.lit(type, 0); // a - a -> 0
826 case wrap::mul: break;
827 case wrap::shl: break;
828 }
829 }
830 // clang-format on
831
832 if (auto res = reassociate<wrap>(id, world, callee, a, b)) return res;
833
834 return world.raw_app(type, callee, {a, b});
835}

837template<div id>
838const Def* normalize_div(const Def* full_type, const Def*, const Def* arg) {
839 auto& world = full_type->world();
840 auto [mem, ab] = arg->projs<2>();
841 auto [a, b] = ab->projs<2>();
842 auto [_, type] = full_type->projs<2>(); // peel off actual type
843 auto make_res = [&, mem = mem](const Def* res) { return world.tuple({mem, res}); };
844
845 if (auto result = fold<div, id>(world, type, a, b)) return make_res(result);
846
847 if (auto la = Lit::isa(a)) {
848 if (*la == 0) return make_res(a); // 0 / b -> 0 and 0 % b -> 0
849 }
850
851 if (auto lb = Lit::isa(b)) {
852 if (*lb == 0) return make_res(world.bot(type)); // a / 0 -> ⊥ and a % 0 -> ⊥
853
854 if (*lb == 1) {
855 switch (id) {
856 case div::sdiv: return make_res(a); // a / 1 -> a
857 case div::udiv: return make_res(a); // a / 1 -> a
858 case div::srem: return make_res(world.lit(type, 0)); // a % 1 -> 0
859 case div::urem: return make_res(world.lit(type, 0)); // a % 1 -> 0
860 }
861 }
862 }
863
864 if (a == b) {
865 switch (id) {
866 case div::sdiv: return make_res(world.lit(type, 1)); // a / a -> 1
867 case div::udiv: return make_res(world.lit(type, 1)); // a / a -> 1
868 case div::srem: return make_res(world.lit(type, 0)); // a % a -> 0
869 case div::urem: return make_res(world.lit(type, 0)); // a % a -> 0
870 }
871 }
872
873 return {};
874}

876template<conv id>
877const Def* normalize_conv(const Def* dst_t, const Def*, const Def* x) {
878 auto& world = dst_t->world();
879 auto s_t = x->type()->as<App>();
880 auto d_t = dst_t->as<App>();
881 auto s = s_t->arg();
882 auto d = d_t->arg();
883 auto ls = Lit::isa(s);
884 auto ld = Lit::isa(d);
885
886 if (s_t == d_t) return x;
887 if (x->isa<Bot>()) return world.bot(d_t);
888
889 if (auto l = Lit::isa(x); l && ls && ld) {
890 if constexpr (id == conv::u) {
891 if (*ld == 0) return world.lit(d_t, *l); // I64
892 return world.lit(d_t, *l % *ld);
893 }
894
895 return world.lit(d_t, idx_from_signed_mod(*ld, idx_sext(*ls, *l)));
896 }
897
898 if (ls && ld)
899 if (auto c1 = Axm::isa(id, x)) {
900 auto x1 = c1->arg();
901 if (auto ls1 = Lit::isa(x1->type()->as<App>()->arg()))
902 if (*ls > *ls1 || *ls == 0) // the intermediate conv is widening
903 if (*ld == *ls1) return x1; // conv(conv(x)) -> x
904 }
905
906 return {};
907}

940 } else if (auto w = math::isa_f(type)) {

942 case 16: return world.lit_nat(2);
943 case 32: return world.lit_nat(4);

946 }
947 } else if (type->isa<Sigma>() || type->isa<Meet>()) {

949 u64 align = 1;

951 auto a = Lit::isa(core::op(trait::align, t));
952 auto s = Lit::isa(core::op(trait::size, t));

956 offset = pad(offset, *a) + *s;
957 }
958
959 offset = pad(offset, align);
960 u64 size = std::max(1_u64, offset);
961
962 switch (id) {
963 case trait::align: return world.lit_nat(align);
964 case trait::size: return world.lit_nat(size);
965 }
966 } else if (auto arr = type->isa_imm<Arr>()) {
967 auto align = op(trait::align, arr->body());
968 if constexpr (id == trait::align) return align;
969 auto b = op(trait::size, arr->body());
970 if (b->isa<Lit>()) return world.call(nat::mul, Defs{arr->arity(), b});
971 } else if (auto join = type->isa<Join>()) {
972 if (auto sigma = convert(join)) return core::op(id, sigma);
973 }
974
975 return {};
976}
