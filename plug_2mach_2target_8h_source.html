16/// What an instruction on the chain prescribes about the registers of its operands; registers are hardware encodings.
17struct Constraint {
18 DefVec ops; ///< See Target::operands.
19 fe::Vector<std::pair<size_t, size_t>> fixed; ///< Non-literal operands that have to be in a given register.
20 fe::Vector<std::pair<size_t, size_t>> results; ///< Results, see Target::result, that land in a given register.
21 std::optional<size_t> tied; ///< The operand whose register the (only) result takes.
22 fe::Vector<size_t> clobbers; ///< Registers written while the operands still occupy theirs.
23};

25/// The fixed operands with their registers, and the tied one with no register if it is @p live_after.
26inline fe::Vector<std::pair<size_t, std::optional<size_t>>> copies(const Constraint& c, auto live_after) {
27 auto res = fe::Vector<std::pair<size_t, std::optional<size_t>>>();
28 for (auto [j, r] : c.fixed)
29 res.emplace_back(j, r);
30 if (c.tied && live_after(c.ops[*c.tied])) res.emplace_back(*c.tied, std::nullopt);
31 return res;
32}

34/// The machine facts of a native backend that the phases of `mach` need.
35/// A value's class is `Flags` for `mach.Flags`, `Stack` for `mach.Spill T`, `Fpr` for `math.F pe`, and `Gpr` otherwise.
36/// A backend derives from it and adds its instructions, register file, ABI, and constraints.

78 /// The number of registers of @p cls that register allocation may assign.

80 size_t n = 0;
81 for (size_t r = 0, e = num_regs(cls); r != e; ++r)
82 n += !reserved(cls, r);
83 return n;

87 size_t n = 0;
88 for (size_t r = 0, e = num_regs(cls); r != e; ++r)

94 /// @name Instructions
95 ///@{
96 /// The register class of the value @p def, from its type.
97 Class cls(const Def* def) const {
98 auto type = def->type();
99 if (Axm::isa<mach::Flags>(type)) return Flags;
100 if (Axm::isa<Spill>(type)) return Stack;
101 return math::isa_f(type) ? Fpr : Gpr;

103 /// Is @p app an instruction of this backend, other than `mach.sched`, that belongs on a `mem.M 0` chain?
104 /// By default, an application of an axiom of the backend or of `mach` that is no `Pi` and no `mach.in`.
105 virtual bool is_machine_insn(const App* app) const {
106 auto axm = app->axm();
107 return axm && (axm->plugin() == plugin_ || axm->plugin() == Plugin_Id) && !app->type()->isa<Pi>()
108 && !Axm::isa<mach::in>(app);
