144 /// Walks @p decl and @p inst in lockstep; a `Sigma` may have normalized to an `Arr` in @p inst.
145 void adopt_names(const Def* decl, const Def* inst, DefSet& visited) {
146 if (decl == inst || decl->isa<Nom>() || inst->isa<Nom>() || !visited.emplace(decl).second) return;
147 if (auto i = def2sym2idx_.find(decl); i != def2sym2idx_.end()) {
148 auto sym2idx = i->second; // add_names may rehash def2sym2idx_
149 add_names(inst, sym2idx);
150 }
151 if (decl->node() == inst->node() && decl->num_ops() == inst->num_ops()) {
152 for (size_t k = 0, n = decl->num_ops(); k != n; ++k)
153 adopt_names(decl->op(k), inst->op(k), visited);
154 } else if (auto sigma = decl->isa<Sigma>()) {
155 if (auto n = sigma->num_ops(); Lit::isa(inst->arity()) == n)
156 for (size_t k = 0; k != n; ++k)
157 adopt_names(sigma->op(k), inst->proj(n, k), visited);
158 }
159 }

268const Def* TuplePtrn::emit_body(Emitter& e, const Def* decl) const {
269 auto _ = e.world().push(loc());
270 auto n = num_ptrns();
271 Sigma* sigma;
272 if (decl) {
273 sigma = decl->as_mut<Sigma>();
274 } else {
275 auto type = e.world().type_infer_univ();
276 sigma = e.world().mut_sigma(type, n);
277 }
278 auto var = sigma->var();
279 auto sym2idx = fe::SymMap<size_t>();
280
281 for (size_t i = 0; i != n; ++i) {
282 sigma->set(i, ptrn(i)->emit_type(e));
283 ptrn(i)->emit_proj(e, var, n, i);
284 if (auto p = ptrn(i); (p->isa<IdPtrn>() || p->isa<GrpPtrn>()) && !p->dbg().is_anon())
285 sym2idx[p->dbg().sym()] = i;
286 }
287
288 e.add_names(sigma, sym2idx);
289 auto imm = sigma->immutabilize();
290 // A 1-Sigma collapses to its field's type, which must not inherit the name.
291 if (imm && n > 1) e.add_names(imm, sym2idx);
292 return imm ? imm : sigma;

372/// If @p type is a `math.F` type of known precision/exponent, yields its bit width.
373/// Note that libmim must not depend on the generated math plugin header, so lookup the Axm at runtime instead.
374static std::optional<nat_t> isa_math_f(Emitter& e, const Def* type) {
375 auto math_f = e.world().annex(e.world().sym("math.F"));
376 if (auto app = type->zonk()->isa<App>(); math_f && app && app->callee() == math_f) {
377 if (auto [p, ex] = app->arg()->projs<2>([](auto op) { return Lit::isa(op); }); p && ex) {
378 if (*p == 10 && *ex == 5) return 16;
379 if (*p == 23 && *ex == 8) return 32;
380 if (*p == 52 && *ex == 11) return 64;
381 }
382 }
383 return {};

502/// `T#C` or `T#i` selects case `C` or `i` of the Variant `T`: a constructor function - or its value for a `[]` payload.
503/// If @p nom is given, `T` is its body and the constructor wraps its result into @p nom.
504static const Def* emit_ctor(Emitter& e, const Variant* variant, const Expr* index, const Nom* nom = nullptr) {
505 auto& w = e.world();
506 auto i = std::optional<nat_t>();
507
508 if (auto path = index->isa<PathExpr>(); path && path->path()->dbgs().size() == 1) {
509 i = e.find_name(variant, path->dbg());
510 if (!i && !path->decl())
511 e.error()
512 .e(path->dbg().loc(), "variant `{}` has no constructor `{}`", e.variant_str(variant), path->dbg())
513 .bail();
514 }
515 if (!i) i = Lit::isa(index->emit(e));
516 if (!i) e.error().e(index->loc(), "a case of a variant must be selected by name or by literal index").bail();
517 if (*i >= variant->num_ops())
518 e.error()

520 variant->num_ops(), *i)
521 .bail();
522
523 auto wrap = [&](const Def* val) { return nom ? w.name(nom, val) : val; };
524 auto payload = variant->op(*i);
525 if (payload == w.sigma()) return wrap(w.inj(variant, *i, w.tuple()));
526 auto ctor = w.mut_lam(w.pi(payload, nom ? static_cast<const Def*>(nom) : variant));
527 return ctor->set(true, wrap(w.inj(variant, *i, ctor->var())));
528}

530const Def* InfixExpr::emit_(Emitter& e) const {
531 auto& w = e.world();
532
533 switch (op().tag()) {
534 case Tag::T_union: {
535 DefVec types;
536 emit_union(e, this, types);
537 return w.join(types);
538 }
539 case Tag::T_extract: {
540 auto tup = lhs()->emit(e);
541 if (auto variant = tup->isa<Variant>()) return emit_ctor(e, variant, rhs());
542 if (auto nom = tup->isa<Nom>()) {
543 if (auto variant = nom->op()->isa<Variant>()) {
544 check_nom_scope(e, loc(), nom);
545 e.adopt_names(nom);
546 return emit_ctor(e, variant, rhs(), nom);
547 }
548 }
549 return w.extract(tup, emit_index(e, tup));
550 }
551 case Tag::T_arrow_l: {
552 fe::Vector<const InfixExpr*> exs;
553 auto base = lhs();
554 while (auto ex = InfixExpr::isa_op(Tag::T_extract, base)) {
555 exs.emplace_back(ex);
556 base = ex->lhs();
557 }
558
559 if (exs.empty())
560 e.error()
561 .e(lhs()->loc(), "expected `#` on the left-hand side of `{}`", Tok::tag2str(op().tag()))
562 .n("an update needs a component, as in `tuple#index {} value`", Tok::tag2str(op().tag()))
563 .bail();
564
565 auto tup = base->emit(e);
566 DefVec tups, idxs;
567 for (auto ex : exs | std::views::reverse) {
568 auto idx = ex->emit_index(e, tup);
569 tups.emplace_back(tup);
570 idxs.emplace_back(idx);
571 tup = w.extract(tup, idx);
572 }
573
574 auto val = rhs()->emit(e);
575 for (size_t i = tups.size(); i-- != 0;)
576 val = w.insert(tups[i], idxs[i], val);
577 return val;
578 }
579 default: break;
580 }
581
582 auto c = callee() ? callee()->emit(e) : nullptr;
583 auto l = lhs()->emit(e);
584 auto r = rhs()->emit(e);
585
586 switch (op().tag()) {
587 case Tag::T_arrow_r: return w.pi(l, r);
588 case Tag::T_at: return w.app(l, r);
589 case Tag::K_inj:
590 if (auto nom = r->isa<Nom>()) check_nom_scope(e, loc(), nom);
591 return w.inj(r, l);
592 default: return w.implicit_app(c, w.tuple({l, r})); // MIM_INFIX_SUGAR
593 }

615const Def* MatchExpr::emit_variant(Emitter& e, const Def* scrutinee, const Variant* variant) const {
616 auto sel = DefVec(variant->num_ops());
617 for (auto arm : arms()) {
618 auto i = size_t();
619 if (auto index = arm->index()) {
620 if (*index >= variant->num_ops())
621 e.error()
622 .e(arm->loc(), "variant `{}` has {} cases but case {} was requested", e.variant_str(variant),
623 variant->num_ops(), *index)
624 .bail();
625 i = *index;
626 } else if (auto id = arm->ctor()) {
627 i = e.ctor(variant, id->dbg());
628 } else {
629 e.error()
630 .e(arm->ptrn()->loc(), "an arm of a match on variant `{}` must name a constructor",
631 e.variant_str(variant))
632 .bail();
633 }
634 if (sel[i]) {
635 e.error().w(arm->loc(), "this arm is unreachable: constructor {} is already handled",
636 e.ctor_names(variant, i));
637 continue;
638 }
639 sel[i] = arm->emit(e, variant->op(i));
640 }
641
642 auto missing = std::vector<std::string>();
643 for (size_t i = 0, n = sel.size(); i != n; ++i)
644 if (!sel[i]) missing.emplace_back(e.ctor_names(variant, i));
645 if (!missing.empty()) e.error().e(loc(), "match expression has no arm for {}", fe::Join(missing, ", ")).bail();
646
647 auto ops = DefVec{scrutinee};
648 ops.append_range(sel);
649 return e.world().match(ops);
650}

652const Def* MatchExpr::emit_(Emitter& e) const {
653 DefVec ops;
654 ops.emplace_back(scrutinee()->emit(e));
655 if (auto variant = ops.front()->isa_type<Variant>()) return emit_variant(e, ops.front(), variant);
656
657 for (auto arm : arms()) {
658 if (arm->payload() || arm->index())
659 e.error()
660 .e(arm->loc(), "a constructor pattern needs a variant scrutinee")
661 .n("but the scrutinee has type `{}`", type_of(ops.front()))
662 .bail();
663 ops.emplace_back(arm->emit(e));
664 }
665 auto res = e.world().match(ops);
666
667 // Only a *source* arm is unreachable by mistake; substitution legitimately kills arms of a polymorphic match.
668 auto cases = Match::cases(ops.front());
669 for (size_t i = 0, n = num_arms(); i != n; ++i)
670 if (std::ranges::none_of(cases, [&](const Def* c) { return Match::accepts(ops[i + 1], c); }))
671 e.error().w(arm(i)->loc(), "this arm is unreachable: no case of `{}` matches `{}`", type_of(ops.front()),
672 ops[i + 1]->isa_type<Pi>()->dom());
673
674 return res;
675}

957 auto _ = e.world().push(loc());
958 bool is_cps = !ISA(tag_, C_DS);
959
960 // Iterate over all doms: Build a Lam for curr dom, by first building a curried Pi for the remaining doms.
961 for (size_t i = 0, n = num_doms(); i != n; ++i) {
962 for (auto dom : doms() | std::views::drop(i))
963 dom->emit_type(e);
964
965 auto cod = codom() ? codom()->emit(e) : is_cps ? e.world().type_bot() : e.world().mut_hole_type();
966 for (auto dom : doms() | std::views::drop(i) | std::views::reverse)
967 cod = dom->pi_->set_codom(cod);
968
969 auto cur = dom(i);
970 auto lam = cur->emit_value(e);
971 if (auto filter = cur->filter()) {
972 auto _filter = e.world().push(filter->loc());
973 lam->set_filter(filter->emit(e));
974 } else {
975 lam->set_filter(i + 1 == n && is_cps ? e.world().lit_ff() : e.world().lit_tt());
976 }
977
978 if (i == 0)
979 def_ = lam->set(dbg().sym());
980 else
981 dom(i - 1)->lam_->set_body(lam);
982 }

985void LamDecl::emit_body(Emitter& e) const {
986 if (!body()) return; // extern forward declaration: the implementation lives in a native translation unit
987
988 auto _ = e.world().push(loc());
989 {
990 auto _body = e.world().push(body()->loc());
991 doms().back()->lam_->set_body(body()->emit(e));
992 }
993
994 // rewrite holes
995 for (size_t i = 0, n = num_doms(); i != n; ++i) {
996 auto rw = VarRewriter(e.world());
997 auto lam = dom(i)->lam_;
998 auto pi = lam->type()->as_mut<Pi>();
999 for (auto dom : doms() | std::views::drop(i)) {
1000 // A Lam's Var that was never created cannot occur in the codomain.
1001 if (auto var = pi->has_var())
1002 if (auto lam_var = dom->lam_->has_var()) rw.add(lam_var, var);
1003 auto cod = pi->codom();
1004 if (!cod || !cod->isa_mut<Pi>()) break;
1005 pi = cod->as_mut<Pi>();
1006 }
1007
1008 if (auto cod = pi->codom(); cod && cod->has_dep(Dep::Hole)) pi->set(pi->dom(), rw.rewrite(cod));
1009 }
1010
1011 for (auto dom : doms() | std::views::reverse) {
1012 if (auto imm = dom->pi_->immutabilize()) {
1013 auto f = dom->lam_->filter();
1014 auto b = dom->lam_->body();
1015 dom->lam_->unset()->set_type(imm)->as<Lam>()->set(f, b);
1016 }
1017 }
1018
1019 if (is_extern()) {
1020 auto lam = doms().front()->lam_;
1021 if (!lam->is_closed())
1022 e.error()
1023 .e(loc(),
1024 "external function `{}` is not closed: its inferred type escapes into the scope of `{}`. This "
1025 "usually means an unannotated parameter's type could only be inferred to depend on a variable bound "
1026 "in an inner/sibling scope; add an explicit type annotation to the offending parameter.",
1027 dbg().sym(), lam->free_vars().min()->binder()->sym())
1028 .bail();
1029 if (auto prev = e.world().externals()[dbg().sym()])
1030 e.error()
1031 .e(loc(), "external function `{}` is already defined", dbg().sym())
1032 .n(prev->loc(), "previous definition here")
1033 .bail();
1034 lam->externalize();
1035 }
1036 e.attach(annex_, sub_, dbg().sym(), def_);
1037}
