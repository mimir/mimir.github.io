36 void add_names(const Def* def, const fe::SymMap<size_t>& sym2idx) {
37 auto& names = def2sym2idx_[def];

39 if (auto [j, ins] = names.emplace(sym, i); !ins && j->second != i) j->second = Ambiguous;

42 std::optional<size_t> find_name(const Def* def, Dbg dbg) {
43 auto i = def2sym2idx_.find(def);
44 if (i == def2sym2idx_.end()) return {};
45 auto j = i->second.find(dbg.sym());
46 if (j == i->second.end()) return {};

48 if (def->isa<Variant>())

183const Def* TuplePtrn::emit_body(Emitter& e, const Def* decl) const {
184 auto _ = e.world().push(loc());
185 auto n = num_ptrns();
186 Sigma* sigma;
187 if (decl) {
188 sigma = decl->as_mut<Sigma>();
189 } else {
190 auto type = e.world().type_infer_univ();
191 sigma = e.world().mut_sigma(type, n);
192 }
193 auto var = sigma->var();
194 auto sym2idx = fe::SymMap<size_t>();
195
196 for (size_t i = 0; i != n; ++i) {
197 sigma->set(i, ptrn(i)->emit_type(e));
198 ptrn(i)->emit_proj(e, var, n, i);
199 if (auto p = ptrn(i); (p->isa<IdPtrn>() || p->isa<GrpPtrn>()) && !p->dbg().is_anon())
200 sym2idx[p->dbg().sym()] = i;
201 }

204 auto imm = sigma->immutabilize();

206 if (imm && n > 1) e.add_names(imm, sym2idx);

280/// If @p type is a `math.F` type of known precision/exponent, yields its bit width.
281/// Note that libmim must not depend on the generated math plugin header, so lookup the Axm at runtime instead.
282static std::optional<nat_t> isa_math_f(Emitter& e, const Def* type) {
283 auto math_f = e.world().annex(e.world().sym("math.F"));
284 if (auto app = type->zonk()->isa<App>(); math_f && app && app->callee() == math_f) {

377/// `T#C` or `T#i` selects case `C` or `i` of the Variant `T`: a constructor function - or its value for a `[]` payload.
378static const Def* emit_ctor(Emitter& e, const Variant* variant, const Expr* index) {
379 auto& w = e.world();
380 auto i = std::optional<nat_t>();
381
382 if (auto path = index->isa<PathExpr>(); path && path->path()->dbgs().size() == 1) {
383 i = e.find_name(variant, path->dbg());
384 if (!i && !path->decl())
385 e.error().e(path->dbg().loc(), "variant `{}` has no constructor `{}`", variant, path->dbg()).bail();
386 }
387 if (!i) i = Lit::isa(index->emit(e));
388 if (!i) e.error().e(index->loc(), "a case of a variant must be selected by name or by literal index").bail();
389 if (*i >= variant->num_ops())
390 e.error()
391 .e(index->loc(), "variant `{}` has {} cases but case {} was requested", variant, variant->num_ops(), *i)
392 .bail();
393
394 auto payload = variant->op(*i);
395 if (payload == w.sigma()) return w.inj(variant, *i, w.tuple());
396 auto ctor = w.mut_lam(w.pi(payload, variant));
397 return ctor->set(true, w.inj(variant, *i, ctor->var()));

400const Def* InfixExpr::emit_(Emitter& e) const {
401 auto& w = e.world();
402
403 switch (op().tag()) {
404 case Tag::T_union: {
405 DefVec types;
406 emit_union(e, this, types);
407 return w.join(types);
408 }
409 case Tag::T_extract: {
410 auto tup = lhs()->emit(e);
411 if (auto variant = tup->isa<Variant>()) return emit_ctor(e, variant, rhs());
412 return w.extract(tup, emit_index(e, tup));
413 }
414 case Tag::T_arrow_l: {
415 fe::Vector<const InfixExpr*> exs;
416 auto base = lhs();
417 while (auto ex = InfixExpr::isa_op(Tag::T_extract, base)) {
418 exs.emplace_back(ex);
419 base = ex->lhs();
420 }
421
422 if (exs.empty())
423 e.error()
424 .e(lhs()->loc(), "expected `#` on the left-hand side of `{}`", Tok::tag2str(op().tag()))
425 .n("an update needs a component, as in `tuple#index {} value`", Tok::tag2str(op().tag()))
426 .bail();
427
428 auto tup = base->emit(e);
429 DefVec tups, idxs;
430 for (auto ex : exs | std::views::reverse) {
431 auto idx = ex->emit_index(e, tup);
432 tups.emplace_back(tup);
433 idxs.emplace_back(idx);
434 tup = w.extract(tup, idx);
435 }
436
437 auto val = rhs()->emit(e);
438 for (size_t i = tups.size(); i-- != 0;)
439 val = w.insert(tups[i], idxs[i], val);
440 return val;
441 }
442 default: break;
443 }
444
445 auto c = callee() ? callee()->emit(e) : nullptr;
446 auto l = lhs()->emit(e);
447 auto r = rhs()->emit(e);
448
449 switch (op().tag()) {
450 case Tag::T_arrow_r: return w.pi(l, r);
451 case Tag::T_at: return w.app(l, r);
452 case Tag::K_inj: return w.inj(r, l);
453 default: return w.implicit_app(c, w.tuple({l, r})); // MIM_INFIX_SUGAR
454 }

476const Def* MatchExpr::emit_variant(Emitter& e, const Def* scrutinee, const Variant* variant) const {
477 auto sel = DefVec(variant->num_ops());
478 for (auto arm : arms()) {
479 auto id = arm->ctor();
480 if (!id)
481 e.error()
482 .e(arm->ptrn()->loc(), "an arm of a match on variant `{}` must name a constructor", variant)
483 .bail();
484 auto i = e.ctor(variant, id->dbg());
485 if (sel[i]) {
486 e.error().w(arm->loc(), "this arm is unreachable: constructor `{}` is already handled", id->dbg());
487 continue;
488 }
489 sel[i] = arm->emit(e, variant->op(i));
490 }
491
492 auto missing = std::vector<std::string>();
493 for (size_t i = 0, n = sel.size(); i != n; ++i)
494 if (!sel[i]) missing.emplace_back(e.ctor_names(variant, i));
495 if (!missing.empty()) e.error().e(loc(), "match expression has no arm for {}", fe::Join(missing, ", ")).bail();
496
497 auto ops = DefVec{scrutinee};
498 ops.append_range(sel);
499 return e.world().match(ops);
500}

502const Def* MatchExpr::emit_(Emitter& e) const {
503 DefVec ops;
504 ops.emplace_back(scrutinee()->emit(e));
505 if (auto variant = ops.front()->isa_type<Variant>()) return emit_variant(e, ops.front(), variant);
506
507 for (auto arm : arms()) {
508 if (arm->payload())
509 e.error()
510 .e(arm->loc(), "a constructor pattern needs a variant scrutinee")
511 .n("but the scrutinee has type `{}`", type_of(ops.front()))
512 .bail();
513 ops.emplace_back(arm->emit(e));
514 }
515 auto res = e.world().match(ops);
516
517 // Only a *source* arm is unreachable by mistake; substitution legitimately kills arms of a polymorphic match.
518 auto cases = Match::cases(ops.front());
519 for (size_t i = 0, n = num_arms(); i != n; ++i)
520 if (std::ranges::none_of(cases, [&](const Def* c) { return Match::accepts(ops[i + 1], c); }))
521 e.error().w(arm(i)->loc(), "this arm is unreachable: no case of `{}` matches `{}`", type_of(ops.front()),
522 ops[i + 1]->isa_type<Pi>()->dom());
523
524 return res;
525}

771void LamDecl::emit_decl(Emitter& e) const {
772 auto _ = e.world().push(loc());
773 bool is_cps = !ISA(tag_, C_DS);
774
775 // Iterate over all doms: Build a Lam for curr dom, by first building a curried Pi for the remaining doms.
776 for (size_t i = 0, n = num_doms(); i != n; ++i) {
777 for (auto dom : doms() | std::views::drop(i))
778 dom->emit_type(e);
779
780 auto cod = codom() ? codom()->emit(e) : is_cps ? e.world().type_bot() : e.world().mut_hole_type();
781 for (auto dom : doms() | std::views::drop(i) | std::views::reverse)
782 cod = dom->pi_->set_codom(cod);
783
784 auto cur = dom(i);
785 auto lam = cur->emit_value(e);
786 if (auto filter = cur->filter()) {
787 auto _filter = e.world().push(filter->loc());
788 lam->set_filter(filter->emit(e));
789 } else {
790 lam->set_filter(i + 1 == n && is_cps ? e.world().lit_ff() : e.world().lit_tt());
791 }
792
793 if (i == 0)
794 def_ = lam->set(dbg().sym());
795 else
796 dom(i - 1)->lam_->set_body(lam);
797 }

800void LamDecl::emit_body(Emitter& e) const {
801 if (!body()) return; // extern forward declaration: the implementation lives in a native translation unit
802
803 auto _ = e.world().push(loc());
804 {
805 auto _body = e.world().push(body()->loc());
806 doms().back()->lam_->set_body(body()->emit(e));
807 }
808
809 // rewrite holes
810 for (size_t i = 0, n = num_doms(); i != n; ++i) {
811 auto rw = VarRewriter(e.world());
812 auto lam = dom(i)->lam_;
813 auto pi = lam->type()->as_mut<Pi>();
814 for (auto dom : doms() | std::views::drop(i)) {
815 // A Lam's Var that was never created cannot occur in the codomain.
816 if (auto var = pi->has_var())
817 if (auto lam_var = dom->lam_->has_var()) rw.add(lam_var, var);
818 auto cod = pi->codom();
819 if (!cod || !cod->isa_mut<Pi>()) break;
820 pi = cod->as_mut<Pi>();
821 }
822
823 if (auto cod = pi->codom(); cod && cod->has_dep(Dep::Hole)) pi->set(pi->dom(), rw.rewrite(cod));
824 }
825
826 for (auto dom : doms() | std::views::reverse) {
827 if (auto imm = dom->pi_->immutabilize()) {
828 auto f = dom->lam_->filter();
829 auto b = dom->lam_->body();
830 dom->lam_->unset()->set_type(imm)->as<Lam>()->set(f, b);
831 }
832 }
833
834 if (is_extern()) {
835 auto lam = doms().front()->lam_;
836 if (!lam->is_closed())
837 e.error()
838 .e(loc(),
839 "external function `{}` is not closed: its inferred type escapes into the scope of `{}`. This "
840 "usually means an unannotated parameter's type could only be inferred to depend on a variable bound "
841 "in an inner/sibling scope; add an explicit type annotation to the offending parameter.",
842 dbg().sym(), lam->free_vars().min()->binder()->sym())
843 .bail();
844 if (auto prev = e.world().externals()[dbg().sym()])
845 e.error()
846 .e(loc(), "external function `{}` is already defined", dbg().sym())
847 .n(prev->loc(), "previous definition here")
848 .bail();
849 lam->externalize();
850 }
851 e.attach(annex_, sub_, dbg().sym(), def_);
852}
