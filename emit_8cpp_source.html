151 /// Walks @p decl and @p inst in lockstep; a `Sigma` may have normalized to an `Arr` in @p inst.
152 void adopt_names(const Def* decl, const Def* inst, DefSet& visited) {
153 if (decl == inst || decl->isa<Nom>() || inst->isa<Nom>() || !visited.emplace(decl).second) return;
154 if (auto i = def2sym2idx_.find(decl); i != def2sym2idx_.end()) {
155 auto sym2idx = i->second; // add_names may rehash def2sym2idx_
156 add_names(inst, sym2idx);
157 }
158 if (decl->node() == inst->node() && decl->num_ops() == inst->num_ops()) {
159 for (size_t k = 0, n = decl->num_ops(); k != n; ++k)
160 adopt_names(decl->op(k), inst->op(k), visited);
161 } else if (auto sigma = decl->isa<Sigma>()) {
162 if (auto n = sigma->num_ops(); Lit::isa(inst->arity()) == n)
163 for (size_t k = 0; k != n; ++k)
164 adopt_names(sigma->op(k), inst->proj(n, k), visited);
165 }
166 }

292 auto _ = e.world().push(loc());

294 Sigma* sigma;
295 if (decl)

392/// If @p type is a `math.F` type of known precision/exponent, yields its bit width.
393/// Note that libmim must not depend on the generated math plugin header, so lookup the Axm at runtime instead.

395 auto math_f = e.world().annex(e.world().sym("math.F"));
396 if (auto app = type->zonk()->isa<App>(); math_f && app && app->callee() == math_f) {

399 if (*p == 23 && *ex == 8) return 32;

574const Def* InfixExpr::emit_(Emitter& e) const {
575 auto& w = e.world();
576
577 switch (op().tag()) {
578 case Tag::T_union: {
579 DefVec types;
580 emit_union(e, this, types);
581 return w.join(types);
582 }
583 case Tag::T_extract: {
584 auto tup = lhs()->emit(e);
585 if (auto variant = tup->isa<Variant>()) return emit_ctor(e, variant, rhs());
586 if (auto nom = tup->isa<Nom>()) {
587 if (auto variant = nom->op()->isa<Variant>()) {
588 open_nom(e, loc(), nom);
589 return emit_ctor(e, variant, rhs(), nom);
590 }
591 }
592 if (auto value = unwrap_field(e, this, tup)) tup = value;
593 return w.extract(tup, emit_index(e, tup));
594 }
595 case Tag::T_arrow_l: {
596 fe::Vector<const InfixExpr*> exs;
597 auto base = lhs();
598 while (auto ex = InfixExpr::isa_op(Tag::T_extract, base)) {
599 exs.emplace_back(ex);
600 base = ex->lhs();
601 }
602
603 if (exs.empty())
604 e.error()
605 .e(lhs()->loc(), "expected `#` on the left-hand side of `{}`", Tok::tag2str(op().tag()))
606 .n("an update needs a component, as in `tuple#index {} value`", Tok::tag2str(op().tag()))
607 .bail();
608
609 auto tup = base->emit(e);
610 DefVec tups, idxs;
611 fe::Vector<const Nom*> noms;
612 for (auto ex : exs | std::views::reverse) {
613 auto value = unwrap_field(e, ex, tup);
614 noms.emplace_back(value ? tup->isa_type<Nom>() : nullptr);
615 if (value) tup = value;
616 auto idx = ex->emit_index(e, tup);
617 tups.emplace_back(tup);
618 idxs.emplace_back(idx);
619 tup = w.extract(tup, idx);
620 }
621
622 auto val = rhs()->emit(e);
623 for (size_t i = tups.size(); i-- != 0;) {
624 val = w.insert(tups[i], idxs[i], val);
625 if (noms[i]) val = w.name(noms[i], val);
626 }
627 return val;
628 }
629 default: break;
630 }
631
632 auto c = callee() ? callee()->emit(e) : nullptr;
633 auto l = lhs()->emit(e);
634 auto r = rhs()->emit(e);
635
636 switch (op().tag()) {
637 case Tag::T_arrow_r: return w.pi(l, r);
638 case Tag::T_at: return w.app(l, r);
639 case Tag::K_inj:
640 if (auto nom = r->isa<Nom>()) check_nom_scope(e, loc(), nom);
641 return w.inj(r, l);
642 default: return w.implicit_app(c, w.tuple({l, r})); // MIM_INFIX_SUGAR
643 }

661const Def* MatchExpr::emit_variant(Emitter& e, const Def* scrutinee, const Variant* variant) const {
662 auto sel = DefVec(variant->num_ops());
663 for (auto arm : arms()) {
664 auto i = size_t();
665 if (auto index = arm->index()) {
666 if (*index >= variant->num_ops())
667 e.error()
668 .e(arm->loc(), "variant `{}` has {} cases but case {} was requested", e.variant_str(variant),
669 variant->num_ops(), *index)
670 .bail();
671 i = *index;
672 } else if (auto ctor = arm->ctor()) {
673 i = e.ctor(variant, ctor);
674 } else if (auto id = arm->ptrn()->isa<IdPtrn>(); id && !id->type()) {
675 i = e.ctor(variant, id->dbg());
676 } else {
677 e.error()
678 .e(arm->ptrn()->loc(), "an arm of a match on variant `{}` must name a constructor",
679 e.variant_str(variant))
680 .bail();
681 }
682 if (sel[i]) {
683 e.error().w(arm->loc(), "this arm is unreachable: constructor {} is already handled",
684 e.ctor_names(variant, i));
685 continue;
686 }
687 sel[i] = arm->emit(e, variant->op(i));
688 }
689
690 auto missing = std::vector<std::string>();
691 for (size_t i = 0, n = sel.size(); i != n; ++i)
692 if (!sel[i]) missing.emplace_back(e.ctor_names(variant, i));
693 if (!missing.empty()) e.error().e(loc(), "match expression has no arm for {}", fe::Join(missing, ", ")).bail();
694
695 auto ops = DefVec{scrutinee};
696 ops.append_range(sel);
697 return e.world().match(ops);
698}

700const Def* MatchExpr::emit_(Emitter& e) const {
701 DefVec ops;
702 ops.emplace_back(scrutinee()->emit(e));
703 if (auto variant = ops.front()->isa_type<Variant>()) return emit_variant(e, ops.front(), variant);
704 if (auto nom = ops.front()->isa_type<Nom>())
705 if (auto variant = nom->op()->isa<Variant>();
706 variant && std::ranges::any_of(arms(), [](const auto& arm) { return arm->index() || arm->ctor(); })) {
707 open_nom(e, loc(), nom);
708 return emit_variant(e, e.world().struc(ops.front()), variant);
709 }
710
711 for (auto arm : arms()) {
712 if (arm->index() || arm->ctor())
713 e.error()
714 .e(arm->loc(), "a constructor pattern needs a variant scrutinee")
715 .n("but the scrutinee has type `{}`", type_of(ops.front()))
716 .bail();
717 ops.emplace_back(arm->emit(e));
718 }
719 auto res = e.world().match(ops);
720
721 // Only a *source* arm is unreachable by mistake; substitution legitimately kills arms of a polymorphic match.
722 auto cases = Match::cases(ops.front());
723 for (size_t i = 0, n = num_arms(); i != n; ++i)
724 if (std::ranges::none_of(cases, [&](const Def* c) { return Match::accepts(ops[i + 1], c); }))
725 e.error().w(arm(i)->loc(), "this arm is unreachable: no case of `{}` matches `{}`", type_of(ops.front()),
726 ops[i + 1]->isa_type<Pi>()->dom());
727
728 return res;
729}

1053void LamDecl::emit_body(Emitter& e) const {
1054 if (!body()) return; // extern forward declaration: the implementation lives in a native translation unit
1055
1056 auto _ = e.world().push(loc());
1057 {
1058 auto _body = e.world().push(body()->loc());
1059 doms().back()->lam_->set_body(body()->emit(e));
1060 }
1061
1062 // rewrite holes
1063 for (size_t i = 0, n = num_doms(); i != n; ++i) {
1064 auto rw = VarRewriter(e.world());
1065 auto lam = dom(i)->lam_;
1066 auto pi = lam->type()->as_mut<Pi>();
1067 for (auto dom : doms() | std::views::drop(i)) {
1068 // A Lam's Var that was never created cannot occur in the codomain.
1069 if (auto var = pi->has_var())

1071 auto cod = pi->codom();

1073 pi = cod->as_mut<Pi>();
1074 }
1075
1076 if (auto cod = pi->codom(); cod && cod->has_hole()) pi->set(pi->dom(), rw.rewrite(cod));
1077 }
1078
1079 for (auto dom : doms() | std::views::reverse) {
1080 if (auto imm = dom->pi_->immutabilize()) {
1081 auto f = dom->lam_->filter();

1084 }
1085 }
1086
1087 if (is_extern()) {
1088 auto lam = doms().front()->lam_;
1089 if (!lam->is_closed())
1090 e.error()

1093 "usually means an unannotated parameter's type could only be inferred to depend on a variable bound "

1100 .n(prev->loc(), "previous definition here")
1101 .bail();
1102 lam->externalize();
1103 }
1104 e.attach(annex_, sub_, dbg().sym(), def_);
1105}
