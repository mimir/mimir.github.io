16// During dumping, we classify Defs according to the following logic:
17// * Inline: These Defs are *always* displayed with all of its operands "inline".
18// E.g.: (1, 2, 3).
19// * All other Defs are referenced by its name/unique_name (see id) when they appear as an operand.
20// * Mutables are either classifed as "decl" (see isa_decl).
21// In this case, recursing through the Defs' operands stops and this particular Decl is dumped as its own thing.
22// * Or - if they are not a "decl" - they are basicallally handled like immutables.

28Def* isa_decl(const Def* def) {
29 if (auto mut = def->isa_mut()) {
30 if (mut->isa<Hole>()) return nullptr; // a Hole stands for what it was set to, or for `?`
31 if (mut->is_external() || mut->isa<Lam>() || (mut->sym() && mut->sym() != '_')) return mut;
32 }
33 return nullptr;
34}

36/// What the running dump decided about the Def%s it emits; `nullptr` where a diagnostic formats a bare Def.
37struct Ctx {
38 /// Names the dump binds itself: a `fun` calls its `ret` continuation `return`, whatever its Def::sym.
39 /// Keyed by Var and index, because the projection itself need not exist as a Def while the World is frozen.

42 DefSet opaque; ///< Def%s no pattern took apart, so their components have no name of their own.

46/// Def::unique_name - or the plain Def::sym while a diagnostic is being formatted, where a gid is noise.
47/// @p ctx is `nullptr` outside a running dump: there are no dump-bound names to consult then.
48std::string name(Ctx* ctx, const Def* def) {
49 if (ctx) {
50 if (auto i = ctx->names.find(def); i != ctx->names.end()) return i->second;
51 if (auto ex = def->isa<Extract>())
52 if (auto i = ctx->ret_projs.find(ex->tuple()); i != ctx->ret_projs.end())
53 if (Lit::isa(ex->index()) == i->second) return "return"s;
54 }
55 if (auto sym = def->sym(); sym && sym != '_' && PlainNames::claim(def->world().driver(), sym, def->gid()))
56 return sym.str();
57 return def->unique_name();
58}

75Prec def2prec(const Def* def) {
76 // A Var's projection prints as its name and hence is atomic; see the Extract case in operator<<.
77 if (auto ex = def->isa<Extract>())
78 return ex->tuple()->isa<Var>() && ex->index()->isa<Lit>() ? Prec::Lit : Prec::Extract;
79 if (def->isa<Insert>()) return Prec::Ins;
80 if (def->isa<Join>()) return Prec::Union;
81 if (def->isa<Inj>()) return Prec::Inj;
82 if (def->isa<Reform>()) return Prec::App;
83 // `Cn e` parses its domain at Prec::Bot, so it swallows whatever follows and only ever fits a closed context.
84 if (auto pi = def->isa<Pi>()) return Pi::isa_cn(pi) ? Prec::Bot : Prec::Arrow;
85 if (def->isa<Lam>()) return Prec::Bot; // a λ-expression swallows what follows it, too
86 if (auto app = def->isa<App>()) {
87 if (auto size = Idx::isa(app)) {
88 if (auto l = Lit::isa(size)) {
89 // clang-format off
90 switch (*l) {
91 case 0x0'0000'0002_n:
92 case 0x0'0000'0100_n:
93 case 0x0'0001'0000_n:
94 case 0x1'0000'0000_n:
95 case 0_n: return Prec::Lit;
96 default: break;
97 }
98 // clang-format on
99 }
100 }
101 return Prec::App;
102 }
103
104 return Prec::Lit;
105}

107/// This is a wrapper to dump a Def.
108class Op {
109public:
110 Op(Ctx* ctx, const Def* def, Prec prec = Prec::Bot, bool is_left = false)
111 : ctx_(ctx)
112 , def_(def)
113 , prec_(prec)
114 , is_left_(is_left) {}
115
116 /// @name Sub-expressions
117 /// They inherit this Op's Ctx - which is how the running dump reaches the whole recursion.
118 ///@{
119 Op op(const Def* def, Prec prec = Prec::Bot, bool is_left = false) const { return {ctx_, def, prec, is_left}; }
120 Op l(const Def* def, Prec prec = Prec::Bot) const { return {ctx_, def, prec, true}; }
121 Op r(const Def* def, Prec prec = Prec::Bot) const { return {ctx_, def, prec, false}; }
122 ///@}
123
124 static auto map(Ctx* ctx, const auto& range, const char* sep = ", ", Prec prec = Prec::Bot) {
125 return fe::Join(range | std::views::transform([ctx, prec](auto op) { return Op(ctx, op, prec); }), sep);
126 }
127
128 /// @name Getters
129 ///@{
130 Ctx* ctx() const { return ctx_; }
131 Prec prec() const { return prec_; }
132 bool is_left() const { return is_left_; }
133 const Def* def() const { return def_; }
134 const Def* operator->() const { return def_; }
135 const Def* operator*() const { return def_; }
136 explicit operator bool() const { return def_ != nullptr; }
137 ///@}
138
139private:
140 Ctx* ctx_;
141 const Def* def_;
142 Prec prec_;
143 bool is_left_;
144
145 /// This will stream @p def as an operand.
146 /// This is usually `id(def)` unless it can be displayed Inline.
147 friend std::ostream& operator<<(std::ostream&, Op);

150/// This is a wrapper to dump a Def "inline" and print it with all of its operands.
151class Full : public Op {
152public:
153 Full(Ctx* ctx, const Def* def, Prec prec = Prec::Bot, bool is_left = false)
154 : Op(ctx, def, prec, is_left) {}
155 Full(Op op)
156 : Full(op.ctx(), op.def(), op.prec(), op.is_left()) {}
157
158 explicit operator bool() const { return is_inline(); }
159
160 bool is_inline() const {
161 if (ctx() && ctx()->inlined.contains(def())) return true;
162 if (auto mut = def()->isa_mut()) {
163 if (isa_decl(mut)) return false;
164 return true;
165 }
166
167 if (def()->is_closed()) return true;
168
169 if (auto app = def()->isa<App>()) {
170 if (app->type()->isa<Pi>()) return true; // curried apps are printed inline
171 if (app->type()->isa<Type>()) return true;
172 if (app->callee()->isa<Axm>()) return app->callee_type()->num_doms() <= 1;
173 return false;
174 }
175
176 return true;
177 }
178
179 bool needs_parens() const {
180 if (!is_inline()) return false;
181
182 auto child_prec = def2prec(def());
183 if (child_prec < prec()) return true;
184 if (child_prec > prec()) return false;
185
186 switch (prec_assoc(prec())) {
187 case Assoc::R: return is_left();
188 case Assoc::L: return !is_left();
189 case Assoc::N: return false;
190 }
191 fe::unreachable();
192 }
193
194 friend std::ostream& operator<<(std::ostream&, Full);
195};

208/// Def::num_tprojs, except that a *fused* Seq stays one binder: dumping freezes the World and its sub-Seqs
209/// need not exist as Def%s.
210nat_t num_binders(const Def* type) {
211 if (auto seq = type->isa<Seq>(); seq && seq->shape().is_fused()) return 1;
212 return type->num_tprojs();
213}

215/// A Seq's shape the way the surface syntax spells it: one axis per dimension, `2, 3` for a fused `(2, 3)`.
216/// Falls back to the shape as a whole - also a legal spelling - if an axis does not exist as a Def.
217std::string shape(Ctx* ctx, const Seq* seq) {
218 auto s = seq->shape();
219 auto r = s.rank();
220 if (!r || *r <= 1) return std::format("{}", Op(ctx, *s));
221
222 auto axes = s->projs(*r);
223 if (std::ranges::any_of(axes, [](auto a) { return !a; })) return std::format("{}", Op(ctx, *s));
224 return std::format("{}", Op::map(ctx, axes));
225}
226
227/// As above but each axis binds its index - `i: 2, j: 3`; a fused Seq binds one Var per axis.
228std::string shape(Ctx* ctx, const Seq* seq, const Def* var) {
229 auto s = seq->shape();
230 auto r = s.rank();
231 if (!r || *r <= 1) return std::format("{}: {}", Op(ctx, var), Op(ctx, *s));
232
233 auto res = std::string();
234 for (auto sep = ""; auto i : std::views::iota(nat_t(0), *r)) {
235 auto axis = var->proj(*r, i);
236 auto extent = s[i];
237 if (!extent) return std::format("{}: {}", Op(ctx, var), Op(ctx, *s));
238 // A projection the body never mentions does not exist as a Def and hence has no name of its own.
239 res += std::format("{}{}: {}", sep, axis ? std::format("{}", Op(ctx, axis)) : "_"s, Op(ctx, extent));
240 sep = ", ";
241 }
242 return res;
243}

245/// May @p def of @p type be taken apart into @p n components?
246/// A component the frozen World does not hand out has no name, so it has to fall back to @p type - and that
247/// spells a *dependent* component with the domain's binders instead of this pattern's. That is where
248/// destructuring stops: @p def stays opaque and its components print as `def#i` rather than by a dangling name.
249bool destructible(Ctx* ctx, const Def* def, const Def* type, size_t n) {
250 if (!def || n <= 1) return false;
251 auto var = type->isa_mut<Sigma>() ? type->has_var() : nullptr;
252
253 for (size_t i = 0; i != n; ++i) {
254 if (def->proj(n, i)) continue; // it exists and hence brings its own name and type
255 auto t = type->proj(n, i);
256 if (!t || (var && t->has_free_var(var))) {
257 if (ctx) ctx->opaque.emplace(def);
258 return false;
259 }
260 }
261 return true;
262}

264/// @p brckt selects the `[]` sub-patterns of a type over the `()` sub-patterns of a Lam's parameter list.
265void ptrn(std::ostream& os, Ctx* ctx, const Def* def, const Def* type, bool brckt = false) {
266 if (!def) return std::print(os, "_: {}", Op(ctx, type));
267
268 auto n = def->num_tprojs();
269 if (!destructible(ctx, def, type, n)) return std::print(os, "{}: {}", name(ctx, def), Op(ctx, type));
270
271 os << (brckt ? '[' : '(');
272 for (auto sep = ""; auto i : std::views::iota(size_t(0), n)) {
273 auto proj = def->proj(n, i);
274 os << sep;
275 // A projection's own type is the one where the binder has already been substituted for this Var.
276 ptrn(os, ctx, proj, proj ? proj->type() : type->proj(n, i), brckt);
277 sep = ", ";
278 }
279 // The alias keeps a name for the entity as a whole - which the body may well refer to.
280 std::print(os, "{} as {}", brckt ? ']' : ')', name(ctx, def));
281}

283/// A Pi's domain the way the surface syntax binds it - a pattern, once the Var's projections are in use.
284void dom_ptrn(std::ostream& os, Ctx* ctx, const Def* var, const Def* type, bool implicit) {
285 auto l = implicit ? '{' : '[';
286 auto r = implicit ? '}' : ']';
287 auto n = var->num_tprojs();
288 if (!destructible(ctx, var, type, n)) return std::print(os, "{}{}: {}{}", l, name(ctx, var), Op(ctx, type), r);
289
290 os << l;
291 for (auto sep = ""; auto i : std::views::iota(size_t(0), n)) {
292 os << sep;
293 auto proj = var->proj(n, i);
294 ptrn(os, ctx, proj, proj ? proj->type() : type->proj(n, i), true);
295 sep = ", ";
296 }
297 std::print(os, "{} as {}", r, name(ctx, var));
298}

305/// @p num is how many binders the domain has; @p limit how many of them to print - a returning Lam hides its last.
306void curry(std::ostream& os,
307 Ctx* ctx,
308 const Def* def,
309 const Def* type,
310 bool implicit,
311 size_t num,
312 size_t limit,
313 bool alias) {
314 auto l = implicit ? '{' : '(';
315 auto r = implicit ? '}' : ')';
316
317 if (limit == 0) return (void)(os << l << r);
318 // Same as in ptrn: a parameter list only comes apart if every parameter is there to be named.
319 if (limit == num && !destructible(ctx, def, type, num)) {
320 if (def) return std::print(os, "{}{}: {}{}", l, name(ctx, def), Op(ctx, type), r);
321 return std::print(os, "{}_: {}{}", l, Op(ctx, type), r);
322 }
323
324 os << l;
325 for (auto sep = ""; auto i : std::views::iota(size_t(0), limit)) {
326 os << sep;
327 auto proj = def ? def->proj(num, i) : nullptr;
328 bndr(os, ctx, proj, proj ? proj->type() : type->proj(num, i));
329 sep = ", ";
330 }
331 os << r;
332 if (alias && def) std::print(os, " as {}", name(ctx, def));
333}

335/// Is @p lam sugar for a `fun`?
336/// That sugar folds the domain into `[<params>, Cn <ret>]`, so a *flat* domain with a trailing Cn stays a `con`.
337bool isa_fun(const Lam* lam) {
338 auto pi = lam->type();
339 return Lam::isa_returning(lam) && num_binders(pi->dom()) == 2 && Pi::isa_basicblock(pi->dom(2, 1));
340}

342/// A Lam's parameter list, plus its filter where that is not the default for its position.
343/// @p last marks a curried chain's final Lam: only that one hides a `fun`'s `ret` and defaults its filter to `ff`.
344void lam_ptrn(std::ostream& os, Ctx* ctx, Lam* lam, bool fun, bool con, bool last, bool alias) {
345 auto num = num_binders(lam->type()->dom());
346 auto limit = fun && last ? num - 1 : num;
347 curry(os, ctx, lam->has_var(), lam->type()->dom(), lam->type()->is_implicit(), num, limit, alias);
348 auto& w = lam->world();
349 if (auto dflt = last && (fun || con) ? w.lit_ff() : w.lit_tt(); lam->filter() != dflt)
350 std::print(os, "@({})", Op(ctx, lam->filter()));
351}

367/// Streams @p d with all of its operands; Full::is_inline decides who gets here.
368void full(std::ostream& os, Full d) {
369 if (auto hole = d->isa_mut<Hole>()) {
370 if (hole->is_set()) return std::print(os, "{}", d.op(hole->op()));
371 return std::print(os, "?");
372 }
373 if (auto mut = d->isa_mut(); mut && !mut->is_set()) return std::print(os, "unset");
374 if (d.needs_parens()) return std::print(os, "({})", Full(d.ctx(), *d));
375
376 bool ascii = d->world().flags().ascii;
377 auto arw = ascii ? "->" : "→";
378 auto al = ascii ? "<<" : "«";
379 auto ar = ascii ? ">>" : "»";
380 auto pl = ascii ? "(<" : "‹";
381 auto pr = ascii ? ">)" : "›";
382 auto bot = ascii ? "bot" : "⊥";
383 auto top = ascii ? "top" : "⊤";
384
385 if (auto type = d->isa<Type>()) {
386 if (auto level = Lit::isa(type->level()); level && !ascii) {
387 if (level == 0) return std::print(os, "*");
388 if (level == 1) return std::print(os, "□");
389 }
390 return std::print(os, "Type {}", d.r(type->level(), Prec::App));
391 } else if (auto reform = d->isa<Reform>()) {
392 return std::print(os, "Rule {}", d.r(reform->dom(), Prec::App));
393 } else if (d->isa<Univ>()) {
394 return std::print(os, "Univ");
395 } else if (d->isa<Nat>()) {
396 return std::print(os, "Nat");
397 } else if (d->isa<Idx>()) {
398 return std::print(os, "Idx");
399 } else if (auto ext = d->isa<Ext>()) {
400 return std::print(os, "{}:{}", ext->isa<Bot>() ? bot : top, d.r(ext->type(), Prec::Lit));
401 } else if (auto axm = d->isa<Axm>()) {
402 return std::print(os, "{}", axm->sym());
403 } else if (auto lit = d->isa<Lit>()) {
404 if (lit->type()->isa<Nat>()) {
405 // clang-format off
406 switch (lit->get()) {
407 case 0x0'0000'0100_n: return std::print(os, "i8");
408 case 0x0'0001'0000_n: return std::print(os, "i16");
409 case 0x1'0000'0000_n: return std::print(os, "i32");
410 default: return std::print(os, "{}", lit->get());
411 }
412 // clang-format on
413 } else if (auto size = Idx::isa(lit->type())) {
414 if (auto s = Lit::isa(size)) {
415 // clang-format off
416 switch (*s) {
417 case 0x0'0000'0002_n: return std::print(os, "{}", lit->get<bool>() ? "tt" : "ff");
418 case 0x0'0000'0100_n: return std::print(os, "{}I8" , lit->get());
419 case 0x0'0001'0000_n: return std::print(os, "{}I16", lit->get());
420 case 0x1'0000'0000_n: return std::print(os, "{}I32", lit->get());
421 case 0_n: return std::print(os, "{}I64", lit->get());
422 default: {
423 std::print(os, "{}", lit->get());
424 std::vector<uint8_t> digits;
425 for (auto z = *s; z; z /= 10) digits.emplace_back(z % 10);
426
427 // Raw UTF-8 units of a subscript - bytes to stream, not numbers to format.
428 if (ascii) {
429 os << '_';
430 for (auto d : digits | std::views::reverse)
431 os << char('0' + d);
432 } else {
433 for (auto d : digits | std::views::reverse)
434 os << uint8_t(0xE2) << uint8_t(0x82) << (uint8_t(0x80 + d));
435 }
436 return;
437 }
438 }
439 // clang-format on
440 }
441 }
442 return std::print(os, "{}:{}", lit->get(), d.r(lit->type(), Prec::Lit));
443 } else if (auto ex = d->isa<Extract>()) {
444 // A Var's component prints by name - unless the pattern that would have bound that name kept the Var whole.
445 auto opaque = d.ctx() && d.ctx()->opaque.contains(ex->tuple());
446 if (!opaque && ex->tuple()->isa<Var>() && ex->index()->isa<Lit>())
447 return std::print(os, "{}", name(d.ctx(), ex));
448 return std::print(os, "{}#{}", d.l(ex->tuple(), Prec::Extract), d.r(ex->index(), Prec::Extract));
449 } else if (auto ins = d->isa<Insert>()) {
450 auto tup = d.l(ins->tuple(), Prec::Extract);
451 // `←` updates the whole `#`-path, so an Extract target needs parens to re-parse.
452 if (auto ex = ins->tuple()->isa<Extract>(); ex && !(ex->tuple()->isa<Var>() && ex->index()->isa<Lit>()))
453 std::print(os, "({})", tup);
454 else
455 std::print(os, "{}", tup);
456 return std::print(os, "#{} ← {}", d.r(ins->index(), Prec::Extract), d.r(ins->value(), Prec::Ins));
457 } else if (auto var = d->isa<Var>()) {
458 return std::print(os, "{}", name(d.ctx(), var));
459 } else if (auto [pi, var] = d->isa_binder<Pi>(); pi) {
460 dom_ptrn(os, d.ctx(), var, pi->dom(), pi->is_implicit());
461 return std::print(os, " {} {}", arw, d.r(pi->codom(), Prec::Arrow));
462 } else if (auto pi = d->isa<Pi>()) {
463 if (Pi::isa_cn(pi)) return std::print(os, "Cn {}", d.op(pi->dom()));
464 if (pi->is_implicit())
465 return std::print(os, "{{_: {}}} {} {}", d.op(pi->dom()), arw, d.r(pi->codom(), Prec::Arrow));
466 return std::print(os, "{} {} {}", d.l(pi->dom(), Prec::Arrow), arw, d.r(pi->codom(), Prec::Arrow));
467 } else if (auto lam = d->isa_mut<Lam>()) {
468 // A Lam that has no declaration of its own is a λ-expression.
469 auto fun = isa_fun(lam);
470 auto con = Lam::isa_cn(lam) && !fun;
471 std::print(os, "{}", fun ? "fn " : con ? "cn " : "λ ");
472 lam_ptrn(os, d.ctx(), lam, fun, con, true, true);
473 lam_codom(os, d.ctx(), lam, fun, con);
474 // Like any tail, the body is spelled out - there is no block here that could `let`-bind it.
475 if (isa_decl(lam->body())) return std::print(os, " = {}", d.op(lam->body()));
476 return std::print(os, " = {}", Full(d.ctx(), lam->body()));
477 } else if (auto app = d->isa<App>()) {
478 if (auto size = Idx::isa(app)) {
479 if (auto l = Lit::isa(size)) {
480 // clang-format off
481 switch (*l) {
482 case 0x0'0000'0002_n: return std::print(os, "Bool");
483 case 0x0'0000'0100_n: return std::print(os, "I8");
484 case 0x0'0001'0000_n: return std::print(os, "I16");
485 case 0x1'0000'0000_n: return std::print(os, "I32");
486 case 0_n: return std::print(os, "I64");
487 default: break;
488 }
489 // clang-format on
490 }
491 }
492
493 if (Pi::isa_implicit(app->callee()->unfold_type())) {
494 // An argument that is still a Hole is what the parser inserts anyway - and `?` has no spelling.
495 if (app->arg()->has_dep(Dep::Hole)) return std::print(os, "{}", d.op(app->callee(), d.prec(), d.is_left()));
496 // Spelling out an implicit argument needs `@`; juxtaposition would insert a Hole in front of it.
497 return std::print(os, "{} @ {}", d.l(app->callee(), Prec::App), d.r(app->arg(), Prec::App));
498 }
499 return std::print(os, "{} {}", d.l(app->callee(), Prec::App), d.r(app->arg(), Prec::App));
500 } else if (auto [sigma, var] = d->isa_binder<Sigma>(); sigma) {
501 size_t i = 0;
502 auto elem = fe::StreamFn{[&](std::ostream& os) -> std::ostream& {
503 auto sep = "";
504 for (auto op : sigma->ops()) {
505 os << sep;
506 if (auto v = sigma->var(sigma->num_ops(), i++))
507 std::print(os, "{}: {}", d.op(v), d.op(op));
508 else
509 os << d.op(op);
510 sep = ", ";
511 }
512 return os;
513 }};
514
515 return std::print(os, "[{}]", elem);
516 } else if (auto sigma = d->isa<Sigma>()) {
517 return std::print(os, "[{}]", Op::map(d.ctx(), sigma->ops()));
518 } else if (auto tuple = d->isa<Tuple>()) {
519 return std::print(os, "({})", Op::map(d.ctx(), tuple->ops()));
520 } else if (auto [arr, var] = d->isa_binder<Arr>(); arr) {
521 return std::print(os, "{}{}; {}{}", al, shape(d.ctx(), arr, var), d.op(arr->body()), ar);
522 } else if (auto arr = d->isa<Arr>()) {
523 return std::print(os, "{}{}; {}{}", al, shape(d.ctx(), arr), d.op(arr->body()), ar);
524 } else if (auto [pack, var] = d->isa_binder<Pack>(); pack) {
525 return std::print(os, "{}{}; {}{}", pl, shape(d.ctx(), pack, var), d.op(pack->body()), pr);
526 } else if (auto pack = d->isa<Pack>()) {
527 return std::print(os, "{}{}; {}{}", pl, shape(d.ctx(), pack), d.op(pack->body()), pr);
528 } else if (auto proxy = d->isa<Proxy>()) {
529 return std::print(os, "(proxy#{} {})", proxy->tag(), Op::map(d.ctx(), proxy->ops()));
530 } else if (auto bound = d->isa<Bound>()) {
531 auto op = bound->isa<Join>() ? "∪" : "∩"; // TODO ascii
532 if (auto mut = d->isa_mut()) std::print(os, "{}{}: {}", op, name(d.ctx(), mut), d.op(mut->type()));
533 if (!bound->isa<Join>()) return std::print(os, "{}({})", op, Op::map(d.ctx(), bound->ops()));
534 return std::print(os, "{}", Op::map(d.ctx(), bound->ops(), " ∪ ", Prec::Union));
535 } else if (auto inj = d->isa<Inj>()) {
536 return std::print(os, "{} inj {}", d.l(inj->value(), Prec::Inj), d.r(inj->type(), Prec::Inj));
537 } else if (auto single = d->isa<Single>()) {
538 return std::print(os, "{}{}{}", al, d.op(single->op()), ar);
539 } else if (auto wrap = d->isa<Wrap>()) {
540 return std::print(os, "{}{}{}", pl, d.op(wrap->op()), pr);
541 }
542
543 // other
544 auto tag = d->flags() == 0 ? std::string(d->node_name()) : std::format("{}#{}", d->node_name(), d->flags());
545 if (d->ops().empty()) return std::print(os, "({})", tag);
546 std::print(os, "({} {})", tag, Op::map(d.ctx(), d->ops(), " "));
547}

558/// The Lam%s a curried declaration folds into one: `lam f (a) (b) = e`.
559fe::Vector<Lam*> curry_chain(Lam* lam) {
560 auto chain = fe::Vector<Lam*>();
561 for (auto* curr = lam;;) {
562 chain.emplace_back(curr);
563 if (auto body = curr->body())
564 if (auto next = body->isa_mut<Lam>(); next && !next->is_external()) {
565 curr = next;
566 continue;
567 }
568 break;
569 }
570 return chain;
571}

573/// Emits Def%s as Mim declarations.
574///
575/// Dump::Expr and Dump::Local walk Def::deps and nothing else - which is what makes them safe to call from a
576/// debugger, where the World is usually half-built. Dump::Scope and Dump::All buy a better layout with a Nest:
577/// it nests the mutables, and Scheduler::smart places everything else. Should that analysis choke on the very
578/// program you wanted to look at, World::dot still shows it - its tooltips only ever use Dump::Expr.
579class Dumper {
580public:
581 /// @p srcs are the files an `import` pulls in: they declare their own content, so a dump must not repeat it.
582 Dumper(std::ostream& os, Dump mode, bool typed_let = false, absl::flat_hash_set<const fe::Src*> srcs = {})
583 : os_(os)
584 , mode_(mode)
585 , typed_let_(typed_let)
586 , srcs_(std::move(srcs)) {}
587
588 /// @name dump
589 ///@{
590 /// @p def and whatever Dumper::mode_ reaches from it.
591 void dump(const Def* def) {
592 if (auto mut = isa_decl(def)) return dump_muts(mut);
593
594 if (mode_ != Dump::Local)
595 for (auto mut : def->local_muts())
596 if (isa_decl(mut)) dump_muts(mut);
597
598 block_ = nullptr; // a non-decl opens no block, so nothing can be entered or placed into one
599 emit_block(def);
600 sep(Prev::Let);
601 emit_tail(def, "");
602 }
603
604 ///@}
605
606private:
607 /// What the current block emitted last; only a declaration is worth setting apart.
608 enum class Prev { None, Let, Decl };
609
610 /// One scope: the Nest of a single closed mutable - which is the only thing a Scheduler can place into.
611 void visit_scope(const Nest& nest) {
612 auto root = nest.root()->mut();
613 // An `import` re-declares what it *exports* - but a plain `let` in that file stays local to it, so the
614 // dump has to spell that one out or nothing binds the name it refers to.
615 if (root->is_external() && srcs_.contains(root->loc().src)) return;
616
617 auto sched = Scheduler(nest);
618 auto _ = fe::Restore(nest_, &nest);
619 auto __ = fe::Restore(sched_, &sched);
620 block_ = root;
621 emit_block(root);
622 }
623
624 /// Schedules @p root into the current Dumper::block_ and emits whatever landed there.
625 void emit_block(const Def* root) {
626 schedule_(root, nullptr);
627 bucket();
628 emit(nullptr);
629 }
630
631 /// @name schedule
632 ///@{
633 /// The closed mutables @p mut reaches, callees first: Mim binds a name before its uses.
634 void dump_muts(Def* mut) {
635 auto descend = [this](Def*) { return mode_ == Dump::All || mode_ == Dump::Local; };
636 auto collect = [this](Def* mut) { return mut->is_closed() || mode_ == Dump::Local; };
637 auto todo = fe::Vector<Def*>();
638 post_order(mut, scheduled_, todo, descend, collect);
639
640 for (auto curr : todo) {
641 if (mode_ == Dump::Local) {
642 block_ = curr;
643 emit_block(curr);
644 } else {
645 visit_scope(Nest(curr));
646 }
647 }
648 }
649
650 /// @name schedule
651 ///@{
652 void schedule_(const Def* def, Def* curr) {
653 if (!def) return;
654 if (auto mut = isa_decl(def)) return schedule_mut(mut, curr);
655 if (!done_.emplace(def).second) return;
656 for (auto op : def->deps())
657 schedule_(op, curr);
658 if (!Full(&ctx_, def)) order_.emplace_back(def, curr);
659 }
660
661 void schedule_mut(Def* mut, Def* curr) {
662 if (open_.contains(mut)) recursive_.emplace(mut);
663 if (!enter(mut)) return;
664 if (!done_.emplace(mut).second) return;
665
666 auto chain = mut->isa_mut<Lam>() ? curry_chain(mut->as_mut<Lam>()) : fe::Vector<Lam*>();
667 open_.emplace(mut);
668
669 if (chain.empty()) {
670 for (auto op : mut->deps())
671 schedule_(op, mut);
672 } else {
673 for (auto* lam : chain)
674 if (lam != mut) done_.emplace(lam), absorbed_.emplace(lam, mut);
675 for (auto* lam : chain) {
676 schedule_(lam->type(), mut);
677 schedule_(lam->filter(), mut);
678 if (lam == chain.back()) schedule_tail(lam->body(), mut);
679 }
680 }
681
682 open_.erase(mut);
683 order_.emplace_back(mut, curr);
684 }
685
686 /// A Lam's body is the tail of its block, so it is emitted there instead of as a `let` of its own.
687 void schedule_tail(const Def* def, Def* curr) {
688 if (!def || isa_decl(def)) return schedule_(def, curr);
689 if (!done_.emplace(def).second) return;
690 for (auto op : def->deps())
691 schedule_(op, curr);
692 }
693
694 /// Is @p mut ours to emit - or does it only get referenced by name?
695 bool enter(Def* mut) const { return nest_ ? (*nest_)[mut] != nullptr : mut == block_; }
696
697 /// Turns the schedule into one list per block: the Nest nests the mutables, Scheduler::smart places the rest.
698 void bucket() {
699 for (auto [def, curr] : order_) {
700 auto mut = isa_decl(def);
701 auto key = curr; // Dump::Local has one block per root and asks no analysis where anything belongs
702 if (nest_) {
703 // Nest::idom, not Nest::inest: a mutable only one sibling reaches belongs into *its* block.
704 auto in = curr ? curr : block_;
705 key = mut ? owner((*nest_)[mut]->idom()) : owner(sched_->smart(in, def));
706 // A binder that prints inline has no block of its own, so what belongs into it is inlined as well.
707 if (key && inlines(key) && !(mut && recursive_.contains(mut))) {
708 ctx_.inlined.emplace(def);
709 continue;
710 }
711 }
712 (key ? bucket_[key] : top_).emplace_back(def);
713 }
714 order_.clear();
715 }
716
717 /// Does @p mut print inline - be it one itself, or because it sits inside one that does?
718 bool inlines(Def* mut) const {
719 for (auto node = (*nest_)[mut]; node; node = node->inest())
720 if (auto m = owner(node); m && !isa_decl(m)) return true;
721 return false;
722 }
723
724 /// The mutable whose block @p node stands for; `nullptr` for the top level.
725 Def* owner(const Nest::Node* node) const {
726 if (!node) return nullptr;
727 auto mut = node->mut();
728 if (!mut) return nullptr; // a *virtual* root is the top level
729 if (auto i = absorbed_.find(mut); i != absorbed_.end()) return i->second; // a curried chain shares a block
730 return mut;
731 }
732 ///@}
733
734 /// @name emit
735 ///@{
736 void emit(Def* key) {
737 auto defs = key ? std::move(bucket_[key]) : std::move(top_);
738 for (auto def : defs) {
739 auto mut = isa_decl(def);
740 sep(mut ? Prev::Decl : Prev::Let);
741 if (mut)
742 emit_decl(mut);
743 else
744 emit_let(def);
745 }
746 }
747
748 /// A declaration spans several lines, so a blank line sets it apart from its neighbors within the same block.
749 void sep(Prev curr) {
750 if (prev_ == Prev::Decl || (prev_ == Prev::Let && curr == Prev::Decl)) os_ << '\n';
751 prev_ = curr;
752 }
753
754 void emit_let(const Def* def) {
755 auto type = typed_let_ ? std::format(": {}", Op(&ctx_, def->type())) : std::string();
756 std::println(os_, "{}let {}{} = {};", tab_, name(&ctx_, def), type, Full(&ctx_, def));
757 }
758
759 void emit_decl(Def* mut) {
760 if (auto lam = mut->isa_mut<Lam>()) return emit_lam(lam);
761 if (!mut->is_set()) return emit_unset(mut);
762 // `rec` binds the name for the body - which only a self-referential mutable needs; `extern` is out either way.
763 std::println(os_, "{}{} {} = {};", tab_, recursive_.contains(mut) ? "rec" : "let", id(&ctx_, mut),
764 Full(&ctx_, mut));
765 }
766
767 /// Nothing declares a mutable that was never set, so leave a trace instead of an unreadable dump.
768 void emit_unset(Def* mut) {
769 std::println(os_, "{}// `{}: {}` is unset", tab_, id(&ctx_, mut), Op(&ctx_, mut->type()));
770 }
771
772 void emit_lam(Lam* lam) {
773 auto chain = curry_chain(lam);
774 auto last = chain.back();
775 auto fun = isa_fun(last) && !shadows_ret(last);
776 auto con = Lam::isa_cn(last) && !fun;
777
778 // A `fun` binds its `ret` continuation as `return`, so that is the name its Var has to print with.
779 if (fun) {
780 if (auto var = last->has_var()) {
781 if (auto num = num_binders(last->type()->dom()); num == 1)
782 ctx_.names.emplace(var, "return");
783 else
784 ctx_.ret_projs.emplace(var, num - 1);
785 }
786 }
787 // A Lam without a body forward-declares what a native translation unit provides - and only an `extern`
788 // may go without one.
789 if (!last->is_set()) return emit_bodyless(lam, chain, fun, con);
790
791 std::print(os_, "{}{}{} {}", tab_, external(lam), fun ? "fun" : con ? "con" : "lam", id(&ctx_, lam));
792 for (auto* c : chain) {
793 os_ << ' ';
794 lam_ptrn(os_, &ctx_, c, fun, con, c == last, !fun || c != last);
795 }
796
797 lam_codom(os_, &ctx_, last, fun, con);
798 os_ << " =\n";
799
800 ++tab_;
801 rets_.emplace_back(fun ? last->has_var() : nullptr);
802 auto _ = fe::Restore(prev_, Prev::None);
803 emit(lam);
804 sep(Prev::Let);
805 emit_tail(last->body(), ";");
806 rets_.pop_back();
807 --tab_;
808 }
809
810 /// A Lam without a body only ever declares what some backend provides: `extern con f [mem.M 0, I32];`.
811 void emit_bodyless(Lam* lam, const fe::Vector<Lam*>& chain, bool fun, bool con) {
812 auto last = chain.back();
813 auto dom = last->type()->dom();
814 std::print(os_, "{}extern {} {}", tab_, fun ? "fun" : con ? "con" : "lam", id(&ctx_, lam));
815 auto num = num_binders(dom);
816 if (fun) {
817 os_ << '[';
818 for (auto sep = ""; auto i : std::views::iota(size_t(0), num - 1)) {
819 std::print(os_, "{}{}", sep, Op(&ctx_, dom->proj(num, i)));
820 sep = ", ";
821 }
822 std::println(os_, "]: {};", Op(&ctx_, last->ret_dom()));
823 } else {
824 std::println(os_, "[{}]{};", Op::map(&ctx_, dom->projs(num)),
825 con ? std::string() : std::format(": {}", Op(&ctx_, last->type()->codom())));
826 }
827 }
828
829 /// Tail position: what a block ends with is spelled out - it has no `let` of its own.
830 void emit_tail(const Def* def, std::string_view end) {
831 if (isa_decl(def))
832 std::println(os_, "{}{}{}", tab_, Op(&ctx_, def), end);
833 else
834 std::println(os_, "{}{}{}", tab_, Full(&ctx_, def), end);
835 }
836
837 /// Would a `fun` here hide the `return` of an enclosing one @p lam might still refer to?
838 /// The `return` is a projection of its Lam's Var, so capturing that Var at all is as precise as this gets.
839 bool shadows_ret(Lam* lam) const {
840 for (auto* var : rets_)
841 if (var && lam->has_free_var(var)) return true;
842 return false;
843 }
844 ///@}
845
846 std::ostream& os_;
847 Dump mode_;
848 bool typed_let_;
849 const Nest* nest_ = nullptr; ///< Nests the mutables of the scope being emitted.
850 Scheduler* sched_ = nullptr; ///< Places everything else in it.
851 Def* block_ = nullptr; ///< The mutable whose block we are filling; the only one in Dump::Local.
852 Prev prev_ = Prev::None;
853 fe::Tab tab_ = fe::Tab::spaces();
854 fe::Vector<const Var*> rets_;
855 fe::Vector<std::pair<const Def*, Def*>> order_; ///< What to emit, in dependency order, with its block's mutable.
856 DefVec top_;
857 MutMap<DefVec> bucket_;
858 MutMap<Def*> absorbed_; ///< Inner Lam of a curried chain -> the chain's outermost one.
859 DefSet done_;
860 MutSet open_; ///< On the schedule stack - a Def reaching one of these is recursive.
861 MutSet recursive_;
862 Ctx ctx_;
863 absl::flat_hash_set<const fe::Src*> srcs_;
864 MutSet scheduled_;
865};

904void World::dump(std::ostream& os) {
905 auto _ = freeze();
906 auto old_gid = curr_gid();
907 auto srcs = absl::flat_hash_set<const fe::Src*>();
908 for (const auto& import : driver().imports())
909 srcs.emplace(import.src);
910 for (const auto& import : driver().imports()) {
911 auto kw = import.tag == ast::Tok::Tag::K_plugin ? "plugin" : "import";
912 // The spelling was relative to the importing file; only the resolved path re-parses from here.
913 // Generic format: a native Windows `\` would lex as an escape sequence inside the string literal.
914 if (import.path)
915 std::print(os, "{} \"{}\";\n", kw, ast::Lexer::escape(import.src->path().generic_string()));
916 else
917 std::print(os, "{} {};\n", kw, import.sym);
918 }
919
920 if (!driver().imports().entries().empty() && externals().size() != 0) os << '\n';
921
922 // The local dump keeps every mutable to itself: no Nest that a broken program could trip over.
923 auto mode = flags().mim_local ? Def::Dump::Local : Def::Dump::All;
924 auto dumper = Dumper(os, mode, flags().mim_typed_let, std::move(srcs));
925 for (auto mut : externals().muts())
926 dumper.dump(mut);
927
928 assertf(old_gid == curr_gid(), "new nodes created during dump. old_gid: {}; curr_gid: {}", old_gid, curr_gid());
929}
