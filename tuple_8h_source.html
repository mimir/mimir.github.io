105/// Filters the components of an aggregate and keeps track of where the survivors end up.
106/// This is what a Phase needs that *drops* components of a Sigma - or the parameters of a Lam:
107/// * Sieve::gather rebuilds an immutable aggregate,
108/// * Sieve::new2old rebuilds a mutable one via Rewriter::rewrite_stub, and
109/// * Sieve::operator[] adjusts the index of every Extract / Insert into it.
110///
111/// @note Only a Sigma ever needs that last adjustment: an Arr's components all share one type - so it is dropped as a
112/// whole - and World::extract_fused splits a fused index at the Arr boundary, which leaves a Sigma's index scalar.
113///
114/// A rewrite that also *adds* components - `mem::SEO` threads fresh φs through a Lam - Sieve::cat%s them as further
115/// candidates: they extend the very same index space, so one Sieve still answers the whole new layout.
116class Sieve {
117public:
118 static constexpr size_t Gone = size_t(-1); ///< @see Sieve::operator[]
119
120 /// @name Construction
121 ///@{
122 /// Keeps all @p num components.
123 explicit Sieve(size_t num)
124 : num_(num)
125 , new2old_(num, [](size_t i) { return i; }) {}
126
127 /// Keeps those of @p num candidates that satisfy @p pred.
128 Sieve(size_t num, std::predicate<size_t> auto pred)
129 : num_(num) {
130 new2old_.reserve(num);
131 for (size_t i = 0; i != num; ++i)
132 if (pred(i)) new2old_.emplace_back(i);
133 }
134
135 /// Keeps those @p ops that satisfy @p pred.
136 Sieve(Defs ops, std::predicate<const Def*> auto pred)
137 : Sieve(ops.size(), [ops, pred](size_t i) { return pred(ops[i]); }) {}
138 ///@}
139
140 /// @name Concat
141 /// Candidates the rewrite *adds*; they have no old counterpart and simply continue the candidate indices.
142 ///@{
143 /// Concats @p num candidates behind the old ones and keeps those that satisfy @p pred.
144 /// @note A concat'ed component has no old op to rewrite from, so this rules out Sieve::new2old.
145 void cat(size_t num, std::predicate<size_t> auto pred) {
146 new2old_.reserve(new2old_.size() + num);
147 for (size_t i = 0; i != num; ++i)
148 if (pred(i)) {
149 new2old_.emplace_back(num_ + i);
150 ++num_cat_new_;
151 }
152 num_ += num;
153 num_cat_ += num;
154 }
155
156 size_t num_cat() const { return num_cat_; } ///< Number of concat'ed candidates.
157 size_t cat(size_t i) const { return (*this)[num_old() + i]; } ///< As Sieve::operator[] but for concat'ed @p i.
158 ///@}
159
160 /// @name Getters
161 ///@{
162 size_t num_old() const { return num_ - num_cat_; } ///< Number of old components.
163 size_t num_new() const { return new2old_.size(); } ///< Number of survivors.
164 /// Is the new aggregate the old one - nothing dropped and nothing concat'ed?
165 bool all() const { return num_new() == num_old() && num_cat_new_ == 0; }
166 /// The old index of the @p i th survivor; this is what Rewriter::rewrite_stub expects.
167 fe::View<size_t> new2old() const {
168 assert(num_cat_ == 0 && "rewrite_stub cannot rewrite a concat'ed component from an old op");
169 return new2old_;
170 }
171 /// The new index of the candidate @p i - or Sieve::Gone, if it didn't survive.
172 size_t operator[](size_t i) const {
173 auto j = std::ranges::lower_bound(new2old_, i);
174 return j != new2old_.end() && *j == i ? size_t(j - new2old_.begin()) : Gone;
175 }
176 /// The surviving components of @p ops - one per candidate, concat'ed ones included.
177 DefVec gather(Defs ops) const {
178 assert(ops.size() == num_);
179 return DefVec(num_new(), [this, ops](size_t i) { return ops[new2old_[i]]; });
180 }
181 ///@}
182
183private:
184 size_t num_; ///< Candidates - old ones followed by concat'ed ones.
185 size_t num_cat_ = 0;
186 size_t num_cat_new_ = 0; ///< Survivors among the concat'ed candidates.
187 fe::Vector<size_t> new2old_;
188};
