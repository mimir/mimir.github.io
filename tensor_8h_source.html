70/// Recognizes the (rebuilt) `tensor_copy` combiner `(acc, ys) ↦ ys#0`: the result is exactly the

94/// If `value` is a pure re-indexed read — a copy-combiner map_reduce without reduction loops that
95/// writes its full loop domain through the identity output map (reshape/transpose/slice/flip/repeat
96/// lower to these) — returns its single access map and source.
97/// `fuse_tensor`'s read-through absorbs exactly these into the consuming op's access maps.
98inline std::optional<PureRead> is_pure_read(const Def* value) {
99 auto mr = Axm::isa<tensor::map_reduce_post>(value);
100 if (!mr) return {};
101 auto [nis_nps, meta, shapes, in_tys, comb_init, map_out, maps_all, is_all] = mr->uncurry_args<8>();
102
103 auto [nis, nps] = nis_nps->projs<2>();
104 // No reduction loops: the total loop count Rn equals the output rank Ro.
105 if (Lit::isa(nis) != 1 || Lit::isa(nps) != 0 || meta->proj(5, 2) != meta->proj(5, 3)) return {};
106 auto [So, Sr, sched] = shapes->projs<3>();
107 if (Sr != So) return {};
108 if (!is_identity_map(map_out)) return {};
109 auto [comb, init, post] = comb_init->projs<3>();
110 if (!is_copy_comb(comb) || !is_identity_post(post)) return {};
111
112 auto sole = [](const Def* d, u64 n, u64 i) { return d->proj(n, i)->proj(1, 0); };
113 return PureRead{sole(is_all, 2, 0), sole(maps_all, 2, 0), sole(in_tys, 6, 0), sole(in_tys, 6, 1),
114 sole(in_tys, 6, 2)};
115}

151/// Counts the consumers of every def of @p world matched by @p pred.
152/// Tuples and packs are transparent argument wrappers, so a wrapped def is charged to the enclosing
153/// non-tuple consumer - a shared argument tuple charges each of its users, and a def used twice in one
154/// argument list counts twice.
155/// A phase whose world does not track uses needs this up front.
156DefMap<u64> count_consumers(const World& world, std::predicate<const Def*> auto pred) {
157 auto counts = DefMap<u64>();
158 auto charge = [&](this auto&& charge, const Def* d) -> void {
159 if (pred(d))
160 ++counts[d];
161 else if (d->isa<Tuple>() || d->isa<Pack>())
162 for (auto op : d->ops())
163 if (op) charge(op);
164 };
165
166 auto wl = fe::BFSWorklist<DefSet>();
167 wl.push(world.externals().muts());
168
169 while (!wl.empty()) {
170 auto def = wl.pop();
171 auto transparent = def->isa<Tuple>() || def->isa<Pack>();
172 for (auto op : def->ops())
173 if (op) {
174 if (!transparent) charge(op);
175 wl.push(op);
176 }
177 if (def->type()) wl.push(def->type());
178 }
179
180 return counts;
181}
