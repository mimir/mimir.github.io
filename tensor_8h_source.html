12/// Recognizes the (rebuilt) `tensor_copy` combiner `(acc, ys) ↦ ys#0`: the result is exactly the
13/// single input element, so a map_reduce built on it is a pure re-indexed read of that input.
14inline bool is_copy_comb(const Def* comb) {
15 auto lam = comb->isa_mut<Lam>();
16 if (!lam || !lam->is_set()) return false;
17 auto app = lam->body()->isa<App>();
18 return app && app->callee() == lam->var(1) && app->arg() == lam->var(0)->proj(2, 1)->proj(1, 0);
19}

21/// Is `post` the (rebuilt) CPS identity `%tensor.id`, i.e. a lam `(x, extras) ↦ x` that returns its
22/// first argument (and hence has no epilogue inputs)?
23inline bool is_identity_post(const Def* post) {
24 auto lam = post->isa_mut<Lam>();
25 if (!lam || !lam->is_set()) return false;
26 auto app = lam->body()->isa<App>();
27 return app && app->callee() == lam->var(1) && app->arg() == lam->var(0)->proj(2, 0);
28}

40/// If `value` is a pure re-indexed read — a copy-combiner map_reduce without reduction loops that
41/// writes its full loop domain through the identity output map (reshape/transpose/slice/flip/repeat
42/// lower to these) — returns its single access map and source.
43/// `fuse_tensor`'s read-through absorbs exactly these into the consuming op's access maps.
44inline std::optional<PureRead> is_pure_read(const Def* value) {
45 auto mr = Axm::isa<tensor::map_reduce_post>(value);
46 if (!mr) return {};
47 auto [nis_nps, meta, shapes, in_tys, comb_init, map_out, maps_all, is_all] = mr->uncurry_args<8>();
48 auto [nis_l, nps_l] = nis_nps->projs<2>([](const Def* d) { return Lit::isa<u64>(d); });
49 // No reduction loops: the total loop count Rn equals the output rank Ro.
50 if (!nis_l || *nis_l != 1 || !nps_l || *nps_l != 0 || meta->proj(5, 2) != meta->proj(5, 3)) return {};
51 auto [So, Sr, sched] = shapes->projs<3>();
52 if (Sr != So) return {};
53 auto id_lam = map_out->isa_mut<Lam>();
54 if (!id_lam || !id_lam->is_set() || id_lam->body() != id_lam->var()) return {};
55 auto [comb, init, post] = comb_init->projs<3>();
56 if (!is_copy_comb(comb) || !is_identity_post(post)) return {};
57
58 return PureRead{is_all->proj(2, 0)->proj(1, 0), maps_all->proj(2, 0)->proj(1, 0), in_tys->proj(6, 0)->proj(1, 0),
59 in_tys->proj(6, 1)->proj(1, 0), in_tys->proj(6, 2)->proj(1, 0)};
60}
