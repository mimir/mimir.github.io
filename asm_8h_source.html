17/// The `emit` Phase of a native backend: writes GNU assembler code for each external function and each function it
18/// calls. The output goes to `<world>.s` (or `a.s` if the World is unnamed), or to `<file>` of `-X <plugin>:o=<file>`.
19class AsmEmit : public Phase {
20public:
21 AsmEmit(World& world, flags_t annex)
22 : Phase(world, annex) {}
23
24 void start() final {
25 auto path = std::string(arg_value(args(), "o", "output").value_or(""));
26 if (path.empty()) path = (world().name() ? world().name().str() : std::string("a")) + ".s";
27 auto out = Out(path);
28 auto& os = *out.os();
29
30 auto muts = world().externals().mutate();
31 std::ranges::sort(muts, {}, [](Def* mut) { return mut->sym().str(); });
32
33 std::println(os, "\t.text");
34 for (auto mut : muts) {
35 auto lam = mut->isa<Lam>();
36 if (!lam) mut->blame("{} backend: cannot emit `{}`", Target::of(driver()).name(), mut).bail();
37 if (lam->is_set() && seen_.emplace(lam).second) emit(os, lam, true);
38 }
39 while (!todo_.empty()) {
40 auto lam = todo_.front();
41 todo_.pop_front();
42 emit(os, lam, false);
43 }
44 finish(os);
45 if (Target::is_elf()) std::println(os, "\t.section .note.GNU-stack,\"\",@progbits");
46 }
47
48 /// The assembler symbol of @p lam; queues an internal function for emission.
49 std::string symbol(Lam* lam) {
50 if (lam->is_external() || !lam->is_set()) return std::format("{}{}", Target::global_prefix(), lam->sym());
51 if (seen_.emplace(lam).second) todo_.emplace_back(lam);
52 return lam->unique_name();
53 }
54
55protected:
56 virtual void emit(std::ostream& os, Lam* fun, bool global) = 0;
57 /// Writes what all functions together need after them, e.g. a constant pool.
58 virtual void finish(std::ostream&) {}

63};
