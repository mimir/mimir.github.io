44const Def* Join2Variant::rewrite_imm_Match(const Match* match) {
45 auto scrutinee = match->scrutinee();
46 auto join = scrutinee->unfold_type()->isa<Join>();

49
50 auto& w = new_world();

53 for (size_t i = 0, e = cases.size(); i != e; ++i) {
54 auto c = cases[i];
55 auto arm = join ? *std::ranges::find_if(match->arms(), [c](const Def* arm) { return Match::accepts(arm, c); })
56 : match->arm(i);
57 auto pi = arm->isa_type<Pi>();

59 ops.emplace_back(rewrite(arm));
60 } else {

62 auto res = w.app(rewrite(arm), lift(pi->dom(), c, lam->var()));
