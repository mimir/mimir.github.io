50/// Does core.icmp @p id give the same result on the canonical values of any `Idx s`?
51inline bool is_unsigned(icmp id) {
52 switch (id) {
53 case icmp::f:
54 case icmp::e:
55 case icmp::ug:
56 case icmp::uge:
57 case icmp::ul:
58 case icmp::ule:
59 case icmp::ne:
60 case icmp::t: return true;
61 default: return false;
62 }
63}

65/// The isa_odd_idx size whose modulus the core operation @p def depends on.
66/// Neither core.wrap without unsigned wrap-around nor core.idx with a mode depends on it.
67/// core.conv.u into such an `Idx` does, but core.legalize_idx emits that itself for values in range.
68inline std::optional<nat_t> isa_odd_idx_op(const Def* def) {
69 auto mode = [](const App* op) { return Lit::isa(op->decurry()->decurry()->arg()); };
70 if (auto op = Axm::isa<wrap>(def)) {
71 if (auto m = mode(op); m && (*m & nat_t(Mode::nuw))) return {};
72 return isa_odd_idx(def->type());
73 }
74 if (auto op = Axm::isa<idx>(def)) return mode(op) == 0 ? isa_odd_idx(def->type()) : std::nullopt;
75 if (Axm::isa<bitcast>(def)) return isa_odd_idx(def->type());
76 if (auto op = Axm::isa<conv>(def); op && op.id() == conv::s)
77 return isa_odd_idx(def->type()).or_else([&] { return isa_odd_idx(op->arg()->type()); });
78 if (auto op = Axm::isa<icmp>(def); op && !is_unsigned(op.id())) return isa_odd_idx(op->arg(2, 0)->type());
79 if (auto op = Axm::isa<shr>(def); op && op.id() == shr::a) return isa_odd_idx(def->type());
80 if (auto op = Axm::isa<extrema>(def); op && (op.id() == extrema::smin || op.id() == extrema::smax))
81 return isa_odd_idx(def->type());
82 if (auto op = Axm::isa<abs>(def)) return isa_odd_idx(op->arg(2, 1)->type());
83 if (auto op = Axm::isa<div>(def); op && (op.id() == div::sdiv || op.id() == div::srem))
84 return isa_odd_idx(op->arg(2, 1)->proj(2, 0)->type());
85 return {};
86}

139/// Returns the (first) element of type `core.M a` from the given tuple.
140inline const Def* mem_def(const Def* def) {
141 if (isa_mem(def)) return def;
142 if (def->isa<Arr>() || def->type()->isa<Arr>()) return {}; // don't look into possibly gigantic arrays

144 if (def->num_projs() > 1) {

163/// Removes recusively all occurences of mem from a type (sigma).
164inline const Def* strip_mem_ty(const Def* def) {
165 auto& world = def->world();
166
167 if (auto sigma = def->isa<Sigma>()) {
168 DefVec new_ops;
169 for (auto op : sigma->ops())
170 if (auto new_op = strip_mem_ty(op); new_op != world.sigma()) new_ops.push_back(new_op);
171
172 return world.sigma(new_ops);
173 } else if (Axm::isa<M>(def)) {
174 return world.sigma();
175 }
176
177 return def;
178}
