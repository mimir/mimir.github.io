11const Def* LowerMatrixHighLevelMapRed::rewrite_imm_App(const App* app) {
12 if (is_bootstrapping()) return RWPhase::rewrite_imm_App(app);
13 auto& w = new_world();
14
15 // Each high-level op is implemented by a `map_reduce_*` unfolding function of matching type.
16 // We re-apply its (fully-inferred) implicit meta groups explicitly with `w.app` — `w.call` would insert fresh holes
17 // for the implicit domains and misapply the groups — to strip the meta down to the bare CPS function, then hand it
18 // to `cps2ds` together with the operands.
19 if (auto prod = Axm::isa<matrix::prod>(app)) {
20 auto pe = rewrite(prod->decurry()->decurry()->arg()); // {pe}
21 auto dims = rewrite(prod->decurry()->arg()); // {m, k, l}
22 auto spec = w.app(w.app(w.annex<map_reduce_prod>(), pe), dims);
23 return w.call<cps::cps2ds>(spec, rewrite(prod->arg()));
24 } else if (auto sum = Axm::isa<matrix::sum>(app)) {
25 auto pe = rewrite(sum->decurry()->decurry()->arg()); // {pe}
26 auto dims = rewrite(sum->decurry()->arg()); // {n, S}
27 auto spec = w.app(w.app(w.annex<map_reduce_sum>(), pe), dims);
28 return w.call<cps::cps2ds>(spec, rewrite(sum->arg()));
29 } else if (auto transpose = Axm::isa<matrix::transpose>(app)) {
30 auto dims = rewrite(transpose->decurry()->decurry()->arg()); // {k, l}
31 auto T = rewrite(transpose->decurry()->arg()); // {T}
32 auto spec = w.app(w.app(w.annex<map_reduce_transpose>(), dims), T);
33 return w.call<cps::cps2ds>(spec, rewrite(transpose->arg()));
