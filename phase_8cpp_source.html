75void Analysis::start() {
76 curr_sparse_ = !dense_ && !nonlocal_ && !dirty_.empty();
77 nonlocal_ = false;
78 auto seeds = fe::Vector<Def*>(dirty_.begin(), dirty_.end());
79 dirty_.clear();
80
81 prepare();
82
83 if (curr_sparse_) {
84 log().v("sparse round: re-drain {} dirty muts", seeds.size());
85 std::ranges::sort(seeds, GIDLt<Def*>()); // MutSet iteration order is nondeterministic
86
87 // Pre-install what earlier rounds substituted, so this round prunes everything they already settled -
88 // except for mutables, whose map entry doubles as the per-round "already scheduled" marker and would
89 // suppress their drain. Pruning skips rewrite hooks, which is why only a sparse round does it: the
90 // full round that certifies the fixed point always re-derives from the program.
91 old2new().reserve(lattice_.size());
92 for (auto [concr, abstr] : lattice_)
93 if (!concr->isa_mut()) map(concr, abstr);
94
95 for (auto mut : seeds)
96 rewrite(mut);
97 drain();
98 } else {
99 for (auto mut : world().externals().muts())
100 rewrite_external(mut);
101 drain();
102
103 finalize();
104 }
105
106 profile_count(curr_sparse_ ? "rounds.sparse" : "rounds.full");
107 profile_count("muts.drained", std::exchange(num_drained_, size_t(0)));
108
109 // A quiet sparse round only certifies the muts it visited:
110 // force one full round; the fixed point counts only if that one stays quiet, too.
111 if (curr_sparse_ && !todo()) invalidate();
112}

180void RWPhase::start() {
181 auto max_iters = driver().flags().max_fp_iters;
182 bool todo = true;
183 for (uint32_t i = 0; todo; ++i) {
184 if (i >= max_iters) fe::throwf("phase `{}` did not reach a fixed point after {} iterations", name(), max_iters);
185 log().v("iteration {}", i);
186 todo = analyze();
187 }
188
189 auto gid = new_world().curr_gid();
190 for (auto mut : old_world().externals().muts())
191 rewrite_external(mut);
192 finalize();
193 profile_count("rw.defs", new_world().curr_gid() - gid);
194
195 swap(old_world(), new_world());

236const Def* RWPhase::rewrite_imm_Tuple(const Tuple* tuple) {
237 // Only an annotated type constrains the ops; an inferred one follows them.
238 if (has_inferred_type(tuple)) return new_world().tuple(rewrite(tuple->ops()));

240 auto type = rewrite(tuple->type());
241 auto ops = rewrite(tuple->ops());

244 for (size_t i = 0; i != n; ++i)

249const Def* RWPhase::rewrite_stub(Def* old_mut, Def* new_mut) {
250 auto old_lam = old_mut->isa<Lam>();
251 auto new_lam = new_mut->isa<Lam>();

263const Def* RWPhase::coerce(const Def* type, const Def* val) {
264 if (val->type() == type) return val;
265 auto todo = DefVec{val};
266 auto res = reshape(todo, type);
267 while (!todo.empty() && new_world().inhabitant(todo.back()->type()))
268 todo.pop_back();
269 if (!todo.empty()) fe::throwf("cannot coerce `{}` to `{}`", val, type);
270 return res;
271}

273const Def* RWPhase::reshape(DefVec& todo, const Def* type) {
274 auto& w = new_world();
275 auto front = todo.empty() ? nullptr : todo.back();
276 if (front) {
277 if (auto res = Check::coerce(type, front)) return todo.pop_back(), res;
278 if (auto res = eta_coerce(type, front)) return todo.pop_back(), res;
279 }
280
281 if (auto s = w.inhabitant(type)) return s;
282 if (front && w.inhabitant(front->type())) return todo.pop_back(), reshape(todo, type);
283
284 if (splittable(type)) {
285 auto n = type->num_tprojs();
286 auto parts = DefVec();
287 for (size_t i = 0; i != n; ++i)
288 parts.emplace_back(reshape(todo, w.component(type, parts, n, i)));
289 return w.tuple(type, parts);
290 }
291
292 if (front && front->type()->num_tprojs() != 1) {
293 todo.pop_back();
294 auto pieces = front->tprojs();
295 todo.insert(todo.end(), pieces.rbegin(), pieces.rend());
296 return reshape(todo, type);
297 }
298
299 fe::throwf("cannot coerce to `{}`", type);
300}

302const Def* RWPhase::eta_coerce(const Def* type, const Def* val) {
303 auto dst = type->isa<Pi>();
304 auto src = val->type()->isa<Pi>();
305 if (!dst || !src) return nullptr;
306 auto& memo = etas_[type][val];
307 if (memo) return memo;
308
309 auto eta = new_world().mut_lam(dst)->set(val->dbg_key());
310 memo = eta; // before the recursion below may rehash etas_
311 auto arg = coerce(src->dom(), eta->var());
312 auto codom = dst->reduce(eta->var());
313 log().d("coerce {}: {} → {}", val, src, dst);
314 return eta->set(false, coerce(codom, new_world().app(val, arg)));
315}

345void PhaseMan::start() {
346 auto max_iters = driver().flags().max_fp_iters;
347 auto n = phases().size();
348 // A phase's run is a deterministic function of the World's content.
349 // So a phase only needs to run (again) if the World (may have) changed since its last quiet run.
350 auto all = fe::Bitset(n, true);
351 auto stale = all;
352 auto ran = fe::Bitset(n, false);
353
354 for (uint32_t iter = 0; stale.any(); ++iter) {
355 if (iter >= max_iters)
356 fe::throwf("phase `{}` did not reach a fixed point after {} iterations", name(), max_iters);
357 if (fixed_point()) log().v("🔄 fixed-point iteration {}", iter);
358
359 bool todo = false;
360 for (size_t i = 0; i != n; ++i) {
361 auto& phase = phases()[i];
362 if (!stale.test(i)) {
363 log().v("skip `{}`: World unchanged since its last quiet run", phase->name());
364 profile_count("phases.skipped");
365 continue;
366 }
367
368 if (ran.test(i)) phase = phase->recreate(); // re-runs need a fresh instance
369
370 phase->run();
371 ran.set(i);
372 stale.clear(i);
373
374 if (phase->todo()) {
375 todo = true;
376 // The World changed: everyone - including this phase itself - gets another look.
377 stale = all;
378 }
379 }
380
381 todo &= fixed_point();
382 invalidate(todo);
383 if (!fixed_point()) break;
384 }
385}
