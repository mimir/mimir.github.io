59void Analysis::start() {
60 curr_sparse_ = !dense_ && !nonlocal_ && !dirty_.empty();
61 nonlocal_ = false;
62 auto seeds = Vector<Def*>(dirty_.begin(), dirty_.end());
63 dirty_.clear();
64
65 prepare();
66
67 if (curr_sparse_) {
68 bootstrapping_ = false; // a sparse round re-drains program muts - it has no annex half
69 VLOG("sparse round: re-draining {} dirty muts", seeds.size());
70 std::ranges::sort(seeds, GIDLt<Def*>()); // MutSet iteration order is nondeterministic
71
72 // Pre-install what earlier rounds substituted, so this round prunes everything they already settled -
73 // except for mutables, whose map entry doubles as the per-round "already scheduled" marker and would
74 // suppress their drain. Pruning skips rewrite hooks, which is why only a sparse round does it: the
75 // full round that certifies the fixed point always re-derives from the program.
76 for (auto [concr, abstr] : lattice_)
77 if (!concr->isa_mut()) map(concr, abstr);
78
79 for (auto mut : seeds)
80 rewrite(mut);
81 drain();
82 } else {
83 for (const auto& [flags, e] : world().annexes())
84 rewrite_annex(flags, e.sym, e.def);
85 drain();
86
87 bootstrapping_ = false;
88
89 for (auto mut : world().externals().muts())
90 rewrite_external(mut);
91 drain();
92
93 finalize();
94 }
95
96 profile_count(curr_sparse_ ? "rounds.sparse" : "rounds.full");
97 profile_count("muts.drained", std::exchange(num_drained_, size_t(0)));
98
99 // A quiet sparse round only certifies the muts it visited:
100 // force one full round; the fixed point counts only if that one stays quiet, too.
101 if (curr_sparse_ && !todo()) invalidate();
102}

107const Def* Analysis::repr_(const Def* slow, const Def* fast) const {
108 while (slow != fast) {
109 auto next = follow(fast);
110 if (!next) return fast;
111 fast = follow(next);
112 if (!fast) return next;
113 slow = follow(slow);
114 assert(slow && "slow lags fast, so fast already traversed slow's successor");
115 }
116
117 auto res = slow;
118 for (auto def = follow(slow); def != slow; def = follow(def))
119 if (def->gid() < res->gid()) res = def;
120 return res;
121}

152void RWBase::start() {
153 auto max_iters = driver().flags().max_fp_iters;
154 bool todo = true;
155 for (uint32_t i = 0; todo; ++i) {
156 if (i >= max_iters) fe::throwf("phase `{}` did not reach a fixed point after {} iterations", name(), max_iters);
157 VLOG("iteration: {}", i);
158 todo = analyze();
159 }
160
161 // Count the Def%s each half of the walk creates.
162 // For an RWPhase the annex half is a fixed tax proportional to the loaded plugins' annex graph - not to the
163 // program - which is exactly why an InplaceRWPhase skips it by default.
164 auto gid = Rewriter::world().curr_gid();
165 if (rewrite_annexes())
166 for (const auto& [flags, e] : Phase::world().annexes())
167 rewrite_annex(flags, e.sym, e.def);
168 profile_count("rw.defs.annex", Rewriter::world().curr_gid() - gid);
169
170 bootstrapping_ = false;
171
172 gid = Rewriter::world().curr_gid();
173 // mutate(): an in-place rewrite_external may re-externalize, which would invalidate a live iterator.
174 for (auto mut : Phase::world().externals().mutate())
175 rewrite_external(mut);
176 finalize(); // inside the span: work deferred by the root walk belongs to the root walk
177 profile_count("rw.defs.external", Rewriter::world().curr_gid() - gid);
178}

229const Def* InplaceRWPhase::rewrite_mut(Def* mut) {
230 if (auto hole = mut->isa<Hole>()) {
231 auto [last, op] = hole->find();
232 return op ? rewrite(op) : last; // an unresolved Hole stays as is
233 }
234
235 // A mutable's identity is tied to its type, so if the rewrite changes the type, we cannot keep it: fall back to
236 // an RWPhase-style rebuild - Rewriter::rewrite_mut stubs a fresh mutable (in this very World) and maps onto it.
237 if (auto type = mut->type(); type && rewrite(type) != type) {
238 profile_count("inplace.muts.rebuilt");
239 invalidate();
240 return Rewriter::rewrite_mut(mut);
241 }
242
243 map(mut, mut); // keep the identity; doubles as the cycle breaker for recursive mutables
244 if (!mut->is_set()) return mut;
245
246 auto _ = enter(mut);
247 auto new_ops = rewrite(mut->ops());
248 if (!std::ranges::equal(new_ops, mut->ops())) {
249 mut->unset()->set(new_ops);
250 profile_count("inplace.muts.reset");
251 invalidate();
252 }
253
254 return mut;
255}

285void PhaseMan::start() {
286 auto max_iters = driver().flags().max_fp_iters;
287 auto n = phases().size();
288 // A phase's run is a deterministic function of the World's content.
289 // So a phase only needs to run (again) if the World (may have) changed since its last quiet run.
290 auto stale = absl::FixedArray<bool>(n, true);
291 auto ran = absl::FixedArray<bool>(n, false);
292
293 auto any_stale = [&stale]() { return std::ranges::any_of(stale, [](bool b) { return b; }); };
294
295 for (uint32_t iter = 0; any_stale(); ++iter) {
296 if (iter >= max_iters)
297 fe::throwf("phase `{}` did not reach a fixed point after {} iterations", name(), max_iters);
298 if (fixed_point()) VLOG("🔄 fixed-point iteration: {}", iter);
299
300 bool todo = false;
301 for (size_t i = 0; i != n; ++i) {
302 auto& phase = phases()[i];
303 if (!stale[i]) {
304 VLOG("skipping `{}`: World unchanged since its last quiet run", phase->name());
305 profile_count("phases.skipped");
306 continue;
307 }
308
309 if (ran[i]) { // re-runs need a fresh instance
310 auto new_phase = std::unique_ptr<Phase>(static_cast<Phase*>(phase->recreate().release()));
311 swap(new_phase, phase);
312 }
313
314 phase->run();
315 ran[i] = true;
316 stale[i] = false;
317
318 if (phase->todo()) {
319 todo = true;
320 // The World changed: everyone - including this phase itself - gets another look.
321 std::ranges::fill(stale, true);
322 }
323 }
324
325 todo &= fixed_point();
326 invalidate(todo);
327 if (!fixed_point()) break;
328 }
