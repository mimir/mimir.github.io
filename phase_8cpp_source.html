164void PhaseMan::start() {
165 auto max_iters = driver().flags().max_fp_iters;
166 auto n = phases().size();
167 // A phase's run is a deterministic function of the World's content.
168 // So a phase only needs to run (again) if the World (may have) changed since its last quiet run.
169 auto stale = absl::FixedArray<bool>(n, true);
170 auto ran = absl::FixedArray<bool>(n, false);
171
172 auto any_stale = [&stale]() { return std::ranges::any_of(stale, [](bool b) { return b; }); };
173
174 for (uint32_t iter = 0; any_stale(); ++iter) {
175 if (iter >= max_iters) error("phase `{}` did not reach a fixed point after {} iterations", name(), max_iters);
176 if (fixed_point()) VLOG("🔄 fixed-point iteration: {}", iter);
177
178 bool todo = false;
179 for (size_t i = 0; i != n; ++i) {
180 auto& phase = phases()[i];
181 if (!stale[i]) {
182 VLOG("skipping `{}`: World unchanged since its last quiet run", phase->name());
183 profile_count("phases.skipped");
184 continue;
185 }
186
187 if (ran[i]) { // re-runs need a fresh instance
188 auto new_phase = std::unique_ptr<Phase>(static_cast<Phase*>(phase->recreate().release()));
189 swap(new_phase, phase);
190 }
191
192 phase->run();
193 ran[i] = true;
194 stale[i] = false;
195
196 if (phase->todo()) {
197 todo = true;
198 // The World changed: everyone - including this phase itself - gets another look.
199 std::ranges::fill(stale, true);
200 }
201 }
202
203 todo &= fixed_point();
204 invalidate(todo);
205 if (!fixed_point()) break;
206 }
207}
