18 /// The line Loc::begin points at, or empty if the file or that row is unavailable.
19 std::string_view line(Loc loc) {
20 if (!loc || !loc.path) return {};
21
22 auto [i, ins] = path2lines_.try_emplace(loc.path);
23 if (ins) {
24 auto ifs = std::ifstream(*loc.path);
25 for (std::string line; std::getline(ifs, line);) {

29 }

33 }

42/// Streams @p line, then a caret run underlining the columns @p loc spans, both behind a Gutter-wide row number.
43/// A @p loc spanning several rows underlines the remainder of its first one.
44void stream_snippet(std::ostream& os, Loc loc, std::string_view line, fe::term::FG color) {
45 // A col of 0 is Pos's "unknown" sentinel, so there is nothing to point at.
46 if (line.empty() || loc.begin.col == 0) return;
47
48 auto len = fe::utf8::num_code_points(line);
49 auto begin = size_t(loc.begin.col) - 1;
50 if (begin >= len) return;
51
52 auto finis = loc.finis.row != loc.begin.row || loc.finis.col == 0 ? len : size_t(loc.finis.col);
53 finis = std::min(std::max(finis, begin + 1), len);
54
55 os << fe::term::FG::Gray << std::format("{:>{}} | ", loc.begin.row, Gutter) << fe::term::FG::Reset << line << '\n'
56 << fe::term::FG::Gray << std::format("{:>{}} | ", "", Gutter) << color;
57
58 // One blank per *code point*, since that is what Pos::col counts;
59 // a tab is echoed rather than blanked, or the carets drift by its width.
60 for (size_t i = 0, c = 0; c != begin && i < line.size(); ++c) {
61 auto n = fe::utf8::num_bytes(char8_t(line[i]));
62 os << (line[i] == '\t' ? '\t' : ' ');
63 i += n == 0 ? 1 : n;
64 }
65 for (size_t i = begin; i != finis; ++i)
66 os << '^';
67
68 os << fe::term::FG::Reset << '\n';
69}
