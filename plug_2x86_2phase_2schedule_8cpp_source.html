25void Schedule::place(Lam* curr, const Def* def) {
26 if (def->isa_mut()) return;
27
28 auto insn = is_insn(def);
29 auto remat = is_remat(def);
30 if (!remat && !(insn ? placed_ : visited_[curr]).emplace(def).second) return;
31 if (is_mem(def)) mems_[scheduler_->early(def)->mut()->as<Lam>()].emplace_back(def);
32 if (auto node = insn && !remat ? scheduler_->smart(curr, def) : nullptr) curr = node->mut()->as<Lam>();
33
34 auto ops = DefVec(def->ops().begin(), def->ops().end());
35 // A terminator branches on its condition last.
36 if (auto app = def->isa<App>(); app && def == curr->body()) ops = {app->arg(), app->callee()};
37 for (auto op : ops)
38 if (!is_remat(op)) place(curr, op);
39 // Last, so that nothing separates a rematerialized operand from its user.
40 for (auto op : ops)
41 if (is_remat(op)) place(curr, op);
42 if (insn) insns_[curr].emplace_back(def);
43}

72void Schedule::thread(Lam* lam) {
73 auto n = lam->num_vars();
74 auto i = n;
75 for (size_t j = 0; j != n && i == n; ++j)
76 if (is_mem(lam->var(n, j))) i = j;
77
78 // Direct style has no chain, so an instruction in there runs into rewrite_imm_App.
79 if (i == n && Lam::isa_cn(lam))
80 lam->blame(MIM_X86_BE "continuation `{}` has no `mem.M` to schedule on", lam).bail();
81
82 push(Vars(lam->has_var()), Vars(lam->has_var()));
83 if (i != n) chain(lam, i);
84 auto filter = rewrite(lam->filter());
85 auto body = rewrite(lam->body());
86 pop();
87 lam->unset()->set(filter, body);
88}

90void Schedule::chain(Lam* lam, size_t i) {
91 auto& w = world();
92 auto n = lam->num_vars();
93 auto m = lam->var(n, i);
94 auto order = std::move(insns_[lam]);
95 auto mems = std::move(mems_[lam]);
96
97 // Instructions hoisted into lam come after the condition of its branch, so the branch gets a fresh one at the end.
98 if (auto ex = lam->body()->as<App>()->callee()->isa<Extract>(); ex && is_remat(ex->index())) {
99 auto cc = ex->index()->as<App>();
100 if (is_remat(cc->arg())) order.emplace_back(cc->arg());
101 order.emplace_back(cc);
102 }
103
104 for (auto old_def : order) {
105 auto app = old_def->as<App>();
106 if (Axm::isa<sched>(app)) {
107 auto res = w.call<sched>(rewrite(sched_insn(app)), m, rewrite(app->arg()));
108 m = res->proj(2, 0);
109 map(old_def, res);
110 } else if (prev_mem(app)) {
111 auto old_arg = app->arg();
112 auto k = old_arg->num_projs();
113 auto args = DefVec(k, [&](size_t j) {
114 auto a = old_arg->proj(k, j);
115 return is_mem(a) ? m : rewrite(a);
116 });
117 auto new_app = w.app(rewrite(app->callee()), w.tuple(args));
118 if (is_mem(new_app)) {
119 m = new_app;
120 } else {
121 m = new_app->proj(2, 0);
122 map(old_def, new_app);
123 }
124 } else {
125 auto res = w.call<sched>(rewrite(app->callee()), m, rewrite(app->arg()));
126 m = res->proj(2, 0);
127 map(old_def, res->proj(2, 1));
128 }
129 }
130
131 for (auto old_mem : mems)
132 map(old_mem, m);
133 // The whole Var may be passed on, so it must carry the end of the chain as well.
134 if (n > 1) map(lam->var(), w.tuple(DefVec(n, [&](size_t j) { return j == i ? m : lam->var(n, j); })));
135}
