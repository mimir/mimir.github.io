24/// Bound on the magnitude of a proven range: below it, sums and products of proven ranges cannot wrap an i64.
25constexpr int64_t Limit = int64_t{1} << 61;

27std::optional<int64_t> const_val(const Def* d) {
28 if (auto bc = Axm::isa<core::bitcast>(d)) d = bc->arg();
29 if (auto lit = Lit::isa(d); lit && *lit <= u64(Limit)) return int64_t(*lit);
30 return {};

33/// Splits x * k into (x, k).
34std::optional<std::pair<const Def*, int64_t>> as_scaled(const Def* d) {
35 auto wrap = Axm::isa<core::wrap>(d);
36 if (!wrap || wrap.id() != core::wrap::mul) return {};
37 auto [a, b] = wrap->arg()->projs<2>();
38 if (auto k = const_val(b)) return std::pair{a, *k};
39 if (auto k = const_val(a)) return std::pair{b, *k};
40 return {};
41}

43/// True when @p d is a multiple of @p c for every value of its free indices.
44bool divides(const Def* d, int64_t c) {
45 if (c == 0) return false;
46 if (auto k = const_val(d)) return *k % c == 0;
47 if (auto s = as_scaled(d)) return s->second % c == 0 || divides(s->first, c);
48 auto wrap = Axm::isa<core::wrap>(d);
49 if (!wrap || (wrap.id() != core::wrap::add && wrap.id() != core::wrap::sub)) return false;
50 auto [a, b] = wrap->arg()->projs<2>();
51 return divides(a, c) && divides(b, c);
52}

54const Def* times(World& w, const Def* d, int64_t k) {
55 return k == 1 ? d : w.call(core::wrap::mul, core::Mode::none, Defs{d, w.lit_i64(u64(k))});

58/// A linearized index scales whole sums by a stride, so the folds must not depend on how the sum was nested.
59void flatten_add(World& w, const Def* d, int64_t k, DefVec& terms) {
60 if (auto wrap = Axm::isa<core::wrap>(d); wrap && wrap.id() == core::wrap::add) {
61 auto [a, b] = wrap->arg()->projs<2>();
62 flatten_add(w, a, k, terms);
63 flatten_add(w, b, k, terms);
64 } else if (auto s = as_scaled(d); s && s->second != 0 && std::abs(k) <= Limit / std::abs(s->second)) {
65 flatten_add(w, s->first, k * s->second, terms);
66 } else {
67 terms.emplace_back(times(w, d, k));
68 }
69}

71/// @p d divided by @p c, for a @p d that divides established is an exact multiple.
72const Def* exact_div(World& w, const Def* d, int64_t c) {
73 if (c == 1) return d;
74 if (auto k = const_val(d)) return w.lit_i64(u64(*k / c));
75 if (auto s = as_scaled(d)) {
76 auto [x, k] = *s;
77 return k % c == 0 ? times(w, x, k / c) : times(w, exact_div(w, x, c), k);
78 }
79 auto wrap = Axm::isa<core::wrap>(d); // add or sub, both of whose sides divide
80 auto [a, b] = wrap->arg()->projs<2>();
81 return w.call(wrap.id(), core::Mode::none, Defs{exact_div(w, a, c), exact_div(w, b, c)});
82}

84const Def* sum(World& w, Defs terms) {
85 const Def* acc = nullptr;
86 for (auto t : terms)
87 acc = acc ? w.call(core::wrap::add, core::Mode::none, Defs{acc, t}) : t;
88 return acc ? acc : w.lit_i64(0);
89}

93std::optional<LowerIndex::Range> LowerIndex::range_of(const Def* d) const {
94 auto bounded = [](int64_t lo, int64_t hi) -> std::optional<Range> {
95 if (lo < -Limit || hi > Limit) return {};
96 return Range{lo, hi};
97 };
98
99 if (auto k = const_val(d)) return Range{*k, *k};
100
101 if (auto ex = d->isa<Extract>()) {
102 auto i = Lit::isa(ex->index());
103 if (ex->tuple() == dom_.idxs && i && *i < dom_.extents.size())
104 if (auto n = dom_.extents[*i]; n && *n > 0) return Range{0, *n - 1};
105
106 if (auto div = Axm::isa<core::div>(ex->tuple()); div && i == 1) {
107 // `core.div` takes `(mem, (x, c))`.
108 auto [x, c] = div->arg()->proj(2, 1)->projs<2>();
109 auto k = const_val(c);
110 if (!k || *k <= 0) return {};
111 // `urem` is bounded by its divisor whatever its operand; `udiv` only by a non-negative operand's range.
112 if (div.id() == core::div::urem) return Range{0, *k - 1};
113 if (auto l = range_of(x); l && l->lo >= 0 && div.id() == core::div::udiv)
114 return Range{l->lo / *k, l->hi / *k};
115 }
116 }
117
118 if (auto wrap = Axm::isa<core::wrap>(d)) {
119 auto [a, b] = wrap->arg()->projs<2>();
120 auto l = range_of(a), r = range_of(b);
121 if (!l || !r) return {};
122 switch (wrap.id()) {
123 case core::wrap::add: return bounded(l->lo + r->lo, l->hi + r->hi);
124 case core::wrap::sub: return bounded(l->lo - r->hi, l->hi - r->lo);
125 case core::wrap::mul: {
126 std::array cands{std::pair{l->lo, r->lo}, std::pair{l->lo, r->hi}, std::pair{l->hi, r->lo},
127 std::pair{l->hi, r->hi}};
128 for (auto [x, y] : cands)
129 if (x != 0 && std::abs(y) > Limit / std::abs(x)) return {};
130 auto prods = cands | std::views::transform([](auto p) { return p.first * p.second; });
131 return bounded(std::ranges::min(prods), std::ranges::max(prods));
132 }
133 default: break;
134 }
135 }
136
137 return {};
138}

140LowerIndex::Split LowerIndex::split(const Def* x, int64_t c) {
141 DefVec terms;
142 flatten_add(new_world(), x, 1, terms);
143
144 Split split;
145 std::optional<Range> rest = Range{0, 0};
146 for (auto t : terms) {
147 if (divides(t, c)) {
148 split.even.emplace_back(t);
149 } else {
150 split.rest.emplace_back(t);
151 auto r = range_of(t);
152 if (rest && r && std::abs(rest->lo) <= Limit && std::abs(rest->hi) <= Limit)
153 rest = Range{rest->lo + r->lo, rest->hi + r->hi};
154 else
155 rest = std::nullopt;
156 }
157 }
158 split.carries = !rest || rest->lo < 0 || rest->hi >= c;
159 return split;
160}

162/// x floordiv c without a division, or nullptr when the range analysis cannot justify one.
163const Def* LowerIndex::fold_udiv(const Def* x, int64_t c) {
164 auto& w = new_world();
165 if (c <= 0) return nullptr;
166 if (c == 1) return x;
167 if (auto k = const_val(x)) return w.lit_i64(u64(*k / c));
168
169 // The carrier wraps, so a fold needs all of x proven in range; an unknown term keeps the division.
170 auto r = range_of(x);
171 if (!r || r->lo < 0) return nullptr;
172 if (r->hi < c) return w.lit_i64(0);
173
174 auto s = split(x, c);
175 if (s.even.empty() || s.carries) return nullptr;
176 return sum(w, DefVec(s.even.size(), [&](size_t i) { return exact_div(w, s.even[i], c); }));
177}

179/// x mod c without a division, or nullptr when the range analysis cannot justify one.
180const Def* LowerIndex::fold_urem(const Def* x, int64_t c) {
181 auto& w = new_world();
182 if (c <= 0) return nullptr;
183 if (c == 1) return w.lit_i64(0);
184 if (auto k = const_val(x)) return w.lit_i64(u64(*k % c));
185
186 auto r = range_of(x);
187 if (!r || r->lo < 0) return nullptr;
188 if (r->hi < c) return x;
189 if (divides(x, c)) return w.lit_i64(0);
190
191 auto s = split(x, c);
192 return s.even.empty() || s.carries ? nullptr : sum(w, s.rest);
193}

201const Def* LowerIndex::rewrite_imm_App(const App* app) {
202 if (is_bootstrapping()) return RWPhase::rewrite_imm_App(app);
203
204 auto& w = new_world();
205
206 // Emits `core.div.<op> (mem_, (x, c))` on the `Idx 0` carrier, advancing the threaded mem and yielding the value.
207 auto div = [&](core::div op, const Def* x, const Def* c) -> const Def* {
208 auto [m, v] = w.call(op, Defs{mem_, w.tuple({x, c})})->projs<2>();
209 mem_ = m;
210 return v;
211 };
212
213 // The affine index algebra is computed on the wide `Idx 0` carrier with wrap-around (`Mode::none`) arithmetic, so
214 // that negation/subtraction are correct via two's complement; the boundary `affine.map` casts in/out with
215 // `core.conv.u`.
216
217 // affine.lit n ↦ the `Nat` n reinterpreted as an `Idx 0`.
218 if (Axm::isa<affine::lit>(app)) return w.call<core::bitcast>(w.type_i64(), rewrite(app->arg()));
219
220 if (auto op = Axm::isa<affine::op>(app)) {
221 switch (op.id()) {
222 case affine::op::add: {
223 auto [a, b] = rewrite(app->arg())->projs<2>();
224 return w.call(core::wrap::add, core::Mode::none, Defs{a, b});
225 }
226 case affine::op::sub: {
227 auto [a, b] = rewrite(app->arg())->projs<2>();
228 return w.call(core::wrap::sub, core::Mode::none, Defs{a, b});
229 }
230 case affine::op::neg: {
231 auto a = rewrite(app->arg());
232 return w.call(core::wrap::sub, core::Mode::none, Defs{w.lit(w.type_i64(), 0), a});
233 }
234 case affine::op::mul: {
235 fe::throwf("`affine.op.mul` should have been rewritten to `affine.semiop.mul` and then to `core.mul`");
236 }
237 }
238 }
239
240 if (auto semiop = Axm::isa<affine::semiop>(app)) {
241 auto [x, c] = rewrite(app->arg())->projs<2>();
242 switch (semiop.id()) {
243 case affine::semiop::mul: {
244 if (Axm::isa<refly::check>(c))
245 fe::throwf("`affine.semiop.mul` called with non-constant second argument");
246 // `c` is a `Nat` constant; reinterpret it on the `Idx 0` carrier.
247 return w.call(core::wrap::mul, core::Mode::none, Defs{x, w.call<core::bitcast>(w.type_i64(), c)});
248 }
249 case affine::semiop::floordiv: {
250 auto c_idx = w.call<core::bitcast>(w.type_i64(), c);
251 if (auto k = const_val(c_idx))
252 if (auto folded = fold_udiv(x, *k)) return folded;
253 return div(core::div::udiv, x, c_idx);
254 }
255 case affine::semiop::rem: {
256 auto c_idx = w.call<core::bitcast>(w.type_i64(), c);
257 if (auto k = const_val(c_idx))
258 if (auto folded = fold_urem(x, *k)) return folded;
259 return div(core::div::urem, x, c_idx);
260 }
261 case affine::semiop::ceildiv: {
262 auto c_idx = w.call<core::bitcast>(w.type_i64(), c);
263 // ceildiv(x, c) = (x + (c - 1)) / c (unsigned, on the Idx 0 carrier)
264 auto c_1 = w.call(core::wrap::sub, core::Mode::none, Defs{c_idx, w.lit(w.type_i64(), 1)});
265 return div(core::div::udiv, w.call(core::wrap::add, core::Mode::none, Defs{x, c_1}), c_idx);
266 }
267 }
268 }
269
270 // Row-major suffix-product strides of a shape `s` («n; Nat»): `strides#k = ∏_{j>k} s#j` (on `Idx 0`, via
271 // `core.nat`).
272 auto strides = [&](const Def* s, size_t n) {
273 DefVec str(n);
274 if (n) str[n - 1] = w.lit_nat(1);
275 for (size_t k = n - 1; k-- != 0;)
276 str[k] = w.call(core::nat::mul, Defs{str[k + 1], s->proj(n, k + 1)});
277 for (size_t k = 0; k != n; ++k)
278 str[k] = w.call<core::bitcast>(w.type_i64(), str[k]);
279 return str;
280 };
281
282 // affine.linearize (idxs, s) ↦ Σ_k idxs#k · strides#k (on the `Idx 0` carrier).
283 if (Axm::isa<affine::linearize>(app)) {
284 auto [idxs, s] = rewrite(app->arg())->projs<2>();
285 auto xs = idxs->projs();
286 auto str = strides(s, xs.size());
287 const Def* lin = w.lit(w.type_i64(), 0);
288 for (size_t k = 0; k != xs.size(); ++k) {
289 auto term = w.call(core::wrap::mul, core::Mode::none, Defs{xs[k], str[k]});
290 lin = w.call(core::wrap::add, core::Mode::none, Defs{lin, term});
291 }
292 return lin;
293 }
294
295 // affine.delinearize (lin, s) ↦ (lin floordiv strides#d) mod s#d for each d (on the `Idx 0` carrier).
296 if (Axm::isa<affine::delinearize>(app)) {
297 auto [lin, s] = rewrite(app->arg())->projs<2>();
298 auto m = s->num_projs();
299 auto str = strides(s, m);
300 return w.tuple(DefVec(m, [&](size_t d) {
301 const Def* q = nullptr;
302 if (auto k = const_val(str[d])) q = fold_udiv(lin, *k);
303 if (!q) q = div(core::div::udiv, lin, str[d]);
304
305 auto ext = w.call<core::bitcast>(w.type_i64(), s->proj(m, d));
306 if (auto k = const_val(ext))
307 if (auto folded = fold_urem(q, *k)) return folded;
308 return div(core::div::urem, q, ext);
309 }));
310 }
311
312 // affine.map f idxs mem ↦ widen idxs to `Idx 0`, inline f (advancing the threaded mem through any div), and narrow
313 // each result back to its target `Idx (sout#j)`; returns `(mem', narrowed)`.
314 if (Axm::isa<affine::map>(app)) {
315 // Extract f/idxs/sout from the *old* callee; we inline f's body at this call site rather than rewriting it into
316 // a standalone lam, since its body may reference the threaded mem (from the divs) and would otherwise be open.
317 auto [mn, sinout, f, idxs, _] = app->callee()->as<App>()->uncurry_args<5>();
318 auto [sin, sout] = sinout->projs<2>();
319
320 auto __ = fe::Restore(mem_);
321 auto mem = rewrite(app->arg()); // the `affine.map`'s mem operand
322 auto ins = rewrite(idxs)->projs();
323 auto lifted = w.tuple(DefVec(ins.size(), [&](size_t i) { return w.call(core::conv::u, w.lit_i64(), ins[i]); }));
324 auto f_lam = f->isa_mut<Lam>();
325
326 Lam* idx_map_lam = nullptr;
327 Lam* rw_idx_lam = nullptr;
328 if (auto it = specialized_[f_lam].find(sin); it != specialized_[f_lam].end()) {
329 idx_map_lam = it->second;
330 } else if (auto idx_lam = lookup(f_lam)) {
331 // Anything already carrying a mem was rewritten for a *different* `sin` and gets its own specialization
332 // below; anything else was rewritten as part of an annex and needs the mem parameter added.
333 if (auto mut = idx_lam->isa_mut<Lam>();
334 !(mut && mut->num_vars() == 2 && mut->var(2, 0)->type() == mem->type()))
335 rw_idx_lam = idx_lam->as_mut<Lam>();
336 }
337 if (!idx_map_lam) {
338 auto lam_pi = rewrite(f_lam->type())->as<Pi>();
339
340 idx_map_lam = w.mut_lam(w.pi({mem->type(), lam_pi->dom()}, {mem->type(), lam_pi->codom()}));
341 specialized_[f_lam][sin] = idx_map_lam;
342 if (!lookup(f_lam)) map(f_lam, idx_map_lam);
343
344 push();
345 map(f_lam->var(), idx_map_lam->var(2, 1));
346 for (size_t i = 0; i != f_lam->num_vars(); ++i)
347 map(f_lam->var(f_lam->num_vars(), i), idx_map_lam->var(2, 1)->proj(f_lam->num_vars(), i));
348 mem_ = idx_map_lam->var(2, 0);
349
350 auto n = sin->num_projs();
351 auto extents = fe::Vector<std::optional<int64_t>>(n, [&](size_t i) -> std::optional<int64_t> {
352 if (auto e = Lit::isa(sin->proj(n, i)); e && *e <= u64(Limit)) return int64_t(*e);
353 return {};
354 });
355 auto restore_dom = fe::Restore(dom_, Domain{idx_map_lam->var(2, 1), std::move(extents)});
356
357 auto get_body = [&]() -> const Def* {
358 if (rw_idx_lam) return rw_idx_lam->reduce_body(idx_map_lam->var(2, 1));
359 return rewrite(f_lam->body());
360 };
361 idx_map_lam->set(true, w.tuple({mem_, get_body()}))->set(f_lam->dbg_key());
362 pop();
363 }
364
365 auto outs = w.app(idx_map_lam, {mem, lifted});
366
367 auto sout_n = rewrite(sout);
368 auto n_out = sout_n->num_projs();
369 auto narrowed = w.tuple(DefVec(n_out, [&](size_t j) {
370 return w.call(core::conv::u, sout_n->proj(n_out, j), outs->proj(2, 1)->proj(n_out, j));
371 }));
372
373 return w.tuple({outs->proj(2, 0), narrowed});
374 }
375
376 return RWPhase::rewrite_imm_App(app);
377}
