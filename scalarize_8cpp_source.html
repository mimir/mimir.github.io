22bool Scalarize::Analysis::kept(const Pi* pi, size_t dom) const {
23 auto abstr = lattice(pi);
24 if (!abstr) return false;
25 // Too wide for the u64 bitmask: keep() pins the whole Pi for dom >= 64, so a param this wide
26 // can only survive as a u64 entry if it was never recorded - conservatively treat it as kept.
27 if (dom >= BitmaskWidth) return true;
28 if (auto mask = Lit::isa<u64>(abstr)) return (*mask >> dom) & 1; // per-param bitmask
29 return true; // ⊤ sentinel: whole Pi pinned
30}

32void Scalarize::Analysis::keep(const Pi* pi, size_t dom) {
33 // Out of the thresholded arity: plan() only iterates [0, num_tdoms), so such a parameter is never a
34 // split candidate anyway - nothing to record.
35 if (dom >= pi->num_tdoms()) return;
36 auto cur = lattice(pi);
37 if (cur && !Lit::isa<u64>(cur)) return; // already ⊤ - monotone, do not downgrade
38 // Too wide for the u64 bitmask (only with an unusually large scalarize threshold): conservatively pin
39 // the whole Pi rather than risk splitting a dynamically-indexed parameter.
40 if (dom >= BitmaskWidth) return pin(pi);
41 auto mask = cur ? Lit::as<u64>(cur) : u64(0);
42 auto next = mask | (u64(1) << dom);
43 if (next == mask) return;
44 lattice_force(pi, world().lit_nat(next));
45 DLOG("keep: {} #{}", pi, dom);
46}

48void Scalarize::Analysis::pin(const Pi* pi) {
49 auto cur = lattice(pi);
50 if (cur && !Lit::isa<u64>(cur)) return; // already ⊤
51 lattice_force(pi, pi); // ⊤ sentinel: pin the whole Pi
52 DLOG("pin: {}", pi);
53}

75void Scalarize::Analysis::inspect(const Def* def) {
76 // Everything reachable from an annex is interface: normalizers and backends rely on its exact shape.
77 if (is_bootstrapping()) {
78 if (auto pi = isa_flattenable(def)) pin(pi);
79 return;
80 }
81
82 if (auto app = def->isa<App>()) {
83 // The shapes an Axm's signature *itself* dictates are interface - never reshape them
84 // (e.g. `%affine.For`'s body): normalizers and plugin phases match on their exact shape,
85 // and rebuilding the App re-derives them from the Axm's generic type.
86 // Subtrees merely *substituted* in via earlier (type) arguments impose no shape, though -
87 // they rewrite consistently with the rest of the World (e.g. `T` in `%mem.store T`):
88 // seed the walk's visited set with them so they stay flattenable.
89 if (app->uncurry_callee()->isa<Axm>()) {
90 if (auto pi = isa_flattenable(app->callee_type())) pin(pi); // this very application's shape
91 auto skips = DefSet();
92 for (auto d = def; auto a = d->isa<App>(); d = a->callee())
93 collect(skips, a->arg());
94 pin_tree(app->callee_type()->dom(), skips); // what the Axm consumes (e.g. `%affine.For`'s body)
95 pin_tree(app->type(), skips); // what the Axm produces (e.g. `%autodiff.ad f`)
96
97 // An Axm taking a *bare* function argument (`%autodiff.ad f`) is higher-order machinery that
98 // will inspect and call that function by its own convention - keep the function's shape even
99 // where the Axm's signature is fully polymorphic (`{T: *} → T → ...`).
100 // A function buried inside a tuple argument (`%mem.store (mem, ptr, f)`) is just data, though.
101 if (auto lam = app->arg()->isa_mut<Lam>()) pin_tree(lam->type());
102 }
103
104 // Assignability is alpha-equivalence, so an App may connect a dom and an arg whose types are
105 // *distinct* defs (e.g. one with a mutable ret-Pi, one immutable). Flattening decides per def,
106 // which would tear such an edge apart - pin both sides.
107 // Only for immutable Pis: a dependent dom legitimately differs from the instantiated arg type.
108 if (auto pi = app->callee_type(); pi->isa_imm() && pi->dom() != app->arg()->type()) {
109 pin_tree(pi->dom());
110 pin_tree(app->arg()->type());
111 }
112 }
113
114 if (def->isa<Var>()) return; // a Var references its mut as op; that is not a use of the Lam
115
116 // A value inside a *dependently*-typed aggregate - e.g. a typed closure `(T, code, env)` of type
117 // `[T: *, Cn [.., T, ..], T]` - has its type dictated by the Sigma's binder:
118 // reshaping it would tear the aggregate's typing apart (and clos machinery rebuilds against it).
119 if (auto tuple = def->isa<Tuple>(); tuple && tuple->type()->isa_mut())
120 for (auto op : tuple->ops())
121 pin_tree(op->type());
122
123 // An interface Lam (external, annex, or unset declaration) keeps its whole signature:
124 // pin everything its type mentions (its ret Pi, callback params, a polymorphic Lam's inner Cn, ...) -
125 // other code (plugin phases, foreign callers) builds against these shapes.
126 // Only the interface's own (top-level) Pi stays flattenable: it may be shared with internal values
127 // (Scalarize::rewrite_mut_Lam preserves the interface's top level by hand).
128 if (auto lam = def->isa_mut<Lam>(); lam && !isa_optimizable(lam)) {
129 auto visited = DefSet();
130 visited.emplace(lam->type());
131 for (auto d : lam->type()->deps())
132 pin_tree(d, visited);
133 // A curried interface (e.g. `fun extern f {s: Nat} (ab: ...)`) reduces to its inner Lam%s
134 // upon application; their types are the interface's *instantiated* codomains - distinct defs
135 // from the Pi-side codomains pinned above (Lam var vs Pi var) - so pin them whole, too.
136 if (lam->is_set())
137 for (auto inner = lam->body()->isa_mut<Lam>(); inner;) {
138 pin_tree(inner->type(), visited);
139 inner = inner->is_set() ? inner->body()->isa_mut<Lam>() : nullptr;
140 }
141 }
142
143 // As long as an interface Lam is merely called that is a local affair (see Scalarize::rewrite_imm_App),
144 // but escaping as a *value* it would meet flattened values of the same type - so pin the whole type.
145 for (size_t i = 0, e = def->num_ops(); i != e; ++i) {
146 auto op = def->op(i);
147 auto lam = op ? op->isa_mut<Lam>() : nullptr;
148 if (!lam || isa_optimizable(lam)) continue;
149 if (auto pi = isa_flattenable(lam->type()); pi && !(def->isa<App>() && i == 0)) pin(pi);
150 }
151
152 // A parameter that is Extract%ed / Insert%ed via a non-constant index must not be split.
153 const Def* idx_tuple = nullptr;
154 const Def* idx = nullptr;
155 if (auto ex = def->isa<Extract>())
156 idx_tuple = ex->tuple(), idx = ex->index();
157 else if (auto in = def->isa<Insert>())
158 idx_tuple = in->tuple(), idx = in->index();
159 if (!idx_tuple || Lit::isa(idx)) return;
160
161 if (auto var = idx_tuple->isa<Var>()) {
162 if (auto pi = isa_flattenable(var->mut()->type())) pin(pi); // whole var indexed dynamically
163 } else if (auto proj = idx_tuple->isa<Extract>()) {
164 if (auto var = proj->tuple()->isa<Var>()) {
165 if (auto pi = isa_flattenable(var->mut()->type())) {
166 if (auto i = Lit::isa(proj->index()))
167 keep(pi, *i);
168 else
169 pin(pi);
170 }
171 }
172 }

180Vector<bool> Scalarize::Analysis::plan(const Def* type) const {
181 auto mask = Vector<bool>();
182 if (auto pi = isa_flattenable(type)) {
183 auto n = pi->num_tdoms();
184 auto any = false;
185 mask.assign(n, false);
186 for (size_t i = 0; i != n; ++i) {
187 // Only split immutable types: a mutable Sigma's element types may reference
188 // the Sigma's own var (e.g. a typed closure `[T: *, Cn [.., T, ..], T]`),
189 // which splitting would leave unbound.
190 auto t = pi->tdom(i);
191 if (!kept(pi, i) && t->isa_imm() && t->num_tprojs() > 1) mask[i] = any = true;
192 }
193 if (!any) mask.clear();
194 }
195 return mask;
196}

202const Def* Scalarize::rewrite_imm_Pi(const Pi* pi) {
203 auto mask = is_bootstrapping() ? Vector<bool>() : analysis_.plan(pi);
204 if (mask.empty()) return RWPhase::rewrite_imm_Pi(pi);
205
206 // build the flattened (one level) domain
207 auto doms = DefVec();
208 for (size_t i = 0, n = pi->num_tdoms(); i != n; ++i) {
209 auto t = rewrite(pi->tdom(i));
210 if (mask[i]) {
211 auto pieces = t->tprojs();
212 doms.insert(doms.end(), pieces.begin(), pieces.end());
213 } else {
214 doms.emplace_back(t);
215 }
216 }
217
218 auto& w = new_world();
219 auto sca = w.pi(w.sigma(doms), rewrite(pi->codom()), pi->is_implicit());
220 DLOG("scalarize {} ~> {}", pi, sca);
221 invalidate();
222 return sca;
223}

225const Def* Scalarize::rewrite_mut_Lam(Lam* old) {
226 auto mask = is_bootstrapping() ? Vector<bool>() : analysis_.plan(old->type());
227 if (mask.empty()) return RWPhase::rewrite_mut_Lam(old);
228
229 if (!isa_optimizable(old)) {
230 // An interface Lam's signature is ABI: rebuild its Pi via the generic hook -
231 // top-level shape preserved, inner types still flattened - instead of rewrite(),
232 // which would hand back the flattened Pi.
233 auto pi = RWPhase::rewrite_imm_Pi(old->type())->as<Pi>();
234 return rewrite_stub(old, new_world().mut_lam(pi));
235 }
236
237 auto& w = new_world();
238 auto sca = w.mut_lam(rewrite(old->type())->as<Pi>())->set(old->dbg());
239 DLOG("scalarize {} : {} ~> {} : {}", old, old->type(), sca, sca->type());
240 map(old, sca);
241
242 // reassemble the old var one level from the fresh scalar vars
243 auto n = old->num_tvars();
244 auto params = DefVec();
245 params.reserve(n);
246 for (size_t i = 0, v = 0; i != n; ++i) {
247 auto t = rewrite(old->tvar(i)->type());
248 if (mask[i]) {
249 auto pieces = DefVec(t->num_tprojs(), [&](size_t) { return sca->var(v++); });
250 params.emplace_back(w.tuple(t, pieces));
251 } else {
252 params.emplace_back(sca->var(v++));
253 }
254 }
255 map(old->var(), w.tuple(params));
256
257 sca->set(rewrite(old->filter()), rewrite(old->body()));
258 return sca;
259}

265 if (i < mask.size() && mask[i]) {
266 auto pieces = arg->tprojs();
267 args.insert(args.end(), pieces.begin(), pieces.end());
268 } else {

270 }
271 }
