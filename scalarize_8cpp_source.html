49void Scalarize::Analysis::inspect(const Def* def) {
50 // A parameter that is Extract%ed / Insert%ed via a non-constant index must not be split.
51 const Def* idx_tuple = nullptr;
52 const Def* idx = nullptr;
53 if (auto ex = def->isa<Extract>())
54 idx_tuple = ex->tuple(), idx = ex->index();
55 else if (auto in = def->isa<Insert>())
56 idx_tuple = in->tuple(), idx = in->index();
57
58 if (idx_tuple && !Lit::isa(idx)) {
59 if (auto var = idx_tuple->isa<Var>()) {
60 if (auto lam = var->mut()->isa_mut<Lam>()) demote(lam); // whole var indexed dynamically
61 } else if (auto proj = idx_tuple->isa<Extract>()) {
62 if (auto var = proj->tuple()->isa<Var>()) {
63 if (auto lam = var->mut()->isa_mut<Lam>()) {
64 if (auto i = Lit::isa(proj->index()))
65 lock(lam, *i);
66 else
67 demote(lam);
68 }
69 }
70 }
71 }
72
73 if (def->isa<Var>()) return; // a Var references its mut as op; that is not a use of the Lam
74
75 for (size_t i = 0, e = def->num_ops(); i != e; ++i) {
76 auto op = def->op(i);
77 if (!op) continue;
78 auto callee_pos = def->isa<App>() && i == 0;
79
80 if (auto lam = op->isa_mut<Lam>()) {
81 // ops of a lam tuple are judged when the tuple itself is used
82 if (!callee_pos && !is_lam_tuple(def)) demote(lam);
83 } else if (is_lam_tuple(op)) {
84 // all sides of a branch share one expansion decision: only fine if all members
85 // are eligible, agree on their type, are still splittable, and are unlocked.
86 auto extract_pos = def->isa<Extract>() && i == 0;
87 auto consistent = std::all_of(op->ops().begin(), op->ops().end(), [&](const Def* d) {
88 auto lam = d->as_mut<Lam>();
89 auto lck = locked_.find(lam);
90 return d->type() == op->op(0)->type() && eligible(lam) && splittable(lam)
91 && (lck == locked_.end() || lck->second.empty());
92 });
93 if (!extract_pos || !consistent) demote_all(op);
94 } else if (auto proj = op->isa<Extract>(); proj && is_lam_tuple(proj->tuple())) {
95 if (!callee_pos) demote_all(proj->tuple());
96 }
97 }
98}

105Vector<bool> Scalarize::Analysis::plan(Lam* lam) {
106 auto mask = Vector<bool>();
107 if (eligible(lam) && splittable(lam)) {
108 auto n = lam->num_tvars();
109 auto lck = locked_.find(lam);
110 auto any = false;
111 mask.assign(n, false);
112 for (size_t i = 0; i != n; ++i) {
113 auto locked = lck != locked_.end() && lck->second.contains(i);
114 // Only split immutable types: a mutable Sigma's element types may reference
115 // the Sigma's own var (e.g. a typed closure `[T: *, Cn [.., T, ..], T]`),
116 // which splitting would leave unbound.
117 auto t = lam->tvar(i)->type();
118 if (!locked && t->isa_imm() && t->num_tprojs() > 1) mask[i] = any = true;
119 }
120 if (!any) mask.clear();
121 }
122 return mask;
123}

129const Def* Scalarize::rewrite_mut_Lam(Lam* old) {
130 auto mask = is_bootstrapping() ? Vector<bool>() : analysis_.plan(old);
131 if (mask.empty()) return RWPhase::rewrite_mut_Lam(old);
132
133 auto& w = new_world();
134 auto n = old->num_tvars();
135
136 // build the flattened (one level) domain
137 auto dtypes = DefVec(n, [&](size_t i) { return rewrite(old->tvar(i)->type()); });
138 auto doms = DefVec();
139 for (size_t i = 0; i != n; ++i) {
140 if (mask[i]) {
141 auto pieces = dtypes[i]->tprojs();
142 doms.insert(doms.end(), pieces.begin(), pieces.end());
143 } else {
144 doms.emplace_back(dtypes[i]);
145 }
146 }
147
148 auto sca = w.mut_lam(w.cn(doms))->set(old->dbg());
149 DLOG("scalarize {} : {} ~> {} : {}", old, old->type(), sca, sca->type());
150 map(old, sca);
151
152 // reassemble the old var one level from the fresh scalar vars
153 auto params = DefVec();
154 params.reserve(n);
155 for (size_t i = 0, v = 0; i != n; ++i) {
156 if (mask[i]) {
157 auto pieces = DefVec(dtypes[i]->num_tprojs(), [&](size_t) { return sca->var(v++); });
158 params.emplace_back(w.tuple(dtypes[i], pieces));
159 } else {
160 params.emplace_back(sca->var(v++));
161 }
162 }
163 map(old->var(), w.tuple(params));
164
165 sca->set(rewrite(old->filter()), rewrite(old->body()));
166 invalidate();
167 return sca;
168}

184const Def* Scalarize::rewrite_imm_App(const App* app) {
185 if (!is_bootstrapping()) {
186 auto& w = new_world();
187 const Def* new_callee = nullptr;
188 auto mask = Vector<bool>();
189
190 if (auto lam = app->callee()->isa_mut<Lam>()) {
191 if (mask = analysis_.plan(lam); !mask.empty()) new_callee = rewrite(lam);
192 } else if (auto proj = app->callee()->isa<Extract>(); proj && is_lam_tuple(proj->tuple())) {
193 auto tuple = proj->tuple();
194 if (mask = analysis_.plan(tuple->op(0)->as_mut<Lam>()); !mask.empty()) {
195 auto members = DefVec(tuple->num_ops(), [&](size_t i) { return rewrite(tuple->op(i)); });
196 new_callee = w.extract(w.tuple(members), rewrite(proj->index()));
197 }
198 }
199
200 if (new_callee) return w.app(new_callee, flatten_args(app, mask));
201 }
202
203 return RWPhase::rewrite_imm_App(app);
