22/// Is the argument of the Axm application @p app typed by one of the Axm's type variables?

119fe::Bitset Scalarize::Analysis::plan(const Def* type) const {
120 auto mask = fe::Bitset();
121 if (auto pi = isa_flattenable(type)) {
122 // Splitting a dependent domain would shift the indices its Var references.
123 if (auto sig = pi->dom()->isa_mut<Sigma>(); sig && sig->has_var()) return mask;
124 for (size_t i = 0, n = pi->num_tdoms(); i != n; ++i) {
125 auto t = pi->tdom(i);
126 if (!kept(pi, i) && (t->isa_imm() || is_value_telescope(t)) && t->num_tprojs() > 1) mask.set(i);
127 }
128 }
129 return mask;
130}

165const Def* Scalarize::flatten(Rewriter& rw, const Pi* pi, const fe::Bitset& mask) {
166 if (mask.none()) return RWPhase::rewrite_imm_Pi(pi);
167
168 auto& w = new_world();
169 auto n = pi->num_tdoms();
170 auto ts = DefVec(n, [&](size_t i) { return rw.rewrite(pi->tdom(i)); });
171 auto num = size_t(0);
172 auto deps = false;
173 for (size_t i = 0; i != n; ++i) {
174 num += mask[i] ? ts[i]->num_tprojs() : 1;
175 deps |= mask[i] && ts[i]->isa_binder<Sigma>().second;
176 }
177
178 const Def* dom;
179 if (deps) {
180 // A telescope's pieces refer to their predecessors, which now live in the new domain's binder.
181 auto sigma = w.mut_sigma(num);
182 for (size_t i = 0, v = 0; i != n; ++i) {
183 if (!mask[i]) {
184 sigma->set(v++, ts[i]);
185 continue;
186 }
187 auto k = ts[i]->num_tprojs();
188 auto parts = DefVec();
189 for (size_t j = 0; j != k; ++j, ++v) {
190 sigma->set(v, w.component(ts[i], parts, k, j));
191 parts.emplace_back(sigma->var(num, v));
192 }
193 }
194 dom = sigma;
195 } else {
196 auto doms = DefVec();
197 for (size_t i = 0; i != n; ++i) {
198 if (mask[i]) {
199 auto pieces = ts[i]->tprojs();
200 doms.insert(doms.end(), pieces.begin(), pieces.end());
201 } else {
202 doms.emplace_back(ts[i]);
203 }
204 }
205 dom = w.sigma(doms);
206 }
207
208 auto sca = w.pi(dom, rw.rewrite(pi->codom()), pi->is_implicit());
209 log().d("scalarize {} → {}", pi, sca);
210 invalidate();
211 return sca;
212}

214Lam* Scalarize::stub(Lam* old, const fe::Bitset& mask) {
215 auto n = old->num_tvars();

220 auto type = mask.any() ? flatten(rw, old->type()->as<Pi>(), mask) : rw.rewrite(old->type());
221 if (old->is_set())
222 for (size_t i = 0; i != n; ++i)
223 if (mask[i]) types[i] = rw.rewrite(old->tvar(i)->type());
224
225 auto& w = new_world();

227 map(old, sca);
228 if (!old->is_set() || mask.none()) return sca;
