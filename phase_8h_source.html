50 auto& world = def->world();

121/// Traverses the current World using Rewriter infrastructure while staying in the same world.
122///
123/// It recursively rewrites
124/// 1. all World::annexes() (during which Analysis::is_bootstrapping() is `true`), and then
125/// 2. all World::externals() (during which it is `false`).
126///
127/// Analysis provides a reusable lattice() mapping old Def%s to abstract values, represented as ordinary MimIR Def%s.
128/// @note You can override
129/// - Rewriter::rewrite(),
130/// - Rewriter::rewrite_imm(),
131/// - Rewriter::rewrite_mut(), etc.
132/// @see @ref phases_analysis
133class Analysis : public Phase, public Rewriter {
134public:
135 /// @name Construction & Destruction
136 ///@{
137 Analysis(World& world, std::string name)
138 : Phase(world, std::move(name))
139 , Rewriter(world) {}
140 Analysis(World& world, flags_t annex)
141 : Phase(world, annex)
142 , Rewriter(world) {}
143
144 /// Clears the rewriter map and resets Phase::todo() for the next fixed-point iteration.
145 /// lattice() is **preserved** across iterations so that abstract values accumulated in earlier
146 /// rounds remain available - this is what makes fixed-point convergence possible.
147 /// @see RWPhase::analyze
148 virtual void reset();
149 ///@}
150
151 /// @name Getters
152 ///@{
153 World& world() { return Phase::world(); }
154 bool is_bootstrapping() const { return bootstrapping_; }
155 ///@}
156
157 /// @name lattice
158 /// Conventions: *absent* = ⊥ (nothing known); `def ↦ def` = ⊤ (keep as is).
159 /// Subclasses may store their own sentinels in between (e.g. SEO's `nullptr` and GVN-bundle Proxy%s).
160 ///@{
161 const auto& lattice() const { return lattice_; }
162
163 /// @returns the abstract value recorded for @p def, or `nullptr` if unknown.
164 const Def* lattice(const Def* def) const {
165 if (auto i = lattice_.find(def); i != lattice_.end()) return i->second;
166 return nullptr;
167 }
168
169 bool is_top(const Def* def) const { return lattice(def) == def; }
170
171 /// Records the abstract value @p abstr for @p concr in both lattice() (the analysis result)
172 /// and map() (so the rewriter short-circuits future rewrites of @p concr to @p abstr).
173 /// Bypasses update() and hence never invalidate()s.
174 const Def* set(const Def* concr, const Def* abstr) {
175 lattice_[concr] = abstr;
176 return map(concr, abstr);
177 }
178
179 /// Monotonically forces @p def to ⊤ (keep as is).
180 const Def* pin_top(const Def* def) {
181 update(def, def);
182 return def;
183 }
184 ///@}
185
186protected:
187 /// @name lattice
188 ///@{
189
190 /// Writes `concr ↦ abstr` into lattice(); does **not** map().
191 /// invalidate()s - and thereby triggers another fixed-point round - iff this changes observable information:
192 /// an existing entry was overwritten, or a fresh fact other than ⊤ was inserted.
193 /// Freshly inserting ⊤ (`concr ↦ concr`) stays silent, as it is indistinguishable from *absent* for consumers.
194 /// @returns the former abstract value:
195 /// - `nullptr`: fresh insert (or the former value was a subclass' `nullptr` sentinel),
196 /// - `== abstr`: unchanged,
197 /// - anything else: overwritten.
198 const Def* update(const Def* concr, const Def* abstr) {
199 auto [i, ins] = lattice_.emplace(concr, abstr);
200 if (ins) {
201 invalidate(concr != abstr);
202 return nullptr;
203 }
204
205 auto old = i->second;
206 if (old != abstr) {
207 i->second = abstr;
208 invalidate();
209 }
210 return old;
211 }
212 ///@}

214 /// @name Rewrite
215 ///@{
216 virtual void prepare() {} ///< Run **before** the main analysis.
217 virtual void finalize() {} ///< Run **after** the main analysis.
218 void start() override;
219 virtual void rewrite_annex(flags_t, Sym, const Def*);
220 virtual void rewrite_external(Def*);

226 Def* rewrite_mut(Def*) override;
227 ///@}
228
229private:
230 /// Walks all enqueued mutables' dependencies - in BFS order - under each mutable's curr_mut() scope.
231 void drain();

238/// Rebuilds old_world() into new_world() and then swaps them.
239///
240/// It recursively rewrites
241/// 1. all old World::annexes() (during which RWPhase::is_bootstrapping() is `true`, and then
242/// 2. all old World::externals() (during which it is `false`).
243///
244/// During bootstrapping, rewrites that depend on other annexes may need to be skipped,
245/// since those annexes might not yet exist in the new world.
246///
247/// If an associated Analysis is provided, the rewrite can query its abstract results through lattice().
248///
249/// @note You can override
250/// - Rewriter::rewrite(),
251/// - Rewriter::rewrite_imm(),
252/// - Rewriter::rewrite_mut(), etc.

255public:

260 , Rewriter(world.inherit())
261 , analysis_(analysis) {}
262 RWPhase(World& world, flags_t annex, Analysis* analysis = nullptr)
263 : Phase(world, annex)
264 , Rewriter(world.inherit())
265 , analysis_(analysis) {}
266 ///@}
267
268 /// @name Analysis
269 ///@{
270 Analysis* analysis() { return analysis_; }

272
273 /// Returns the abstract value computed by the associated Analysis for the given old-world Def, or `nullptr` if no
274 /// value is available.

276
277 /// Returns lattice(@p old_def) if it differs from @p old_def (i.e. we learned something), otherwise `nullptr`.
