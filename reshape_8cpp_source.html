163const Def* Reshape::reshape(DefVec& defs, const Def* T, const Def* mem) {
164 auto& world = T->world();
165 if (should_flatten(T)) {
166 auto tuples = T->projs([&](auto P) { return reshape(defs, P, mem); });
167 return world.tuple(tuples);
168 } else {
169 const Def* def;
170 if (is_mem_ty(T)) {
171 assert(mem != nullptr && "Reshape: mems not found");
172 def = mem;
173 } else {
174 do {
175 assert(defs.size() > 0 && "Reshape: not enough arguments");
176 def = defs.front();
177 defs.erase(defs.begin());
178 } while (is_mem_ty(def->type()));
179 }
180 // For inner function types, we override the type
181 if (!def->type()->isa<Pi>()) {
182 if (!Checker::alpha<Checker::Check>(def->type(), T))
183 world.ELOG("reconstruct T {} from def {}", T, def->type());
184 assert(Checker::alpha<Checker::Check>(def->type(), T) && "Reshape: argument type mismatch");
185 }
186 return def;
187 }
