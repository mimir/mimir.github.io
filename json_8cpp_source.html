126 std::int64_t res = 0;

141 Tok lex() {
142 while (true) {
143 start();
144
145 if (accept(utf8::EoF)) return {loc_, Tag::EoF};
146 if (accept(utf8::isspace)) continue;
147 if (recover_utf8()) continue;
148 if (accept('{')) return {loc_, Tag::D_brace_l};
149 if (accept('}')) return {loc_, Tag::D_brace_r};
150 if (accept('[')) return {loc_, Tag::D_bracket_l};
151 if (accept(']')) return {loc_, Tag::D_bracket_r};
152 if (accept(':')) return {loc_, Tag::T_colon};
153 if (accept(',')) return {loc_, Tag::T_comma};
154 if (accept('"')) return lex_string();
155 if (ahead() == '-' || utf8::isdigit(ahead())) return lex_number();
156
157 if (accept(utf8::isalpha)) {
158 accept_while(utf8::isalpha);
159#define CODE(t, str) \
160 if (view() == str) return {loc_, Tag::t};
161 PBQP_JSON_KEY(CODE)
162#undef CODE
163 error().e(loc_, "unknown literal `{}`", view());
164 continue;
165 }
166
167 recover_char();
168 }
169 }
170
171private:
172 bool accept_digits() { return !accept_while(utf8::isdigit).empty(); }
173
174 Tok lex_number() {
175 accept('-');
176 bool ok = accept_digits();
177 bool integer = true;
178 if (accept('.')) {
179 integer = false;
180 ok = accept_digits() && ok;
181 }
182 if (accept('e') || accept('E')) {
183 integer = false;
184 if (!accept('+')) accept('-');
185 ok = accept_digits() && ok;
186 }
187 if (!ok) error().e(loc_, "malformed number `{}`", view());
188 if (integer && ok) return {loc_, to_i64(view())};
189 return {loc_, Tag::V_num, driver_.sym(view())};
190 }
191
192 /// Escapes stay verbatim; none of the keys or cost spellings needs one.
193 Tok lex_string() {
194 while (true) {
195 accept_while_none_of(u8'"', u8'\\');
196 if (accept('"')) return {loc_, Tag::V_str, driver_.sym(view().substr(1, view().size() - 2))};
197 if (!accept('\\') || ahead() == utf8::EoF) break;
198 next();
199 }
200 error().e(loc_, "unterminated string");
201 return {loc_, Tag::V_str, driver_.sym(view().substr(1))};
202 }

355 std::optional<Field> src, dst;

357 // clang-format off

363 else parse_value("edge");

368 if (!dst) return void(error().e(edge.loc, "edge without `dst`"));
369 if (!edge.costs_loc) return void(error().e(edge.loc, "edge without `costs`"));
370 edge.src = *src;

375 void build(Instance& instance) {
376 Graph& graph = instance.graph;
377 for (const auto& costs : nodes_)
378 graph.add(costs);
379
380 const auto known = [&](const Field& field) {
381 if (field.value < graph.num_nodes()) return true;
382 error().e(field.loc, "unknown node {}", field.value);
383 return false;
384 };
385
386 for (const auto& edge : edges_) {
387 if (!known(edge.src) || !known(edge.dst)) continue;
388 const NodeId src = edge.src.value;
389 const NodeId dst = edge.dst.value;
390 if (src == dst) {
391 error().e(edge.loc, "self-edge on node {}", src);
392 continue;
393 }
394 if (graph.edge(src, dst) != Graph::npos) {
395 error().e(edge.loc, "duplicate edge between nodes {} and {}", src, dst);
396 continue;
397 }
398 const size_t rows = graph.num_options(src);
399 const size_t cols = graph.num_options(dst);
400 if (edge.rows && edge.rows->value != rows)
401 error().e(edge.rows->loc, "`rows` is {}, but node {} has {} options", edge.rows->value, src, rows);
402 else if (edge.cols && edge.cols->value != cols)
403 error().e(edge.cols->loc, "`cols` is {}, but node {} has {} options", edge.cols->value, dst, cols);
404 else if (edge.costs.size() != rows * cols)
405 error().e(edge.costs_loc, "expected {} × {} costs, got {}", rows, cols, edge.costs.size());
406 else
407 graph.link(src, dst, MatrixView(edge.costs.data(), rows, cols));
408 }
409
410 if (rn_priority_loc_ && rn_priority_.size() != graph.num_nodes())
411 error().e(rn_priority_loc_, "expected {} priorities, got {}", graph.num_nodes(), rn_priority_.size());
412 instance.rn_priority = std::move(rn_priority_);
413 }
