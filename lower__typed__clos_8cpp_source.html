15void LowerTypedClos::start() {
16 // Bootstrapping: rebuild all annexes verbatim (see LowerTypedClos::rewrite).
17 for (const auto& [flags, e] : old_world().annexes())
18 rewrite_annex(flags, e.sym, e.def);
19
20 converting_ = true;
21 dummy_ret_ = new_world().bot(new_world().cn(new_world().call<mem::M>(0)));
22
23 for (auto old_mut : old_world().externals().muts()) {
24 auto new_def = rewrite(old_mut);
25 // Converted Lam%s are externalized inside make_stub; other externals carry over here.
26 if (auto new_mut = new_def->isa_mut(); new_mut && old_mut->is_external() && !new_mut->is_external())
27 new_mut->externalize();
28 }
29
30 while (!worklist_.empty()) {
31 auto [lvm, lcm, old_lam, new_lam] = worklist_.front();
32 worklist_.pop();
33 lvm_ = lvm;
34 lcm_ = lcm;
35 DLOG("in {} (lvm={}, lcm={})", new_lam, lvm_, lcm_);
36 if (old_lam->is_set()) new_lam->set(rewrite(old_lam->filter()), rewrite(old_lam->body()));
37 }
38
39 swap(old_world(), new_world());
40}

42Lam* LowerTypedClos::make_stub(Lam* lam, Mode mode, bool adjust_bb_type) {
43 assert(lam && "make_stub: not a lam");
44 if (auto i = lookup(lam); i && i->isa_mut<Lam>()) return i->as_mut<Lam>();
45
46 auto& w = new_world();
47 auto ep = env_param(lam->type()->as<Pi>());
48 auto new_dom = w.sigma(DefVec(lam->num_doms(), [&](auto i) -> const Def* {
49 auto new_dom = rewrite(lam->dom(i));
50 if (i == ep) {
51 if (mode == Unbox) return env_type();
52 if (mode == Box) return w.call<mem::Ptr0>(new_dom);
53 }
54 return new_dom;
55 }));
56 if (Lam::isa_basicblock(lam) && adjust_bb_type) new_dom = insert_ret(new_dom, dummy_ret_->type());
57 auto new_lam = w.mut_lam(w.cn(new_dom))->set(lam->dbg_key());
58 DLOG("stub {} ~> {}", lam, new_lam);
59 if (lam->is_external()) new_lam->externalize();
60
61 auto lcm = mem::mem_var(new_lam);
62 // The environment always lives in slot `ep`; a single-parameter lam has an atomic var (no projection),
63 // so use the whole var there. This selection is independent of `mode` -- the mode only governs how the
64 // environment is subsequently consumed (loaded, bitcast, or passed through), not where it sits.
65 auto env = new_lam->num_vars() < 2 ? new_lam->var() : new_lam->var(ep);
66 if (mode == Box) {
67 // A mem-free closure still has to unbox its heap-allocated environment via a mem.load; if it has no mem of
68 // its own, a throw-away witness is fine here -- if it actually closes over a real mem, that one is recovered
69 // from the unboxed environment below instead (see issue #126).
70 if (!lcm) lcm = w.bot(w.call<mem::M>(0));
71 auto [m, e] = w.call<mem::load>(Defs{lcm, env})->projs<2>();
72 lcm = m->set("mem");
73 env = e->set("closure_env");
74 } else if (mode == Unbox) {
75 env = w.call<core::bitcast>(rewrite(lam->dom(ep)), env)->set("unboxed_env");
76 }
77 auto new_args = w.tuple(DefVec(lam->num_doms(), [&](auto i) {
78 return (i == ep) ? env : (lam->var(i) == mem::mem_var(lam)) ? lcm : new_lam->var(i);
79 }));
80 assert(new_args->num_projs() == lam->num_doms());
81 assert(lam->num_doms() <= new_lam->num_doms());
82 map(lam->var(), new_args);
83
84 // This closure may not have mem as a direct parameter, yet still close over a real one through its captured
85 // environment (e.g. a callback typed without mem that nonetheless uses an outer `mem`). Recover it so
86 // mem-effectful operations inside this closure's own body (e.g. packing a further nested closure) have a real
87 // chain to thread, instead of none (see issue #126).
88 auto lvm = mem::mem_var(lam);
89 // `ep < num_vars()` guards a `Cn []` (or mem-only) closure: there is no environment slot to recover a
90 // mem from, and `lam->var(ep)` would be an out-of-bounds projection reading past the operand array.
91 // Mirrors the `num_vars() < 2` guard applied to `new_lam` above.
92 if (!lvm && ep < lam->num_vars()) {
93 auto old_env = lam->var(ep);
94 if (Axm::isa<mem::M>(old_env->type())) {
95 lvm = old_env;
96 lcm = env;
97 } else if (auto sig = old_env->type()->isa<Sigma>()) {
98 for (size_t i = 0, e = sig->num_ops(); i != e; ++i)
99 if (Axm::isa<mem::M>(sig->op(i))) {
100 lvm = old_env->proj(i);
101 lcm = env->proj(i);
102 break;
103 }
104 }
105 }
106 worklist_.emplace(lvm, lcm, lam, new_lam);
107 map(lam, new_lam);
108 return new_lam;
109}

111const Def* LowerTypedClos::rewrite(const Def* def) {
112 if (!converting_) return RWPhase::rewrite(def); // bootstrapping: rebuild verbatim
113 if (auto new_def = lookup(def)) return new_def;
114
115 assert((!def->isa<Var>() || !def->as<Var>()->binder()->isa_mut<Lam>()) && "Lam vars should appear in a map!");
116
117 auto& w = new_world();
118
119 // Lower a closure type `[Env: *, Cn [Env, Args..], Env]` to an untyped `(code-ptr, env-ptr)` pair type.
120 if (auto ct = isa_clos_type(def)) {
121 auto pi = rewrite(ct->op(1))->as<Pi>();
122 if (Pi::isa_basicblock(pi)) pi = w.cn(insert_ret(pi->dom(), dummy_ret_->type()));
123 auto env_type = rewrite(ct->op(2));
124 return map(def, w.sigma({pi, env_type}));
125 }
126
127 // Project out of a closure: index 0 is the (erased) env type, 1 the code, 2 the env.
128 if (auto proj = def->isa<Extract>(); proj && isa_clos_type(proj->tuple()->type())) {
129 auto idx = Lit::isa(proj->index());
130 assert(idx && *idx <= 2 && "unknown proj from closure tuple");
131 return map(def, *idx == 0 ? env_type() : rewrite(proj->tuple())->proj(*idx - 1));
132 }
133
134 // Lower a closure literal to an untyped `(code-ptr, env-ptr)` pair, boxing/unboxing the environment.
135 if (auto c = isa_clos_lit(def)) {
136 auto new_type = rewrite(def->type());
137 auto env = rewrite(c.env());
138 auto mode = (env->type()->isa<Idx>() || Axm::isa<mem::Ptr>(env->type())) ? Unbox : Box;
139 const Def* fn = make_stub(c.fnc_as_lam(), mode, true);
140 if (env->type() == w.sigma()) {
141 env = w.bot(env_type()); // optimize empty env
142 } else if (mode == Box) {
143 auto [mem, env_ptr] = mem::op_alloc(env->type(), lcm_)->projs<2>();
144 lcm_ = w.call<mem::store>(Defs{mem, env_ptr, env});
145 map(lvm_, lcm_);
146 env = env_ptr;
147 }
148 fn = w.call<core::bitcast>(new_type->op(0), fn);
149 env = w.call<core::bitcast>(new_type->op(1), env);
150 return map(def, w.tuple({fn, env}));
151 }
152
153 if (auto lam = def->isa_mut<Lam>()) return make_stub(lam, No_Env, false);
154 if (def->isa_mut()) {
155 assert(!isa_clos_type(def));
156 return RWPhase::rewrite_mut(const_cast<Def*>(def));
157 }
158 if (auto var = def->isa<Var>()) return map(def, w.var(rewrite(var->binder())->as_mut()));
159
160 // Leaves and axioms need no mem threading; let the base rebuild them.
161 switch (def->node()) {
162 case Node::Bot:
163 case Node::Top:
164 case Node::Type:
165 case Node::Univ:
166 case Node::Nat:
167 case Node::Axm: return RWPhase::rewrite_imm(def);
168 default: break;
169 }
170
171 // Generic immutable: rebuild, add a dummy return continuation to first-class BBs, then thread the mem token.
172 auto new_type = rewrite(def->type());
173 auto new_ops = DefVec(def->num_ops(), [&](auto i) { return rewrite(def->op(i)); });
174 if (auto app = def->isa<App>())
175 if (auto p = app->callee()->isa<Extract>();
176 p && isa_clos_type(p->tuple()->type()) && Pi::isa_basicblock(app->callee_type()))
177 new_ops[1] = insert_ret(new_ops[1], dummy_ret_);
178 auto new_def = def->rebuild(w, new_type, new_ops);
179
180 // Rethread the operand that consumed this function's current mem chain end (lvm_): boxing above may have
181 // advanced lcm_ past what the operand was rewritten to. Only that operand - other mem operands may belong to
182 // another function's chain (shared subgraphs) and rethreading them would corrupt it.
183 for (size_t i = 0, e = new_def->num_ops(); i != e; ++i)
184 if (def->op(i) == lvm_ && new_def->op(i)->type() == w.call<mem::M>(0)) new_def = new_def->refine(i, lcm_);
185
186 if (new_type == w.call<mem::M>(0)) { // :store
187 lcm_ = new_def;
188 lvm_ = def;
189 } else if (new_type->isa<Sigma>()) { // :alloc, :slot, ...
190 for (size_t i = 0, e = new_type->num_ops(); i != e; ++i)
191 if (new_type->op(i) == w.call<mem::M>(0)) {
192 lcm_ = w.extract(new_def, i); // new-world mem chain
193 lvm_ = old_world().extract(def, i); // old-world marker, compared against old ops
194 break;
195 }
196 }
197
198 return map(def, new_def);
199}
