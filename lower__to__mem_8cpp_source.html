20/// Does `t` (recursively through immutable sigmas) contain a Pi?
21bool contains_pi(const Def* t) {
22 if (t->isa<Pi>()) return true;
23 if (auto sig = t->isa_imm<Sigma>())
24 for (auto op : sig->ops())
25 if (contains_pi(op)) return true;
26 return false;
27}

29/// Peels nested `Pack`s off `d`; returns the innermost body iff `d` is a constant splat — every axis a
30/// `Pack` (so `d` is uniform, not e.g. a `Tuple` of distinct rows), bottoming out in a closed scalar
31/// (index-independent, non-array). So `‹784, 1024; 1e-3›` yields the scalar `1e-3`, while an index-dependent
32/// pack `‹i; f i›` or a genuine literal `((1, 2), (3, 4))` yields `nullptr`.
33const Def* splat_scalar(const Def* d) {
34 if (!d->isa<Pack>()) return nullptr;
35 while (auto pack = d->isa<Pack>())
36 d = pack->body();
37 return (d->is_closed() && !d->type()->isa<Arr>()) ? d : nullptr;
38}

53void LowerToMem::collect_tensor_types() {
54 // The default pipeline lowers tensors exclusively through buffers — there is no value-semantics
55 // fallback. A program shape the conversion cannot handle is a hard error, not silent residue.
56 auto gate = [](const char* why, const Def* culprit) { fe::throwf("cannot bufferize: {} (`{}`)", why, culprit); };
57 // Fully folded shapes (`«1; T»` ≡ `T`) denote plain scalars — recording them would poison every
58 // function whose signature mentions the element type, so only genuine array types count as tensors.
59 auto add_tensor_ty = [this](const Def* t) {
60 if (t->isa<Arr>()) tensor_ty_.emplace(t);
61 };
62
63 unique_queue<DefSet> wl;
64 auto push = [&wl](const Def* d) {
65 if (d) wl.push(d);
66 };
67
68 for (auto mut : old_world().externals().muts())
69 push(mut);
70
71 while (!wl.empty()) {
72 auto def = wl.pop();
73
74 if (auto app = def->isa<App>()) {
75 if (auto [axm, curry, trip] = Axm::get(app); axm && curry == 0 && axm->plugin() == tensor::Plugin_Id)
76 ops_seen_ = true;
77 if (Axm::isa<tensor::get>(app) || Axm::isa<tensor::set>(app)) {
78 // get/set: the tensor `arr` is the *second* explicit argument (the first one is `index`).
79 auto [T, r, s] = app->callee()->as<App>()->args<3>();
80 if (T->isa<Arr>()) gate("tensor with array element type", T);
81 add_tensor_ty(app->arg()->proj(1)->type());
82 } else if (Axm::isa<tensor::splat>(app)) {
83 auto [T, r] = app->callee()->as<App>()->args<2>();
84 if (T->isa<Arr>()) gate("tensor with array element type", T);
85 add_tensor_ty(app->type());
86 } else if (Axm::isa<tensor::map_reduce_post>(app)) {
87 // result and each of the `nis` inputs / `nps` epilogue inputs are tensors.
88 add_tensor_ty(app->type());
89 auto [nis_nps, meta, shapes, in_tys, comb_init, acc_out, accs]
90 = app->callee()->as<App>()->uncurry_args<7>();
91 if (meta->proj(5, 0)->isa<Arr>()) gate("tensor with array element type", meta->proj(5, 0));
92 if (meta->proj(5, 1)->isa<Arr>()) gate("tensor with array element type", meta->proj(5, 1));
93 if (auto nis_l = Lit::isa<u64>(nis_nps->proj(2, 0))) {
94 auto Tis = in_tys->proj(6, 0);
95 for (u64 i = 0; i < *nis_l; ++i) {
96 if (Tis->proj(*nis_l, i)->isa<Arr>()) gate("tensor with array element type", Tis);
97 add_tensor_ty(app->arg()->proj(2, 0)->proj(*nis_l, i)->type());
98 }
99 }
100 if (auto nps_l = Lit::isa<u64>(nis_nps->proj(2, 1))) {
101 auto Tps = in_tys->proj(6, 3);
102 for (u64 j = 0; j < *nps_l; ++j) {
103 if (Tps->proj(*nps_l, j)->isa<Arr>()) gate("tensor with array element type", Tps);
104 add_tensor_ty(app->arg()->proj(2, 1)->proj(*nps_l, j)->type());
105 }
106 }
107 } else if (Axm::isa<tensor::broadcast>(app)) {
108 // result «s_out; T» and input «s_in; T» (the 3rd argument) are tensors.
109 auto [T, r] = app->callee()->as<App>()->args<2>();
110 if (T->isa<Arr>()) gate("tensor with array element type", T);
111 add_tensor_ty(app->type());
112 add_tensor_ty(app->arg()->proj(2)->type());
113 } else if (Axm::isa<tensor::pad>(app)) {
114 // callee: pad {T, r} [s_in] [mode, lo, hi]; result «s_out; T» and `input` are tensors.
115 auto [Tr, s_in, params] = app->callee()->as<App>()->uncurry_args<3>();
116 auto [T, r] = Tr->projs<2>();
117 if (T->isa<Arr>()) gate("tensor with array element type", T);
118 if (!Lit::isa<u64>(r) || !Lit::isa<u64>(params->proj(3, 0)))
119 gate("non-literal rank/mode of `%tensor.pad`", app);
120 add_tensor_ty(app->type());
121 add_tensor_ty(app->arg()->proj(0)->type());
122 } else if (Axm::isa<tensor::concat>(app)) {
123 // callee: concat {T, nis, r} [ax] {Sis}; result «s_out; T» and each input are tensors.
124 auto [TnisR, ax, Sis] = app->callee()->as<App>()->uncurry_args<3>();
125 auto [T, nis, r] = TnisR->projs<3>();
126 if (T->isa<Arr>()) gate("tensor with array element type", T);
127 auto nis_l = Lit::isa<u64>(nis);
128 auto r_l = Lit::isa<u64>(r);
129 auto ax_l = Lit::isa<u64>(ax);
130 if (!nis_l || !r_l || !ax_l) {
131 gate("non-literal nis/rank/axis of `%tensor.concat`", app);
132 } else {
133 add_tensor_ty(app->type());
134 for (u64 i = 0; i < *nis_l; ++i) {
135 // The loop generation needs literal extents along `ax` for the prefix offsets.
136 if (!Lit::isa<u64>(Sis->proj(*nis_l, i)->proj(*r_l, *ax_l)))
137 gate("non-literal extent along the concat axis", app);
138 add_tensor_ty(app->arg()->proj(*nis_l, i)->type());
139 }
140 }
141 } else if (Axm::isa<tensor::if_static>(app)) {
142 // Value-level binding-time dispatch; this phase's rewrite residualizes it to its
143 // dynamic branch - not a tensor op.
144 } else if (auto [axm, curry, trip] = Axm::get(app);
145 axm && curry == 0 && axm->plugin() == tensor::Plugin_Id) {
146 // Any other tensor op (a symbolic `shape`, …) has no buffer-world lowering.
147 gate("unbufferizable tensor op", app);
148 }
149 // Lams passed inside a tensor op's curry chain (combiners, affine index maps, schedule
150 // nests) are element-level — TRANSITIVELY, including their local helper lams: e.g. the
151 // loop-vector prefixes «r; I32» inside a schedule nest must not be mistaken for value
152 // tensors of a recorded «r; I32» tensor type.
153 if (is_tensor_op(app)) {
154 unique_queue<DefSet> wl3;
155 for (const App* a = app; a; a = a->callee()->isa<App>())
156 wl3.push(a->arg());
157 while (!wl3.empty()) {
158 auto d = wl3.pop();
159 if (auto k = d->isa_mut<Lam>()) op_args_.emplace(k);
160 for (auto op : d->ops())
161 if (op) wl3.push(op);
162 }
163 }
164 }
165
166 for (auto op : def->ops())
167 push(op);
168 push(def->type());
169 }
170
171 for (auto mut : old_world().externals().muts())
172 if (auto lam = mut->isa_mut<Lam>(); lam && is_tensor_fn(lam)) tensor_fns_.emplace(lam);
173 // No tensor boundaries AND no tensor ops: nothing for the sweeps below to check.
174 // (Ops without boundaries still bufferize: their value-world operands are materialized.)
175 if (tensor_fns_.empty() && !ops_seen_) return;
176
177 // Higher-order bufferized functions: a continuation parameter whose domain itself contains a Pi would
178 // need boundary conversion inside nested continuation types, which `conv_boundary` does not perform.
179 for (auto old_fn : tensor_fns_) {
180 auto dom = old_fn->type()->dom();
181 auto n = dom->num_projs();
182 for (size_t i = 0; i != n; ++i)
183 if (auto pi = Pi::isa_cn(dom->proj(n, i)); pi && contains_pi(pi->dom()))
184 return gate("higher-order bufferized function", old_fn);
185 }
186
187 // Second sweep: shapes the conversion cannot adapt — the value-semantics path lowers everything instead.
188 unique_queue<DefSet> wl2;
189 for (auto mut : old_world().externals().muts())
190 if (mut) wl2.push(mut);
191 while (!wl2.empty()) {
192 auto def = wl2.pop();
193
194 for (auto op : def->ops()) {
195 if (!op) continue;
196 if (auto fn = op->isa_mut<Lam>()) {
197 // A bufferized function referenced as a value (not as the callee of a call, and not the
198 // binder back-reference of its own variable) cannot be adapted.
199 if (tensor_fns_.contains(fn))
200 if (!(def->isa<App>() && def->as<App>()->callee() == op) && !def->isa<Var>())
201 return gate("bufferized function used as a value", fn);
202 if (!tensor_fns_.contains(fn) && !op_args_.contains(fn) && mentions_tensor(fn->type()->dom())) {
203 // A tensor-typed function the conversion does not rewrite: an unset external keeps its
204 // value ABI, a direct-style local cannot be rebuilt as a continuation — either way a
205 // bufferized caller would pass a buffer against a value-array signature.
206 if (fn->is_external() || !Pi::isa_cn(fn->type()))
207 return gate("unconvertible tensor-typed function", fn);
208 }
209 }
210 wl2.push(op);
211 }
212 if (def->type()) wl2.push(def->type());
213 }

216void LowerToMem::start() {
217 collect_tensor_types(); // hard-errors on shapes the conversion cannot handle
218 // Nothing tensor-related in the program: skip the whole-world rebuild entirely.
219 if (tensor_fns_.empty() && !ops_seen_) return;
220 RWPhase::start();
221 // Every fresh-memory continuation must have found an enclosing lam to be chained into.
222 assert(pending_.empty());

232 }

256void LowerToMem::wrap_fresh_mem(Lam* new_lam) {
257 auto& w = new_world();
258 auto filter = new_lam->filter();
259 auto body = new_lam->body();
260 new_lam->unset();
261 // The last-minted continuation carries the original body; the lam ends up requesting the first memory.
262 for (auto k : pending_ | std::views::reverse) {
263 k->set(true, body); // filter `tt`: k vanishes as soon as AddMem substitutes the real memory
264 body = w.app(w.annex<mem::fresh>(), w.tuple({w.lit_nat_0(), k}));
265 }
266 new_lam->set(filter, body);
267}

269const Def* LowerToMem::rewrite(const Def* old_def) {
270 // An op lowering that consumes a fresh memory references its receiving continuation's var (see
271 // fresh_mem). The global memo would share such a lowering with every other function that mentions the
272 // same (closed) old op - where that var would dangle - so these ops are memoized per enclosing lam.
273 if (!is_bootstrapping()) {
274 if (auto app = old_def->isa<App>(); app && wants_fresh_mem(app)) {
275 if (auto i = fresh_memo_.find(app); i != fresh_memo_.end()) return i->second;
276 auto new_def = rewrite_imm_App(app);
277 fresh_memo_.emplace(app, new_def);
278 return new_def;
279 }
280 }
281 return RWPhase::rewrite(old_def);
282}

311const Def* LowerToMem::rewrite_mut_Lam(Lam* lam) {
312 if (is_bootstrapping()) return RWPhase::rewrite_mut_Lam(lam);
313
314 // Scope the fresh-memory bookkeeping: ops lowered while this body is rewritten mint their receiving
315 // continuations into pending_, which are chained in front of the finished body. Nested lams anchor
316 // their own requests (and their own per-lam op memo).
317 auto p = Restore(pending_, {});
318 auto m = Restore(fresh_memo_, {});
319 auto new_def = conv_mut_Lam(lam);
320 if (!pending_.empty()) wrap_fresh_mem(new_def->as_mut<Lam>());
321 return new_def;

324const Def* LowerToMem::conv_mut_Lam(Lam* lam) {
325 // A bufferized function: convert tensor-typed parameters to `%buffer.Buf`, including inside grouped
326 // sigma parameters and continuation domains. No memory is introduced here — AddMem does that.
327 if (is_tensor_fn(lam)) {
328 auto& w = new_world();
329 auto dom = lam->type()->dom();
330 auto n = dom->num_projs();
331
332 DefVec doms(n);
333 for (size_t i = 0; i != n; ++i) {
334 auto d = dom->proj(n, i);
335 if (auto pi = Pi::isa_cn(d))
336 doms[i] = w.cn(conv_boundary(pi->dom()));
337 else
338 doms[i] = conv_boundary(d);
339 }
340
341 auto new_lam = w.mut_con(doms)->set(lam->dbg_key());
342 map(lam, new_lam);
343 if (lam->num_vars() != 0) map(lam->var(), new_lam->var());
344 new_lam->set(rewrite(lam->filter()), rewrite(lam->body()));
345 return new_lam;
346 }
347
348 // A local continuation carrying tensor types (a return continuation of a bufferized call, a join point,
349 // an error continuation): convert its domain the same way. This is context-independent, so the order in
350 // which references reach it does not matter.
351 if (!lam->is_external() && lam->is_set() && !op_args_.contains(lam)) {
352 if (auto pi = Pi::isa_cn(lam->type()); pi && mentions_tensor(pi->dom())) {
353 auto& w = new_world();
354 auto new_lam = w.mut_con(conv_boundary(pi->dom()))->set(lam->dbg_key());
355 map(lam, new_lam);
356 if (lam->num_vars() != 0) map(lam->var(), new_lam->var());
357 new_lam->set(rewrite(lam->filter()), rewrite(lam->body()));
358 return new_lam;
359 }
360 }
361
362 return RWPhase::rewrite_mut_Lam(lam);
363}

365const Def* LowerToMem::rewrite_imm_App(const App* app) {
366 if (is_bootstrapping()) return RWPhase::rewrite_imm_App(app);
367 // A `%tensor.if_static` still stuck at lowering time guards a runtime value: residualize to
368 // its dynamic branch.
369 if (Axm::isa<tensor::if_static>(app)) return rewrite(app->arg(3, 2));
370 if (Axm::isa<tensor::get>(app)) return lower_get(app);
371 if (Axm::isa<tensor::set>(app)) return lower_set(app);
372 if (Axm::isa<tensor::splat>(app)) return lower_splat(app);
373 if (Axm::isa<tensor::broadcast>(app)) return lower_broadcast(app);
374 if (Axm::isa<tensor::map_reduce_post>(app)) return lower_map_reduce(app);
375 if (Axm::isa<tensor::pad>(app)) return lower_pad(app);
376 if (Axm::isa<tensor::concat>(app)) return lower_concat(app);
377
378 // Call of a bufferized function: adapt the call site.
379 if (auto callee = app->callee()->isa_mut<Lam>(); callee && tensor_fns_.contains(callee))
380 return lower_call(app, callee);
381
382 // Call of a converted continuation (a local lam or a parameter var whose domain mentions a tensor):
383 // materialize value-world tensor arguments into buffers. Element-level lams (op_args_) — and their
384 // continuation PARAMETERS (e.g. a schedule nest's `cell`, whose «r; I32» loop vector may look like
385 // a recorded tensor type) — keep value ABI.
386 if (auto pi = Pi::isa_cn(app->callee()->type()); pi && mentions_tensor(pi->dom())) {
387 auto elementwise = [&](const Def* callee) {
388 if (auto lam = callee->isa_mut<Lam>()) return op_args_.contains(lam);
389 for (auto d = callee; d;) {
390 if (auto ex = d->isa<Extract>()) {
391 d = ex->tuple();
392 continue;
393 }
394 if (auto var = d->isa<Var>())
395 if (auto lam = var->binder()->isa_mut<Lam>()) return op_args_.contains(lam);
396 break;
397 }
398 return false;
399 };
400 if (elementwise(app->callee())) return RWPhase::rewrite_imm_App(app);
401 auto& w = new_world();
402 return w.app(rewrite(app->callee()), materialize(pi->dom(), app->arg()));
403 }
404
405 return RWPhase::rewrite_imm_App(app);
406}

408const Def* LowerToMem::lower_call(const App* app, Lam* old_callee) {
409 auto& w = new_world();
410 auto new_callee = rewrite(old_callee);
411 auto dom = old_callee->type()->dom();
412 auto n = dom->num_projs();
413
414 DefVec args(n);
415 for (size_t i = 0; i != n; ++i) {
416 auto d = dom->proj(n, i);
417 auto a = app->arg()->proj(n, i);
418 // Continuations pass through: their domains are converted by `rewrite_mut_Lam` to exactly the
419 // domain the callee's new signature expects.
420 args[i] = Pi::isa_cn(d) ? rewrite(a) : materialize(d, a);
421 }
422 return w.app(new_callee, w.tuple(args));
423}

425const Def* LowerToMem::splat_buffer(const Def* arr_ty, const Def* scalar) {
426 // `%buffer.lit` sets every element to `scalar`; `%btensor.lower_map_reduce` fills it with a loop rather
427 // than storing a monolithic literal array (which the LLVM backend cannot digest for large shapes).
428 auto [bro, bso, boT] = Axm::isa<buffer::Buf>(buf_of(arr_ty))->args<3>();
429 auto [m, out] = buffer::op_lit(bro, bso, boT, bot_mem(), scalar)->projs<2>();
430 return out;
431}

433const Def* LowerToMem::materialize(const Def* old_ty, const Def* old_arg) {
434 auto& w = new_world();
435 if (tensor_ty_.contains(old_ty)) {
436 // A constant splat `‹s; c›` (e.g. a learning-rate or bias literal): a `%buffer.init` would store the
437 // whole array as one giant LLVM constant. Emit `%buffer.lit` (lowered to a fill loop) instead.
438 if (auto c = splat_scalar(old_arg)) return splat_buffer(old_ty, rewrite(c));
439 auto v = rewrite(old_arg);
440 if (Axm::isa<buffer::Buf>(v->type())) return v; // already a buffer
441 auto [br, bs, bT] = Axm::isa<buffer::Buf>(buf_of(old_ty))->args<3>();
442 auto [m, buf] = buffer::op_init(br, bs, bT, bot_mem(), v)->projs<2>();
443 return buf;
444 }
445 if (auto sig = old_ty->isa_imm<Sigma>(); sig && mentions_tensor(sig)) {
446 auto n = sig->num_ops();
447 DefVec ops(n);
448 for (size_t i = 0; i != n; ++i)
449 ops[i] = materialize(sig->op(i), old_arg->proj(n, i));
450 return w.tuple(ops);
451 }
452 return rewrite(old_arg);
453}

455const Def* LowerToMem::lower_get(const App* app) {
456 auto c = rewrite(app->callee())->as<App>();
457 auto arg = rewrite(app->arg());
458 auto [index, arr] = arg->projs<2>();
459 auto [T, r, s] = c->args<3>();
460 auto buf = Axm::isa<buffer::Buf>(arr->type());
461 // A `get` on a value-world tensor (e.g. a literal): materialize it into a buffer.
462 if (!buf) {
463 arr = materialize(app->arg()->proj(1)->type(), app->arg()->proj(1));
464 buf = Axm::isa<buffer::Buf>(arr->type());
465 if (!buf) return RWPhase::rewrite_imm_App(app); // not a recorded tensor type: leave it alone
466 }
467 auto [br, bs, bT] = buf->args<3>(); // actual (folded) buffer metadata
468
469 auto [m, v] = buffer::op_read(br, bs, bT, bot_mem(), arr, fold_index(s, index))->projs<2>();
470 return v; // the loaded value
471}

473const Def* LowerToMem::lower_set(const App* app) {
474 auto c = rewrite(app->callee())->as<App>();
475 auto arg = rewrite(app->arg());
476 auto [index, arr, x] = arg->projs<3>();
477 auto [T, r, s] = c->args<3>();
478 auto buf = Axm::isa<buffer::Buf>(arr->type());
479 // A `set` on a value-world tensor (e.g. a literal): materialize it into a buffer.
480 if (!buf) {
481 arr = materialize(app->arg()->proj(1)->type(), app->arg()->proj(1));
482 buf = Axm::isa<buffer::Buf>(arr->type());
483 if (!buf) return RWPhase::rewrite_imm_App(app); // not a recorded tensor type: leave it alone
484 }
485 auto [br, bs, bT] = buf->args<3>(); // actual (folded) buffer metadata
486 auto fidx = fold_index(s, index);
487
488 if (reuse_in_place(app)) {
489 auto [m, buf2] = buffer::op_write(br, bs, bT, fresh_mem(), arr, fidx, x)->projs<2>();
490 return buf2;
491 }
492
493 // AlwaysAllocate policy: allocate a fresh buffer, copy the source in, then write the element.
494 // This local chain is properly threaded; AddMem splices its placeholder root into the global chain.
495 auto [m1, q] = buffer::op_alloc(br, bs, bT, fresh_mem())->projs<2>();
496 auto m2 = buffer::op_copy(br, bs, bT, m1, q, arr);
497 auto [m3, out] = buffer::op_write(br, bs, bT, m2, q, fidx, x)->projs<2>();
498 return out;
499}

501const Def* LowerToMem::lower_splat(const App* app) {
502 auto value = app->arg()->proj(2, 1);
503
504 // MimIR folds rank-zero tensors and tensors whose literal dimensions are all
505 // one into their scalar element type. Such results have no tensor boundary to
506 // bufferize, so lower the splat directly to its scalar value.
507 if (!app->type()->isa<Arr>()) return rewrite(value);
508 return splat_buffer(app->type(), rewrite(value));
509}

511const Def* LowerToMem::lower_broadcast(const App* app) {
512 // Thin bufferization: map the SSA `tensor.broadcast` onto the buffer-world `btensor.broadcast`.
513 // The loop generation lives in the btensor plugin (`%btensor.lower_map_reduce`).
514 auto& w = new_world();
515 auto c = rewrite(app->callee())->as<App>();
516 auto arg = rewrite(app->arg());
517 auto [s_in, s_out, input] = arg->projs<3>();
518 auto [T, r] = c->args<2>();
519
520 // No-op broadcast (already normalized away in most cases).
521 if (s_in == s_out) return input;
522
523 // A broadcast of a value-world tensor (e.g. a literal): materialize it into a buffer.
524 auto in_buf = Axm::isa<buffer::Buf>(input->type());
525 if (!in_buf) {
526 input = materialize(app->arg()->proj(2)->type(), app->arg()->proj(2));
527 in_buf = Axm::isa<buffer::Buf>(input->type());
528 }
529
530 // Rank-0 source: an all-size-1 input shape folds to a plain scalar that is never recorded as a tensor
531 // type, so `materialize` leaves it as a value. Broadcasting a scalar fills every element with it.
532 if (!in_buf) return splat_buffer(app->type(), input);
533
534 // Actual (size-1-folded) input/output buffer shapes — `btensor.broadcast` is parameterised by them.
535 auto [bri, bsi, biT] = in_buf->args<3>();
536 auto [bro, bso, boT] = Axm::isa<buffer::Buf>(buf_of(app->type()))->args<3>();
537
538 // NB: no `w.call` here — `{ro, so}` (the *output* buffer shape) occur nowhere but in the result type, so
539 // there is no argument for inference to read them off.
540 auto op = w.annex<btensor::broadcast>();
541 op = w.app(op, w.tuple({T, bri, bsi, bro, bso, r}));
542 op = w.app(op, w.tuple({s_in, s_out}));
543 auto [m, out] = w.app(op, w.tuple({fresh_mem(), input}))->projs<2>();
544 return out;
545}

547const Def* LowerToMem::lower_map_reduce(const App* app) {
548 // Thin bufferization: map the SSA `tensor.map_reduce_post` onto the buffer-world `btensor.map_reduce_post`,
549 // reusing the (rewritten) meta. The loop generation lives in the btensor plugin (`%btensor.lower_map_reduce`).
550 auto& w = new_world();
551 auto c = rewrite(app->callee())->as<App>();
552 auto args = rewrite(app->arg()); // (is, post_is): the (bufferized) input buffers
553 auto [is, post_is] = args->projs<2>();
554
555 auto [nis_nps, meta, shapes, in_tys, comb_init, acc_out, accs] = c->uncurry_args<7>();
556 auto [nis, nps] = nis_nps->projs<2>();
557 auto [comb, init, post] = comb_init->projs<3>();
558
559 // Value-world tensor inputs (e.g. literals): materialize them into buffers. Re-tuple the lists:
560 // the generic rewrite rebuilds the argument tuple with its stale value-array type even when its
561 // elements were converted to buffers, which would not be assignable to the op's
562 // `«nis; %buffer.Buf …»` domain.
563 auto bufferize_list = [&](const Def* list, const Def* old_list, const Def* n) -> const Def* {
564 auto n_l = Lit::isa<u64>(n);
565 if (!n_l) return list;
566 DefVec ins(*n_l);
567 for (u64 i = 0; i < *n_l; ++i) {
568 ins[i] = list->proj(*n_l, i);
569 if (!Axm::isa<buffer::Buf>(ins[i]->type())) {
570 auto old_in = old_list->proj(*n_l, i);
571 ins[i] = materialize(old_in->type(), old_in);
572 if (!Axm::isa<buffer::Buf>(ins[i]->type())) return nullptr; // not a recorded tensor type
573 }
574 }
575 return w.tuple(ins);
576 };
577 auto [old_is, old_post_is] = app->arg()->projs<2>();
578 is = bufferize_list(is, old_is, nis);
579 post_is = bufferize_list(post_is, old_post_is, nps);
580 if (!is || !post_is) return RWPhase::rewrite_imm_App(app); // leave it alone
581
582 // Wrap the pure tensor combiner `Fn [To, «nis; Tis»] → To` into the mem-threaded combiner
583 // `Fn [%mem.M 0, To, «nis; Tis»] → [%mem.M 0, To]` that `btensor.map_reduce_post` expects.
584 auto mem_ty = w.call<mem::M>(0);
585 auto inner = comb->type()->as<Pi>()->dom()->proj(0); // [To, «nis; Tis»]
586 auto [cTo, ins_ty] = inner->projs<2>();
587 auto memcomb = w.mut_fun(w.sigma({mem_ty, cTo, ins_ty}), w.sigma({mem_ty, cTo}))->set("memComb");
588 auto [cm, cacc, cins] = memcomb->var(0_n)->projs<3>();
589 auto cret = memcomb->var(1);
590 auto after = w.mut_con(cTo)->set("afterComb");
591 after->app(true, cret, w.tuple({cm, after->var(0_n)}));
592 memcomb->set(true, w.app(comb, w.tuple({w.tuple({cacc, cins}), after})));
593
594 // Likewise wrap the pure epilogue `Fn [To, «nps; Tps»] → Tp` into
595 // `Fn [%mem.M 0, To, «nps; Tps»] → [%mem.M 0, Tp]`.
596 auto pTp = meta->proj(5, 1);
597 auto pin = post->type()->as<Pi>()->dom()->proj(2, 0); // [To, «nps; Tps»]
598 auto [pTo, pexts_ty] = pin->projs<2>();
599 auto mempost = w.mut_fun(w.sigma({mem_ty, pTo, pexts_ty}), w.sigma({mem_ty, pTp}))->set("memPost");
600 auto [pm, pacc, pexts] = mempost->var(0_n)->projs<3>();
601 auto pret = mempost->var(1);
602 auto pafter = w.mut_con(pTp)->set("afterPost");
603 pafter->app(true, pret, w.tuple({pm, pafter->var(0_n)}));
604 mempost->set(true, w.app(post, w.tuple({w.tuple({pacc, pexts}), pafter})));
605
606 // NB: no `w.call` here — the meta groups are *forwarded* from the tensor op on purpose. Leaving them to
607 // inference would re-derive `{Tis, Ris, Sis}` from the `%buffer.Buf` operands, whose literal size-1 axes are
608 // already folded away, so they would no longer agree with the logical shapes the loop generation iterates.
609 auto op = w.annex<btensor::map_reduce_post>();
610 op = w.app(op, nis_nps);
611 op = w.app(op, meta);
612 op = w.app(op, shapes);
613 op = w.app(op, in_tys);
614 op = w.app(op, w.tuple({memcomb, init, mempost}));
615 op = w.app(op, acc_out);
616 op = w.app(op, accs);
617 auto [m, out] = w.app(op, w.tuple({fresh_mem(), is, post_is}))->projs<2>();
618 return out;
619}

621const Def* LowerToMem::lower_pad(const App* app) {
622 // Thin bufferization: map the SSA `tensor.pad` onto the buffer-world `btensor.pad`.
623 // The loop generation lives in the btensor plugin (`%btensor.lower_map_reduce`).
624 auto& w = new_world();
625 auto c = rewrite(app->callee())->as<App>();
626 auto [input, value] = rewrite(app->arg())->projs<2>();
627
628 auto [Tr, s_in, params] = c->uncurry_args<3>();
629 auto [T, r] = Tr->projs<2>();
630 auto [mode, lo, hi] = params->projs<3>();
631
632 // A pad of a value-world tensor (e.g. a literal): materialize it into a buffer.
633 if (!Axm::isa<buffer::Buf>(input->type())) {
634 input = materialize(app->arg()->proj(0)->type(), app->arg()->proj(0));
635 if (!Axm::isa<buffer::Buf>(input->type()))
636 return RWPhase::rewrite_imm_App(app); // not a recorded tensor type: leave it alone
637 }
638
639 auto r_l = Lit::isa<u64>(r);
640 if (!r_l) return RWPhase::rewrite_imm_App(app);
641
642 // The LOGICAL output shape `s_out#d = lo#d + s_in#d + hi#d` — the loop generation iterates it, so it
643 // must keep size-1 axes (the result type's `Buf` folds them away and cannot be used here).
644 DefVec so(*r_l);
645 for (u64 d = 0; d < *r_l; ++d)
646 so[d] = w.call(core::nat::add, DefVec{w.call(core::nat::add, DefVec{lo->proj(*r_l, d), s_in->proj(*r_l, d)}),
647 hi->proj(*r_l, d)});
648 auto s_out = w.tuple(so);
649
650 // `{T, r}` is inferred: `s_in` pins `r`, and `input`'s `%buffer.Buf (r, s_in, T)` pins `T`.
651 auto [m, out] = w.call<btensor::pad>(s_in, Defs{mode, lo, hi}, s_out, Defs{fresh_mem(), input, value})->projs<2>();
652 return out;
653}

655const Def* LowerToMem::lower_concat(const App* app) {
656 // Thin bufferization: map the SSA `tensor.concat` onto the buffer-world `btensor.concat`.
657 // The loop generation lives in the btensor plugin (`%btensor.lower_map_reduce`).
658 auto& w = new_world();
659 auto c = rewrite(app->callee())->as<App>();
660 auto arg = rewrite(app->arg());
661
662 auto [TnisR, ax, Sis] = c->uncurry_args<3>();
663 auto [T, nis, r] = TnisR->projs<3>();
664
665 auto nis_l = Lit::isa<u64>(nis);
666 auto r_l = Lit::isa<u64>(r);
667 auto ax_l = Lit::isa<u64>(ax);
668 if (!nis_l || !r_l || !ax_l) return RWPhase::rewrite_imm_App(app);
669
670 // Value-world tensor inputs (e.g. literals): materialize them into buffers. Re-tuple the projected
671 // inputs so the tuple type is re-inferred from the converted elements (see `lower_map_reduce`).
672 DefVec ins(*nis_l);
673 for (u64 i = 0; i < *nis_l; ++i) {
674 ins[i] = arg->proj(*nis_l, i);
675 if (!Axm::isa<buffer::Buf>(ins[i]->type())) {
676 auto old_in = app->arg()->proj(*nis_l, i);
677 ins[i] = materialize(old_in->type(), old_in);
678 if (!Axm::isa<buffer::Buf>(ins[i]->type()))
679 return RWPhase::rewrite_imm_App(app); // not a recorded tensor type: leave it alone
680 }
681 }
682 auto inputs = w.tuple(ins);
683
684 // The LOGICAL output shape: the summed extent along `ax` (literal, gated in `collect_tensor_types`),
685 // the shared extents elsewhere — the loop generation iterates it, so it must keep size-1 axes (the
686 // result type's `Buf` folds them away and cannot be used here).
687 u64 sum_ax = 0;
688 for (u64 i = 0; i < *nis_l; ++i) {
689 auto e = Lit::isa<u64>(Sis->proj(*nis_l, i)->proj(*r_l, *ax_l));
690 if (!e) return RWPhase::rewrite_imm_App(app);
691 sum_ax += *e;
692 }
693 DefVec so(*r_l);
694 for (u64 d = 0; d < *r_l; ++d)
695 so[d] = d == *ax_l ? w.lit_nat(sum_ax) : Sis->proj(*nis_l, 0)->proj(*r_l, d);
696 auto s_out = w.tuple(so);
697
698 // NB: `{T, nis, r}` and `{Sis}` cannot be left to inference (hence no `w.call` here). `Sis` holds the
699 // *logical* per-input shapes, but the operands are `%buffer.Buf` handles with their literal size-1 axes
700 // already folded away — unifying `%buffer.Buf (r, Sis#i, T)` against a folded handle fails, because the
701 // left-hand side only folds once `Sis#i` is known. Passing the logical shapes makes both sides fold alike.
702 auto op = w.annex<btensor::concat>();
703 op = w.app(op, w.tuple({T, nis, r}));
704 op = w.app(op, ax);
705 op = w.app(op, Sis);
706 op = w.app(op, s_out);
707 auto [m, out] = w.app(op, w.tuple({fresh_mem(), inputs}))->projs<2>();
708 return out;
709}
