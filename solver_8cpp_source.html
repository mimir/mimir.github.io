37/// By value, as GCC fails to vectorize `std::min` over references to a class.
38Cost min(Cost lhs, Cost rhs) { return lhs.value() < rhs.value() ? lhs : rhs; }
39
40/// Only valid for normalized, hence non-negative, costs: their sum fits into 32 unsigned bits.
41Cost add_normalized(Cost lhs, Cost rhs) {
42 const auto sum = static_cast<std::uint32_t>(lhs.value()) + static_cast<std::uint32_t>(rhs.value());
43 return static_cast<Cost::Rep>(std::min(sum, static_cast<std::uint32_t>(Cost::inf().value())));
44}
45
46/// Only valid for normalized costs with `minimum <= value`; an inf minimum means the whole row was moved elsewhere.
47Cost sub_normalized(Cost value, Cost minimum) {
48 if (minimum.is_inf()) return 0;
49 if (value.is_inf()) return Cost::inf();
50 return static_cast<Cost::Rep>(std::min<int64_t>(int64_t{value.value()} - minimum.value(), Cost::inf().value()));
51}

53PBQP_DISPATCH void add_into(Cost* __restrict dst, const Cost* __restrict src, size_t count) {
54 for (size_t i = 0; i < count; ++i)
55 dst[i] = add_normalized(dst[i], src[i]);

58PBQP_DISPATCH void fold_row(Cost* __restrict dst, const Cost* __restrict row, Cost cost, size_t count) {
59 for (size_t i = 0; i < count; ++i)
60 dst[i] = min(dst[i], add_normalized(cost, row[i]));

96 Reducer(const Graph& graph, fe::View<size_t> rn_priority)
97 : rn_priority_(rn_priority) {
98 nodes_.resize(graph.num_nodes());
99 for (const auto n : graph.nodes()) {
100 const auto costs = graph.node_costs(n);
101 nodes_[n].costs.assign(costs.begin(), costs.end());
102 nodes_[n].original.resize(costs.size());

105 nodes_[n].alive = true;

108 && "solve: rn_priority does not cover every node");

123 /// Exact reductions first; RN takes the highest priority, then degree, then rn_score.
124 NodeId next() {
125 while (!dirty_.empty()) {
126 const NodeId n = dirty_.back();
127 dirty_.pop_back();
128 nodes_[n].dirty = false;
129 if (nodes_[n].alive) prune(n);
130 }
131 while (!low_.empty()) {
132 const NodeId n = low_.back();
133 low_.pop_back();
134 if (nodes_[n].alive) return n;
135 }
136 while (!high_.empty()) {
137 const auto candidate = fe::pop(high_);
138 const Node& node = nodes_[candidate.node];
139 if (node.alive && node.version == candidate.version) return candidate.node;
140 }
141 return Graph::npos;
142 }

144 void reduce(NodeId n, fe::Vector<Subst>& stack) {
145 if (degree(n) > 2 && options(n) > 1) {
146 bool merged = false;
147 while (merge_neighbour(n, stack))
148 merged = true;
149 if (merged) {
150 touch(n);
151 return;
152 }
153 }
154
155 stack.emplace_back(record(n));
156 const auto mark = scratch_.state();
157 switch (const auto reduction = snapshot(n, stack.back()); reduction.edges.size()) {
158 case 0: ++statistics_.r0; break;
159 case 1:
160 ++statistics_.r1;
161 fold_into_neighbour(reduction);
162 break;
163 case 2:
164 ++statistics_.r2;
165 contract(reduction);
166 break;
167 default:
168 ++(options(n) == 1 ? statistics_.single_option : statistics_.rn);
169 fold_locally_best_option(reduction);
170 break;
171 }
172 scratch_.deallocate(mark);
173 remove_node(n);
174 }

176 /// Exact for degree <= 2 nodes; RN nodes may still improve on their RN choice here.
177 [[nodiscard]] OptionId choose(const Subst& subst, const Choices& choices) const {
178 const auto original = [&](OptionId x) { return subst.original ? subst.original[x] : x; };
179
180 OptionId best = original(0);
181 Cost best_total = Cost::inf();
182 for (OptionId x = 0; x < subst.costs.size(); ++x) {
183 Cost total = subst.costs[x];
184 for (const Link& link : subst.links) {
185 if (total.is_inf()) break;
186 const OptionId chosen = choices[link.neighbour];
187 const OptionId y = link.neighbour_working ? link.neighbour_working[chosen] : chosen;
188 assert(y != Graph::npos && "choose: a neighbour picked an option it had already lost");
189 total += link.matrix[x, y];
190 }
191 if (total < best_total) {
192 best_total = total;
193 best = original(x);
194 }
195 }
196 return best;
197 }

252 /// Cost spread per incident edge: what a wrong RN decision is likely to cost.
253 [[nodiscard]] double rn_score(NodeId n) const {
254 const auto [low, high] = std::ranges::minmax(nodes_[n].costs);
255 return static_cast<double>(high.value() - low.value()) / static_cast<double>(std::max<size_t>(1, degree(n)));

258 /// Single-option nodes are reduced exactly at any degree, so they belong on the low worklist.
259 void touch(NodeId n) {
260 Node& node = nodes_[n];
261 ++node.version;
262 if (!node.dirty) dirty_.emplace_back(n);
263 node.dirty = true;
264 if (degree(n) <= 2 || options(n) == 1) {
265 if (!node.low) low_.emplace_back(n);
266 node.low = true;
267 } else {
268 high_.push(Candidate{.priority = rn_priority_.empty() ? 0 : rn_priority_[n],
269 .degree = degree(n),
270 .score = rn_score(n),
271 .node = n,
272 .version = node.version});
273 }

276 void normalize_node(NodeId n) {
277 Costs& costs = nodes_[n].costs;
278 const Cost minimum = std::accumulate(costs.begin(), costs.end(), Cost::inf(), min);
279 if (minimum == 0 || minimum.is_inf()) return;
280 for (auto& cost : costs)
281 cost = sub_normalized(cost, minimum);
282 }

284 /// Moves row and column minima into the node vectors and drops the edge if nothing is left.
285 void normalize_edge(EdgeId e) {
286 const NodeId src = edges_[e].src;
287 const NodeId dst = edges_[e].dst;
288 Costs& src_costs = nodes_[src].costs;
289 Costs& dst_costs = nodes_[dst].costs;
290 const size_t rows = src_costs.size();
291 const size_t cols = dst_costs.size();
292
293 for (size_t r = 0; r < rows; ++r) {
294 const Cost* const row = view(e).row(r);
295 const Cost minimum = std::accumulate(row, row + cols, Cost::inf(), min);
296 if (minimum == 0) continue;
297 src_costs[r] += minimum;
298 const MatrixSpan matrix = writable(e);
299 for (size_t c = 0; c < cols; ++c)
300 matrix[r, c] = sub_normalized(matrix[r, c], minimum);
301 }
302 const auto mark = scratch_.state();
303 Cost* const minima = scratch_.allocate<Cost>(cols);
304 std::fill_n(minima, cols, Cost::inf());
305 for (size_t r = 0; r < rows; ++r)
306 fold_row(minima, view(e).row(r), 0, cols);
307 if (std::ranges::any_of(fe::View<Cost>{minima, cols}, [](Cost minimum) { return minimum != 0; })) {
308 for (size_t c = 0; c < cols; ++c)
309 dst_costs[c] += minima[c];
310 const MatrixSpan matrix = writable(e);
311 for (size_t r = 0; r < rows; ++r)
312 for (size_t c = 0; c < cols; ++c)
313 matrix[r, c] = sub_normalized(matrix[r, c], minima[c]);
314 }
315 scratch_.deallocate(mark);
316 if (std::ranges::all_of(view(e), [](Cost cost) { return cost == 0; })) {
317 erase(nodes_[src].edges, e);
318 erase(nodes_[dst].edges, e);
319 ++statistics_.dropped_edges;
320 }
321 }

338 /// Deletes forbidden options, always keeping at least one.
339 void prune(NodeId n) {
340 bool changed = false;
341 for (OptionId a = 0; a < options(n) && options(n) > 1;) {
342 if (nodes_[n].costs[a].is_inf()) {

353 for (const auto e : incident)
354 normalize_edge(e);
355 normalize_node(n);

429 /// RI: c'_n(y) = c_n(y) + min_x ( c(x) + M(x, y) )
430 void fold_into_neighbour(const Reduction& reduction) {
431 const TrimmedEdge& edge = reduction.edges.front();
432 const size_t y_dim = edge.matrix.cols();
433
434 Cost* const best = scratch_.allocate<Cost>(y_dim);
435 std::fill_n(best, y_dim, Cost::inf());
436 for (size_t x = 0; x < reduction.costs.size(); ++x)
437 fold_row(best, edge.matrix.row(x), reduction.costs[x], y_dim);
438 add_into(nodes_[edge.neighbour].costs.data(), best, y_dim);
439 normalize_node(edge.neighbour);
440 }

442 /// RII: M'_mn(i, j) = min_x ( c(x) + M_m(x, i) + M_n(x, j) )
443 void contract(const Reduction& reduction) {
444 const TrimmedEdge* to_m = &reduction.edges[0];
445 const TrimmedEdge* to_n = &reduction.edges[1];
446 const EdgeId e = find_edge(to_m->neighbour, to_n->neighbour);
447 // fold_row fills whole rows, so they must be the rows of the edge's storage.
448 if (e != Graph::npos && edges_[e].src != to_m->neighbour) std::swap(to_m, to_n);
449 const size_t m_dim = to_m->matrix.cols();
450 const size_t n_dim = to_n->matrix.cols();
451
452 const auto state = arena_.state();
453 const MatrixSpan merged = allocate_matrix(arena_, m_dim, n_dim);
454 std::ranges::fill(merged, Cost::inf());
455 for (size_t x = 0; x < reduction.costs.size(); ++x) {
456 const Cost* const n_row = to_n->matrix.row(x);
457 for (size_t i = 0; i < m_dim; ++i)
458 fold_row(merged.row(i), n_row, add_normalized(reduction.costs[x], to_m->matrix[x, i]), n_dim);
459 }
460 add_to_edge(e, to_m->neighbour, to_n->neighbour, merged, state);
461 }

469 /// Adds `matrix`, the newest allocation in the arena since `state`, to `e`, the edge between m and n, if any.
470 /// Its storage is oriented like `e`, so it can become the edge's matrix.
471 void add_to_edge(EdgeId e, NodeId m, NodeId n, MatrixSpan matrix, fe::Arena::State state) {
472 assert((matrix.rows() == options(m) && matrix.cols() == options(n)) && "add_to_edge: matrix shape");
473 assert(((e == Graph::npos || edges_[e].src == m) ? matrix : matrix.transposed()).row_major()
474 && "add_to_edge: storage orientation");
475 if (e == Graph::npos) {
476 e = edges_.size();
477 link(m, n, nullptr, matrix.data());
478 } else if (edges_[e].owned == nullptr) {
479 add_into(matrix.data(), edges_[e].shared, matrix.size());
480 edges_[e].owned = matrix.data();
481 } else {
482 add_into(edges_[e].owned, matrix.data(), matrix.size());
483 arena_.deallocate(state);
484 }
485 normalize_edge(e);
486 normalize_node(m);

490 /// The option of `m` that each option of `n` forces, or null if some option of `n` leaves `m` a choice.
491 [[nodiscard]] const OptionId* implied_options(EdgeId e, NodeId n) {
492 const MatrixView matrix = view(e, n);
493 const Costs& m_costs = nodes_[other_end(e, n)].costs;
494 OptionId* const implied = scratch_.allocate<OptionId>(options(n));
495 std::fill_n(implied, options(n), Graph::npos);
496 for (OptionId i = 0; i < options(n); ++i) {
497 if (nodes_[n].costs[i].is_inf()) continue;
498 for (OptionId j = 0; j < matrix.cols(); ++j) {
499 if (add_normalized(matrix[i, j], m_costs[j]).is_inf()) continue;
500 if (implied[i] != Graph::npos) return nullptr;
501 implied[i] = j;
502 }
503 }
504 return implied;

507 /// RM (Buchwald et al. 2011): moves the other edges of a neighbour whose option `n` determines onto `n`.
508 bool merge_neighbour(NodeId n, fe::Vector<Subst>& stack) {
509 // add_to_edge may grow nodes_[n].edges and dangle this loop's iterator, so a merge must return immediately.
510 for (const auto e : nodes_[n].edges) {
511 const NodeId m = other_end(e, n);
512 if (degree(m) < 2) continue;
513 const auto mark = scratch_.state();
514 const OptionId* const implied = implied_options(e, n);
515 if (implied == nullptr) {
516 scratch_.deallocate(mark);
517 continue;
518 }
519
520 const Edges moved = nodes_[m].edges;
521 for (const auto f : moved) {
522 if (f == e) continue;
523 const NodeId w = other_end(f, m);
524 const MatrixView to_w = view(f, m);
525 const EdgeId to_n_w = find_edge(n, w);
526
527 const auto state = arena_.state();
528 const MatrixSpan delta = allocate_edge_matrix(to_n_w, n, w);
529 std::ranges::fill(delta, Cost{0});
530 for (OptionId i = 0; i < options(n); ++i)
531 if (implied[i] != Graph::npos)
532 for (OptionId j = 0; j < options(w); ++j)
533 delta[i, j] = to_w[implied[i], j];
534 erase(nodes_[m].edges, f);
535 erase(nodes_[w].edges, f);
536 add_to_edge(to_n_w, n, w, delta, state);
537 touch(w);
538 }
539 scratch_.deallocate(mark);
540 ++statistics_.rm;
541 reduce(m, stack);
542 return true;
543 }
544 return false;
545 }

547 /// RN: commit to the locally best option and charge its matrix rows to the neighbours.
548 void fold_locally_best_option(const Reduction& reduction) {
549 OptionId best_option = 0;
550 Cost best_total = Cost::inf();
551 for (OptionId x = 0; x < reduction.costs.size(); ++x) {
552 Cost total = reduction.costs[x];
553 for (const auto& edge : reduction.edges) {
554 const Costs& neighbour_costs = nodes_[edge.neighbour].costs;
555 Cost cheapest = Cost::inf();
556 for (size_t y = 0; y < edge.matrix.cols(); ++y)
557 cheapest = min(cheapest, add_normalized(neighbour_costs[y], edge.matrix[x, y]));
558 total += cheapest;
559 }
560 if (total < best_total) {
561 best_total = total;
562 best_option = x;
563 }
564 }
565 for (const auto& edge : reduction.edges) {
566 add_into(nodes_[edge.neighbour].costs.data(), edge.matrix.row(best_option), edge.matrix.cols());
567 normalize_node(edge.neighbour);
568 }
569 }

584Cost evaluate(const Graph& graph, fe::View<OptionId> choices) {
585 Cost total = 0;
586
587 for (const auto n : graph.nodes()) {
588 assert(n < choices.size() && "evaluate: choices is too short");
589 const auto costs = graph.node_costs(n);
590 assert(choices[n] < costs.size() && "evaluate: option index out of range");
591 total += costs[choices[n]];
592 }
593 for (const auto e : graph.edges()) {
594 const auto [src, dst] = graph.edge(e);
595 total += graph.edge_matrix(e)[choices[src], choices[dst]];
596 }
597 return total;
598}

600fe::Bitset conflicts(const Graph& graph, fe::View<OptionId> choices) {
601 fe::Bitset result;
602 for (const auto n : graph.nodes()) {
603 assert(n < choices.size() && "conflicts: choices is too short");
604 if (graph.node_costs(n)[choices[n]].is_inf()) result.set(n);
605 }
606 for (const auto e : graph.edges()) {
607 const auto [src, dst] = graph.edge(e);
608 if (graph.edge_matrix(e)[choices[src], choices[dst]].is_inf()) result.set(src).set(dst);

622Solution solve(const Graph& graph, fe::View<size_t> rn_priority) {
623 Reducer reducer{graph, rn_priority};
624 fe::Vector<Subst> stack;
625 stack.reserve(graph.num_nodes());
626
627 for (NodeId n = reducer.next(); n != Graph::npos; n = reducer.next())
628 reducer.reduce(n, stack);
629
630 Choices choices(graph.num_nodes(), 0);
631 for (const auto& subst : std::views::reverse(stack))
632 choices[subst.node] = reducer.choose(subst, choices);
633
634 const Cost cost = evaluate(graph, choices);
635 return finish(
636 graph,
637 Solution{.choices = std::move(choices), .cost = cost, .conflicts = {}, .statistics = reducer.statistics()});
638}

642/// Mixed-radix layout of the assignment space; digit n belongs to node n.
643struct Enumeration {
644 fe::Vector<size_t, 16> options;
645 size_t combinations = 1;
646};
647
648std::optional<Enumeration> make_enumeration(const Graph& graph, size_t max_combinations) {
649 Enumeration enumeration;
650 for (const auto n : graph.nodes())
651 enumeration.options.emplace_back(graph.num_options(n));
652 for (const auto options : enumeration.options) {
653 if (enumeration.combinations > max_combinations / options) return std::nullopt;
654 enumeration.combinations *= options;
655 }
656 return enumeration;
657}
658
659/// The smallest index wins ties so that the result does not depend on the partitioning.
660Solution search_range(const Graph& graph, const Enumeration& enumeration, size_t begin, size_t end) {
661 Choices choices(enumeration.options.size(), 0);
662 for (size_t n = 0, rest = begin; n < choices.size(); ++n) {
663 choices[n] = rest % enumeration.options[n];
664 rest /= enumeration.options[n];
665 }
666
667 auto best = Solution{.choices = choices, .cost = Cost::inf(), .conflicts = {}, .statistics = {}};
668 for (size_t index = begin; index < end; ++index) {
669 const auto cost = evaluate(graph, choices);
670 if (cost < best.cost) {
671 best.choices = choices;
672 best.cost = cost;
673 }
674 for (NodeId n = 0; n < choices.size(); ++n) {
675 if (++choices[n] < enumeration.options[n]) break;
676 choices[n] = 0;
677 }
678 }
679 return best;
680}

684std::optional<Solution> solve_exhaustive(const Graph& graph, size_t max_combinations, unsigned num_threads) {
685 const auto made = make_enumeration(graph, max_combinations);
686 if (!made) return std::nullopt;
687 const Enumeration& enumeration = *made;
688 if (enumeration.options.empty()) return Solution{};
689
690 unsigned requested = num_threads;
691 if (requested == 0) requested = std::thread::hardware_concurrency();
692 if (requested == 0) requested = 1;
693 const size_t workers = std::min<size_t>(requested, enumeration.combinations);
694
695 if (workers <= 1) return finish(graph, search_range(graph, enumeration, 0, enumeration.combinations));
696
697 fe::Vector<Solution> partials(workers);
698 {
699 fe::Vector<std::jthread> threads;
700 threads.reserve(workers);
701
702 const size_t base = enumeration.combinations / workers;
703 const size_t extra = enumeration.combinations % workers;
704 size_t begin = 0;
705 for (size_t i = 0; i < workers; ++i) {
706 const size_t end = begin + base + (i < extra ? 1 : 0);
707 threads.emplace_back([&graph, &enumeration, &partials, i, begin, end] {
708 partials[i] = search_range(graph, enumeration, begin, end);
709 });
710 begin = end;
711 }
712 }
713
714 // Ranges ascend, so the first minimum is the one with the smallest index.
715 return finish(graph, std::move(*std::ranges::min_element(partials, {}, &Solution::cost)));
716}
