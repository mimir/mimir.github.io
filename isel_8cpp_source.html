54bool depends(const Def* def, const Def* target, DefSet& done) {

128void ISel::append(const rules::Rewrite& rule, const Def* root, Alts& alts) {
129 auto swaps = rule.swaps;
130 auto copies = ankerl::unordered_dense::map<u32, u32>();
131 // Ascending subsets of `swaps`, so the rival of a swap comes first.
132 for (u32 swap = 0;; swap = ((swap | ~swaps) + 1) & swaps) {
133 auto rival = swap & rule.imm_swaps ? fe::lookup(copies, swap & ~rule.imm_swaps) : nullptr;
134 if (auto c = append(rule, root, swap, rival ? std::optional(*rival) : std::nullopt, alts)) copies[swap] = *c;
135 if (swap == swaps) break;
136 }
137}

139std::optional<u32>
140ISel::append(const rules::Rewrite& r, const Def* root, u32 swap, std::optional<u32> rival, Alts& alts) {
141 const auto& m = *r.match;
142 auto metas = std::array<const Def*, rules::Max_Metas>{};
143 auto refs = std::array<const Def*, rules::Max_Refs>{};
144 if (!m.fn(root, swap, metas.data(), refs.data())) return {};
145
146 auto alt = Alt();
147 alt.rule = &r;
148 alt.covers.resize(m.num_covers);
149 for (size_t j = 0; j != m.num_refs; ++j) {
150 auto info = rules::all_refs[m.refs + j];
151 auto ex = refs[j]->isa<Extract>();
152 auto cover = ex ? ex->tuple() : refs[j];
153 alt.covers[info.cover].def = cover;
154 if (info.kind == rules::Ref::Mem && uses(refs[j]) != 1) return {};
155 }
156
157 // A pure inner node that something else uses is looked through: the match takes its operands but not the node.
158 auto parent = std::array<u8, rules::Max_Covers>{};
159 auto owner = [&](u8 o) {
160 while (o != 0 && alt.covers[o - 1].through)
161 o = parent[o - 1];
162 return o;
163 };
164 auto node = [&](u8 o) { return o == 0 ? root : alt.covers[o - 1].def; };
165 auto demands = [&](u8 o) -> auto& { return o == 0 ? alt.demands : alt.covers[o - 1].demands; };
166 bool imm = false;
167 for (size_t j = 0; j != m.num_uses; ++j) {
168 auto u = rules::all_uses[m.uses + j];
169 auto o = owner(u.owner);
170 if (u.ref) {
171 auto def = refs[u.index];
172 auto c = rules::all_refs[m.refs + u.index].cover;
173 auto& cov = alt.covers[c];
174 parent[c] = o;
175 if (!root_of(cov.def)) return {};
176 if (sole(node(u.owner), def) != cov.def) {
177 if (mem_valued(cov.def) || c + 1u == r.maps) return {};
178 cov.through = true;
179 continue;
180 }
181 demands(o).emplace_back(Demand{def, Want::Reg, size_t(c) + 1});
182 } else if (auto w = want_of(r, u.index); w == Want::Lit) {
183 if (!Lit::isa(metas[u.index])) return {};
184 } else if (w != Want::Param) {
185 auto def = metas[u.index];
186 auto rw = resolve(def, w);
187 if (rules::is_literal(rw) && !literal(def, rw)) return {};
188 imm |= rw != w && rw != Want::Reg;
189 demands(o).emplace_back(Demand{def, rw});
190 }
191 }
192 for (size_t i = 0; i != r.num_metas; ++i)
193 if (r.tied >> i & 1 && overwritten(metas[i])) ++alt.copies;
194 // Without an immediate, such a swap only differs from its rival in which operand it overwrites.
195 if ((swap & r.imm_swaps) && !imm && !(rival && alt.copies < *rival)) return {};
196 if (!holds(alt)) return {};
197 alt.metas = DefVec(metas.begin(), metas.begin() + r.num_metas);
198 auto res = alt.copies;
199 alts.emplace_back(std::move(alt));
200 return res;
201}

203const Def* ISel::build(const Alt& alt) {
204 const auto& r = *alt.rule;
205 auto holes = DefVec(r.num_metas, [&](size_t i) {
206 auto w = want_of(r, i);
207 return rules::is_param(w) ? rewrite(alt.metas[i]) : hole(alt.metas[i], resolve(alt.metas[i], w));
208 });
209 auto res = r.build(new_world(), holes.data());
210 if (!r.maps) return res;
211 map(alt.covers[r.maps - 1].def, res);
212 return res->proj(2, 1);

230bool ISel::holds(const Alt& alt) {
231 // The replacement sits at the inner node whose memory it replaces, so the root's holes must be available there.
232 if (!alt.rule->maps) return true;
233 auto ld = alt.covers[alt.rule->maps - 1].def->as<App>();
234 auto bb = chain(ld->arg(2, 0));
235 if (!bb) return false;
236 // `done` is shared, since a Def that does not reach `ld` never will.
237 auto done = DefSet();
238 for (const auto& d : alt.demands)
239 if (!d.inner && (!d.def->in_scope(bb) || depends(d.def, ld, done))) return false;
240 return true;
241}

247bool ISel::analyze() {
248 for (auto mut : old_world().externals().muts())
249 count(mut);
250
251 auto defs = DefVec(counted_.begin(), counted_.end());
252 std::ranges::sort(defs, [](const Def* a, const Def* b) { return a->gid() < b->gid(); });
253
254 auto nodes = DefVec();
255 for (auto def : defs) {
256 auto alts = Alts();
257 match(def, alts);
258 if (alts.empty()) continue;
259 alts_[def] = std::move(alts);
260 nodes.emplace_back(def);
261 }
262
263 // Each inner node of a match becomes an alternative of the node it covers; equal ones are one class.
264 // A cover's operands follow it in preorder, so their classes are known before its own.
265 auto resolve = [](const Alt& alt, fe::Vector<Demand>& demands) {
266 for (auto& d : demands)
267 if (d.inner) d.inner = alt.covers[d.inner - 1].inner;
268 };
269 for (auto def : nodes)
270 for (auto& alt : alts_[def]) {
271 for (size_t j = alt.covers.size(); j-- != 0;) {
272 auto& cover = alt.covers[j];
273 if (cover.through) continue;
274 resolve(alt, cover.demands);
275 auto& inner = alts_.at(cover.def);
276 auto same = [&](const Alt& a) { return !a.rule && std::ranges::equal(a.demands, cover.demands); };
277 auto i = std::ranges::find_if(inner, same);
278 cover.inner = size_t(i - inner.begin()) + 1;
279 if (i == inner.end()) {
280 auto a = Alt();
281 a.demands = cover.demands;
282 inner.emplace_back(std::move(a));
283 }
284 }
285 resolve(alt, alt.demands);
286 }
287
288 auto graph = pbqp::Graph();
289 auto ids = DefMap<pbqp::NodeId>();
290 for (auto def : nodes) {
291 auto costs = pbqp::Costs();
292 for (const auto& alt : alts_[def]) {
293 auto c = alt.rule ? pbqp::Cost(alt.rule->cost + alt.copies * insn::Copy) : pbqp::Cost(0);
294 for (const auto& d : alt.demands)
295 if (!alts_.contains(key(d.def))) c += d.inner ? pbqp::Cost::inf() : leaf(d.def, d.want);
296 costs.emplace_back(c);
297 }
298 ids[def] = graph.add(costs);
299 }
300
301 for (auto def : nodes) {
302 const auto& alts = alts_[def];
303 auto keys = fe::Vector<DefVec>(alts.size());
304 auto operands = DefVec();
305 auto seen = DefSet();
306 for (size_t i = 0; i != alts.size(); ++i)
307 for (const auto& d : alts[i].demands) {
308 auto op = keys[i].emplace_back(key(d.def));
309 if (alts_.contains(op) && seen.emplace(op).second) operands.emplace_back(op);
310 }
311 for (auto op : operands) {
312 const auto& op_alts = alts_[op];
313 auto costs = pbqp::Matrix(alts.size(), op_alts.size(), [&](size_t i, size_t j) {
314 const auto& o = op_alts[j];
315 auto c = pbqp::Cost(0);
316 for (size_t k = 0; k != alts[i].demands.size(); ++k) {
317 if (keys[i][k] != op) continue;
318 const auto& d = alts[i].demands[k];
319 if (d.inner)
320 c += !o.rule && j + 1 == d.inner ? 0 : pbqp::Cost::inf();
321 else
322 c += o.rule ? conv(o.rule->type, d.want) : pbqp::Cost::inf();
323 }
324 return c;
325 });
326 if (std::ranges::any_of(costs, [](pbqp::Cost c) { return c != 0; }))
327 graph.add_edge_costs(ids[def], ids[op], costs);
328 }
329 }
330
331 auto fallback = pbqp::Choices(graph.num_nodes(), 0);
332 for (auto def : nodes) {
333 auto i = std::ranges::find_if(alts_[def], [](const Alt& a) { return a.rule && atomic(*a.rule); });
334 auto found = i != alts_[def].end();
335 fallback[ids[def]] = found ? size_t(i - alts_[def].begin()) : 0;
336#ifdef MIM_ENABLE_CHECKS
337 if (!found) fe::throwf("x86 isel: `{}` has no atomic rule", def);
338#endif
339 }
340#ifdef MIM_ENABLE_CHECKS
341 if (pbqp::evaluate(graph, fallback).is_inf()) fe::throwf("the atomic rules must cover the program");
342#endif
343 auto sol = pbqp::solve(graph);
344 log().d("x86 isel: {} nodes, {} edges, cost {}", graph.num_nodes(), graph.num_edges(), sol.cost.value());
345 // RN is a heuristic, but the atomic rules always remain a valid cover.
346 const auto& picked = sol.feasible() ? sol.choices : fallback;
347 if (auto d = arg_value(args(), "isel-dot")) {
348 auto out = Out(std::string(*d));
349 auto& os = *out.os();
350 for (auto def : nodes)
351 std::println(os, "// n{} = {}", ids[def], def);
352 pbqp::write_dot(os, graph, picked);
353 }
354 for (auto def : nodes) {
355 auto c = picked[ids[def]];
356 choices_[def] = c;
357 for (const auto& cover : alts_[def][c].covers)
358 if (!cover.through) roots_[cover.def] = def;
359 }
360 return false;
361}

418const Def* ISel::convert(const Def* x, Ty from, Want want) {
419 if (rules::is_literal(want)) return x;
420 auto& w = new_world();
421 if (Lit::isa(x) && (from == Ty::Reg || from == Ty::FReg)) from = Ty::Lit;
422 auto to = rules::chains.target(from, want).first;
423 for (auto t = from; t != to;) {
424 auto c = &rules::conversions[rules::chains.first[size_t(t)][size_t(to)]];
425 auto holes = x->projs(c->arity);
426 x = c->build(w, holes.data());
427 t = c->to;
428 }
429 return x;
