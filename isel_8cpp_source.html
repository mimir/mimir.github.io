208/// The continuation whose `mem.M` chain @p m is on.
209Lam* chain(const Def* m) {
210 for (;;) {
211 if (auto ex = m->isa<Extract>()) m = ex->tuple();
212 if (auto var = m->isa<Var>()) return var->binder()->isa<Lam>();
213 auto app = m->isa<App>();
214 if (!app) return nullptr;
215 auto arg = app->arg();
216 auto n = arg->num_projs();
217 auto next = (const Def*)nullptr;
218 for (size_t i = 0; i != n && !next; ++i)
219 if (auto a = arg->proj(n, i); Axm::isa<core::M>(a->type())) next = a;
220 if (!next) return nullptr;
221 m = next;
222 }
223}

226bool depends(const Def* def, const Def* target, DefSet& done) {

253/// The constant byte offset of the `mem.lea` @p lea, unless its index is dynamic.
254std::optional<nat_t> offset(const App* lea) {
255 auto [p, i] = Axm::as<mem::lea>(lea)->args<2>();
256 auto [pointee, _] = Axm::as<core::Ptr>(p->type())->args<2>();
257 auto l = Lit::isa(i);
258 if (pointee->isa<Arr>()) return l ? std::optional(*l * elem_size(lea)) : std::nullopt;
259 if (!l) i->blame(MIM_X86_BE "index `{}` into `{}` must be a literal", i, pointee).bail();
260 nat_t off = 0;
261 for (nat_t j = 0; j <= *l; ++j) {
262 auto field = pointee->proj(pointee->num_projs(), j);
263 auto align = trait(core::trait::align, field);
264 off = (off + align - 1) / align * align;
265 if (j != *l) off += trait(core::trait::size, field);
266 }
267 return off;
268}

290bool ISel::supports(flags_t insn, nat_t size) {
291 if (auto res = supports_.find({insn, size}); res != supports_.end()) return res->second;
292 auto& w = new_world();
293 auto pi = w.annex(insn)->type()->as<Pi>();
294 auto var = pi->has_var();
295 auto done = DefSet();
296 // The `refly.check` in its codomain states the sizes it supports.
297 auto cond = size_check(pi->codom(), done);
298 return supports_[{insn, size}] = !var || !cond || VarRewriter(var, w.lit_nat(size)).rewrite(cond) == w.lit_tt();
299}

306const Def* ISel::call_insn(const Insn& insn, Want want, const Def* a, const Def* b) {
307 auto& w = new_world();
308 auto call = [&](const Def* x, const Def* y) {
309 auto r = w.call(insn.id, Defs{reg(w, x), is_literal(want) ? y : reg(w, y)});
310 return insn.res < 0 ? r : r->proj(2, insn.res);
311 };
312 auto size = Idx::isa_lit(a->type());
313 if (!size || supports(insn.id, core::widen_idx(*size))) return call(a, b);
314 auto ext = [&](const Def* x) { return insn.sign ? w.call<movsx>(w.lit_i32(), reg(w, x)) : zext(w.lit_i32(), x); };
315 return w.call<trunc>(*size, call(ext(a), ext(b)));
316}

349template<class... S>
350constexpr Rule rule(Op op, Ty type, pbqp::Cost cost, std::string_view name, BuildFn build, S... slots) {
351 static_assert(sizeof...(S) <= Rule::Max_Slots);
352 return Rule{op, type, cost, name, build, Cond::None, u8(sizeof...(S)), {Slot(slots)...}};
353}

355/// The end of the subtree of slots that starts at @p i, or beyond `num_slots` if the slots are malformed.
356constexpr size_t skip(const Rule& r, size_t i) {
357 if (i >= r.num_slots) return r.num_slots + 1;
358 auto n = r.slots[i++].arity;
359 while (n-- != 0)
360 i = skip(r, i);
361 return i;
362}
363
364/// Are the two operand subtrees starting at @p i equal, so that swapping the operands yields no other match?
365constexpr bool symmetric(const Rule& r, size_t i) {
366 auto j = skip(r, i);
367 auto k = skip(r, j);
368 return std::equal(r.slots.begin() + i, r.slots.begin() + j, r.slots.begin() + j, r.slots.begin() + k);
369}
370
371constexpr bool well_formed(const Rule& r) {
372 size_t i = 0;
373 while (i < r.num_slots)
374 i = skip(r, i);
375 return i == r.num_slots;
376}

378/// An atomic rule has no inner nodes and only holes that every rule type can fill.
379constexpr bool atomic(const Rule& r) {
380 for (size_t i = 0; i != r.num_slots; ++i) {
381 auto s = r.slots[i];
382 if (!s.is_hole() || (s.want != Want::Reg && s.want != Want::Base && s.want != Want::Addr)) return false;
383 }
384 return true;
385}

585const Def* ISel::build_select(const Def* root, const Alt& alt) {
586 auto& w = new_world();

604std::optional<ISel::Node> ISel::decompose(const Def* def) const {
605 auto ptrn = pattern(def);
606 if (!ptrn) return {};
607 auto app = def->isa<App>();
608 auto node = [&](DefVec ops) { return Node{ptrn, ptrn->op, std::move(ops)}; };
609
610 switch (ptrn->tag) {
611 case Tag::Select: {
612 auto ex = def->as<Extract>();
613 auto index = ex->index();
614 if (!xsize(ex->type()) || Idx::isa_lit(index->type()) != 2 || Lit::isa(index)
615 || ex->tuple()->num_projs() != 2)
616 return {};
617 auto [x, y] = ex->tuple()->projs<2>();
618 return node({x, y, index});
619 }
620 case Tag::Lea: return node({app->arg(2, 0)});
621 case Tag::Load:
622 if (!xsize(app->type()->proj(2, 1))) return {};
623 return node({app->arg(2, 1)});
624 case Tag::Store: {
625 auto [_, p, x] = app->args<3>();
626 if (!xsize(x->type())) return {};
627 return node({p, x});
628 }
629 case Tag::Bin: {
630 auto [a, b] = app->args<2>();
631 return node({a, b});
632 }
633 case Tag::Unary: return node({app->arg()});
634 case Tag::Div: {
635 auto [x, y] = app->arg(2, 1)->projs<2>();
636 return node({x, y});
637 }
638 case Tag::None: return node({});
639 }
640 std::unreachable();
641}

656void ISel::descend(const Rule& rule,
657 const Def* root,
658 size_t i,
659 Alt alt,
660 fe::Vector<Frame> frames,
661 Frame frame,
662 const Node& node,
663 Alts& alts) {
664 if (node.ptrn->commutes && frame.ops.size() == 2 && !symmetric(rule, i)) {
665 auto swapped = frames;
666 swapped.emplace_back(Frame{frame.node, {frame.ops[1], frame.ops[0]}, 0, frame.cover});
667 frames.emplace_back(std::move(frame));
668 bind(rule, root, i, alt, std::move(frames), alts);
669 bind(rule, root, i, std::move(alt), std::move(swapped), alts);
670 } else {
671 frames.emplace_back(std::move(frame));
672 bind(rule, root, i, std::move(alt), std::move(frames), alts);
673 }
674}

676void ISel::bind(const Rule& rule, const Def* root, size_t i, Alt alt, fe::Vector<Frame> frames, Alts& alts) {
677 while (!frames.empty() && frames.back().next == frames.back().ops.size())
678 frames.pop_back();
679 if (frames.empty()) {
680 if (i == rule.num_slots && holds(root, alt)) alts.emplace_back(std::move(alt));
681 return;
682 }
683 if (i == rule.num_slots) return;
684
685 auto& f = frames.back();
686 auto def = f.ops[f.next++];
687 auto slot = rule.slots[i];
688 auto& demands = f.cover == size_t(-1) ? alt.demands : alt.covers[f.cover].demands;
689 if (slot.is_hole()) {
690 if (is_literal(slot.want) && !literal(def, slot.want)) return;
691 demands.emplace_back(Demand{def, slot.want});
692 return bind(rule, root, i + 1, std::move(alt), std::move(frames), alts);
693 }
694
695 auto app = sole(f, def);
696 auto node = app ? decompose(app) : std::nullopt;
697 if (!node || node->ops.size() != slot.arity) return;
698 if (slot.node == Op::Alu ? !has_mem_form(*node->ptrn, app, true) : node->op != slot.node) return;
699 auto c = alt.covers.size();
700 demands.emplace_back(Demand{def, Want::Reg, c + 1});
701 alt.covers.emplace_back(Cover{app, 0, {}});
702 descend(rule, root, i + 1, std::move(alt), std::move(frames), Frame{app, node->ops, 0, c}, *node, alts);
703}

705const App* ISel::sole(const Frame& parent, const Def* def) const {
706 if (auto app = parent.node->isa<App>(); !app || uses(app->arg()) != 1) return nullptr;
707 if (std::ranges::count(parent.ops, def) != 1) return nullptr;

709 return uses(def) == 1 ? def->isa<App>() : nullptr;

712bool ISel::holds(const Def* root, const Alt& alt) {
713 switch (alt.rule->cond) {
714 case Cond::None: return true;
715 case Cond::Same:
716 case Cond::Narrow:
717 case Cond::Widen: {
718 auto src = new_size(alt.demands[0].def->type());
719 auto dst = new_size(root->type());
720 if (!src || !dst) return false;
721 if (*src == *dst) return alt.rule->cond == Cond::Same;
722 return alt.rule->cond == (reg_width(*src) >= reg_width(*dst) ? Cond::Narrow : Cond::Widen);
723 }
724 case Cond::Scale: {
725 auto size = elem_size(root->as<App>());
726 return size == 1 || size == 2 || size == 4 || size == 8;
727 }
728 case Cond::Fold: {
729 if (!has_mem_form(*pattern(root), root, false)) return false;
730 // The replacement sits at the load, so the other operands must be available there.
731 auto ld = alt.covers.back().def->as<App>();
732 auto bb = chain(ld->arg(2, 0));
733 if (!bb) return false;
734 for (const auto& d : alt.demands) {
735 auto done = DefSet();
736 if (!d.inner && (!d.def->in_scope(bb) || depends(d.def, ld, done))) return false;
737 }
738 return true;
739 }
740 case Cond::Rmw: {
741 // The load's memory has no other user, so nothing lies between load and store.
742 auto [m, p, _] = root->as<App>()->args<3>();
743 auto ld = alt.covers.back().def->as<App>();
744 return ld->arg(2, 1) == p && ld->proj(2, 0) == m && uses(m) == 1;
745 }
746 }
747 std::unreachable();
748}

754bool ISel::analyze() {
755#ifdef MIM_ENABLE_CHECKS
756 check_rules();
757#endif
758 for (auto mut : old_world().externals().muts())
759 count(mut);
760
761 auto defs = DefVec(counted_.begin(), counted_.end());
762 std::ranges::sort(defs, [](const Def* a, const Def* b) { return a->gid() < b->gid(); });
763
764 auto nodes = DefVec();
765 for (auto def : defs) {
766 auto alts = Alts();
767 match(def, alts);
768 if (alts.empty()) continue;
769 alts_[def] = std::move(alts);
770 nodes.emplace_back(def);
771 }
772
773 // Each inner node of a match becomes an alternative of the node it covers; equal ones are one class.
774 // A cover's operands follow it in preorder, so their classes are known before its own.
775 auto resolve = [](const Alt& alt, fe::Vector<Demand>& demands) {
776 for (auto& d : demands)
777 if (d.inner) d.inner = alt.covers[d.inner - 1].inner;
778 };
779 for (auto def : nodes)
780 for (auto& alt : alts_[def]) {
781 for (size_t j = alt.covers.size(); j-- != 0;) {
782 auto& cover = alt.covers[j];
783 resolve(alt, cover.demands);
784 auto& inner = alts_.at(cover.def);
785 auto same = [&](const Alt& a) { return !a.rule && std::ranges::equal(a.demands, cover.demands); };
786 auto i = std::ranges::find_if(inner, same);
787 cover.inner = size_t(i - inner.begin()) + 1;
788 if (i == inner.end()) {
789 auto a = Alt();
790 a.demands = cover.demands;
791 inner.emplace_back(std::move(a));
792 }
793 }
794 resolve(alt, alt.demands);
795 }
796
797 auto graph = pbqp::Graph();
798 auto ids = DefMap<pbqp::NodeId>();
799 for (auto def : nodes) {
800 auto costs = pbqp::Costs();
801 for (const auto& alt : alts_[def]) {
802 auto c = alt.cost;
803 for (const auto& d : alt.demands)
804 if (!alts_.contains(key(d.def))) c += d.inner ? pbqp::Cost::inf() : leaf(d.def, d.want);
805 costs.emplace_back(c);
806 }
807 ids[def] = graph.add(costs);
808 }
809
810 for (auto def : nodes) {
811 const auto& alts = alts_[def];
812 auto operands = DefVec();
813 for (const auto& alt : alts)
814 for (const auto& d : alt.demands)
815 if (auto op = key(d.def); alts_.contains(op) && std::ranges::find(operands, op) == operands.end())
816 operands.emplace_back(op);
817 for (auto op : operands) {
818 const auto& op_alts = alts_[op];
819 auto costs = pbqp::Matrix(alts.size(), op_alts.size(), [&](size_t i, size_t j) {
820 const auto& o = op_alts[j];
821 auto c = pbqp::Cost(0);
822 for (const auto& d : alts[i].demands) {
823 if (key(d.def) != op) continue;
824 if (d.inner)
825 c += !o.rule && j + 1 == d.inner ? 0 : pbqp::Cost::inf();
826 else
827 c += o.rule ? conv(o.rule->type, d.want) : pbqp::Cost::inf();
828 }
829 return c;
830 });
831 if (std::ranges::any_of(costs, [](pbqp::Cost c) { return c != 0; }))
832 graph.add_edge_costs(ids[def], ids[op], costs);
833 }
834 }
835
836 auto fallback = pbqp::Choices(graph.num_nodes(), 0);
837 for (auto def : nodes) {
838 auto i = std::ranges::find_if(alts_[def], [](const Alt& a) { return a.rule && atomic(*a.rule); });
839 auto found = i != alts_[def].end();
840 fallback[ids[def]] = found ? size_t(i - alts_[def].begin()) : 0;
841#ifdef MIM_ENABLE_CHECKS
842 if (!found) fe::throwf("x86 isel: `{}` has no atomic rule", def);
843#endif
844 }
845#ifdef MIM_ENABLE_CHECKS
846 if (pbqp::evaluate(graph, fallback).is_inf()) fe::throwf("the atomic rules must cover the program");
847#endif
848 auto sol = pbqp::solve(graph);
849 log().d("x86 isel: {} nodes, {} edges, cost {}", graph.num_nodes(), graph.num_edges(), sol.cost.value());
850 // RN is a heuristic, but the atomic rules always remain a valid cover.
851 const auto& picked = sol.feasible() ? sol.choices : fallback;
852 if (auto d = arg_value(args(), "isel-dot")) {
853 auto out = Out(std::string(*d));
854 auto& os = *out.os();
855 for (auto def : nodes)
856 std::println(os, "// n{} = {}", ids[def], def);
857 pbqp::write_dot(os, graph, picked);
858 }
859 for (auto def : nodes) {
860 auto c = picked[ids[def]];
861 choices_[def] = c;
862 for (const auto& cover : alts_[def][c].covers)
863 roots_[cover.def] = def;
864 }
865 return false;
866}

869void ISel::check_rules() const {
870 for (const auto& r : rules_) {
871 if (conv(r.type, Want::Reg).is_inf())
872 fe::throwf("x86 isel: rule `{}` yields a value that does not convert to a register", r.name);
873 for (size_t i = 0; i != r.num_slots; ++i) {
874 auto s = r.slots[i];
875 if (!s.is_hole()) {
876 if (s.node != Op::Alu && !has_atomic(s.node))
877 fe::throwf("x86 isel: `{}` folds `{}`, but no atomic rule selects it", r.name, op_name(s.node));
878 for (const auto& p : patterns_)
879 if (s.node == Op::Alu && p.insn.mem == Mem::Rmw && !has_atomic(p.op))
880 fe::throwf("x86 isel: `{}` folds `{}`, but no atomic rule selects it", r.name, op_name(p.op));
881 continue;
882 }
883 if (is_literal(s.want)) continue; // satisfied by a literal, not a rule type
884 bool ok = false;
885 for (auto ty : {Ty::Reg, Ty::Base, Ty::Index})
886 if (!conv(ty, s.want).is_inf()) ok = true;
887 if (!ok) fe::throwf("x86 isel: no rule type converts to a hole of `{}`", r.name);
888 }
889 }
890
891 for (const auto& r : rules_)
892 if (!has_atomic(r.op)) fe::throwf("x86 isel: no atomic rule for `{}`", op_name(r.op));
893}
