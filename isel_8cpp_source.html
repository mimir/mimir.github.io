114/// The x86 instruction for the `core` operation @p def if it has a memory-operand form (read-modify-write if @p rmw).
115std::optional<flags_t> alu_id(const Def* def, bool rmw) {
116 auto s = Idx::isa_lit(def->type());
117 if (!s || !is_reg_size(*s)) return {};
118 if (auto op = Axm::isa<core::wrap>(def)) {
119 switch (op.id()) {
120 case core::wrap::add: return Annex::Base<add>;
121 case core::wrap::sub: return Annex::Base<sub>;
122 // `imul` has neither an 8-bit nor a read-modify-write form.
123 case core::wrap::mul: return !rmw && reg_width(*s) >= 16 ? std::optional(Annex::Base<imul>) : std::nullopt;
124 default: return {};
125 }
126 } else if (auto op = Axm::isa<core::bit2>(def)) {
127 switch (op.id()) {
128 case core::bit2::and_: return Annex::Base<and_>;
129 case core::bit2::or_: return Annex::Base<or_>;
130 case core::bit2::xor_: return Annex::Base<xor_>;
131 default: return {};
132 }
133 }
134 return {};
135}

142/// The continuation whose `mem.M` chain @p m is on.
143Lam* chain(const Def* m) {
144 for (;;) {
145 if (auto ex = m->isa<Extract>()) m = ex->tuple();
146 if (auto var = m->isa<Var>()) return var->binder()->isa<Lam>();
147 auto app = m->isa<App>();
148 if (!app) return nullptr;
149 auto arg = app->arg();
150 auto n = arg->num_projs();
151 auto next = (const Def*)nullptr;
152 for (size_t i = 0; i != n && !next; ++i)
153 if (auto a = arg->proj(n, i); Axm::isa<core::M>(a->type())) next = a;
154 if (!next) return nullptr;
155 m = next;
156 }
157}

159/// Does @p def depend on @p target without going through a mutable?
160bool depends(const Def* def, const Def* target, DefSet& done) {
161 if (def == target) return true;
162 if (def->isa_mut() || !done.emplace(def).second) return false;
163 for (auto op : def->ops())
164 if (depends(op, target, done)) return true;
165 return false;
166}

184/// The constant byte offset of the `mem.lea` @p lea, unless its index is dynamic.
185std::optional<nat_t> offset(const App* lea) {
186 auto [p, i] = Axm::as<mem::lea>(lea)->args<2>();
187 auto [pointee, _] = Axm::as<core::Ptr>(p->type())->args<2>();
188 auto l = Lit::isa(i);
189 if (pointee->isa<Arr>()) return l ? std::optional(*l * elem_size(lea)) : std::nullopt;
190 if (!l) i->blame(MIM_X86_BE "index `{}` into `{}` must be a literal", i, pointee).bail();
191 nat_t off = 0;
192 for (nat_t j = 0; j <= *l; ++j) {
193 auto field = pointee->proj(pointee->num_projs(), j);
194 auto align = trait(core::trait::align, field);
195 off = (off + align - 1) / align * align;
196 if (j != *l) off += trait(core::trait::size, field);
197 }
198 return off;
199}

209ISel::wide(const Def* (*insn)(World&, const Def*, const Def*), const Def* type, bool sign, const Def* a, const Def* b) {
210 auto& w = new_world();
211 auto size = Idx::isa_lit(type);
212 if (!size || reg_width(*size) >= 16) return insn(w, a, b);
213 auto ext = [&](const Def* x) { return sign ? w.call<movsx>(w.lit_i32(), reg(w, x)) : zext(w.lit_i32(), x); };
214 return w.call<trunc>(*size, insn(w, ext(a), ext(b)));
215}

223/// Cost of converting a replacement of type @p from into a hole of type @p want.
224pbqp::Cost conv(Ty from, Want want) {
225 switch (want) {
226 case Want::Reg: return from == Ty::Reg ? 0 : 1;
227 case Want::Imm: return pbqp::Cost::inf();
228 case Want::Base: return from == Ty::Index ? 1 : 0;
229 case Want::Index: return from == Ty::Index ? 0 : pbqp::Cost::inf();
230 case Want::Addr: return 0;
231 }
232 std::unreachable();
233}

235/// Cost of the operand @p def, which is no node, in a hole of type @p want: a literal that is no immediate and not
236/// part of an address needs a `mov`.
237pbqp::Cost leaf(const Def* def, Want want) {
238 if (!Lit::isa(def)) return conv(Ty::Reg, want);
239 return want == Want::Reg ? 1 : want == Want::Index ? pbqp::Cost::inf() : 0;
240}

297const Def* ISel::build_bin(const Def* root, const Alt& alt) {
298 auto& w = new_world();
299 auto app = root->as<App>();
300 auto a = hole(alt.demands[0].def, alt.demands[0].want);
301 auto b = hole(alt.demands[1].def, alt.demands[1].want);
302 if (auto op = Axm::isa<core::wrap>(app)) {
303 switch (op.id()) {
304 case core::wrap::add: return bin<add>(w, a, b);
305 case core::wrap::sub: return bin<sub>(w, a, b);
306 // The low bits of a product do not depend on how the operands were extended.
307 case core::wrap::mul: return wide(bin<imul>, app->type(), false, a, b);
308 case core::wrap::shl: return shift<shl>(w, a, b);
309 }
310 } else if (auto op = Axm::isa<core::bit2>(app)) {
311 // clang-format off
312 switch (op.id()) {
313 case core::bit2::and_: return bin<and_>(w, a, b);
314 case core::bit2::or_: return bin<or_> (w, a, b);
315 case core::bit2::xor_: return bin<xor_>(w, a, b);
316 case core::bit2::nand: return bin<xor_>(w, bin<and_>(w, a, b), ones(w, a->type()));
317 case core::bit2::nor: return bin<xor_>(w, bin<or_ >(w, a, b), ones(w, a->type()));
318 case core::bit2::nxor: return bin<xor_>(w, bin<xor_>(w, a, b), ones(w, a->type()));
319 default: return nullptr;
320 }
321 // clang-format on
322 }
323 auto op = Axm::as<core::shr>(app);
324 return op.id() == core::shr::a ? shift<sar>(w, a, b) : shift<shr>(w, a, b);
325}

359const Def* ISel::build_unary(const Def* root, const Alt& alt) {
360 auto& w = new_world();
361 auto app = root->as<App>();
362 auto x = hole(alt.demands[0].def);
363 if (auto op = Axm::isa<core::conv>(app)) {
364 auto src = Idx::isa_lit(x->type());
365 auto dst = Idx::isa_lit(rewrite(app->type()));
366 if (!src || !dst) return nullptr;
367 if (*src == *dst) return x;
368 if (reg_width(*src) >= reg_width(*dst)) return w.call<trunc>(*dst, x);
369 return op.id() == core::conv::s ? w.call<movsx>(*dst, x) : w.call<movzx>(*dst, x);
370 } else if (Axm::isa<core::idx>(app)) {
371 return rewrite(app->type()) == w.type_i64() ? x : nullptr;
372 } else if (Axm::isa<core::idx2nat>(app)) {
373 return Idx::isa_lit(x->type()) == 0 ? x : zext(w.lit_i64(), x);
374 } else if (Axm::isa<core::bitcast>(app)) {
375 auto src = Idx::isa_lit(x->type());
376 auto dst = Idx::isa_lit(rewrite(app->type()));
377 if (x->type() == rewrite(app->type())) return x;
378 if (!src || !dst) return nullptr;
379 return reg_width(*src) >= reg_width(*dst) ? w.call<trunc>(*dst, x) : w.call<movzx>(*dst, x);
380 }
381 auto op = Axm::as<core::rem>(app);
382 auto d = w.lit(x->type(), *Lit::isa(op->decurry()->arg()));
383 return wide(divrem<div, 1>, x->type(), false, x, d);
384}

387 auto& w = new_world();

415const Def* ISel::build_select(const Def* root, const Alt& alt) {
416 auto& w = new_world();
417 auto size = *xsize(root->type());
418 auto x = hole(alt.demands[0].def);
419 auto y = hole(alt.demands[1].def);
420 auto c = hole(alt.demands[2].def);
421 auto code = 5_n;
422 auto fl = rr<test>(w, c, c);
423 if (auto app = c->isa<App>(); app && app->axm() && app->axm()->flags() == Annex::Base<x86::cc>)
424 code = *Lit::isa(app->callee()->as<App>()->arg()), fl = app->arg();
425 // `cmov` has no 8-bit form.
426 if (reg_width(size) < 16) {
427 auto ext = [&](const Def* v) { return reg(w, zext(w.lit_i32(), v)); };
428 return w.call<trunc>(size, w.call<cmovcc>(code, Defs{fl, ext(x), ext(y)}));
429 }
430 return w.call<cmovcc>(code, Defs{fl, x, y});
431}

455/// Appends the matches rooted at @p def to @p alts, an atomic one first.
456void ISel::match(const Def* def, Alts& alts) const {
457 if (auto ex = def->isa<Extract>()) {
458 auto index = ex->index();
459 if (xsize(ex->type()) && Idx::isa_lit(index->type()) == 2 && !Lit::isa(index)
460 && ex->tuple()->num_projs() == 2) {
461 auto [x, y] = ex->tuple()->projs<2>();
462 alts.emplace_back(root(select_, 1, {{x}, {y}, {index}}));
463 }
464 return;
465 }
466
467 auto app = def->isa<App>();
468 if (!app) return;
469
470 if (auto lea = Axm::isa<mem::lea>(app)) {
471 auto [p, i] = lea->args<2>();
472 if (offset(app)) {
473 alts.emplace_back(root(lea_off_base, 0, {{p, Want::Base}}));
474 alts.emplace_back(root(lea_off_index, 0, {{p, Want::Index}}));
475 } else {
476 auto size = elem_size(app);
477 auto cost = size == 1 || size == 2 || size == 4 || size == 8 ? 0 : 1;
478 alts.emplace_back(root(lea_index, cost, {{p, Want::Base}, {i}}));
479 }
480 } else if (auto ld = Axm::isa<mem::load>(app)) {
481 if (xsize(app->type()->proj(2, 1))) alts.emplace_back(root(load_, 1, {{ld->arg(2, 1), Want::Addr}}));
482 } else if (auto st = Axm::isa<mem::store>(app)) {
483 auto [m, p, x] = st->args<3>();
484 if (!xsize(x->type())) return;
485 alts.emplace_back(root(store_, 1, {{p, Want::Addr}, {x}}));
486 if (ri(x) == Want::Imm) alts.emplace_back(root(store_, 1, {{p, Want::Addr}, {x, Want::Imm}}));
487 if (auto u = x->isa<App>(); u && uses(u) == 1 && alu_id(u, true) && uses(u->arg()) == 1) {
488 auto [a, b] = u->args<2>();
489 for (auto [v, other] : {std::pair{a, b}, std::pair{b, a}}) {
490 if (v == b && !commutes(u)) break;
491 // The load's memory has no other user, so nothing lies between load and store.
492 if (auto ld = sole_load(v);
493 ld && v != other && ld->arg(2, 1) == p && ld->proj(2, 0) == m && uses(m) == 1) {
494 auto inner = (*alu_id(u, true) << 2) | (v == b ? 2 : 0) | (ri(other) == Want::Imm ? 1 : 0);
495 auto alt = root(store_mr, 1, {{p, Want::Addr}, {x, Want::Reg, inner}});
496 alt.covers.emplace_back(Cover{u, inner, {{v, Want::Reg, Load}, {other, ri(other)}}});
497 alt.covers.emplace_back(Cover{ld, Load, {{p, Want::Addr}}});
498 alts.emplace_back(std::move(alt));
499 }
500 }
501 }
502 } else if (Axm::isa<core::wrap>(app) || Axm::isa<core::bit2>(app) || Axm::isa<core::shr>(app)) {
503 auto w = Axm::isa<core::wrap>(app);
504 auto b2 = Axm::isa<core::bit2>(app);
505 auto [a, b] = app->args<2>();
506 auto sh = !b2 && (!w || w.id() == core::wrap::shl);
507 alts.emplace_back(root(bin_, 1, {{a}, {b}}));
508 if (Lit::isa(b) && (sh || fits_imm(b)))
509 alts.emplace_back(root(bin_, 1, {{a}, {b, Want::Imm}}));
510 else if (!sh && commutes(app) && ri(a) == Want::Imm)
511 alts.emplace_back(root(bin_, 1, {{b}, {a, Want::Imm}}));
512
513 if (!alu_id(app, false) || uses(app->arg()) != 1) return;
514 for (auto [v, other] : {std::pair{b, a}, std::pair{a, b}}) {
515 if (v == a && !commutes(app)) break;
516 auto ld = sole_load(v);
517 if (!ld || v == other) continue;
518 // The replacement sits at the load, so the other operand must be available there.
519 auto bb = chain(ld->arg(2, 0));
520 auto done = DefSet();
521 if (bb && other->in_scope(bb) && !depends(other, ld, done)) {
522 auto alt = root(bin_rm, 1, {{other}, {v, Want::Reg, Load}});
523 alt.covers.emplace_back(Cover{ld, Load, {{ld->arg(2, 1), Want::Addr}}});
524 alts.emplace_back(std::move(alt));
525 }
526 }
527 } else if (Axm::isa<core::icmp>(app) || Axm::isa<core::ncmp>(app)) {
528 auto ic = Axm::isa<core::icmp>(app);
529 auto nc = Axm::isa<core::ncmp>(app);
530 if (!(ic ? cc(ic.id()) : cc(nc.id()))) return;
531 auto [a, b] = app->args<2>();
532 alts.emplace_back(root(cmp_, 2, {{a}, {b}}));
533 if (ri(b) == Want::Imm) alts.emplace_back(root(cmp_, 2, {{a}, {b, Want::Imm}}));
534 } else if (auto op = Axm::isa<core::nat>(app)) {
535 auto [a, b] = app->args<2>();
536 auto imm = op.id() == core::nat::add || op.id() == core::nat::mul || op.id() == core::nat::sub;
537 alts.emplace_back(root(nat_, 1, {{a}, {b}}));
538 if (imm && ri(b) == Want::Imm)
539 alts.emplace_back(root(nat_, 1, {{a}, {b, Want::Imm}}));
540 else if (imm && op.id() != core::nat::sub && ri(a) == Want::Imm)
541 alts.emplace_back(root(nat_, 1, {{b}, {a, Want::Imm}}));
542 } else if (Axm::isa<core::conv>(app) || Axm::isa<core::idx>(app) || Axm::isa<core::idx2nat>(app)
543 || Axm::isa<core::bitcast>(app) || Axm::isa<core::rem>(app)) {
544 alts.emplace_back(root(unary, 1, {{app->arg()}}));
545 } else if (auto op = Axm::isa<core::div>(app)) {
546 auto [x, y] = op->arg(2, 1)->projs<2>();
547 alts.emplace_back(root(div_, 1, {{x}, {y}}));
548 } else if (Axm::isa<mem::mslot>(app)) {
549 alts.emplace_back(root(slot_, 1, {}));
550 } else if (Axm::isa<mem::assume>(app)) {
551 alts.emplace_back(root(assume_, 0, {}));
552 }
553}

559bool ISel::analyze() {
560 for (auto mut : old_world().externals().muts())
561 count(mut);
562
563 auto defs = DefVec(counted_.begin(), counted_.end());
564 std::ranges::sort(defs, [](const Def* a, const Def* b) { return a->gid() < b->gid(); });
565
566 auto nodes = DefVec();
567 for (auto def : defs) {
568 auto alts = Alts();
569 match(def, alts);
570 if (alts.empty()) continue;
571 assert(alts.front().covers.empty() && "the first alternative must be atomic");
572 alts_[def] = std::move(alts);
573 nodes.emplace_back(def);
574 }
575
576 // Each inner node of a match becomes an alternative of the node it covers; equal ones are one class.
577 for (auto def : nodes)
578 for (size_t i = 0, e = alts_[def].size(); i != e; ++i)
579 for (size_t j = 0, n = alts_[def][i].covers.size(); j != n; ++j) {
580 auto cover = alts_[def][i].covers[j];
581 auto& inner = alts_.at(cover.def);
582 if (std::ranges::none_of(inner, [&](const Alt& a) { return !a.rule && a.inner == cover.inner; })) {
583 auto alt = Alt();
584 alt.inner = cover.inner;
585 alt.demands = std::move(cover.demands);
586 inner.emplace_back(std::move(alt));
587 }
588 }
589
590 auto graph = pbqp::Graph();
591 auto ids = DefMap<pbqp::NodeId>();
592 for (auto def : nodes) {
593 auto costs = pbqp::Costs();
594 for (const auto& alt : alts_[def]) {
595 auto c = alt.cost;
596 for (const auto& d : alt.demands)
597 if (!alts_.contains(key(d.def))) c += d.inner ? pbqp::Cost::inf() : leaf(d.def, d.want);
598 costs.emplace_back(c);
599 }
600 ids[def] = graph.add(costs);
601 }
602
603 for (auto def : nodes) {
604 const auto& alts = alts_[def];
605 auto operands = DefVec();
606 for (const auto& alt : alts)
607 for (const auto& d : alt.demands)
608 if (auto op = key(d.def); alts_.contains(op) && std::ranges::find(operands, op) == operands.end())
609 operands.emplace_back(op);
610 for (auto op : operands) {
611 auto costs = pbqp::Costs();
612 for (const auto& alt : alts)
613 for (const auto& o : alts_[op]) {
614 pbqp::Cost c = 0;
615 for (const auto& d : alt.demands) {
616 if (key(d.def) != op) continue;
617 if (d.inner)
618 c += !o.rule && o.inner == d.inner ? 0 : pbqp::Cost::inf();
619 else
620 c += o.rule ? conv(o.rule->type, d.want) : pbqp::Cost::inf();
621 }
622 costs.emplace_back(c);
623 }
624 if (std::ranges::any_of(costs, [](pbqp::Cost c) { return c != 0; }))
625 graph.add_edge_costs(ids[def], ids[op], costs);
626 }
627 }
628
629 auto atomic = pbqp::Choices(graph.num_nodes(), 0);
630#ifdef MIM_ENABLE_CHECKS
631 if (pbqp::evaluate(graph, atomic).is_inf()) fe::throwf("the atomic rules must cover the program");
632#endif
633 auto sol = pbqp::solve(graph);
634 log().d("x86 isel: {} nodes, {} edges, cost {}", graph.num_nodes(), graph.num_edges(), sol.cost.value());
635 // RN is a heuristic, but the atomic rules always remain a valid cover.
636 const auto& picked = sol.feasible() ? sol.choices : atomic;
637 if (auto d = arg_value(args(), "isel-dot")) {
638 auto out = Out(std::string(*d));
639 auto& os = *out.os();
640 for (auto def : nodes)
641 std::println(os, "// n{} = {}", ids[def], def);
642 pbqp::write_dot(os, graph, picked);
643 }
644 for (auto def : nodes) {
645 auto c = picked[ids[def]];
646 choices_[def] = c;
647 for (const auto& cover : alts_[def][c].covers)
648 roots_[cover.def] = def;
649 }
650 return false;
651}

728 auto ix = Axm::isa<index>(addr);
729 auto [disp, base] = Axm::as<x86::addr>(ix ? ix->decurry()->arg() : addr)->args<2>();

739const Def* ISel::addr(Address a) {
740 auto& w = new_world();
741 if (auto l = Lit::isa(a.base)) a.disp += *l, a.base = w.lit_i64(0);
742 if (a.index)
743 if (auto l = Lit::isa(a.index)) a.disp += *l * a.scale, a.index = nullptr;
744 if (!fits_i32(a.disp)) a = {materialize(a)};
745 auto res = w.call<x86::addr>(Defs{w.lit_i32(u32(a.disp)), a.base});
746 return a.index ? w.call<index>(res, Defs{a.index, w.lit_i2(std::countr_zero(a.scale))}) : res;
747}

749const Def* ISel::materialize(Address a) {
750 auto& w = new_world();
751 if (!a.index && Lit::isa(a.base)) return w.lit_i64(*Lit::isa(a.base) + a.disp);
752 if (!fits_i32(a.disp)) a = {bin<add>(w, a.base, w.lit_i64(a.disp)), a.index, a.scale};
753 if (!a.index && a.disp == 0) return a.base;
754 return w.call<lea>(addr(a));
755}
