36/// Does @p def depend on @p target without going through a mutable?
37bool depends(const Def* def, const Def* target, DefSet& done) {
38 if (def == target) return true;
39 if (def->isa_mut() || !done.emplace(def).second) return false;
40 for (auto op : def->ops())
41 if (depends(op, target, done)) return true;

99void ISel::append(const rules::Rewrite& rule, const Def* root, Alts& alts) {
100 auto swaps = rule.swaps;
101 auto copies = ankerl::unordered_dense::map<u32, u32>();
102 // Ascending subsets of `swaps`, so the rival of a swap comes first.
103 for (u32 swap = 0;; swap = ((swap | ~swaps) + 1) & swaps) {
104 auto rival = swap & rule.imm_swaps ? fe::lookup(copies, swap & ~rule.imm_swaps) : nullptr;
105 if (auto c = append(rule, root, swap, rival ? std::optional(*rival) : std::nullopt, alts)) copies[swap] = *c;
106 if (swap == swaps) break;
107 }
108}

110std::optional<u32>
111ISel::append(const rules::Rewrite& r, const Def* root, u32 swap, std::optional<u32> rival, Alts& alts) {
112 const auto& m = *r.match;
113 auto metas = DefVec(table_.max_metas);
114 auto refs = DefVec(table_.max_refs);
115 if (!m.fn(root, swap, metas.data(), refs.data())) return {};
116
117 auto alt = Alt();
118 alt.rule = &r;
119 alt.covers.resize(m.num_covers);
120 for (size_t j = 0; j != m.num_refs; ++j) {
121 auto info = table_.refs[m.refs + j];
122 auto ex = refs[j]->isa<Extract>();
123 auto cover = ex ? ex->tuple() : refs[j];
124 alt.covers[info.cover].def = cover;
125 if (info.kind == rules::Ref::Mem && uses(refs[j]) != 1) return {};
126 }
127
128 // A pure inner node that something else uses is looked through: the match takes its operands but not the node.
129 auto parent = fe::Vector<u8>(table_.max_covers);
130 auto owner = [&](u8 o) {
131 while (o != 0 && alt.covers[o - 1].through)
132 o = parent[o - 1];
133 return o;
134 };
135 auto node = [&](u8 o) { return o == 0 ? root : alt.covers[o - 1].def; };
136 auto demands = [&](u8 o) -> auto& { return o == 0 ? alt.demands : alt.covers[o - 1].demands; };
137 bool imm = false;
138 for (size_t j = 0; j != m.num_uses; ++j) {
139 auto u = table_.uses[m.uses + j];
140 auto o = owner(u.owner);
141 if (u.ref) {
142 auto def = refs[u.index];
143 auto c = table_.refs[m.refs + u.index].cover;
144 auto& cov = alt.covers[c];
145 parent[c] = o;
146 if (!root_of(cov.def)) return {};
147 if (sole(node(u.owner), def) != cov.def) {
148 if (mem_valued(cov.def) || c + 1u == r.maps) return {};
149 cov.through = true;
150 continue;
151 }
152 demands(o).emplace_back(Demand{def, Want::Reg, size_t(c) + 1});
153 } else if (auto w = want_of(r, u.index); w == Want::Lit) {
154 if (!Lit::isa(metas[u.index])) return {};
155 } else if (w != Want::Param) {
156 auto def = metas[u.index];
157 auto rw = resolve(def, w);
158 if (rules::is_literal(rw) && !literal(def, rw)) return {};
159 imm |= rw != w && rw != Want::Reg;
160 demands(o).emplace_back(Demand{def, rw});
161 }
162 }
163 for (size_t i = 0; i != r.num_metas; ++i)
164 if (r.tied >> i & 1 && overwritten(metas[i])) ++alt.copies;
165 // Without an immediate, such a swap only differs from its rival in which operand it overwrites.
166 if ((swap & r.imm_swaps) && !imm && !(rival && alt.copies < *rival)) return {};
167 if (!holds(alt)) return {};
168 alt.metas = DefVec(metas.begin(), metas.begin() + r.num_metas);
169 auto res = alt.copies;
170 alts.emplace_back(std::move(alt));
171 return res;
172}

174const Def* ISel::build(const Alt& alt) {

176 auto holes = DefVec(r.num_metas, [&](size_t i) {

201bool ISel::holds(const Alt& alt) {
202 // The replacement sits at the inner node whose memory it replaces, so the root's holes must be available there.
203 if (!alt.rule->maps) return true;
204 auto ld = alt.covers[alt.rule->maps - 1].def->as<App>();
205 auto bb = chain(ld->arg(2, 0));
206 if (!bb) return false;
207 // `done` is shared, since a Def that does not reach `ld` never will.
208 auto done = DefSet();
209 for (const auto& d : alt.demands)
210 if (!d.inner && (!d.def->in_scope(bb) || depends(d.def, ld, done))) return false;
211 return true;
212}

218bool ISel::analyze() {
219 for (auto mut : old_world().externals().muts())
220 count(mut);
221
222 auto defs = DefVec(counted_.begin(), counted_.end());
223 std::ranges::sort(defs, [](const Def* a, const Def* b) { return a->gid() < b->gid(); });
224
225 auto nodes = DefVec();
226 for (auto def : defs) {
227 auto alts = Alts();
228 match(def, alts);
229 if (alts.empty()) continue;
230 alts_[def] = std::move(alts);
231 nodes.emplace_back(def);
232 }
233
234 // Each inner node of a match becomes an alternative of the node it covers; equal ones are one class.
235 // A cover's operands follow it in preorder, so their classes are known before its own.
236 auto resolve = [](const Alt& alt, fe::Vector<Demand>& demands) {
237 for (auto& d : demands)
238 if (d.inner) d.inner = alt.covers[d.inner - 1].inner;
239 };
240 for (auto def : nodes)
241 for (auto& alt : alts_[def]) {
242 for (size_t j = alt.covers.size(); j-- != 0;) {
243 auto& cover = alt.covers[j];
244 if (cover.through) continue;
245 resolve(alt, cover.demands);
246 auto& inner = alts_.at(cover.def);
247 auto same = [&](const Alt& a) { return !a.rule && std::ranges::equal(a.demands, cover.demands); };
248 auto i = std::ranges::find_if(inner, same);
249 cover.inner = size_t(i - inner.begin()) + 1;
250 if (i == inner.end()) {
251 auto a = Alt();
252 a.demands = cover.demands;
253 inner.emplace_back(std::move(a));
254 }
255 }
256 resolve(alt, alt.demands);
257 }
258
259 auto graph = pbqp::Graph();
260 auto ids = DefMap<pbqp::NodeId>();
261 for (auto def : nodes) {
262 auto costs = pbqp::Costs();
263 for (const auto& alt : alts_[def]) {
264 auto c = alt.rule ? pbqp::Cost(alt.rule->cost + alt.copies * rules::Copy) : pbqp::Cost(0);
265 for (const auto& d : alt.demands)
266 if (!alts_.contains(key(d.def))) c += d.inner ? pbqp::Cost::inf() : leaf(d.def, d.want);
267 costs.emplace_back(c);
268 }
269 ids[def] = graph.add(costs);
270 }
271
272 for (auto def : nodes) {
273 const auto& alts = alts_[def];
274 auto keys = fe::Vector<DefVec>(alts.size());
275 auto operands = DefVec();
276 auto seen = DefSet();
277 for (size_t i = 0; i != alts.size(); ++i)
278 for (const auto& d : alts[i].demands) {
279 auto op = keys[i].emplace_back(key(d.def));
280 if (alts_.contains(op) && seen.emplace(op).second) operands.emplace_back(op);
281 }
282 for (auto op : operands) {
283 const auto& op_alts = alts_[op];
284 auto costs = pbqp::Matrix(alts.size(), op_alts.size(), [&](size_t i, size_t j) {
285 const auto& o = op_alts[j];
286 auto c = pbqp::Cost(0);
287 for (size_t k = 0; k != alts[i].demands.size(); ++k) {
288 if (keys[i][k] != op) continue;
289 const auto& d = alts[i].demands[k];
290 if (d.inner)
291 c += !o.rule && j + 1 == d.inner ? 0 : pbqp::Cost::inf();
292 else
293 c += o.rule ? conv(o.rule->type, d.want) : pbqp::Cost::inf();
294 }
295 return c;
296 });
297 if (std::ranges::any_of(costs, [](pbqp::Cost c) { return c != 0; }))
298 graph.add_edge_costs(ids[def], ids[op], costs);
299 }
300 }
301
302 auto fallback = pbqp::Choices(graph.num_nodes(), 0);
303 for (auto def : nodes) {
304 auto i = std::ranges::find_if(alts_[def], [this](const Alt& a) { return a.rule && atomic(*a.rule); });
305 auto found = i != alts_[def].end();
306 fallback[ids[def]] = found ? size_t(i - alts_[def].begin()) : 0;
307#ifdef MIM_ENABLE_CHECKS
308 if (!found) fe::throwf("{} isel: `{}` has no atomic rule", target_.name(), def);
309#endif
310 }
311#ifdef MIM_ENABLE_CHECKS
312 if (pbqp::evaluate(graph, fallback).is_inf()) fe::throwf("the atomic rules must cover the program");
313#endif
314 auto sol = pbqp::solve(graph);
315 log().d("{} isel: {} nodes, {} edges, cost {}", target_.name(), graph.num_nodes(), graph.num_edges(),
316 sol.cost.value());
317 // RN is a heuristic, but the atomic rules always remain a valid cover.
318 const auto& picked = sol.feasible() ? sol.choices : fallback;
319 if (auto d = arg_value(args(), "isel-dot")) {
320 auto out = Out(std::string(*d));
321 auto& os = *out.os();
322 for (auto def : nodes)
323 std::println(os, "// n{} = {}", ids[def], def);
324 pbqp::write_dot(os, graph, picked);
325 }
326 for (auto def : nodes) {
327 auto c = picked[ids[def]];
328 choices_[def] = c;
329 for (const auto& cover : alts_[def][c].covers)
330 if (!cover.through) roots_[cover.def] = def;
331 }
332 return false;
333}
