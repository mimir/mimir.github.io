20/// If `e` reads coordinate `var#i` injectively, returns `i`:
21/// a plain extract, possibly strided (`%affine.semiop.mul` by a non-zero literal) and/or shifted
22/// (`%affine.op.add`/`sub` with a loop-invariant `%affine.constant` on the other side).
23/// Everything else — `mod`/`div`, sums of two loop indices (convolution windows), symbolic strides
24/// (which may be 0 at runtime) — yields nothing.
25static std::optional<u64> injective_coord(const Def* var, const Def* e) {
26 // A one-loop domain «1; %affine.index» collapses to a plain %affine.index, so the var itself is coordinate 0.
27 if (e == var) return 0;
28 if (auto ex = e->isa<Extract>(); ex && ex->tuple() == var) return Lit::isa<u64>(ex->index());
29
30 if (auto semiop = Axm::isa(affine::semiop::mul, e)) {
31 auto [x, c] = semiop->args<2>();
32 if (auto lit = Lit::isa<u64>(c); lit && *lit != 0) return injective_coord(var, x);
33 return {};
34 }
35
36 if (auto op = Axm::isa<affine::op>(e)) {
37 if (op.id() != affine::op::add && op.id() != affine::op::sub) return {};
38 auto [a, b] = op->args<2>();
39 bool a_const = static_cast<bool>(Axm::isa<affine::constant>(a));
40 bool b_const = static_cast<bool>(Axm::isa<affine::constant>(b));
41 if (a_const == b_const) return {};
42 return injective_coord(var, a_const ? b : a);
43 }
44
45 return {};
46}

48/// Checks that `map` provably reads through *every* loop index of its domain: its body is the
49/// identity, or each result coordinate is an injective read of one loop index (see `injective_coord`)
50/// and together they cover all indices.
51/// Such a map is injective over the iteration domain, so each element of the input behind it is read
52/// (and, after fusion, computed) *at most* once — strided/shifted reads skip elements entirely, so
53/// fusing behind them even drops computations the consumer never looks at.
54/// Anything not provably injective — a dropped loop index, a wrapped coordinate (`mod`, ...), a
55/// non-literal rank — is conservatively rejected.
56static bool reads_injectively(const Def* map) {
57 auto lam = map->isa_mut<Lam>();
58 if (!lam || !lam->is_set()) return false;
59 auto var = lam->var();
60 auto body = lam->body();
61 if (body == var) return true; // identity: every loop index passes through
62
63 auto n = Lit::isa<u64>(var->type()->arity());
64 if (!n) return false;
65
66 Vector<bool> used(*n, false);
67 auto mark = [&](const Def* elem) {
68 auto i = injective_coord(var, elem);
69 if (!i || *i >= *n) return false;
70 used[*i] = true;
71 return true;
72 };
73
74 if (auto tuple = body->isa<Tuple>()) {
75 for (auto elem : tuple->ops())
76 if (!mark(elem)) return false;
77 } else if (auto pack = body->isa<Pack>()) {
78 // A pack repeats a single coordinate, so it can only cover a one-loop domain.
79 if (!pack->is_set() || !mark(pack->body())) return false;
80 } else if (!mark(body)) {
81 return false;
82 }
83
84 return std::ranges::all_of(used, std::identity{});
85}

87/// `inner ∘ outer`: feeds the outer op's read coordinates for one input into the inner op's access map.
88static const Def* compose_map(World& w, const Def* inner, const Def* outer) {
89 auto dom = outer->type()->as<Pi>()->dom();
90 auto codom = inner->type()->as<Pi>()->codom();
91 auto lam = w.mut_lam(dom, codom)->set("fused_map");
92 lam->set(true, w.app(inner, w.app(outer, lam->var())));
93 return lam;
94}

106/// If `value` is a pure re-indexed read — a copy-combiner map_reduce (reshape/transpose/slice/
107/// flip/repeat lower to these) or a `%tensor.broadcast` — returns its source and access map, to be
108/// composed behind the consuming slot's map. Such reads perform no computation, so reading through
109/// them needs neither an injectivity gate nor a consumer count.
110static std::optional<ReadThrough> read_through(World& w, const Def* value, const Def* slot_map) {
111 if (auto pr = is_pure_read(value)) {
112 // A source whose axes are all size 1 type-collapses to a plain scalar («1; T» ≡ T): the
113 // fused op would carry an input the bufferization cannot represent - keep the copy instead.
114 if (!pr->src->type()->isa<Arr>()) return {};
115
116 return ReadThrough{pr->src, pr->map, pr->T, pr->R, pr->S};
117 }
118
119 if (auto bc = Axm::isa<tensor::broadcast>(value)) {
120 auto [T, r] = bc->callee()->as<App>()->args<2>();
121 auto [s_in, s_out, input] = bc->arg()->projs<3>();
122 auto r_l = Lit::isa<u64>(r);
123 if (!r_l) return {};
124 // See above: an all-size-1 source is a type-collapsed scalar - not readable through.
125 if (!input->type()->isa<Arr>()) return {};
126
127 // Source axis d reads o#d where the sizes agree and index 0 where the source axis is 1
128 // (expressed as `o#d · 0`, like `bid_map`). Bail on axes where neither is provable.
129 auto vec_ty = slot_map->type()->as<Pi>()->codom(); // «r; %affine.index»
130 auto lam = w.mut_lam(vec_ty, vec_ty)->set("bcast_map");
131 DefVec elems(*r_l);
132 for (u64 d = 0; d < *r_l; ++d) {
133 auto in_d = s_in->proj(*r_l, d);
134 auto o_d = lam->var()->proj(*r_l, d);
135 if (in_d == s_out->proj(*r_l, d))
136 elems[d] = o_d;
137 else if (auto l = Lit::isa<u64>(in_d); l && *l == 1)
138 elems[d] = w.call(affine::semiop::mul, Defs{o_d, w.lit_nat(0)});
139 else
140 return {};
141 }
142 lam->set(true, w.tuple(elems));
143
144 return ReadThrough{input, lam, T, r, s_in};
145 }
146
147 return {};
148}

150// Fuses an outer `tensor.map_reduce` with any number of its inputs — and, recursively, any
151// fusible inputs of those inputs — whenever each such input is itself a `tensor.map_reduce`
152// without reduction loops (`Rr = 0`) that writes its full loop domain through the identity output
153// map (`Sr = So`, `map_out = %affine.id`), *and* the outer reads that input injectively (its
154// access map uses every loop index, see `reads_injectively`). Reading such an inner tensor at a
155// position is then just a single call to the inner combination function, with each inner access
156// map composed behind the outer's access map for that input; injectivity guarantees that this
157// inlining performs each inner computation at most once, so fusion never duplicates work.
158//
159// Outer: map_reduce nis_o (To, Ro, Rr) (So, Sr) (Tis_o, Ris_o, Sis_o) (f_o, init_o) map_out maps_o is_o
160// Inner: map_reduce nis_k (To_k, Ro_k, 0) (So_k, So_k) (Tis_k, Ris_k, Sis_k) (f_k, init_k) id maps_k is_k
161// for every fusible input — possibly nested inside another fusible input
162//
163// Result: map_reduce nis_new (To, Ro, Rr) (So, Sr) (Tis_new, Ris_new, Sis_new) (f_new, init_o)
164// map_out maps_new is_new
165//
166// The collection phase walks the tree of fusible inner ops below `app` once, producing a flat list
167// of *leaves* (the surviving tensor inputs of the fused op) and *inner nodes* (the inner combiners
168// that must run before `f_o`). Each fusible input is replaced by its inner's inputs, with access
169// maps composed behind the outer map at that position; the composition nests across levels. The
170// new combination function `f_new` invokes every inner combiner in post-order — each starting from
171// its own init — and finally invokes `f_o`, threading inner results into the corresponding outer
172// input slots.

242 if (!inner_nps_nat || *inner_nps_nat != 0) continue;

245 // loop domain through the identity output map. In that case the inner tensor at any
246 // position is just a single call of `inner_comb` at that position.
247 // The identity map (`%affine.id`) is recognized structurally (a lam returning its own var),

285 // slot range in the fused list.

320 new_Ris_vec[pos] = Ris->proj(nis_nat, i);
321 new_Sis_vec[pos] = Sis->proj(nis_nat, i);

323 new_is_vec[pos] = is->proj(nis_nat, i);

328 auto new_Ris = w.tuple(new_Ris_vec);

330 auto new_maps = w.tuple(new_maps_vec);
331 auto new_is = w.tuple(new_is_vec);

335 // Build the fused combination function:
336 //
337 // cn f_new(data: [To, [new_Tis ...]], ret: cn To) =
338 // cn inner_ret_<r>(value_<r>: inner_To_<r>) = ...
339 // f_<fused[0]>((init_<fused[0]>, inner_inputs_<fused[0]>), inner_ret_0)
340 //
341 // inner_ret_<r>(value_<r>):

343 // f_<fused[r+1]>((init_<fused[r+1]>, inner_inputs_<fused[r+1]>), inner_ret_<r+1>)
344 // else:

381 for (size_t r = 0; r < fused_indices.size(); ++r) {

384 auto new_inner_init = infos[k].init;
385
386 DefVec inner_inputs_vec(infos[k].nis);
387 for (u64 l = 0; l < infos[k].nis; ++l)
388 inner_inputs_vec[l] = new_in->proj(new_nis_nat, new_pos[k] + l);
389
390 Lam* caller = (r == 0) ? new_comb : inner_rets[r - 1];
391 caller->app(true, new_inner_comb, {w.tuple({new_inner_init, w.tuple(inner_inputs_vec)}), inner_rets[r]});
392 }
393
394 // After every inner combiner has produced its value, call the outer combiner.
395 inner_rets.back()->app(true, comb, {w.tuple({new_acc, w.tuple(outer_inputs_vec)}), new_ret});

397 // Construct the fused map_reduce; the loop domain, output map, init and epilogue (including the
398 // epilogue inputs) are the outer's.
399 auto mr = w.annex<tensor::map_reduce_post>();
400 mr = w.app(mr, {new_nis_def, nps});
401 mr = w.app(mr, meta);
402 mr = w.app(mr, shapes);
403 mr = w.app(mr, {new_Tis, new_Ris, new_Sis, Tps, Rps, Sps});
404 mr = w.app(mr, {new_comb, init, post});
405 mr = w.app(mr, map_out);
406 mr = w.app(mr, {new_maps, post_maps});
407 mr = w.app(mr, {new_is, post_is});
408
409 return mr;

412// Fuses a trailing elementwise map into the reducing `tensor.map_reduce` producing one of its
413// inputs — the reverse of `fuse_map_reduce` and the direction the producer gate deliberately
414// rejects for `Rr > 0`: a reduction must not be inlined into a consumer's combiner (it would rerun
415// per fold step), but it *can* absorb the consumer into its per-output-cell `post` epilogue (the
416// GEMM-epilogue pattern, e.g. `relu(add(conv, bias))`).
417//
418// The producer input `k0` must be read elementwise over the map's whole domain (identity access
419// map, `Sis#k0 = So`, with the map's `Rr = 0`, `Sr = So` and identity `map_out`) — anything else
420// changes coordinates or drops cells and would need map inversion. All *other* inputs of the map —
421// and the map's own epilogue inputs — become epilogue inputs of the producer, read once per output
422// cell at their (arbitrary) access maps: those maps take the map's loop vector, which under the
423// identity-read gate is exactly the producer's output-cell coordinate system.
424//
425// Result: the producer with
426// post := (x, extras) ↦ post_o(f_o(init_o, extras[k0 ↦ post_i(x, extras_i)]), extras_o)
427// post_is := inner post_is ++ outer is \ k0 ++ outer post_is (maps concatenated likewise)
428//
429// The producer must be consumed by this map alone (checked via its old-world consumer count):
430// otherwise it stays materialized for the other consumers and additionally runs fused —
431// duplicating the whole reduction loop nest.
432//
433// `callee`/`arg` are new-world (already rewritten): the caller iterates this on freshly fused apps.
434/// Is `map` the row-major reshape read `%tensor.reshape_map (s_in, s_out)` — the map that reads a
435/// PACKED producer (its output strip-mined to `s_in`) at the unpacked coordinates `s_out`?
436/// Decided by normalization: both `map` and the canonical unpack map are applied to the same probe
437/// variable; the reduced bodies are hash-consed, so pointer equality decides alpha-equivalence.
438static bool
439is_unpack_read(World& w, const Def* map, const Def* r_in, const Def* s_in, const Def* r_out, const Def* s_out) {
440 auto pi = map->type()->isa<Pi>();
441 if (!pi) return false;
442 auto expected = w.app(w.app(w.annex<tensor::reshape_map>(), {r_in, r_out}), {s_in, s_out});
443 auto epi = expected->type()->isa<Pi>();
444 if (!epi || epi->dom() != pi->dom() || epi->codom() != pi->codom()) return false;
445 auto probe = w.mut_lam(pi->dom(), pi->codom()); // scratch binder: its var stands in for the cell vector
446 return w.app(map, probe->var()) == w.app(expected, probe->var());
447}

449// Is `d` (a new-world map_reduce) consumed by exactly one node? Resolved via the old-world app it
450// replaces; `new2old_` covers every map_reduce the phase has rewritten. Defs in `shared_` gained
451// consumers by a read-through of a shared read, which the old-world count cannot see.
452bool Fuse::sole_consumer(const Def* d) const {
453 if (shared_.contains(d)) return false;
454 auto old_it = new2old_.find(d);
455 if (old_it == new2old_.end()) return false;
456 auto cnt = mr_consumers_.find(old_it->second);
457 return cnt != mr_consumers_.end() && cnt->second == 1;
458}

460const Def* Fuse::fuse_epilogue(const App* callee, const Def* arg) {

463 auto [nis, nps] = nis_nps->projs<2>();

466 auto [comb, init, post] = comb_init->projs<3>();
467 auto [Tis, Ris, Sis, Tps, Rps, Sps] = in_tys->projs<6>();
468 auto [maps, post_maps] = maps_all->projs<2>();

470 auto& w = new_world();
471
472 auto nis_lit = Lit::isa<u64>(nis);
473 auto nps_lit = Lit::isa<u64>(nps);
474 auto ro_lit = Lit::isa<u64>(Ro);
475 auto rn_lit = Lit::isa<u64>(Rn);
476 if (!nis_lit || !nps_lit || !ro_lit || !rn_lit || *ro_lit != *rn_lit) return nullptr;
477 auto nis_nat = *nis_lit;
478 auto nps_nat = *nps_lit;
479 if (nis_nat == 0) return nullptr;
480 if (Sr != So) return nullptr;
481
482 auto id_out = map_out->isa_mut<Lam>();
483 if (!id_out || !id_out->is_set() || id_out->body() != id_out->var()) return nullptr;
484
485 // A consumer that computes nothing (a pure re-indexed read) is read-through's job — sinking it
486 // would re-wrap the producer forever (the unpack path below emits exactly such a copy on top).
487 if (nis_nat == 1 && nps_nat == 0 && is_copy_comb(comb) && is_identity_post(post)) return nullptr;
488
489 auto [is, ps] = arg->projs<2>();
490
491 // Find the producer: the first input that is a map_reduce read elementwise over the whole
492 // domain — at identity coordinates (input shape == So), or through the row-major reshape that
493 // unpacks a strip-mined (packed) producer — and consumed by this map alone.
494 u64 k0 = 0;
495 const Def* inner_def = nullptr;
496 bool unpack = false;
497 for (u64 k = 0; k < nis_nat && !inner_def; ++k) {
498 auto cand = is->proj(nis_nat, k);
499 if (!Axm::isa<tensor::map_reduce_post>(cand)) continue;
500 if (!sole_consumer(cand)) continue;
501
502 auto m = maps->proj(nis_nat, k);
503 auto id_in = m->isa_mut<Lam>();
504 if (id_in && id_in->is_set() && id_in->body() == id_in->var() && Sis->proj(nis_nat, k) == So) {
505 k0 = k;
506 inner_def = cand;
507 } else if (Sis->proj(nis_nat, k) != So
508 && is_unpack_read(w, m, Ris->proj(nis_nat, k), Sis->proj(nis_nat, k), Ro, So)) {
509 k0 = k;
510 inner_def = cand;
511 unpack = true;
512 }
513 }
514 if (!inner_def) return nullptr;
515
516 // Reading through the unpack: the fused epilogue runs at the producer's PACKED cell
517 // coordinates, so every map carried over from this consumer — its other combiner inputs and
518 // its own epilogue inputs, all taking unpacked cell coordinates — is composed behind the
519 // inverse reshape (packed cell → unpacked cell).
520 const Def* to_cells = nullptr;
521 if (unpack)
522 to_cells
523 = w.app(w.app(w.annex<tensor::reshape_map>(), {Ro, Ris->proj(nis_nat, k0)}), {So, Sis->proj(nis_nat, k0)});
524 auto inner = Axm::isa<tensor::map_reduce_post>(inner_def);
525
526 auto [i_nis_nps, i_meta, i_shapes, i_in_tys, i_comb_init, i_map_out, i_maps_all, i_is_all]
527 = inner->uncurry_args<8>();
528 auto [i_nis, i_nps] = i_nis_nps->projs<2>();
529 auto [i_To, i_Tp, i_Ro, i_Rn, i_TSched] = i_meta->projs<5>();
530 auto [i_Tis, i_Ris, i_Sis, i_Tps, i_Rps, i_Sps] = i_in_tys->projs<6>();
531 auto [i_comb, i_init, i_post] = i_comb_init->projs<3>();
532 auto [i_maps, i_post_maps] = i_maps_all->projs<2>();
533 auto [i_is, i_ps] = i_is_all->projs<2>();
534
535 auto i_nps_lit = Lit::isa<u64>(i_nps);
536 if (!i_nps_lit) return nullptr;
537 auto i_nps_nat = *i_nps_lit;
538
539 DLOG("fusing trailing map {} {} into the epilogue of {}", callee, arg, inner_def);
540
541 // Concatenated epilogue inputs: the inner's own, then the map's other combiner inputs, then the
542 // map's epilogue inputs. The maps' access maps carry over verbatim — their domain (the map's
543 // loop vector) is the producer's output-cell coordinate system.
544 auto new_nps = i_nps_nat + (nis_nat - 1) + nps_nat;
545 DefVec nTps, nRps, nSps, nPmaps, nPis;
546 nTps.reserve(new_nps), nRps.reserve(new_nps), nSps.reserve(new_nps);
547 nPmaps.reserve(new_nps), nPis.reserve(new_nps);
548 for (u64 j = 0; j < i_nps_nat; ++j) {
549 nTps.emplace_back(i_Tps->proj(i_nps_nat, j));
550 nRps.emplace_back(i_Rps->proj(i_nps_nat, j));
551 nSps.emplace_back(i_Sps->proj(i_nps_nat, j));
552 nPmaps.emplace_back(i_post_maps->proj(i_nps_nat, j));
553 nPis.emplace_back(i_ps->proj(i_nps_nat, j));
554 }
555 for (u64 i = 0; i < nis_nat; ++i) {
556 if (i == k0) continue;
557 nTps.emplace_back(Tis->proj(nis_nat, i));
558 nRps.emplace_back(Ris->proj(nis_nat, i));
559 nSps.emplace_back(Sis->proj(nis_nat, i));
560 nPmaps.emplace_back(to_cells ? compose_map(w, maps->proj(nis_nat, i), to_cells) : maps->proj(nis_nat, i));
561 nPis.emplace_back(is->proj(nis_nat, i));
562 }
563 for (u64 j = 0; j < nps_nat; ++j) {
564 nTps.emplace_back(Tps->proj(nps_nat, j));
565 nRps.emplace_back(Rps->proj(nps_nat, j));
566 nSps.emplace_back(Sps->proj(nps_nat, j));
567 nPmaps.emplace_back(to_cells ? compose_map(w, post_maps->proj(nps_nat, j), to_cells)
568 : post_maps->proj(nps_nat, j));
569 nPis.emplace_back(ps->proj(nps_nat, j));
570 }
571
572 // The composed epilogue as a CPS chain: inner post → map combiner → map post. Identity hops are
573 // called through anyway — their always-true filters inline them.
574 auto fused_post = w.mut_con({w.sigma({i_To, w.sigma(nTps)}), w.cn(Tp)})->set("fused_post");
575 auto after_ip = w.mut_con(i_Tp)->set("afterInnerPost");
576 auto after_comb = w.mut_con(To)->set("afterComb");
577 auto [x, extras] = fused_post->var(0)->projs<2>();
578
579 DefVec i_extras(i_nps_nat);
580 for (u64 j = 0; j < i_nps_nat; ++j)
581 i_extras[j] = extras->proj(new_nps, j);
582 fused_post->app(true, i_post, {w.tuple({x, w.tuple(i_extras)}), after_ip});
583
584 DefVec comb_inputs(nis_nat);
585 {
586 u64 r = 0;
587 for (u64 i = 0; i < nis_nat; ++i)
588 comb_inputs[i] = (i == k0) ? after_ip->var(0) : extras->proj(new_nps, i_nps_nat + r++);
589 }
590 after_ip->app(true, comb, {w.tuple({init, w.tuple(comb_inputs)}), after_comb});
591
592 DefVec o_extras(nps_nat);
593 for (u64 j = 0; j < nps_nat; ++j)
594 o_extras[j] = extras->proj(new_nps, i_nps_nat + (nis_nat - 1) + j);
595 after_comb->app(true, post, {w.tuple({after_comb->var(0), w.tuple(o_extras)}), fused_post->var(1)});
596
597 // The producer, keeping its loop nest untouched; only the epilogue (inputs) and the out-element
598 // type change.
599 auto mr = w.annex<tensor::map_reduce_post>();
600 mr = w.app(mr, {i_nis, w.lit_nat(new_nps)});
601 mr = w.app(mr, {i_To, Tp, i_Ro, i_Rn, i_TSched});
602 mr = w.app(mr, i_shapes);
603 mr = w.app(mr, {i_Tis, i_Ris, i_Sis, w.tuple(nTps), w.tuple(nRps), w.tuple(nSps)});
604 mr = w.app(mr, {i_comb, i_init, fused_post});
605 mr = w.app(mr, i_map_out);
606 mr = w.app(mr, {i_maps, w.tuple(nPmaps)});
607 mr = w.app(mr, {i_is, w.tuple(nPis)});
608
609 if (unpack) {
610 // The fused op still materializes PACKED; re-wrap it in the unpacking reshape — expanded
611 // via the impl so the wrapper is again a pure copy-mr that consumers read through (or a
612 // plain copy at a graph boundary). The copy-consumer guard above keeps this wrapper from
613 // being sunk right back.
614 auto impl = w.annex<tensor::reshape_impl>();
615 impl = w.app(impl, {Tp, Ris->proj(nis_nat, k0), Ro});
616 impl = w.app(impl, Sis->proj(nis_nat, k0));
617 impl = w.app(impl, So);
618 mr = w.app(impl, mr);
619 }
620
621 return mr;
622}

624// Rewires every input slot — combiner and epilogue alike — that is a pure re-indexed read
625// (a copy-combiner map_reduce or a `%tensor.broadcast`, see `read_through`) to read the underlying
626// source directly, with the read's access map composed behind the slot's map. This absorbs
627// reshape/transpose/slice/flip/repeat/broadcast chains into the access maps of the consuming
628// map_reduce, so they never materialize; the bypassed op dies with cleanup unless someone else
629// still reads it (in which case the rewiring is still free — it removes no sharing, only a copy).
630const Def* Fuse::fuse_read_through(const App* callee, const Def* arg) {
631 auto [nis_nps, meta, shapes, in_tys, comb_init, map_out, maps_all] = callee->uncurry_args<7>();
632
633 auto [nis, nps] = nis_nps->projs<2>();
634 auto [Tis, Ris, Sis, Tps, Rps, Sps] = in_tys->projs<6>();
635 auto [maps, post_maps] = maps_all->projs<2>();
636
637 auto nis_lit = Lit::isa<u64>(nis);
638 auto nps_lit = Lit::isa<u64>(nps);
639 if (!nis_lit || !nps_lit) return nullptr;
640 auto nis_nat = *nis_lit;
641 auto nps_nat = *nps_lit;
642
643 auto& w = new_world();
644
645 auto [is, ps] = arg->projs<2>();
646
647 bool changed = false;
648 auto rewire = [&](DefVec& T, DefVec& R, DefVec& S, DefVec& m, DefVec& v, u64 n, const Def* Ts, const Def* Rs,
649 const Def* Ss, const Def* ms, const Def* vs) {
650 for (u64 i = 0; i < n; ++i) {
651 T[i] = Ts->proj(n, i);
652 R[i] = Rs->proj(n, i);
653 S[i] = Ss->proj(n, i);
654 m[i] = ms->proj(n, i);
655 v[i] = vs->proj(n, i);
656 if (auto rt = read_through(w, v[i], m[i])) {
657 DLOG("reading input {} of {} through {}", i, callee, v[i]);
658 // The bypassed read forwards its consumers to the source: a shared (or uncounted,
659 // e.g. broadcast) read leaves the source multiply-consumed for the epilogue guard.
660 if (!sole_consumer(v[i])) shared_.insert(rt->value);
661 m[i] = compose_map(w, rt->map, m[i]);
662 v[i] = rt->value;
663 T[i] = rt->T;
664 R[i] = rt->R;
665 S[i] = rt->S;
666 changed = true;
667 }
668 }
669 };
670
671 DefVec nTis(nis_nat), nRis(nis_nat), nSis(nis_nat), nMaps(nis_nat), nIs(nis_nat);
672 DefVec nTps(nps_nat), nRps(nps_nat), nSps(nps_nat), nPmaps(nps_nat), nPis(nps_nat);
673 rewire(nTis, nRis, nSis, nMaps, nIs, nis_nat, Tis, Ris, Sis, maps, is);
674 rewire(nTps, nRps, nSps, nPmaps, nPis, nps_nat, Tps, Rps, Sps, post_maps, ps);
675 if (!changed) return nullptr;
676
677 auto mr = w.annex<tensor::map_reduce_post>();
678 mr = w.app(mr, nis_nps);
679 mr = w.app(mr, meta);
680 mr = w.app(mr, shapes);
681 mr = w.app(mr, {w.tuple(nTis), w.tuple(nRis), w.tuple(nSis), w.tuple(nTps), w.tuple(nRps), w.tuple(nSps)});
682 mr = w.app(mr, comb_init);
683 mr = w.app(mr, map_out);
684 mr = w.app(mr, {w.tuple(nMaps), w.tuple(nPmaps)});
685 mr = w.app(mr, {w.tuple(nIs), w.tuple(nPis)});
686
687 return mr;
688}

694/// Does `d` — descending only through tuple/pack wrappers — contain a `map_reduce_post` app?
695bool contains_mr(const Def* d, DefMap<bool>& memo) {
696 if (is_mr(d)) return true;
697 if (!d->isa<Tuple>() && !d->isa<Pack>()) return false;
698 if (auto it = memo.find(d); it != memo.end()) return it->second;
699 bool res = false;
700 for (auto op : d->ops())
701 if (op && contains_mr(op, memo)) {
702 res = true;
703 break;
704 }
705 return memo[d] = res;
706}

708/// Attributes consumption of a map_reduce to the enclosing non-tuple node: tuples and packs are
709/// transparent argument wrappers, so recurse through them, charging each contained map_reduce.
710void count_mr_consumers(const Def* d, DefMap<u64>& counts, DefMap<bool>& memo) {
711 if (is_mr(d)) {
712 ++counts[d];
713 return;
714 }
715 if ((d->isa<Tuple>() || d->isa<Pack>()) && contains_mr(d, memo))
716 for (auto op : d->ops())
717 if (op) count_mr_consumers(op, counts, memo);
718}

722void Fuse::start() {
723 // The epilogue direction needs to know whether a map_reduce is consumed by exactly one other
724 // node; the old world does not track uses, so count consumers in one pass up front. Tuples and
725 // packs are transparent argument wrappers: consumption is attributed to the enclosing non-tuple
726 // node, so a shared argument tuple correctly charges each of its users, and using the same op
727 // twice in one argument list counts twice.
728 DefMap<bool> memo;
729 unique_queue<DefSet> wl;
730 for (auto root : old_world().roots())
731 wl.push(root);
732 while (!wl.empty()) {
733 auto def = wl.pop();
734 if (!def->isa<Tuple>() && !def->isa<Pack>())
735 for (auto op : def->ops())
736 if (op) count_mr_consumers(op, mr_consumers_, memo);
737 for (auto op : def->ops())
738 if (op) wl.push(op);
739 if (def->type()) wl.push(def->type());
740 }
741 RWPhase::start();
742}

744const Def* Fuse::rewrite_imm_App(const App* app) {
745 if (Axm::isa<tensor::map_reduce_post>(app)) {
746 const App* cur = nullptr;
747 if (auto res = fuse_map_reduce(app)) {
748 DLOG("Fused map_reduce at {} into a new map_reduce {}", app, res);
749 cur = res->as<App>();
750 }
751 // Read-throughs and the epilogue direction, to a fixpoint: each round may absorb pure
752 // re-indexed reads into access maps or sink the current map — first the plainly rewritten
753 // one, then the producer-fused result, then each fused result — into the producer of one
754 // of its inputs. Terminates quickly: rewiring walks strictly down the producer DAG, and a
755 // fused reduction fails the Rr = 0 outer gate on the next round.
756 auto callee = cur ? cur->callee()->as<App>() : rewrite(app->callee())->as<App>();
757 auto arg = cur ? cur->arg() : rewrite(app->arg());
758 for (bool progress = true; progress;) {
759 progress = false;
760 const Def* res = fuse_read_through(callee, arg);
761 if (!res) {
762 res = fuse_epilogue(callee, arg);
763 if (res) DLOG("Fused the trailing map at {} into its producer's epilogue: {}", app, res);
764 }
765 if (res) {
766 cur = res->as<App>();
767 callee = cur->callee()->as<App>();
768 arg = cur->arg();
769 progress = true;
770 }
771 }
772 auto result = cur ? cur : RWPhase::rewrite_imm_App(app);
773 // Remember which old app this (possibly fused) map_reduce replaces, so later epilogue
774 // rounds of its consumers can look up its consumer count.
775 if (is_mr(result)) {
776 new2old_[result] = app;
777 // An epilogue sink into a PACKED producer wraps the fused op in an unpack copy (see
778 // fuse_epilogue). Register the wrapped producer under the same old app — its sole
779 // consumer is the wrapper, which stands for this app — so the NEXT trailing map (which
780 // producer-fuses the wrapper away and reads the packed op directly) can sink too.
781 if (auto pr = is_pure_read(result); pr && is_mr(pr->src) && !new2old_.contains(pr->src))
782 new2old_[pr->src] = app;
783 }
784 return result;
785 }
786 return RWPhase::rewrite_imm_App(app);
787}
