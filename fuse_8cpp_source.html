14// Fuses an outer `tensor.map_reduce` with any number of its inputs — and, recursively, any
15// fusible inputs of those inputs — whenever each such input is itself a `tensor.map_reduce`
16// whose subs only reference output indices (i.e. the inner reduction is empty, so reading the
17// inner tensor at a position is just a single call to the inner combination function).
18//
19// Outer: map_reduce nis_o (To, Ro) So (Tis_o, Ris_o, Sis_o) (f_o, init_o) subs_o is_o
20// Inner: map_reduce nis_k (To_k, Ro_k) So_k (Tis_k, Ris_k, Sis_k) (f_k, init_k) subs_k is_k
21// for every fusible input — possibly nested inside another fusible input
22//
23// Result: map_reduce nis_new (To, Ro) So (Tis_new, Ris_new, Sis_new) (f_new, init_o) subs_new
24// is_new
25//
26// The collection phase walks the tree of fusible inner map_reduces below `app` once, producing
27// a flat list of *leaves* (the surviving tensor inputs of the fused mr) and *inner nodes* (the
28// inner combiners that must run before `f_o`). Each fusible input is replaced by its inner's
29// inputs, with subs remapped through the outer subs at that position; the remapping composes
30// across nested levels. The new combination function `f_new` invokes every inner combiner in
31// post-order — each starting from its own init — and finally invokes `f_o`, threading inner
32// results into the corresponding outer input slots.

92
93 // We can only fuse when the inner has no reduction dimensions, i.e. all subs are < Ro_i.
94 // In that case the inner tensor at any position is just a single call of `inner_comb`.
95 Vector<u64> inner_Ris_nats(*inner_nis_nat);
96 bool fusible = true;
97 for (u64 l = 0; l < *inner_nis_nat && fusible; ++l) {
98 auto Ris_l_lit = Lit::isa<u64>(inner_Ris->proj(*inner_nis_nat, l));
99 if (!Ris_l_lit) {
100 fusible = false;
101 break;
102 }
103 inner_Ris_nats[l] = *Ris_l_lit;
104 auto inner_subs_l = inner_subs->proj(*inner_nis_nat, l);
105 for (u64 j = 0; j < inner_Ris_nats[l]; ++j) {
106 auto idx_lit = Lit::isa<u64>(inner_subs_l->proj(inner_Ris_nats[l], j));
107 if (!idx_lit || *idx_lit >= *inner_Ro_nat) {
108 fusible = false;
109 break;
110 }
111 }
112 }
113 if (!fusible) continue;
114
115 // We need the outer subs for input `k` to be indexable by literal positions.
116 auto Ris_k_lit = Lit::isa<u64>(Ris->proj(nis_nat, k));
117 if (!Ris_k_lit) continue;
118
119 auto& info = infos[k];
120 info.fusible = true;
121 info.subs = inner_subs;
122 info.comb = inner_comb;
123 info.init = inner_init;
124 info.Tis = inner_Tis;
125 info.Ris = inner_Ris;
126 info.Sis = inner_Sis;
127 info.To = inner_To;
128 info.nis = *inner_nis_nat;
129 info.is = inner_is;
130 info.Ris_nats = std::move(inner_Ris_nats);
131 info.outer_Ris_nat = *Ris_k_lit;
132 info.outer_subs = subs->proj(nis_nat, k);
133 any_fusible = true;
134 }

140 // Each fusible outer input k is replaced by `infos[k].nis` slots in the fused input list;
141 // every non-fusible input retains exactly one slot. `new_pos[i]` is the start of input i's
142 // slot range in the fused list.

144 u64 new_nis_nat = 0;
145 for (u64 i = 0; i < nis_nat; ++i) {
146 new_pos[i] = new_nis_nat;

162 new_Ris_vec[pos] = info.Ris->proj(info.nis, l);
163 new_Sis_vec[pos] = info.Sis->proj(info.nis, l);
164 new_is_vec[pos] = info.is->proj(info.nis, l);

168 for (u64 j = 0; j < info.Ris_nats[l]; ++j) {

170 // outer index space via the outer subs at position `i`.

172 subs_l_vec[j] = rewrite(info.outer_subs->proj(info.outer_Ris_nat, inner_idx));
173 }
174 new_subs_vec[pos] = w.tuple(subs_l_vec);
175 }
176 } else {

179 new_Ris_vec[pos] = Ris->proj(nis_nat, i);
180 new_Sis_vec[pos] = Sis->proj(nis_nat, i);

182 new_is_vec[pos] = is->proj(nis_nat, i);
183 }

186 auto new_Tis = w.tuple(new_Tis_vec);
187 auto new_Ris = w.tuple(new_Ris_vec);
188 auto new_Sis = w.tuple(new_Sis_vec);
189 auto new_subs = w.tuple(new_subs_vec);

192 auto new_nis_def = w.lit_nat(new_nis_nat);

194 // Build the fused combination function:
195 //
196 // cn f_new(data: [To, [new_Tis ...]], ret: cn To) =
197 // cn inner_ret_<r>(value_<r>: inner_To_<r>) = ...
198 // f_<fused[0]>((init_<fused[0]>, inner_inputs_<fused[0]>), inner_ret_0)
199 //
200 // inner_ret_<r>(value_<r>):
201 // if r is not the last fused input:
202 // f_<fused[r+1]>((init_<fused[r+1]>, inner_inputs_<fused[r+1]>), inner_ret_<r+1>)
203 // else:
204 // f_o((acc, outer_inputs), ret)
205 //
206 // `outer_inputs[i]` is `value_<r>` when input i is the r-th fused input, and the
207 // corresponding `new_in` slot otherwise. Each `inner_ret_<r>` closes over the prior
208 // `value_<j>`s as free variables — those are bound by the dynamic call chain.
209 auto inputs_sigma = w.sigma(new_Tis_vec);
210 auto data_sigma = w.sigma({To, inputs_sigma});
211 auto ret_cn_type = w.cn(To);
212 auto new_comb = w.mut_con({data_sigma, ret_cn_type})->set("fused_comb");
213 auto new_data = new_comb->var(0);
214 auto new_ret = new_comb->var(1);
215 auto new_acc = new_data->proj(2, 0);
216 auto new_in = new_data->proj(2, 1);
217
218 Vector<u64> fused_indices;
219 for (u64 i = 0; i < nis_nat; ++i)
220 if (infos[i].fusible) fused_indices.emplace_back(i);
221
222 Vector<Lam*> inner_rets(fused_indices.size());
223 Vector<const Def*> inner_values(fused_indices.size());
224 for (size_t r = 0; r < fused_indices.size(); ++r) {
225 auto new_inner_To = infos[fused_indices[r]].To;
226 inner_rets[r] = w.mut_con(new_inner_To)->set("inner_ret");
227 inner_values[r] = inner_rets[r]->var(0);
228 }
229
230 // Map each outer input position to its value at the f_o call site.
231 DefVec outer_inputs_vec(nis_nat);
232 {
233 size_t r = 0;
234 for (u64 i = 0; i < nis_nat; ++i)
235 if (infos[i].fusible)
236 outer_inputs_vec[i] = inner_values[r++];
237 else
238 outer_inputs_vec[i] = new_in->proj(new_nis_nat, new_pos[i]);
239 }
240
241 // Chain: caller for fused step r is new_comb (r==0) or inner_rets[r-1] (otherwise).
242 for (size_t r = 0; r < fused_indices.size(); ++r) {
243 auto k = fused_indices[r];
244 auto new_inner_comb = infos[k].comb;
245 auto new_inner_init = infos[k].init;
246
247 DefVec inner_inputs_vec(infos[k].nis);
248 for (u64 l = 0; l < infos[k].nis; ++l)
249 inner_inputs_vec[l] = new_in->proj(new_nis_nat, new_pos[k] + l);
250
251 Lam* caller = (r == 0) ? new_comb : inner_rets[r - 1];
252 caller->app(true, new_inner_comb, {w.tuple({new_inner_init, w.tuple(inner_inputs_vec)}), inner_rets[r]});
253 }
254
255 // After every inner combiner has produced its value, call the outer combiner.
256 inner_rets.back()->app(true, comb, {w.tuple({new_acc, w.tuple(outer_inputs_vec)}), new_ret});
257
258 // Construct the fused map_reduce.
259 auto mr = w.annex<tensor::map_reduce>();
260 mr = w.app(mr, new_nis_def);
261 mr = w.app(mr, {To, Ro});
262 mr = w.app(mr, So);
263 mr = w.app(mr, {new_Tis, new_Ris, new_Sis});
264 mr = w.app(mr, {new_comb, init});
265 mr = w.app(mr, new_subs);
266 mr = w.app(mr, new_is);
267
268 return mr;
