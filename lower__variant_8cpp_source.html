31bool LowerVariant::finite(Layout& l) {
32 auto& w = new_world();
33 auto offsets = fe::Vector<nat_t>();
34 nat_t total = 0, widest = 0;
35 for (auto c : l.cases) {
36 auto s = w.inhabitant(c) ? std::optional<nat_t>(1) : idx_size(c);
37 if (!s || total > ~nat_t(0) - *s) return false;
38 if (*s != 1) widest = std::max(widest, size(c).value_or(0));
39 offsets.emplace_back(total);
40 total += *s;
41 }
42 if (widest != 0 && size(w.type_idx(total)) > widest) return false;
43
44 l.kind = Layout::Kind::Finite;
45 l.type = w.type_idx(total);
46 l.offsets = std::move(offsets);
47 return true;
48}

50bool LowerVariant::word(Layout& l) {
51 auto& w = new_world();
52 auto ptrs = false;
53 auto packs = std::ranges::all_of(l.cases, [&](const Def* c) {
54 if (w.inhabitant(c)) return true;
55 if (auto ptr = Axm::isa<Ptr>(c)) {
56 auto align = Lit::isa(op(trait::align, ptr->arg(2, 0)));
57 return ptrs = align && *align >> l.bits() != 0 && std::has_single_bit(*align);
58 }
59 auto s = idx_size(c);
60 return s && Idx::size2bitwidth(*s) <= 64 - l.bits();
61 });
62 if (!packs || !ptrs) return false;
63
64 l.kind = Layout::Kind::Word;
65 l.type = w.type_i64();
66 return true;
67}

78const Def* LowerVariant::encode(const Layout& l, size_t i, const Def* value) {
79 auto& w = new_world();
80 auto c = l.cases[i];
81 switch (l.kind) {
82 case Layout::Kind::Finite: {
83 auto offset = w.lit_idx(l.total(), l.offsets[i]);
84 if (w.inhabitant(c)) return offset;
85 return w.call(wrap::add, Mode::nuw, Defs{offset, w.call(conv::u, w.lit_nat(l.total()), value)});
86 }
87 case Layout::Kind::Word: {
88 auto tag = w.lit_i64(i);
89 if (w.inhabitant(c)) return tag;
90 auto bits = Axm::isa<Ptr>(c) ? w.call<bitcast>(w.type_i64(), value)
91 : w.call(wrap::shl, Mode::none, Defs{zext(w, value), w.lit_i64(l.bits())});
92 return w.call(bit2::or_, 0_n, Defs{bits, tag});
93 }
94 case Layout::Kind::Tagged: {
95 auto box = l.type->op(1);
96 auto tag = w.lit_idx(l.cases.size(), i);
97 return w.tuple({tag, w.inhabitant(c) ? w.bot(box) : w.call<bitcast>(box, value)});
98 }
99 }
100 std::unreachable();
101}

103const Def* LowerVariant::tag(const Layout& l, const Def* x) {
104 auto& w = new_world();
105 auto n = l.cases.size();
106 switch (l.kind) {
107 case Layout::Kind::Finite: {
108 if (l.total() == n) return x;
109 const Def* t = w.lit_idx(n, 0);
110 for (size_t i = 1; i != n; ++i)
111 t = w.select(w.call(icmp::uge, Defs{x, w.lit_idx(l.total(), l.offsets[i])}), w.lit_idx(n, i), t);
112 return t;
113 }
114 case Layout::Kind::Word: {
115 auto mask = w.lit_i64(l.mask());
116 return w.call(conv::u, w.lit_nat(n), w.call(bit2::and_, 0_n, Defs{x, mask}));
117 }
118 case Layout::Kind::Tagged: return x->proj(2, 0);
119 }
120 std::unreachable();
121}

123const Def* LowerVariant::decode(const Layout& l, size_t i, const Def* operand) {
124 auto& w = new_world();
125 auto c = l.cases[i];
126 if (auto x = w.inhabitant(c)) return x;
127 switch (l.kind) {
128 case Layout::Kind::Finite: {
129 // The operand is at least the offset, so subtracting in `I64` needs no reduction.
130 auto wide = zext(w, operand);
131 auto rel = w.call(wrap::sub, Mode::none, Defs{wide, w.lit_i64(l.offsets[i])});
132 return w.call(conv::u, w.lit_nat(*idx_size(c)), rel);
133 }
134 case Layout::Kind::Word: {
135 if (Axm::isa<Ptr>(c)) {
136 auto mask = w.lit_i64(~l.mask());
137 return w.call<bitcast>(c, w.call(bit2::and_, 0_n, Defs{operand, mask}));
138 }
139 auto bits = w.call(shr::l, Defs{operand, w.lit_i64(l.bits())});
140 return w.call(conv::u, w.lit_nat(*idx_size(c)), bits);
141 }
142 case Layout::Kind::Tagged: return w.call<bitcast>(c, operand);
143 }
144 std::unreachable();
145}

160const Def* LowerVariant::rewrite_imm_Match(const Match* match) {
161 auto variant = match->scrutinee()->unfold_type()->isa_imm<Variant>();
162 if (!variant) return RWPhase::rewrite_imm_Match(match);
163
164 auto& w = new_world();
165 auto l = layout(variant);
166 auto x = rewrite(match->scrutinee());
167 auto t = tag(l, x);
168 auto op = l.kind == Layout::Kind::Tagged ? x->proj(2, 1) : x;
169
170 auto n = l.cases.size();
171 auto arms = DefVec(n);
172 if (std::ranges::all_of(match->arms(), [](const Def* arm) { return Pi::isa_cn(arm->type()); })) {
173 for (size_t i = 0; i != n; ++i) {
174 auto arm = rewrite(match->arm(i));
175 if (l.cases[i] == op->type()) {
176 arms[i] = arm;
177 } else {
178 auto lam = w.mut_con(op->type());
179 arms[i] = lam->app(false, arm, decode(l, i, lam->var()));
180 }
181 }
182 return w.app(w.extract(w.tuple(arms), t), op);
183 }
184
185 // Evaluating every arm is only sound as long as none of them has an effect.
186 if (std::ranges::any_of(match->arms(),
187 [](const Def* arm) { return mentions(arm, [](const Def* d) { return Axm::isa<M>(d); }); }))
188 fe::throwf("cannot lower effectful direct-style match `{}`", match);
189 for (size_t i = 0; i != n; ++i)
190 arms[i] = w.app(rewrite(match->arm(i)), decode(l, i, op));
191 return w.extract(w.tuple(arms), t);
192}
