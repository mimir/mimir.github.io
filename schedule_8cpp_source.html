101static void post_order(const Nest& nest, const Nest::Node* node, Scheduler::Schedule& res, MutSet& done) {
102 // The mut of a virtual root is a nullptr
103 if (!node->mut()) {
104 for (auto child : node->children().nodes())
105 post_order(nest, child, res, done);
106 return;
107 }
108
109 if (!node->mut()->isa<Lam>()) return;
110 if (auto [_, ins] = done.emplace(node->mut()); !ins) return;
111
112 for (auto op : node->mut()->deps()) {
113 for (auto mut : op->local_muts())
114 if (auto next = nest[mut]) post_order(nest, next, res, done);
115 }
116
117 res.emplace_back(node->mut());
118}
