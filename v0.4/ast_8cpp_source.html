202void AST::bootstrap_py(Sym plugin, std::ostream& h) {
203 fe::Tab tab;
204 plugin_t plugin_id = *Annex::mangle(plugin);
205
206 std::print(h, "from enum import IntEnum\n\n");
207 std::println(h, "class {}(IntEnum):", plugin);
208 ++tab;
209 std::println(h, "{}ID = 0x{:x}", tab, plugin_id);
210 std::vector<const AnnexInfo*> annexes_with_subs;
211
212 const auto& unordered = plugin2annexes(plugin);
213 std::deque<std::pair<Sym, AnnexInfo>> infos(unordered.begin(), unordered.end());
214 std::ranges::sort(infos, [&](const auto& p1, const auto& p2) { return p1.second.id.tag < p2.second.id.tag; });
215 for (const auto& [key, annex] : infos) {
216 const auto& sym = annex.sym;
217 if (sym.plugin != plugin) continue;
218
219 flags_t ax_id = annex.base();
220
221 if (auto& subs = annex.subs; subs.empty())
222 std::println(h, "{}{} = 0x{:x}", tab, escape(sym.tag, Py_Keywords), ax_id);
223 else
224 annexes_with_subs.emplace_back(&annex);
225 }
226 for (const auto& [alias, annex, sub] : plugin2foreign_aliases(plugin))
227 std::println(h, "{}{} = 0x{:x}", tab, escape(alias, Py_Keywords), annex->base() + sub);
228 std::print(h, "\n");
229 --tab;
230
231 for (auto annex : annexes_with_subs) {
232 flags_t ax_id = annex->base();
233 auto tag = escape(annex->sym.tag, Py_Keywords);
234 std::println(h, "class _{}_{}(IntEnum):", plugin, tag);
235 ++tab;
236
237 for (const auto& aliases : annex->subs) {
238 auto id = ax_id++;
239 for (const auto alias : aliases)
240 std::println(h, "{}{} = 0x{:x}", tab, escape(alias, Py_Keywords), id);
241 }
242
243 --tab;
244 std::println(h, "\n{}.{} = _{}_{}\n", plugin, tag, plugin, tag);
245 }
246
247 for (const auto& [key, annex] : infos) {
248 if (annex.sym.plugin != plugin || annex.names.empty()) continue;
249 auto type = std::format("{}_{}", annex.sym.tag, kind(annex));
250 std::println(h, "class _{}_{}(IntEnum):", plugin, type);
251 ++tab;
252 for (size_t i = 0, e = annex.names.size(); i != e; ++i)
253 if (auto name = annex.names[i]) std::println(h, "{}{} = {}", tab, escape(name, Py_Keywords), i);
254 --tab;
255 std::println(h, "\n{}.{} = _{}_{}\n", plugin, type, plugin, type);
256 }

293AST load_plugins(World& world, fe::View<std::string> plugins) {
294 auto tag = world.driver().flags().bootstrap ? Tok::Tag::K_import : Tok::Tag::K_plugin;
295 auto ast = AST(world);
296 auto parser = Parser(ast);
297
298 if (!plugins.empty()) {
299 auto imports = parser.import_plugins(plugins, tag);
300 auto decls = Ptrs<ValDecl>();
301 for (auto import : imports)
302 decls.emplace_back(import);
303
304 // No Loc: this File spans no source, and hulling the imports would mix Loc%s of different files.
305 auto file = ast.ptr<File>(Loc(), ast.scope(), ast.copy(decls));
306 file->compile(ast);
307 }
308
309 return ast;
310}
