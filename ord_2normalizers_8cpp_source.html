16/// Decides whether the keys @p k1 and @p k2 are equal; `std::nullopt` means "don't know".
17/// Distinct literals are distinct keys, as the key's `lt` is assumed to be a strict total order.
18std::optional<bool> key_eq(const Def* k1, const Def* k2) {
19 if (k1 == k2) return true;
20 if (k1->isa<Lit>() && k2->isa<Lit>()) return false;
21 return {};

26template<init id>
27const Def* normalize_init(const Def* type, const Def* callee, const Def* arg) {
28 auto& w = type->world();
29 if (auto n = Lit::isa(callee->as<App>()->arg())) {
30 auto key = [](const Def* kv) { return id == init::map ? kv->proj(2, 0) : kv; };
31 auto kvs = DefVec();
32 for (auto kv : arg->projs(*n))
33 if (auto i = std::ranges::find_if(kvs, [&](const Def* cur) { return key(cur) == key(kv); }); i != kvs.end())
34 *i = kv;
35 else
36 kvs.emplace_back(kv);
37 if (kvs.size() != *n) return w.app(w.app(callee->as<App>()->callee(), w.lit_nat(kvs.size())), kvs);
38 }
39 return w.raw_app(type, callee, arg);
40}

48const Def* normalize_get(const Def* type, const Def* callee, const Def* arg) {

51

104template<insert id>
105const Def* normalize_insert(const Def* type, const Def* callee, const Def* arg) {
106 auto& w = type->world();
107 auto [ms, kv] = arg->projs<2>();
108
109 if (auto init = Axm::isa<ord::init>(ms)) {
110 auto [n, kvs] = init->uncurry_args<2>();
111 if (auto l = Lit::isa(n)) {
112 auto key = id == ord::insert::map ? kv->proj(2, 0) : kv;
113 auto new_ops = DefVec();
114 bool updated = false;
115 for (auto cur : kvs->projs(*l)) {
116 auto cur_key = id == ord::insert::map ? cur->proj(2, 0) : cur;
117 if (key == cur_key) updated = true;
118 new_ops.emplace_back(key == cur_key ? kv : cur);
119 }
120 if (!updated) new_ops.emplace_back(kv);
121 return w.app(w.app(init->decurry()->callee(), w.lit_nat(new_ops.size())), new_ops);
122 }
123
124 if (auto pack = init->arg()->isa_imm<Pack>()) w.log().w("Pack not yet implemented: {}", pack);
125 }
126
127 // insert(insert(m, k'), k) = insert(m, k) if k' = k
128 // = insert(insert(m, k), k') if k' ≠ k and k < k' (canonical order of literal keys)
129 if (auto ins = Axm::isa(id, ms)) {
130 auto [m, kv_] = ins->args<2>();
131 auto key = [](const Def* kv) { return id == ord::insert::map ? kv->proj(2, 0) : kv; };
132 if (auto eq = key_eq(key(kv_), key(kv))) {
133 if (*eq) return w.app(callee, w.tuple({m, kv}));
134 if (*Lit::isa(key(kv)) < *Lit::isa(key(kv_)))
135 return w.app(callee, w.tuple({w.app(callee, w.tuple({m, kv})), kv_}));
136 }
137 }
138
139 return {};
140}
