19/// Sequentializes the parallel copies @p moves between registers and @p slot_moves between spill slots.
20void Emitter::parallel(Moves moves, fe::Vector<std::pair<size_t, size_t>> slot_moves) {
21 std::erase_if(slot_moves, [](auto m) { return m.first == m.second; });
22 for (auto [_, from] : slot_moves)
23 push_slot(from);
24 for (auto [to, _] : slot_moves | std::views::reverse)
25 pop_slot(to);
26
27 auto imms = Moves();
28 for (auto& m : moves)
29 if (m.second.imm) imms.emplace_back(m);
30 std::erase_if(moves, [](const auto& m) { return m.second.imm || m.first == m.second.reg; });
31
32 while (!moves.empty()) {
33 auto free = std::ranges::find_if(moves, [&](const auto& m) {
34 return std::ranges::none_of(moves, [&](const auto& n) { return n.second.reg == m.first; });
35 });
36 if (free != moves.end()) {
37 move(free->first, free->second.reg);
38 moves.erase(free);
39 continue;
40 }
41 // Only cycles are left.
42 auto [d, s] = moves.back();
43 moves.pop_back();
44 if (auto tmp = scratch(d.cls)) {
45 move(*tmp, d);
46 for (auto& [_, from] : moves)
47 if (from.reg == d) from.reg = *tmp;
48 move(d, s.reg);
49 } else {
50 swap(s.reg, d);
51 for (auto& [_, from] : moves)
52 if (from.reg == d)
53 from.reg = s.reg;
54 else if (from.reg == s.reg)
55 from.reg = d;
56 std::erase_if(moves, [](const auto& m) { return m.first == m.second.reg; });
57 }
58 }
59
60 for (auto [dst, s] : imms)
61 mov_imm(dst, s.imm);
62}

64void Emitter::copy(Lam* target, const Def* arg) {
65 auto moves = Moves();
66 auto slot_moves = fe::Vector<std::pair<size_t, size_t>>();
67 auto n = target->num_vars();
68 for (size_t i = 0; i != n; ++i) {
69 auto x = target->var(n, i);
70 auto a = arg->proj(n, i);
71 if (Target::is_mem(x)) continue;
72 if (auto cls = target_.cls(x); cls == Target::Stack)
73 slot_moves.emplace_back(spill_slot(x), spill_slot(a));
74 else if (auto r = fe::lookup(regs_, x))
75 moves.emplace_back(Reg{*r, cls}, source(a));
76 }
77 parallel(std::move(moves), std::move(slot_moves));
78}

84void Emitter::call(Lam* callee, const Def* arg) {
85 auto n = arg->num_projs();
86 auto moves = Moves();
87 auto num = std::array<size_t, Target::Num_Classes>{};
88 for (size_t i = 0; i + 1 < n; ++i) {
89 if (auto x = arg->proj(n, i); !Target::is_mem(x)) {
90 auto cls = target_.cls(Target::strip(x));
91 auto& c = num[cls];
92 if (c == target_.num_arg_regs(cls))
93 arg->blame("{} backend: too many arguments are not supported yet", backend()).bail();
94 moves.emplace_back(Reg{target_.arg_reg(cls, c++), cls}, source(x));
95 }
96 }
97 parallel(std::move(moves), {});
98 call_insn(symbol(callee), num);
99
100 auto ret = arg->proj(n, n - 1);
101 if (ret == ret_) return epilogue();
102
103 auto k = Lam::isa_mut_basicblock(ret);
104 auto m = k->num_vars();
105 auto back = Moves();
106 for (size_t i = 0; i != m; ++i)
107 if (auto x = k->var(m, i); !Target::is_mem(x)) {
108 auto cls = target_.cls(x);
109 if (auto r = fe::lookup(regs_, x)) back.emplace_back(Reg{*r, cls}, Src{.reg = {target_.ret_reg(cls), cls}});
110 }
111 parallel(std::move(back), {});
112 jump(k);
113}

115void Emitter::insn(const App* app) {
116 if (Axm::isa<sched>(app)) {
117 auto f = Target::sched_insn(app);
118 auto res = app->proj(2, 1);
119 if (Axm::isa<mach::spill, 1>(f)) return spill(source(app->arg()), app->arg()->type(), spill_slot(res));
120 if (Axm::isa<mach::reload, 1>(f))
121 return reload(Reg{def_reg(res), target_.cls(res)}, res->type(), spill_slot(app->arg()));
122 } else if (Axm::isa<slot>(app)) {
123 auto first = alloc_slots(nat(app->callee()->as<App>()->arg()));
124 return slot_addr(def_reg(app->proj(2, 1)), first, num_slots_ - first);
125 }

129void Emitter::block(Lam* lam, const fe::Vector<const App*>& insns) {
130 for (auto app : insns)
131 insn(app);
132
133 auto term = lam->body()->as<App>();
134 auto arg = term->arg();
135 auto callee = term->callee();
136 if (callee == ret_) {
137 auto moves = Moves();
138 for (auto a : term->args()) {
139 if (Target::is_mem(a)) continue;
140 auto cls = target_.cls(Target::strip(a));
141 moves.emplace_back(Reg{target_.ret_reg(cls), cls}, source(a));
142 }
143 if (moves.size() > 1) lam->blame("{} backend: `{}` returns more than one value", backend(), lam).bail();
144 parallel(std::move(moves), {});
145 epilogue();
146 } else if (auto f = Lam::isa_mut_returning(callee)) {
147 call(f, arg);
148 } else if (auto ex = callee->isa<Extract>()) {
149 auto targets = ex->tuple();
150 if (targets->num_projs() != 2)
151 lam->blame("{} backend: only two-way branches are supported yet", backend()).bail();
152 auto [e, t] = targets->projs<2>([](const Def* d) { return Lam::isa_mut_basicblock(d); });
153 if (!e || !t) lam->blame("{} backend: `{}` branches to something other than blocks", backend(), lam).bail();
154 auto c = ex->index();
155 auto trampoline = takes_values(t);
156 auto t_lbl = trampoline ? label(lam) + ".t" : label(t);
157 if (auto app = target_.isa_sched_cc(c)) {
158 use_flags(app->arg());
159 branch_cc(std::get<0>(Axm::get(Target::sched_insn(app)))->sub(), t_lbl);
160 } else if (Lit::isa(c)) {
161 c->blame("{} backend: a branch on the literal `{}`", backend(), c).bail();
162 } else {
163 branch_if(use_reg(c), t_lbl);
164 }
165 copy(e, arg);
166 if (trampoline) {
167 jmp(label(e));
168 label_here(t_lbl);
169 copy(t, arg);
170 jump(t);
171 } else {
172 jump(e);
173 }
174 } else if (auto bb = Lam::isa_mut_basicblock(callee)) {
175 copy(bb, arg);
176 jump(bb);
177 } else {
178 lam->blame("{} backend: cannot emit the jump to `{}`", backend(), callee).bail();
179 }

182void Emitter::run(bool global) {
183 auto live = Liveness(target_, fun_);
184 if (!ret_) fun_->blame("{} backend: `{}` does not end in a return continuation", backend(), fun_).bail();
185
186 auto done = DefSet();
187 for (const auto& b : live.blocks()) {
188 for (auto insn : b.insns) {
189 collect(insn->callee(), done);
190 collect(insn->arg(), done);
191 for (auto [j, r] : target_.constraint(insn).results)
192 regs_.emplace(Target::result(insn, j), r);
193 }
194 collect(b.lam->body(), done);
195 }
196 auto used = std::array<fe::Vector<bool>, Target::Num_Classes>{};
197 for (size_t c = 0; c != Target::Num_Classes; ++c)
198 used[c].resize(target_.num_regs(Target::Class(c)));
199 for (auto [def, r] : regs_)
200 if (auto& u = used[target_.cls(def)]; r < u.size()) u[r] = true;
201 for (size_t c = 0; c != Target::Num_Classes; ++c) {
202 auto cls = Target::Class(c);
203 for (size_t r = 0; r != used[c].size(); ++r)
204 if (used[c][r] && !target_.caller_saved(cls, r) && !target_.reserved(cls, r)) saved_[c].emplace_back(r);
205 }
206
207 auto order = layout(live);
208
209 auto n = fun_->num_vars();
210 auto moves = Moves();
211 auto num = std::array<size_t, Target::Num_Classes>{};
212 for (size_t i = 0; i + 1 < n; ++i) {
213 if (auto x = fun_->var(n, i); !Target::is_mem(x)) {
214 auto cls = target_.cls(x);
215 auto& c = num[cls];
216 if (c == target_.num_arg_regs(cls))
217 fun_->blame("{} backend: too many arguments are not supported yet", backend()).bail();
218 if (auto r = fe::lookup(regs_, x))
219 moves.emplace_back(Reg{*r, cls}, Src{.reg = {target_.arg_reg(cls, c), cls}});
220 ++c;
221 }
222 }
223 parallel(std::move(moves), {});
224
225 for (size_t i = 0; i != order.size(); ++i) {
226 auto b = order[i];
227 next_ = i + 1 != order.size() ? order[i + 1]->lam : nullptr;
228 if (i != 0) label_here(label(b->lam));
229 flags_ = nullptr;
230 block(b->lam, b->insns);
231 }
232
233 auto sym = symbol(fun_);
234 if (global) std::println(os_, "\t.globl {}", sym);
235 if (Target::is_elf()) std::println(os_, "\t.type {}, {}", sym, function_type());
236 std::println(os_, "{}:", sym);
237 prologue(os_);
238 os_ << body_.view();
239 if (Target::is_elf()) std::println(os_, "\t.size {}, .-{}", sym, sym);
240}
