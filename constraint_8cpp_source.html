31 // clang-format off
32 if (id == Annex::Base<div> || id == Annex::Base<idiv>) {
33 res.fixed.emplace_back(0, w.annex(reg::rax));
34 res.results.emplace_back(0, w.annex(reg::rax));
35 res.results.emplace_back(1, w.annex(reg::rdx));
36 res.clobbers.emplace_back(w.annex(reg::rdx));
37 } else if (id == Annex::Base<shl> || id == Annex::Base<shr> || id == Annex::Base<sar>) {
