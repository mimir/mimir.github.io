115void LitExpr::stream(fe::Tab& tab, std::ostream& os) const {
116 switch (tag()) {
117 case Tag::L_i: {
118 auto [size, val] = tok().lit_i();
119 if (size == 0 || (size > 1 && std::has_single_bit(size)))
120 std::print(os, "{}I{}", val, Idx::size2bitwidth(size));
121 else
122 std::print(os, "{}_{}", val, size);
123 break;
124 }
125 case Tag::L_s: os << std::bit_cast<s64>(tok().lit_u()); break;
126 case Tag::L_u: os << tok().lit_u(); break;
127 case Tag::L_f: {
128 auto str = std::format("{}", std::bit_cast<f64>(tok().lit_u()));
129 // The shortest round-tripping spelling may lack both `.` and exponent, which would lex as an integer.
130 if (str.find_first_of(".ein") == std::string::npos) str += ".0";
131 os << str;
132 break;
133 }
134 case Tag::L_c: std::print(os, "'{}'", escape_char(tok().lit_c())); break;
135 case Tag::L_str: std::print(os, "\"{}\"", Lexer::escape(tok().sym().view())); break;
136 case Tag::T_bot:
137 case Tag::T_top: os << tag(); break;
138 default: fe::unreachable();
139 }
140 if (type()) std::print(os, ":{}", S(tab, type()));
141}

257/// The members of an `axm` group in @p decls, starting at @p i with its owning AxmDecl; see Parser::parse_axm_group.
258static size_t axm_group_end(fe::View<Ptr<ValDecl>> decls, size_t i) {
259 auto owner = decls[i]->as<AxmDecl>();
260 auto prev = owner->dbg().sym();
261 auto is_alias = [&](const ValDecl* decl) {
262 auto alias = decl->isa<AliasDecl>();
263 return alias && alias->vis() == Vis::Pub && alias->path()->dbgs().size() == 1
264 && alias->path()->front().sym() == prev;
265 };
266 for (++i; i != decls.size(); ++i)
267 if (auto sibling = decls[i]->isa<AxmDecl::Sibling>(); sibling && sibling->owner() == owner)
268 prev = sibling->dbg().sym();
269 else if (!is_alias(decls[i].get()))
270 break;
271 return i;
272}

282/// `axm [mod.](tag_0 = alias, ..., tag_n-1): ...;` of the `axm` group `decls[begin, end)`.
283static void
284stream_axm_group(fe::Tab& tab, std::ostream& os, fe::View<Ptr<ValDecl>> decls, size_t begin, size_t end, Dbg mod = {}) {
285 if (decls[begin]->vis() == Vis::Priv) os << "priv ";
286 os << "axm ";
287 if (mod) std::print(os, "{}.", mod);
288 os << '(';
289 for (auto i = begin; i != end; ++i)
290 if (decls[i]->isa<AliasDecl>())
291 std::print(os, " = {}", decls[i]->dbg());
292 else
293 std::print(os, "{}{}", i == begin ? "" : ", ", decls[i]->dbg());
294 os << ')';
295 stream_axm_tail(tab, os, decls[begin]->as<AxmDecl>());
296}

298static void stream_decls(fe::Tab& tab, std::ostream& os, fe::View<Ptr<ValDecl>> decls) {
299 for (size_t i = 0, end; i != decls.size(); i = end) {
300 // Siblings share their owner's type, so they must stay in the group that introduces them.
301 end = decls[i]->isa<AxmDecl>() ? axm_group_end(decls, i) : i + 1;
302 os << tab;
303 if (end != i + 1)
304 stream_axm_group(tab, os, decls, i, end), os << '\n';
305 else
306 std::println(os, "{}", S(tab, decls[i].get()));
307 }
308}
