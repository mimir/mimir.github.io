50void ADCE::Analysis::analyze(const Def* def) {
51 // A Var's lattice spells all of its projections; the live ones already reach us via their Extract.
52 if (def->isa<Var>()) return;
53 if (auto [_, ins] = visited_.emplace(def); !ins) return;
54 if (auto l = lookup(def); l && l != def) return analyze(l);
55
56 if (auto dead = def->isa<Proxy>()) {
57 if (dead->tag() == Proxy_Dead && pin(dead->op(0))) log().d("live: {}", dead->op(0));
58 return; // never walk a proxy's deps: its var would look live
59 }
60
61 if (auto app = def->isa<App>()) {
62 if (auto lam = Lam::isa_rewritable(app->callee())) {
63 analyze(app->type());
64
65 auto n = lam->num_tvars();
66 for (size_t i = 0; i != n; ++i)
67 if (!is_dead(lam->tvar(i))) analyze(app->arg(n, i));
68
69 for (auto d : lam->deps())
70 analyze(d);
71 return;
72 }
73 } else if (auto [lam, var] = def->isa_binder<Lam>(); lam && lam->is_rewritable()) {
74 log().d("unknown lam: {}", lam); // reached as a value, so its signature must stay untouched
75 for (auto v : lam->tvars())
76 pin(v);
77 }
78
79 for (auto d : def->deps())
80 analyze(d);
81}
