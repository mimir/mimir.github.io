217Kernel mk_kernel(const Op& op, nat_t n_groups, nat_t n_items, const Def* shared_ty, std::string name) {
218 auto& w = op.w;
219 auto nis = op.ins.size();
220 auto nps = op.posts.size();
221
222 auto global_ty = w.annex<gpu::GlobalM>();
223 auto shared_m = w.annex<gpu::SharedM>();
224 auto const_ty = w.annex<gpu::ConstM>();
225 auto local_ty = w.annex<gpu::LocalM>();
226
227 DefVec arg_tys;
228 for (auto& in : op.ins)
229 arg_tys.emplace_back(in.dptr->type());
230 for (auto& in : op.posts)
231 arg_tys.emplace_back(in.dptr->type());
232 arg_tys.emplace_back(op.out_dptr->type());
233
234 auto shared_ptr_ty = shared_ty ? w.app(w.annex<gpu::SharedPtr>(), shared_ty) : w.sigma();
235 auto lam = w.mut_con(Defs{global_ty, shared_m, const_ty, local_ty, w.type_idx(n_groups), w.type_idx(n_items),
236 shared_ptr_ty, w.sigma(arg_tys), w.cn({global_ty, shared_m, const_ty, local_ty})})
237 ->set(name);
238 auto [global, shared, const_tok, local, group, item, shared_ptr, args, ret] = lam->vars<9>();
239
240 auto n = nis + nps + 1;
241 Kernel k{lam, global, shared, const_tok, local, group,
242 item, shared_ptr, ret, {}, {}, args->proj(n, nis + nps),
243 n_groups, n_items, shared_ty};
244 for (size_t i = 0; i != nis; ++i)
245 k.ins.emplace_back(args->proj(n, i));
246 for (size_t j = 0; j != nps; ++j)
247 k.posts.emplace_back(args->proj(n, nis + j));
248 return k;
249}

269/// Fills @p at with the epilogue: applies `post` to @p acc at the output cell of the parallel coordinates @p par,
270/// stores the result and returns from the kernel.
271void emit_epilogue(const Op& op,
272 const Kernel& k,
273 Lam* at,
274 const Def* global,
275 const Def* shared,
276 const Def* acc,
277 const DefVec& par) {

279 auto& mr = op.mr;
280
281 DefVec idx = par;

283 idx.emplace_back(op.zero(op.ro + j));

285 = btensor::op_affine_map(mr.map_out, mr.Ro, op.n(), mr.Sr, mr.So, w.tuple(idx), global)->projs<2>();

292 auto [post_mem, elem_post] = after_post->vars<2>();

299/// One thread per output point, reducing sequentially over the `rr` reduction dims.
300Kernel build_kernel(const Op& op) {
301 auto& w = op.w;
302 auto& mr = op.mr;
303 auto global_ty = w.annex<gpu::GlobalM>();
304
305 auto total = product(op.out_dims);
306 auto n_items = std::min<nat_t>(total, 1024);
307 auto k = mk_kernel(op, (total + n_items - 1) / n_items, n_items, nullptr, "mapReduceKernel");
308 auto flat = add_i64(mul_i64(group_i64(k), n_items), item_i64(k));
309
310 auto in_range = w.call(core::icmp::ul, Defs{flat, w.lit_i64(total)});
311 auto early_return = w.mut_con(w.sigma(Defs{}))->set("outOfRange");
312 early_return->app(true, k.ret, Defs{k.global, k.shared, k.const_tok, k.local});
313 auto body = w.mut_con(w.sigma(Defs{}))->set("inRange");
314 k.lam->branch(true, in_range, body, early_return);
315
316 auto write_back = w.mut_con(Defs{global_ty, mr.To})->set("writeBack");
317 auto [wb_mem, acc_final] = write_back->vars<2>();
318 auto [wb_mem2, wb_par] = unflatten_index(w, flat, op.out_dims, wb_mem);
319 emit_epilogue(op, k, write_back, wb_mem2, k.shared, acc_final, wb_par);
320
321 const Def* acc = w.tuple({k.global, mr.init});
322 const Def* cont = write_back;
323 Lam* cur_lam = body;
324 auto red_exts = mr.Sr->projs(op.rn());
325 auto raw = affine::counting_nest(cur_lam, cont, acc, Defs(red_exts).subspan(op.ro, op.rr), "forRed");
326 auto red_iters = btensor::to_idxs(w, Defs(red_exts).subspan(op.ro, op.rr), raw);
327 auto [red_mem, elem_acc] = acc->projs<2>();
328
329 auto [cur, iters_v] = unflatten_index(w, flat, op.out_dims, red_mem);
330 iters_v.insert(iters_v.end(), red_iters.begin(), red_iters.end());
331 auto iters = w.tuple(iters_v);
332
333 DefVec elems(op.ins.size());
334 for (size_t i = 0; i != op.ins.size(); ++i)
335 std::tie(cur, elems[i]) = load_at(op.ins[i], k.ins[i], op.n(), mr.Sr, iters, cur);
336
337 btensor::app_cps(cur_lam, op.comb, {cur, elem_acc, w.tuple(elems)}, cont);
338 return k;
339}

407/// How two inputs that both read every reduction position split the parallel positions between them.
408struct Split {
409 Vector<nat_t> pos0, pos1; ///< Ascending; together they partition `[0, ro)`.
410};
411
412std::optional<Split> split_parallel(const Op& op) {
413 if (op.ins.size() != 2 || op.rr < 1 || op.ro < 2) return {};
414 auto rn = op.rn();
415 for (auto& in : op.ins)
416 for (nat_t p = op.ro; p != rn; ++p)
417 if (!depends_on_axis(in.map, rn, p)) return {};
418
419 Split split;
420 for (nat_t p = 0; p != op.ro; ++p) {
421 auto d0 = depends_on_axis(op.ins[0].map, rn, p);
422 auto d1 = depends_on_axis(op.ins[1].map, rn, p);
423 if (d0 == d1) return {};
424 (d0 ? split.pos0 : split.pos1).emplace_back(p);
425 }
426 if (split.pos0.empty() || split.pos1.empty()) return {};
427 return split;
428}

433/// A shared-memory tiled contraction: input 0 owns the rows `split.pos0`, input 1 the columns `split.pos1`.
434/// Each `Gemm_Tile`² thread block covers one output tile and streams the reduction through shared memory a tile at a
435/// time, so the block reads every row and column from global memory once.
436/// All three flattened extents must be multiples of `Gemm_Tile`.
437Kernel build_kernel_gemm(const Op& op, const Split& split, const Vector<nat_t>& k_ext) {
438 auto& w = op.w;
439 auto& mr = op.mr;
440 auto global_ty = w.annex<gpu::GlobalM>();
441 auto shared_m = w.annex<gpu::SharedM>();
442 auto tile = Gemm_Tile;
443 auto rn = op.rn();
444
445 Vector<nat_t> m_ext, n_ext;
446 for (auto p : split.pos0)
447 m_ext.emplace_back(op.out_dims[p]);
448 for (auto p : split.pos1)
449 n_ext.emplace_back(op.out_dims[p]);
450 auto n_blocks0 = product(m_ext) / tile;
451 auto n_blocks1 = product(n_ext) / tile;
452
453 // The row stride `tile + 1` staggers same-column accesses across shared-memory banks.
454 auto tile_ty = [&](const Def* t) { return w.arr(w.lit_nat(tile), w.arr(w.lit_nat(tile + 1), t)); };
455 auto shared_ty = w.sigma({tile_ty(op.ins[0].t), tile_ty(op.ins[1].t)});
456 auto k = mk_kernel(op, n_blocks0 * n_blocks1, tile * tile, shared_ty, "gemmTiledKernel");
457 auto shared0 = lea_i64(k.shared_ptr, w.lit_i64(0));
458 auto shared1 = lea_i64(k.shared_ptr, w.lit_i64(1));
459
460 auto [g0, block0, block1] = divmod(k.global, group_i64(k), n_blocks1);
461 auto [g1, ty, tx] = divmod(g0, item_i64(k), tile);
462 auto [g2, m_coords] = unflatten_index(w, add_i64(mul_i64(block0, tile), ty), m_ext, g1);
463 auto [entry, n_coords] = unflatten_index(w, add_i64(mul_i64(block1, tile), tx), n_ext, g2);
464
465 auto ty_idx = w.call(core::conv::u, w.lit_nat(tile), ty);
466 auto tx_idx = w.call(core::conv::u, w.lit_nat(tile), tx);
467 auto ty_pad = w.call(core::conv::u, w.lit_nat(tile + 1), ty);
468 auto tx_pad = w.call(core::conv::u, w.lit_nat(tile + 1), tx);
469
470 // A position the input does not read may hold any in-bounds value.
471 auto iters = [&](const Vector<nat_t>& pos, const DefVec& coords, const DefVec& k_vals) {
472 auto iv = DefVec(rn, [&](size_t p) { return op.zero(p); });
473 for (size_t j = 0; j != pos.size(); ++j)
474 iv[pos[j]] = coords[j];
475 for (nat_t j = 0; j != op.rr; ++j)
476 iv[op.ro + j] = k_vals[j];
477 return w.tuple(iv);
478 };
479
480 // The loops below carry only the shared memory: `affine.lower_for` threads a single memory through a loop.
481 auto write_back = w.mut_con(Defs{shared_m, mr.To})->set("gemmWriteBack");
482 auto [wb_shared, wb_acc] = write_back->vars<2>();
483 auto par = DefVec(op.ro);
484 for (size_t j = 0; j != split.pos0.size(); ++j)
485 par[split.pos0[j]] = m_coords[j];
486 for (size_t j = 0; j != split.pos1.size(); ++j)
487 par[split.pos1[j]] = n_coords[j];
488 emit_epilogue(op, k, write_back, entry, wb_shared, wb_acc, par);
489
490 auto [outer, outer_for] = affine::counting_for(w.lit_i64(product(k_ext) / tile), w.tuple({k.shared, mr.init}),
491 write_back, w.sym("gemmKTile"));
492 k.lam->set(true, outer_for);
493 auto [kt, outer_acc, outer_yield] = outer->vars<3>();
494 auto [s0, acc0] = outer_acc->projs<2>();
495 auto k_base = mul_i64(kt, tile);
496
497 auto [g3, k0] = unflatten_index(w, add_i64(k_base, tx), k_ext, entry);
498 auto [g4, val0] = load_at(op.ins[0], k.ins[0], op.n(), mr.Sr, iters(split.pos0, m_coords, k0), g3);
499 auto [g5, k1] = unflatten_index(w, add_i64(k_base, ty), k_ext, g4);
500 auto [g6, val1] = load_at(op.ins[1], k.ins[1], op.n(), mr.Sr, iters(split.pos1, n_coords, k1), g5);
501
502 auto s1 = w.call<mem::store>(Defs{s0, mem::op_lea_tuple(shared0, w.tuple({ty_idx, tx_pad})), val0});
503 auto s2 = w.call<mem::store>(Defs{s1, mem::op_lea_tuple(shared1, w.tuple({tx_idx, ty_pad})), val1});
504 auto [g7, s3] = w.app(w.annex<gpu::sync_work_items>(), Defs{g6, s2})->projs<2>();
505
506 auto after_inner = w.mut_con(Defs{global_ty, mr.To})->set("gemmAfterInner");
507 auto [inner, inner_for]
508 = affine::counting_for(w.lit_i64(tile), w.tuple({g7, acc0}), after_inner, w.sym("gemmInnerK"));
509 outer->set(true, inner_for);
510 auto [kk, inner_acc, inner_yield] = inner->vars<3>();
511 auto [gi, acci] = inner_acc->projs<2>();
512 auto kk_idx = w.call(core::conv::u, w.lit_nat(tile + 1), kk);
513
514 auto [r0, v0] = w.call<mem::load>(Defs{s3, mem::op_lea_tuple(shared0, w.tuple({ty_idx, kk_idx}))})->projs<2>();
515 auto [r1, v1] = w.call<mem::load>(Defs{r0, mem::op_lea_tuple(shared1, w.tuple({tx_idx, kk_idx}))})->projs<2>();
516 btensor::app_cps(inner, op.comb, {gi, acci, w.tuple({v0, v1})}, inner_yield);
517
518 // The block must finish reading the tile before the next round overwrites it.
519 auto [ai_mem, ai_acc] = after_inner->vars<2>();
520 auto s4 = w.app(w.annex<gpu::sync_work_items>(), Defs{ai_mem, s3})->proj(2, 1);
521 after_inner->app(true, outer_yield, Defs{s4, ai_acc});
522
523 return k;
524}

526/// A CUDA block holds at most 1024 threads.
527constexpr nat_t Conv_Max_Ow = 1024;
528/// Only `ow` threads load the filter per round, so a large filter over a narrow row would load for too long.
529constexpr nat_t Conv_Max_Load_Iters = 8;
530
531/// A convolution-shaped reduction: the weight input reads the parallel positions `weight_pos`, the data input the rest.
532/// A thread block is one output row; it loads its whole filter into shared memory once and every thread reads the
533/// weights from there, while the data input is read along the row with unit stride.
534Kernel build_kernel_conv(const Op& op,
535 nat_t weight_idx,
536 const Vector<nat_t>& weight_pos,
537 const Vector<nat_t>& data_pos,
538 const Vector<nat_t>& filter_ext) {
539 auto& w = op.w;
540 auto& mr = op.mr;
541 auto global_ty = w.annex<gpu::GlobalM>();
542 auto shared_m = w.annex<gpu::SharedM>();
543 auto data_idx = 1 - weight_idx;
544 auto& weight = op.ins[weight_idx];
545
546 // A schedule only splits lower positions, so the last data position is still the fastest-varying one.
547 auto ow_pos = data_pos.back();
548 auto ow = op.out_dims[ow_pos];
549
550 auto outer_pos = weight_pos;
551 for (auto p : data_pos)
552 if (p != ow_pos) outer_pos.emplace_back(p);
553 std::ranges::sort(outer_pos);
554 Vector<nat_t> outer_ext;
555 for (auto p : outer_pos)
556 outer_ext.emplace_back(op.out_dims[p]);
557
558 auto filter_total = product(filter_ext);
559 auto filter_strides = row_major_strides(filter_ext);
560 // Padding the cache to whole rounds lets every round load unconditionally; the surplus slots are never read.
561 auto rounds = (filter_total + ow - 1) / ow;
562
563 auto shared_ty = w.arr(w.lit_nat(rounds * ow), weight.t);
564 auto k = mk_kernel(op, product(outer_ext), ow, shared_ty, "convTiledKernel");
565 auto item = item_i64(k);
566
567 auto [entry, outer_coords] = unflatten_index(w, group_i64(k), outer_ext, k.global);
568 auto par = DefVec(op.ro);
569 for (size_t j = 0; j != outer_pos.size(); ++j)
570 par[outer_pos[j]] = outer_coords[j];
571 par[ow_pos] = w.call(core::conv::u, op.extent(ow_pos), item);
572
573 auto iters = [&](const DefVec& red) {
574 auto iv = par;
575 iv.insert(iv.end(), red.begin(), red.end());
576 return w.tuple(iv);
577 };
578
579 // The load loop carries only the shared memory: `affine.lower_for` threads a single memory through a loop.
580 auto after_load = w.mut_con(shared_m)->set("convAfterLoad");
581 auto [load, load_for] = affine::counting_for(w.lit_i64(rounds), k.shared, after_load, w.sym("convWeightLoad"));
582 k.lam->set(true, load_for);
583 auto [round, load_shared, load_yield] = load->vars<3>();
584
585 auto load_flat = add_i64(mul_i64(round, ow), item);
586 auto [lm, load_coords] = unflatten_index(w, load_flat, filter_ext, entry);
587 auto load_val = load_at(weight, k.ins[weight_idx], op.n(), mr.Sr, iters(load_coords), lm).second;
588 load->app(true, load_yield, w.call<mem::store>(Defs{load_shared, lea_i64(k.shared_ptr, load_flat), load_val}));
589
590 auto [bar_mem, bar_shared] = w.app(w.annex<gpu::sync_work_items>(), Defs{entry, after_load->var()})->projs<2>();
591
592 auto write_back = w.mut_con(Defs{global_ty, mr.To})->set("convWriteBack");
593 auto [wb_mem, acc_final] = write_back->vars<2>();
594 emit_epilogue(op, k, write_back, wb_mem, bar_shared, acc_final, par);
595
596 const Def* acc = w.tuple({bar_mem, mr.init});
597 const Def* cont = write_back;
598 Lam* cur_lam = after_load;
599 auto red_exts = mr.Sr->projs(op.rn());
600 auto raw = affine::counting_nest(cur_lam, cont, acc, Defs(red_exts).subspan(op.ro, op.rr), "convRed");
601 auto red = btensor::to_idxs(w, Defs(red_exts).subspan(op.ro, op.rr), raw);
602 auto [red_mem, elem_acc] = acc->projs<2>();
603
604 auto [data_mem, data_val] = load_at(op.ins[data_idx], k.ins[data_idx], op.n(), mr.Sr, iters(red), red_mem);
605
606 const Def* weight_flat = w.lit_i64(0);
607 for (nat_t j = 0; j != op.rr; ++j)
608 weight_flat = add_i64(weight_flat, mul_i64(raw[j], filter_strides[j]));
609 auto weight_val = w.call<mem::load>(Defs{bar_shared, lea_i64(k.shared_ptr, weight_flat)})->proj(2, 1);
610
611 DefVec elems(2);
612 elems[data_idx] = data_val;
613 elems[weight_idx] = weight_val;
614 btensor::app_cps(cur_lam, op.comb, {data_mem, elem_acc, w.tuple(elems)}, cont);
615
616 return k;
617}

619/// Picks the most specific kernel the op's structure allows.
620Kernel select_kernel(const Op& op) {
621 Vector<nat_t> red_ext;
622 for (nat_t j = 0; j != op.rr; ++j)
623 if (auto l = Lit::isa<nat_t>(op.extent(op.ro + j)))
624 red_ext.emplace_back(*l);
625 else
626 return build_kernel(op);
627
628 if (auto split = split_parallel(op)) {
629 auto extent_of = [&](const Vector<nat_t>& pos) {
630 nat_t total = 1;
631 for (auto p : pos)
632 total *= op.out_dims[p];
633 return total;
634 };
635 auto tiles = [](nat_t extent) { return extent != 0 && extent % Gemm_Tile == 0; };
636
637 // Flattening a multi-dim reduction into the tiled K loop is slower than caching the whole filter.
638 if (op.rr == 1 && tiles(extent_of(split->pos0)) && tiles(extent_of(split->pos1)) && tiles(red_ext[0]))
639 return build_kernel_gemm(op, *split, red_ext);
640
641 // A schedule's neutral splits leave size-1 positions behind; the weight reads one real position.
642 auto single = [&](const Vector<nat_t>& pos) {
643 return std::ranges::count_if(pos, [&](nat_t p) { return op.out_dims[p] != 1; }) == 1;
644 };
645 if (op.rr >= 2 && single(split->pos0) != single(split->pos1)) {
646 nat_t weight_idx = single(split->pos0) ? 0 : 1;
647 auto& weight_pos = weight_idx == 0 ? split->pos0 : split->pos1;
648 auto& data_pos = weight_idx == 0 ? split->pos1 : split->pos0;
649 auto ow = op.out_dims[data_pos.back()];
650 auto total = product(red_ext);
651 if (total > 1 && ow <= Conv_Max_Ow && (total + ow - 1) / ow <= Conv_Max_Load_Iters)
652 return build_kernel_conv(op, weight_idx, weight_pos, data_pos, red_ext);
653 }
654 }
655
656 return build_kernel(op);
657}

682const App* LowerMapReduce::chainable(const Def* def, const Chain& chain) {
683 if (!uses_.contains(def)) return nullptr;
684 if (auto ex = def->isa<Extract>())
685 if (auto app = ex->tuple()->isa<App>(); app && lowerable(app)) {
686 // Lowered on its own or read outside the chain: keeps its host result.
687 if (lookup(app) || absorbed_.contains(app) || chain.excluded.contains(app)) return nullptr;
688 // Nobody may observe the destination of an absorbed producer, so it has to be a fresh buffer.
689 auto dst = app->arg(app->num_args(), 1)->isa<Extract>();
690 if (dst && Axm::isa<buffer::alloc>(dst->tuple())) return app;
691 }
692 return nullptr;
693}

695void LowerMapReduce::plan_inputs(Call& call, bool post, const Def* group, Chain& chain) {
696 auto& ins = post ? call.posts : call.ins;
697 auto n = ins.size();
698 // A group tuple shared between two calls would chain the same producer into both.
699 auto shared = n > 1 && uses_[group] != 1;
700 for (size_t i = 0; i != n; ++i) {
701 auto old = group->proj(n, i);
702 auto producer = shared ? nullptr : chainable(old, chain);
703 ins[i] = {producer ? plan(producer, chain) : nullptr, old};
704 }

707LowerMapReduce::Call* LowerMapReduce::plan_concat(const App* app, Chain& chain) {
708 auto& w = new_world();
709 auto [TnisR, ax, Sis, s_out] = rewrite(app->callee())->as<App>()->uncurry_args<4>();
710 auto [T, nis, r] = TnisR->projs<3>();
711
712 auto nis_l = Lit::isa<u64>(nis);
713 auto r_l = Lit::isa<u64>(r);
714 auto ax_l = Lit::isa<u64>(ax);
715 if (!nis_l || !r_l || !ax_l) {
716 log().w("{} doesn't have a lowering-time known input count, rank and axis", app);
717 return nullptr;
718 }
719 auto exts = btensor::concat_extents(Sis, *nis_l, *r_l, *ax_l);
720
721 Vector<nat_t> out_dims(*r_l);
722 for (nat_t d = 0; d != *r_l; ++d)
723 if (auto l = Lit::isa<nat_t>(s_out->proj(*r_l, d)))
724 out_dims[d] = *l;
725 else
726 exts.reset();
727 if (!exts || product(out_dims) == 0) {
728 log().w("{} doesn't have a lowering-time known, non-empty shape", app);
729 return nullptr;
730 }
731
732 btensor::MapRedPost mr{};
733 mr.Ro = mr.Rn = r;
734 mr.So = mr.Sr = s_out;
735 mr.To = mr.Tp = T;
736 mr.Sis = Sis;
737 mr.Ris = w.pack(*nis_l, r);
738 mr.Tis = w.pack(*nis_l, T);
739
740 Call call{app, mr, *r_l, 0, out_dims, nullptr, nullptr, Vector<Call::In>(*nis_l), Vector<Call::In>(0)};
741 call.kind = Call::Kind::Concat;
742 call.ax = *ax_l;
743 call.exts = *exts;
744 plan_inputs(call, false, app->arg(3, 2), chain);
745 return &chain.calls.emplace_back(std::move(call));
746}

749 auto [Tr, s_in, params, s_out] = rewrite(app->callee())->as<App>()->uncurry_args<4>();
750 auto [T, r] = Tr->projs<2>();

752
753 auto r_l = Lit::isa<u64>(r);
754 auto mode_l = Lit::isa<u64>(mode);

756 log().w("{} doesn't have a lowering-time known rank and mode", app);

760 Vector<nat_t> out_dims(*r_l);
761 for (nat_t d = 0; d != *r_l; ++d)
762 if (auto l = Lit::isa<nat_t>(s_out->proj(*r_l, d)))

764 else

782 call.kind = Call::Kind::Pad;
783 call.lo = lo;
784 call.fill = fill;
785 plan_inputs(call, false, app->arg(4, 2), chain);
786 return &chain.calls.emplace_back(std::move(call));

797LowerMapReduce::Call* LowerMapReduce::plan_map_reduce(const App* app, Chain& chain) {
798 auto mr = btensor::MapRedPost::of(rewrite(app->callee())->as<App>());
799 auto ranks = mr.ranks();
800 if (!ranks) {
801 log().w("{} doesn't have lowering-time known rank counts (nis/nps/Ro/Rn)", app);
802 return nullptr;
803 }
804 auto [nis, nps, ro, rr] = *ranks;
805
806 Vector<nat_t> out_dims(ro);
807 for (nat_t d = 0; d != ro; ++d) {
808 auto l = Lit::isa<nat_t>(mr.Sr->proj(ro + rr, d));
809 if (!l) {
810 log().w("{} doesn't have a lowering-time known output (grid) shape", app);
811 return nullptr;
812 }
813 out_dims[d] = *l;
814 }
815 if (product(out_dims) == 0) {
816 log().w("{} has a zero-sized output, skipping GPU lowering", app);
817 return nullptr;
818 }
819
820 auto comb = mr.comb->isa_mut<Lam>();
821 auto post = mr.post->isa_mut<Lam>();
822 if (!comb || !post) {
823 log().w("{} doesn't have a lowering-time known combiner/epilogue", app);
824 return nullptr;
825 }
826
827 auto [mem, dst, is, post_is] = app->args<4>();
828 Call call{app, mr, ro, rr, out_dims, comb, post, Vector<Call::In>(nis), Vector<Call::In>(nps)};
829 plan_inputs(call, false, is, chain);
830 plan_inputs(call, true, post_is, chain);
831 return &chain.calls.emplace_back(std::move(call));
832}

834void LowerMapReduce::commit(Call& call, DefVec& hosts, DefMap<size_t>& host_index, Vector<nat_t>& host_readers) {
835 for (auto group : {&call.ins, &call.posts})
836 for (auto& in : *group)
837 if (in.producer) {
838 if (in.producer->readers++ == 0) {
839 absorbed_.emplace(in.producer->app);
840 commit(*in.producer, hosts, host_index, host_readers);
841 }
842 } else {
843 auto [i, fresh] = host_index.emplace(in.host, hosts.size());
844 if (fresh) {
845 hosts.emplace_back(in.host);
846 host_readers.emplace_back(0);
847 }
848 in.host_idx = i->second;
849 ++host_readers[in.host_idx];
850 }

853const Def* LowerMapReduce::emit(Call& call, Session& s) {
854 if (call.dptr) return call.dptr;
855 auto& w = new_world();
856 auto& mr = call.mr;
857
858 auto resolve = [&](const Vector<Call::In>& ins, const Def* Rs, const Def* Ss, const Def* Ts, const Def* maps) {
859 auto n = ins.size();
860 Slots slots(n);
861 for (size_t i = 0; i != n; ++i) {
862 auto map = maps ? maps->proj(n, i) : nullptr;
863 auto& slot = slots[i] = Slot{Rs->proj(n, i), Ss->proj(n, i), Ts->proj(n, i), map, nullptr};
864 if (auto producer = ins[i].producer) {
865 slot.dptr = emit(*producer, s);
866 } else if (auto& dptr = s.uploaded[ins[i].host_idx]) {
867 slot.dptr = dptr;
868 } else {
869 auto copy = w.app(w.app(w.annex<gpu::buf_alloc_copy>(), {slot.r, slot.s, slot.t}),
870 {s.mem, s.global, s.hosts->proj(s.n_hosts, ins[i].host_idx)});
871 std::tie(s.mem, s.global, slot.dptr) = copy->projs<3>();
872 s.pending[slot.dptr] = s.host_readers[ins[i].host_idx];
873 dptr = slot.dptr;
874 }
875 }
876 return slots;
877 };
878
879 auto ins = resolve(call.ins, mr.Ris, mr.Sis, mr.Tis, mr.maps);
880 auto map_red = call.kind == Call::Kind::MapRed;
881 auto posts = map_red ? resolve(call.posts, mr.Rps, mr.Sps, mr.Tps, mr.post_maps) : Slots();
882
883 auto [out_global, out_dptr]
884 = w.app(w.app(w.annex<gpu::alloc>(gpu::alloc::block), w.arr(mr.So, mr.Tp)), s.global)->projs<2>();
885 s.pending[out_dptr] = call.readers;
886
887 Op op{w,
888 mr,
889 call.ro,
890 call.rr,
891 call.out_dims,
892 std::move(ins),
893 std::move(posts),
894 map_red ? rebuild_lam_global_mem(call.comb, mr.To, w.sym("combGlobal")) : nullptr,
895 map_red ? rebuild_lam_global_mem(call.post, mr.Tp, w.sym("postGlobal")) : nullptr,
896 out_dptr};
897 auto k = call.kind == Call::Kind::Concat ? build_kernel_concat(op, call.ax, call.exts)
898 : call.kind == Call::Kind::Pad ? build_kernel_pad(op, call.lo, call.fill)
899 : select_kernel(op);
900
901 DefVec args;
902 for (auto& in : op.ins)
903 args.emplace_back(in.dptr);
904 for (auto& in : op.posts)
905 args.emplace_back(in.dptr);
906 args.emplace_back(out_dptr);
907 DefVec arg_tys(args.size(), [&](size_t i) { return args[i]->type(); });
908
909 auto launch = w.app(w.annex<gpu::launch>(),
910 Defs{w.lit_nat(k.shared_ty ? 1 : 0), k.shared_ty ? k.shared_ty : w.tuple(), w.sigma(arg_tys)});
911 launch = w.app(launch, Defs{w.lit_nat(k.n_groups), w.lit_nat(k.n_items), w.annex<gpu::default_stream>()});
912 launch = w.app(launch, k.lam);
913 launch = w.app(launch, args);
914
915 auto after_launch
916 = w.mut_con(Defs{w.call<mem::M>(0), w.annex<gpu::GlobalM>(), w.annex<gpu::ConstM>()})->set("afterLaunch");
917 s.cur->set(true, w.app(launch, Defs{w.tuple({s.mem, out_global, s.const_tok}), after_launch}));
918 s.cur = after_launch;
919 std::tie(s.mem, s.global, s.const_tok) = after_launch->vars<3>();
920 // Free each input after its last reader.
921 for (size_t i = 0, e = args.size() - 1; i != e; ++i)
922 if (--s.pending[args[i]] == 0) s.global = w.call(gpu::free::block, Defs{s.global, args[i]});
923 return call.dptr = out_dptr;
924}

926const Def* LowerMapReduce::lower_map_reduce_post(const App* app) {
927 auto& w = new_world();
928 auto n_args = app->num_args();
929 auto old_mem = app->arg(n_args, 0);
930 auto old_dst = app->arg(n_args, 1);
931
932 if (absorbed_.contains(app)) return w.tuple({rewrite(old_mem), rewrite(old_dst)});
933
934 // Replan until no absorbed producer is also read outside the chain.
935 Chain chain;
936 Call* call = nullptr;
937 for (;;) {
938 chain.calls.clear();
939 chain.planned.clear();
940 if (!(call = plan(app, chain))) return Super::rewrite_imm_App(app);
941 auto count = [&](auto& self, Call& c) -> void {
942 for (auto group : {&c.ins, &c.posts})
943 for (auto& in : *group)
944 if (in.producer && (in.producer->result = in.host, in.producer->reads++ == 0))
