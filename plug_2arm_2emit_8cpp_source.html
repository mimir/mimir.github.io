23/// Are the @p bits of a float of width @p w the 8-bit immediate of `fmov`, i.e. `a:¬b:b…b:cdefgh:0…0`?
24bool is_fimm8(u64 bits, nat_t w) {
25 auto [zeros, exp] = w == 64 ? std::pair{48, (bits >> 54) & 0x1ff} : std::pair{19, (bits >> 25) & 0x3f};
26 auto b1 = w == 64 ? 0x0ffu : 0x1fu;
27 return (bits & ((u64(1) << zeros) - 1)) == 0 && (exp == b1 || exp == b1 + 1);
28}

121void Emitter::insn(const Def* res, const Def* f, const Def* x) {
122 auto [axm, curry, _] = Axm::get(f);
123 auto id = axm->base();
124
125 auto three = [&](std::string_view mnemonic, nat_t limit) {
126 auto [a, b] = x->projs<2>();
127 auto w = width(res->type());
128 auto lhs = operand(a, w, Scratch2);
129 auto rhs = operand(a, b, w, limit);
130 line("{} {}, {}, {}", mnemonic, reg(def_reg(res), w), lhs, rhs);
131 };
132 auto fthree = [&](std::string_view mnemonic) {
133 auto [a, b] = x->projs<2>();
134 auto t = res->type();
135 line("{} {}, {}, {}", mnemonic, name(def_reg(res), t), name(use_reg(a), t), name(use_reg(b), t));
136 };
137 auto two = [&](std::string_view mnemonic) {
138 line("{} {}, {}", mnemonic, name(def_reg(res), res->type()), name(use_reg(x), x->type()));
139 };
140 constexpr nat_t Imm12 = 4096;
141
142 if (id == Annex::Base<add>)
143 three("add", Imm12);
144 else if (id == Annex::Base<sub>)
145 three("sub", Imm12);
146 else if (id == Annex::Base<and_>)
147 three("and", 0);
148 else if (id == Annex::Base<or_>)
149 three("orr", 0);
150 else if (id == Annex::Base<xor_>)
151 three("eor", 0);
152 else if (id == Annex::Base<mul>)
153 three("mul", 0);
154 else if (id == Annex::Base<madd>) {
155 auto w = width(res->type());
156 auto scratch = std::array{Scratch2, Scratch};
157 auto next = size_t(0);
158 auto ops = x->projs<3>([&](const Def* op) { return operand(op, w, scratch[Lit::isa(op) ? next++ : 0]); });
159 line("madd {}, {}, {}, {}", reg(def_reg(res), w), ops[0], ops[1], ops[2]);
160 } else if (id == Annex::Base<udiv>)
161 three("udiv", 0);
162 else if (id == Annex::Base<sdiv>)
163 three("sdiv", 0);
164 else if (id == Annex::Base<shl>)
165 three("lsl", width(res->type()));
166 else if (id == Annex::Base<shr>)
167 three("lsr", width(res->type()));
168 else if (id == Annex::Base<sar>)
169 three("asr", width(res->type()));
170 else if (id == Annex::Base<zext> || id == Annex::Base<sext>) {
171 auto ws = width(x->type());
172 auto wd = width(res->type());
173 auto rs = use_reg(x);
174 auto rd = def_reg(res);
175 auto mnem = id == Annex::Base<zext> ? "ubfx" : "sbfx";
176 if (ws == 64) return move(Reg{rd}, Reg{rs});
177 line("{} {}, {}, #0, #{}", mnem, reg(rd, wd), reg(rs, wd), ws);
178 } else if (id == Annex::Base<trunc>) {
179 move(Reg{def_reg(res)}, Reg{use_reg(x)});
180 } else if (id == Annex::Base<cmp>) {
181 auto [a, b] = x->projs<2>();
182 auto w = width(a->type());
183 auto lhs = operand(a, w, Scratch2);
184 auto rhs = operand(a, b, w, Imm12);
185 line("cmp {}, {}", lhs, rhs);
186 set_flags(res);
187 } else if (id == Annex::Base<cc>) {
188 use_flags(x);
189 if (auto r = isa_reg(res)) line("cset w{}, {}", *r, ccs[axm->sub()]);
190 } else if (id == Annex::Base<csel>) {
191 auto [fl, a, b] = x->projs<3>();
192 use_flags(fl);
193 auto w = width(res->type());
194 auto ra = operand(a, w, Scratch2);
195 auto rb = operand(a, b, w, 0);
196 line("csel {}, {}, {}, {}", reg(def_reg(res), w), rb, ra, ccs[axm->sub()]);
197 } else if (id == Annex::Base<fadd>)
198 fthree("fadd");
199 else if (id == Annex::Base<fsub>)
200 fthree("fsub");
201 else if (id == Annex::Base<fmul>)
202 fthree("fmul");
203 else if (id == Annex::Base<fdiv>)
204 fthree("fdiv");
205 else if (id == Annex::Base<fsqrt>)
206 two("fsqrt");
207 else if (id == Annex::Base<fabs>)
208 two("fabs");
209 else if (id == Annex::Base<i2f>)
210 two("scvtf");
211 else if (id == Annex::Base<u2f>)
212 two("ucvtf");
213 else if (id == Annex::Base<f2i>)
214 two("fcvtzs");
215 else if (id == Annex::Base<f2u>)
216 two("fcvtzu");
217 else if (id == Annex::Base<f2f>)
218 two("fcvt");
219 else if (id == Annex::Base<fmov>)
220 mov_imm(Reg{def_reg(res), Target::Fpr}, x);
221 else if (id == Annex::Base<fcmp>) {
222 auto [a, b] = x->projs<2>();
223 line("fcmp {}, {}", name(use_reg(a), a->type()), name(use_reg(b), b->type()));
224 set_flags(res);
225 } else if (id == Annex::Base<fcsel>) {
226 auto [fl, a, b] = x->projs<3>();
227 use_flags(fl);
228 auto t = res->type();
229 line("fcsel {}, {}, {}, {}", name(def_reg(res), t), name(use_reg(b), t), name(use_reg(a), t), ccs[axm->sub()]);
230 } else {
231 f->blame("{} backend: cannot emit `{}`", backend(), axm).bail();
232 }

235void Emitter::fmov_imm(size_t dst, const Def* lit) {
236 auto w = width(lit->type());
237 auto bits = *Lit::isa(lit);
238 auto d = name(dst, lit->type());
239 if (bits == 0) return line("fmov {}, {}", d, w == 64 ? "xzr" : "wzr");
240 if (is_fimm8(bits, w)) {
241 auto imm = std::format("{}", w == 64 ? std::bit_cast<f64>(bits) : f64(std::bit_cast<f32>(u32(bits))));
242 if (imm.find('.') == std::string::npos) imm += ".0";
243 return line("fmov {}, #{}", d, imm);
244 }
245 line("ldr {}, ={}", reg(Scratch2, w), w == 64 ? bits : u32(bits));
246 line("fmov {}, {}", d, reg(Scratch2, w));
247}

249void Emitter::step(const App* app) {
250 if (Axm::isa<mach::sched>(app)) return insn(app->proj(2, 1), Target::sched_insn(app), app->arg());
251
252 if (Axm::isa<load>(app) || Axm::isa<fload>(app)) {
253 auto [_, p] = app->args<2>();
254 auto res = app->proj(2, 1);
255 line("ldr{} {}, [x{}]", suffix(width(res->type())), name(def_reg(res), res->type()), use_reg(p));
256 return;
257 }
258
259 if (Axm::isa<store>(app) || Axm::isa<fstore>(app)) {
260 auto [_, p, v] = app->args<3>();
261 auto w = width(v->type());
262 auto src = Lit::isa(v) ? operand(v, w, Scratch2) : name(use_reg(v), v->type());
263 line("str{} {}, [x{}]", suffix(w), src, use_reg(p));
264 return;
265 }
266
267 app->blame("{} backend: cannot emit `{}`", backend(), app).bail();
268}
