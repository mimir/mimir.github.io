12/// Calls @p f for each block, possibly repeatedly, that the jump @p term passes control or a return continuation to.
13void for_each_succ(const App* term, auto f) {
14 auto succ = [&](const Def* def) {
15 if (auto bb = Lam::isa_mut_basicblock(def); bb && bb->is_set()) f(bb);
16 };
17 succ(term->callee());
18 if (auto ex = term->callee()->isa<Extract>())
19 for (auto target : ex->tuple()->projs())
20 succ(target);
21 for (auto arg : term->args())
22 succ(arg);
23}
