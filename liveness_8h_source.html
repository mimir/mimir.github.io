14/// Calls @p f for each block, possibly repeatedly, that the jump @p term passes control or a return continuation to.
15inline void for_each_succ(const App* term, auto f) {
16 auto succ = [&](const Def* def) {
17 if (auto bb = Lam::isa_mut_basicblock(def); bb && bb->is_set()) f(bb);
18 };
19 succ(term->callee());
20 if (auto ex = term->callee()->isa<Extract>())
21 for (auto target : ex->tuple()->projs())
22 succ(target);
23 for (auto arg : term->args())
24 succ(arg);
25}

129/// The block that a call at the end of @p block returns to, if any.

137/// Calls @p f `(d, v)` for each value `v` that is live at the definition of `d` or defined along with it.
138/// Only values that satisfy @p keep take part.
139template<class Keep, class F>
140void interferences(const Liveness& live, const Liveness::Block& block, Keep keep, F f) {
141 auto alive = DefSet();
142 auto gen = [&](const auto& values) {
143 for (auto v : values)
144 if (keep(v)) alive.emplace(v);
145 };
146 auto def = [&](DefVec ds) {
147 std::erase_if(ds, [&](const Def* d) { return !keep(d); });
148 for (auto d : ds)
149 alive.erase(d);
150 for (size_t i = 0; i != ds.size(); ++i) {
151 for (auto v : alive)
152 f(ds[i], v);
153 for (size_t j = i + 1; j != ds.size(); ++j)
154 f(ds[i], ds[j]);
155 }
156 };
157
158 gen(block.live_out);
159 gen(live.term_uses(block));
160 for (auto insn : block.insns | std::views::reverse) {
161 def(live.defs(insn));
162 gen(live.uses(insn));
163 }
164 def(live.params(block));
165}
