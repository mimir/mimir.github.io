15/// Calls @p f for each block, possibly repeatedly, that the jump @p term passes control or a return continuation to.
16inline void for_each_succ(const App* term, auto f) {
17 auto succ = [&](const Def* def) {
18 if (auto bb = Lam::isa_mut_basicblock(def); bb && bb->is_set()) f(bb);
19 };
20 succ(term->callee());
21 if (auto ex = term->callee()->isa<Extract>())
22 for (auto target : ex->tuple()->projs())

130/// The block that a call at the end of @p block returns to, if any.
131inline const Liveness::Block* call_cont(const Liveness& live, Lam* fun, const Liveness::Block& block) {
132 auto term = block.lam->body()->as<App>();

138/// Calls @p f `(d, v)` for each value `v` that is live at the definition of `d` or defined along with it.
139/// Only values that satisfy @p keep take part.
140template<class Keep, class F>
141void interferences(const Liveness& live, const Liveness::Block& block, Keep keep, F f) {
142 auto alive = DefSet();
143 auto gen = [&](const auto& values) {
144 for (auto v : values)
145 if (keep(v)) alive.emplace(v);
146 };
147 auto def = [&](DefVec ds) {
148 std::erase_if(ds, [&](const Def* d) { return !keep(d); });
149 for (auto d : ds)
150 alive.erase(d);
151 for (size_t i = 0; i != ds.size(); ++i) {
152 for (auto v : alive)
153 f(ds[i], v);
154 for (size_t j = i + 1; j != ds.size(); ++j)
155 f(ds[i], ds[j]);
156 }
157 };
158
159 gen(block.live_out);
160 gen(live.term_uses(block));
161 for (auto insn : block.insns | std::views::reverse) {
162 def(live.defs(insn));
163 gen(live.uses(insn));
164 }
165 def(live.params(block));
166}
